    97→/**
    98→ * Execute detected tool calls, send SSE events, and return results.
    99→ * Returns { newContent: combined display messages, shouldContinue: true if tools were executed }
   100→ */
   101→async function executeToolCallsAndContinue(
   102→  toolCalls: NativeToolCall[],
   103→  availableTools: Array<{ id: string; name: string; label: string; icon: string }>,
   104→  currentRound: number,
   105→  maxRounds: number,
   106→  character: CharacterCard,
   107→  sessionId: string,
   108→  userName: string,
   109→  controller: { enqueue: (chunk: string) => void },
   110→  sessionQuests?: SessionQuestInstance[],
   111→  questTemplates?: QuestTemplate[],
   112→  statsConfig?: CharacterStatsConfig,
   113→  sessionStats?: SessionStats,
   114→  allCharacters?: CharacterCard[],
   115→  characterMemory?: CharacterMemory,
   116→  lorebooks?: Lorebook[],
   117→): Promise<{ newContent: string; shouldContinue: boolean; toolResults: Array<{ success: boolean; displayMessage: string }>; questActivations: QuestActivation[]; toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> }> {
   118→  if (toolCalls.length === 0 || currentRound >= maxRounds) {
   119→    return { newContent: '', shouldContinue: false, toolResults: [], questActivations: [], toolsUsed: [] };
   120→  }
   121→
   122→  const toolResults: Array<{ success: boolean; displayMessage: string }> = [];
   123→  const questActivations: QuestActivation[] = [];
   124→  const toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> = [];
   125→  let allDisplayMessages = '';
   126→
   127→  for (let i = 0; i < toolCalls.length; i++) {
   128→    const tc = toolCalls[i];
   129→    const toolDef = availableTools.find(t => t.name === tc.name);
   130→    const callId = `${Date.now()}_${i}`;
   131→
   132→    // Send tool_call_start event
   133→    controller.enqueue(createSSEJSON({
   134→      type: 'tool_call_start',
   135→      toolName: tc.name,
   136→      toolLabel: toolDef?.label || tc.name,
   137→      toolIcon: toolDef?.icon || 'Wrench',
   138→      params: tc.arguments,
   139→      callId,
   140→    }));
   141→
   142→    console.log(`[Tools] Executing tool call: ${tc.name}`, tc.arguments);
   143→
   144→    // Execute the tool
   145→    const toolResult = await executeTool(
   146→      tc.name,
   147→      tc.arguments,
   148→      {
   149→        characterId: character.id,
   150→        characterName: character.name,
   151→        sessionId,
   152→        userName,
   153→        sessionQuests,
   154→        questTemplates,
   155→        statsConfig: character.statsConfig,
   156→        sessionStats,
   157→        allCharacters,
   158→        characterMemory,
   159→        lorebooks,
   160→      },
   161→    );
   162→
   163→    // Send tool_call_result event
   164→    controller.enqueue(createSSEJSON({
   165→      type: 'tool_call_result',
   166→      toolName: tc.name,
   167→      success: toolResult.success,
   168→      displayMessage: toolResult.displayMessage,
   169→      duration: toolResult.duration || 0,
   170→      callId,
   171→    }));
   172→
   173→    // Check for quest activation and send SSE event
   174→    if (toolResult.questActivation) {
   175→      const activation = toolResult.questActivation;
   176→      console.log(`[Tools] Quest activation from ${tc.name}:`, activation);
   177→      
   178→      controller.enqueue(createSSEJSON({
   179→        type: 'quest_activation',
   180→        toolName: tc.name,
   181→        activationType: activation.type,
   182→        key: activation.key,
   183→        metadata: activation.metadata,
   184→      }));
   185→      
   186→      questActivations.push(activation);
   187→    }
   188→
   189→    // Check for action/skill activation and send SSE event
   190→    if (toolResult.actionActivation) {
   191→      const action = toolResult.actionActivation;
   192→      console.log(`[Tools] Action activation from ${tc.name}:`, action.skillName);
   193→      
   194→      controller.enqueue(createSSEJSON({
   195→        type: 'action_activation',
   196→        toolName: tc.name,
   197→        skillId: action.skillId,
   198→        skillName: action.skillName,
   199→        skillDescription: action.skillDescription,
   200→        skillCompletedDescription: action.skillCompletedDescription,
   201→        activationCosts: action.activationCosts,
   202→        activationRewards: action.activationRewards,
   203→        characterId: action.characterId,
   204→      }));
   205→    }
   206→
   207→    // Check for stat modification from modify_stat tool and send SSE event
   208→    if (toolResult.statActivation) {
   209→      const stat = toolResult.statActivation;
   210→      console.log(`[Tools] Stat activation from ${tc.name}:`, stat.attributeKey, stat.oldValue, '→', stat.newValue);
   211→      
   212→      controller.enqueue(createSSEJSON({
   213→        type: 'stat_activation',
   214→        toolName: tc.name,
   215→        characterId: stat.characterId,
   216→        attributeKey: stat.attributeKey,
   217→        attributeName: stat.attributeName,
   218→        attributeType: stat.attributeType,
   219→        oldValue: stat.oldValue,
   220→        newValue: stat.newValue,
   221→        reason: stat.reason,
   222→      }));
   223→    }
   224→
   225→    // Check for solicitud activation/completion and send SSE event
   226→    if (toolResult.solicitudActivation) {
   227→      const sol = toolResult.solicitudActivation;
   228→      console.log(`[Tools] Solicitud activation from ${tc.name}:`, sol.type, sol.solicitudKey);
   229→      
   230→      controller.enqueue(createSSEJSON({
   231→        type: 'solicitud_activation',
   232→        toolName: tc.name,
   233→        activationType: sol.type,
   234→        solicitudKey: sol.solicitudKey,
   235→        targetCharacterId: sol.targetCharacterId,
   236→        targetCharacterName: sol.targetCharacterName,
   237→        fromCharacterId: sol.fromCharacterId,
   238→        fromCharacterName: sol.fromCharacterName,
   239→        description: sol.description,
   240→        completionDescription: sol.completionDescription,
   241→        peticionKey: sol.peticionKey,
   242→      }));
   243→    }
   244→
   245→    // Check for memory activation (sync to client-side Character Memory)
   246→    if (toolResult.memoryActivation) {
   247→      const mem = toolResult.memoryActivation;
   248→      console.log(`[Tools] Memory activation from ${tc.name}:`, mem.type);
   249→      
   250→      controller.enqueue(createSSEJSON({
   251→        type: 'memory_activation',
   252→        toolName: tc.name,
   253→        activationType: mem.type,
   254→        characterId: mem.characterId,
   255→        eventData: mem.eventData,
   256→        relationshipData: mem.relationshipData,
   257→        noteContent: mem.noteContent,
   258→        deleteEventId: mem.deleteEventId,
   259→        deleteEmbeddingId: mem.deleteEmbeddingId,
   260→      }));
   261→    }
   262→
   263→    console.log(`[Tools] Tool ${tc.name}: success=${toolResult.success} duration=${toolResult.duration}ms`, toolResult.displayMessage);
   264→
   265→    toolResults.push({ success: toolResult.success, displayMessage: toolResult.displayMessage });
   266→    toolsUsed.push({
   267→      name: tc.name,
   268→      label: toolDef?.label || tc.name,
   269→      icon: toolDef?.icon || 'Wrench',
   270→      success: toolResult.success,
   271→    });
   272→    if (toolResult.displayMessage) {
   273→      allDisplayMessages += (allDisplayMessages ? '\n' : '') + toolResult.displayMessage;
   274→    }
   275→  }
   276→
   277→  // Return the display messages as content, tool results, and signal that a follow-up call is needed
   278→  return {
   279→    newContent: allDisplayMessages,
   280→    shouldContinue: true, // Always continue to get the LLM's natural response
   281→    toolResults,
   282→    questActivations,
   283→    toolsUsed,
   284→  };
   285→}

   287→export async function POST(request: NextRequest) {
   288→  try {
   289→    // Capture auth headers from Z.ai gateway for token resolution
   290→    // Only use actual JWT tokens - session IDs are NOT valid X-Tokens
   291→    const incomingXToken = request.headers.get('X-Token');
   292→    const fcSecurityToken = request.headers.get('x-fc-security-token');
   293→
   294→    const body = await request.json();
   295→
   296→    // Validate request (automatically detects request type)
   297→    const validation = validateRequest(null, body);
   298→    if (!validation.success) {
   299→      return createErrorResponse(validation.error, 400);
   300→    }
   301→
   302→    const {
   303→      message,
   304→      character,
   305→      messages = [],
   306→      llmConfig,
   307→      userName = 'User',
   308→      persona,
   309→      sessionStats,
   310→      allCharacters, // All characters for peticiones/solicitudes resolution
   311→    } = validation.data;
   312→
   313→    // Extract lorebooks from body (not validated by validation.ts)
   314→    const lorebooks: Lorebook[] = body.lorebooks || [];
   315→
   316→    // Extract HUD context from body
   317→    const hudContext: HUDContextConfig | undefined = body.hudContext;
   318→
   319→    // Extract Quest data for pre-LLM integration (NEW FORMAT)
   320→    const questTemplates: QuestTemplate[] = body.questTemplates || [];
   321→    const sessionQuests: SessionQuestInstance[] = body.sessionQuests || [];
   322→    const questSettings: QuestSettings = {
   323→      ...DEFAULT_QUEST_SETTINGS,
   324→      ...(body.questSettings || {})
   325→    };
   326→
   327→    // Extract Sound data for {{sonidos}} key resolution
   328→    const soundTriggers: SoundTrigger[] = body.soundTriggers || [];
   329→    const soundSettings = body.settings?.sound;
   330→
   331→    // Extract summary for memory/context compression (single summary from session)
   332→    const summary: SessionSummary | undefined = body.summary;
   333→    
   334→    // Extract embeddings chat settings for automatic context retrieval
   335→    const embeddingsChat: Partial<EmbeddingsChatSettings> = body.embeddingsChat || {};
   336→    const sessionId: string | undefined = body.sessionId;
   337→    const characterId: string | undefined = body.characterId;
   338→    
   339→    // Extract character memory (events, relationships, notes from Zustand store)
   340→    const characterMemory: CharacterMemory | undefined = body.characterMemory;
   341→    
   342→    // Extract tools settings for tool/action system (native tool calling only)
   343→    const toolsSettings: ToolsSettings = {
   344→      enabled: body.toolsSettings?.enabled ?? true,
   345→      maxToolCallsPerTurn: body.toolsSettings?.maxToolCallsPerTurn ?? 4,
   346→      characterConfigs: body.toolsSettings?.characterConfigs || [],
   347→      usePromptBasedFallback: body.toolsSettings?.usePromptBasedFallback ?? false,
   348→      disabledTools: body.toolsSettings?.disabledTools || [],
   349→    };
   350→
   351→    // Extract inventory data for Inventory V2 system
   352→    const inventoryData: InventoryPromptData | undefined = body.inventoryData;
   353→
   354→    if (!llmConfig) {
   355→      return createErrorResponse('No LLM configuration provided', 400);
   356→    }
   357→
   358→    // Validate API key for providers that require one
   359→    const providersRequiringApiKey = ['openai', 'vllm', 'lm-studio', 'custom', 'anthropic', 'grok'];
   360→    if (providersRequiringApiKey.includes(llmConfig.provider) && !llmConfig.apiKey) {
   361→      return createErrorResponse(
   362→        `API Key is required for ${llmConfig.provider} provider. Please configure it in Settings.`,
   363→        400
   364→      );
   365→    }
   366→
   367→    // Validate endpoint for providers that require one
   368→    const providersRequiringEndpoint = ['openai', 'vllm', 'lm-studio', 'custom', 'anthropic', 'ollama', 'text-generation-webui', 'koboldcpp'];
   369→    if (providersRequiringEndpoint.includes(llmConfig.provider) && !llmConfig.endpoint) {
   370→      return createErrorResponse(
   371→        `Endpoint URL is required for ${llmConfig.provider} provider. Please configure it in Settings.`,
   372→        400
   373→      );
   374→    }
   375→
   376→    // If using Z.ai provider, collect gateway-forwarded auth headers
   377→    // The SDK v0.0.17+ reads config from /etc/.z-ai-config and uses the "token" field for X-Token.
   378→    // Gateway headers are passed as runtime overrides to the provider.
   379→    let zaiRuntimeToken: string | undefined;
   380→    if (llmConfig.provider === 'z-ai') {
   381→      // Only use actual auth tokens (not session IDs)
   382→      const gatewayToken = incomingXToken || fcSecurityToken || undefined;
   383→      if (gatewayToken) {
   384→        zaiRuntimeToken = gatewayToken;
   385→        console.log(`[Stream Route] Z.ai runtime token available (${gatewayToken.length} chars, source: ${incomingXToken ? 'X-Token' : 'fc-security-token'})`);
   386→      } else {
   387→        console.log(`[Stream Route] Z.ai: no gateway token available, using config file only`);
   388→      }
   389→    }
   390→
   391→    // Sanitize user message
   392→    const sanitizedMessage = sanitizeInput(message);
   393→
   394→    // Create default character if none provided
   395→    const effectiveCharacter: CharacterCard = character || DEFAULT_CHARACTER;
   396→
   397→    // Get effective user name from persona or use provided userName
   398→    const effectiveUserName = getEffectiveUserName(persona, userName);
   399→
   400→    // Build context configuration from request or use defaults
   401→    const contextConfig: Partial<ContextConfig> = body.contextConfig || {};
   402→
   403→    // Apply sliding window to messages
   404→    const contextWindow = selectContextMessages(messages, llmConfig, contextConfig);
   405→
   406→    // Log context stats (for debugging)
   407→    const stats = getContextStats(messages);
   408→
   409→    // Process lorebooks and get matched entries
   410→    const { plan: lorebookPlan, lorebookAttributeKeys, lorebookEntryKeyMap, lorebookDebugEntries } = buildLorebookSectionForPrompt(
   411→      messages,
   412→      lorebooks,
   413→      {
   414→        scanDepth: contextConfig.scanDepth,
   415→        userName: effectiveUserName,
   416→        charName: effectiveCharacter?.name,
   417→      },
   418→      { sessionStats, characterId: effectiveCharacter?.id, characters: allCharacters }
   419→    );
   420→
   421→    // ========================================
   422→    // DEBUG: Build lorebook debug data for SSE event
   423→    // ========================================
   424→    const lorebookDebugData = {
   425→      lorebookAttributeKeys,
   426→      debugEntries: lorebookDebugEntries,
   427→      availableStats: sessionStats
   428→        ? Object.fromEntries(
   429→            Object.entries(sessionStats.characterStats || {}).map(([charId, cs]) => [
   430→              charId,
   431→              cs?.attributeValues ? { ...cs.attributeValues } : '(no values)',
   432→            ])
   433→          )
   434→        : '(no sessionStats)',
   435→    };
   436→
   437→    // ========================================
   438→    // Embeddings Context Retrieval
   439→    // ========================================
   440→    // Enrich search query with recent context for better semantic matching
   441→    const searchContextDepth = embeddingsChat.searchContextDepth || 0;
   442→    let enrichedSearchQuery = sanitizedMessage;
   443→    if (searchContextDepth > 0) {
   444→      // Collect recent messages from history for context enrichment
   445→      const recentHistory = messages
   446→        .filter(m => !m.isDeleted)
   447→        .slice(-(searchContextDepth * 2 + 1)) // user+assistant pairs
   448→        .map(m => m.content?.trim())
   449→        .filter(Boolean)
   450→        .slice(0, -1); // exclude current message (already in sanitizedMessage)
   451→      
   452→      if (recentHistory.length > 0) {
   453→        enrichedSearchQuery = recentHistory.join(' ') + ' ' + sanitizedMessage;
   454→      }
   455→    }
   456→
   457→    // Smart truncation: limit query to the embedding model's context window.
   458→    // Prioritize the user's CURRENT message over history when truncating.
   459→    // The Ollama client also handles truncation, but this avoids sending huge payloads.
   460→    try {
   461→      const { MODEL_CONTEXT_LENGTHS, DEFAULT_CONTEXT_LENGTH, CHARS_PER_TOKEN } = await import('@/lib/embeddings/types');
   462→      const { loadConfig } = await import('@/lib/embeddings/config-persistence');
   463→      const embConfig = loadConfig();
   464→      const modelKey = embConfig.model || 'bge-m3:567m';
   465→      const modelCtx = MODEL_CONTEXT_LENGTHS[modelKey]
   466→        || MODEL_CONTEXT_LENGTHS[modelKey.split(':')[0]]
   467→        || DEFAULT_CONTEXT_LENGTH;
   468→      const maxQueryChars = Math.floor(modelCtx * 0.75 * CHARS_PER_TOKEN);
   469→
   470→      if (enrichedSearchQuery.length > maxQueryChars) {
   471→        // Smart: keep the user's message intact, truncate history prefix
   472→        const userMsgLen = sanitizedMessage.length;
   473→        if (userMsgLen >= maxQueryChars) {
   474→          // User message alone exceeds limit — truncate it
   475→          enrichedSearchQuery = sanitizedMessage.slice(0, maxQueryChars);
   476→        } else {
   477→          // Keep full user message + as much recent history as fits
   478→          const budgetForHistory = maxQueryChars - userMsgLen - 1; // -1 for separator
   479→          enrichedSearchQuery = enrichedSearchQuery.slice(
   480→            Math.max(0, enrichedSearchQuery.length - maxQueryChars)
   481→          );
   482→        }
   483→        console.warn(
   484→          `[Stream Route] Search query trimmed to ${enrichedSearchQuery.length} chars ` +
   485→          `(model: ${modelKey}, context: ${modelCtx} tokens)`
   486→        );
   487→      }
   488→    } catch { /* fallback: no truncation at this level, Ollama client handles it */ }
   489→
   490→    // Retrieve relevant embeddings based on enriched query and settings
   491→    // Pass Character Memory events for deduplication (avoid duplicate memory in prompt)
   492→    const existingMemoryEvents = characterMemory?.events?.map(e => ({
   493→      content: e.content,
   494→      importance: e.importance,
   495→    }));
   496→
   497→    // Extract last assistant message for bidirectional search
   498→    const lastAssistantMsg = messages
   499→      .filter(m => !m.isDeleted && m.role === 'assistant')
   500→      .pop()?.content;
   501→
   502→    const embeddingsResult = await retrieveEmbeddingsContext(
   503→      enrichedSearchQuery,
   504→      characterId || effectiveCharacter.id,
   505→      sessionId,
   506→      embeddingsChat,
   507→      undefined, // groupId
   508→      existingMemoryEvents, // for deduplication
   509→      lastAssistantMsg, // bidirectional search with last assistant message
   510→    );
   511→    
   512→    if (embeddingsResult.found) {
   513→      console.log(`[Stream Route] Retrieved ${embeddingsResult.count} embeddings from namespaces: ${embeddingsResult.searchedNamespaces.join(', ')}`);
   514→    }
   515→
   516→    // ========================================
   517→    // Build system prompt with unified key resolution
   518→    // ========================================
   519→    // This handles ALL key resolution internally:
   520→    // - Template variables: {{user}}, {{char}}, {{userpersona}}
   521→    // - Stats keys: {{resistencia}}, {{habilidades}}, etc.
   522→    // - Sound keys: {{sonidos}}
   523→    // - All sections including post-history instructions
   524→    const { prompt: systemPrompt, sections: systemSections, lorebookChatInjections, exampleMessages } = buildSystemPrompt(
   525→      effectiveCharacter,
   526→      effectiveUserName,
   527→      persona,
   528→      lorebookPlan,
   529→      sessionStats,
   530→      allCharacters, // Pass all characters for peticiones/solicitudes resolution
   531→      soundTriggers, // Pass soundTriggers for {{sonidos}} resolution
   532→      soundSettings,  // Pass sound settings for {{sonidos}} template
   533→      questTemplates,  // Pass quest templates for {{activeQuests}} key resolution
   534→      sessionQuests,   // Pass session quests for {{activeQuests}} key resolution
   535→      questSettings,    // Pass quest settings for {{activeQuests}} key resolution
   536→      lorebookAttributeKeys,
   537→      inventoryData,    // Pass inventory data for Inventory V2 section
   538→      lorebookEntryKeyMap // Pass lorebook entry key map for {{entryKey}} resolution
   539→    );
   540→
   541→    // Session stats now already include item effects (applied directly to SessionStats
   542→    // when items are activated/equipped in the store). No need for virtual overlay.
   543→    const effectiveSessionStats = sessionStats;
   544→
   545→    // Resolve persona stats first for comprehensive key resolution
   546→    let streamPersonaResolvedStats: import('@/types').ResolvedStats | null = null;
   547→    if (persona?.statsConfig?.enabled && effectiveSessionStats) {
   548→      streamPersonaResolvedStats = resolveStats({
   549→        characterId: '__user__',
   550→        statsConfig: persona.statsConfig,
   551→        sessionStats: effectiveSessionStats,
   552→      });
   553→    }
   554→    const resolvedStats = resolveStats({
   555→      characterId: effectiveCharacter.id,
   556→      statsConfig: effectiveCharacter.statsConfig,
   557→      sessionStats: effectiveSessionStats,
   558→      allCharacters,
   559→      userName: effectiveUserName,
   560→      characterName: effectiveCharacter.name,
   561→      questTemplates,
   562→      personaDescription: persona?.description,
   563→      personaResolvedStats: streamPersonaResolvedStats,
   564→    });
