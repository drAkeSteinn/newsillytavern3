    26→# ─── secure token storage (Windows DPAPI) ─────────────────────────────────────
    27→
    28→class _DATA_BLOB(ctypes.Structure):
    29→    _fields_ = [("cbData", ctypes.wintypes.DWORD),
    30→                ("pbData", ctypes.POINTER(ctypes.c_char))]
    31→
    32→def _token_save(tok: str) -> None:
    33→    p = Path(__file__).parent
    34→    try:
    35→        buf   = ctypes.create_string_buffer(tok.encode("utf-8"))
    36→        b_in  = _DATA_BLOB(len(tok.encode("utf-8")), buf)
    37→        b_out = _DATA_BLOB()
    38→        if not ctypes.windll.Crypt32.CryptProtectData(
    39→                ctypes.byref(b_in), None, None, None, None, 0,
    40→                ctypes.byref(b_out)):
    41→            raise OSError("CryptProtectData failed")
    42→        enc = ctypes.string_at(b_out.pbData, b_out.cbData)
    43→        ctypes.windll.Kernel32.LocalFree(b_out.pbData)
    44→        (p / ".chub_token").write_bytes(base64.b64encode(enc))
    45→    except Exception:
    46→        (p / ".chub_token").write_text(tok, encoding="utf-8")
    47→
    48→def _token_load() -> str:
    49→    path = Path(__file__).parent / ".chub_token"
    50→    if not path.exists():
    51→        return ""
    52→    try:
    53→        raw   = base64.b64decode(path.read_bytes())
    54→        buf   = ctypes.create_string_buffer(raw)
    55→        b_in  = _DATA_BLOB(len(raw), buf)
    56→        b_out = _DATA_BLOB()
    57→        if not ctypes.windll.Crypt32.CryptUnprotectData(
    58→                ctypes.byref(b_in), None, None, None, None, 0,
    59→                ctypes.byref(b_out)):
    60→            raise OSError("CryptUnprotectData failed")
    61→        dec = ctypes.string_at(b_out.pbData, b_out.cbData)
    62→        ctypes.windll.Kernel32.LocalFree(b_out.pbData)
    63→        return dec.decode("utf-8").strip()
    64→    except Exception:
    65→        try:
    66→            return path.read_text(encoding="utf-8").strip()
    67→        except Exception:
    68→            return ""

   503→# ─── message / branch helpers ─────────────────────────────────────────────────
   504→
   505→def _extract_messages(body) -> list:
   506→    if isinstance(body, list):
   507→        return body
   508→    if not isinstance(body, dict):
   509→        return []
   510→    for key in ("chatMessages", "messages", "results", "items"):
   511→        raw = body.get(key)
   512→        if isinstance(raw, list) and raw:
   513→            return raw
   514→        if isinstance(raw, dict) and raw:
   515→            msgs = [v for v in raw.values() if isinstance(v, dict)]
   516→            msgs.sort(key=lambda m: int(m.get("id", 0)))
   517→            return msgs
   518→    return []

   520→def _build_branches(msg_map: dict) -> list:
   521→    if not msg_map:
   522→        return []
   523→    children: dict = {}
   524→    for mid, msg in msg_map.items():
   525→        pid = msg.get("parent_id")
   526→        if pid is None:
   527→            continue
   528→        try:
   529→            pid = int(pid)
   530→        except (ValueError, TypeError):
   531→            continue
   532→        if pid in msg_map:
   533→            children.setdefault(pid, []).append(mid)
   534→
   535→    def _is_root(msg):
   536→        pid = msg.get("parent_id")
   537→        if pid is None:
   538→            return True
   539→        try:
   540→            return int(pid) not in msg_map
   541→        except (ValueError, TypeError):
   542→            return True
   543→
   544→    root_ids = sorted(mid for mid, msg in msg_map.items() if _is_root(msg))
   545→    if not root_ids:
   546→        return [sorted(msg_map.values(), key=lambda m: int(m.get("id", 0)))]
   547→
   548→    branches = []
   549→    stack = [(rid, []) for rid in reversed(root_ids)]
   550→    while stack:
   551→        node_id, path = stack.pop()
   552→        if node_id not in msg_map:
   553→            continue
   554→        new_path = path + [msg_map[node_id]]
   555→        kids = sorted(children.get(node_id, []))
   556→        if not kids:
   557→            branches.append(new_path)
   558→        else:
   559→            for kid in reversed(kids):
   560→                stack.append((kid, new_path))
   561→    return branches or [sorted(msg_map.values(), key=lambda m: int(m.get("id", 0)))]

   563→def _build_main_path(msg_map: dict) -> list:
   564→    """Walk msg_map along the is_main path, returning one entry per message:
   565→    (node, siblings, index_of_node_in_siblings). Siblings are every message
   566→    that shares the same parent (i.e. alternate branches at that point)."""
   567→    if not msg_map:
   568→        return []
   569→    children: dict = {}
   570→    for mid, msg in msg_map.items():
   571→        pid = msg.get("parent_id")
   572→        if pid is None:
   573→            continue
   574→        try:
   575→            pid = int(pid)
   576→        except (ValueError, TypeError):
   577→            continue
   578→        if pid in msg_map:
   579→            children.setdefault(pid, []).append(mid)
   580→    for kids in children.values():
   581→        kids.sort()
   582→
   583→    def _is_root(msg):
   584→        pid = msg.get("parent_id")
   585→        if pid is None:
   586→            return True
   587→        try:
   588→            return int(pid) not in msg_map
   589→        except (ValueError, TypeError):
   590→            return True
   591→
   592→    roots = sorted(mid for mid, msg in msg_map.items() if _is_root(msg))
   593→    if not roots:
   594→        return []
   595→
   596→    def _pick_main(ids):
   597→        for i in ids:
   598→            if msg_map[i].get("is_main"):
   599→                return i
   600→        return ids[0]
   601→
   602→    path = []
   603→    cur = _pick_main(roots)
   604→    while cur is not None:
   605→        node = msg_map[cur]
   606→        pid = node.get("parent_id")
   607→        try:
   608→            pid = int(pid) if pid is not None else None
   609→        except (ValueError, TypeError):
   610→            pid = None
   611→        sibling_ids = roots if (pid is None or pid not in msg_map) else children.get(pid, [cur])
   612→        siblings = [msg_map[s] for s in sibling_ids]
   613→        idx = sibling_ids.index(cur)
   614→        path.append((node, siblings, idx))
   615→        kid_ids = children.get(cur, [])
   616→        cur = _pick_main(kid_ids) if kid_ids else None
   617→    return path

   619→def _fmt_chat_date(iso: str) -> str:
   620→    try:
   621→        dt = _dt.datetime.fromisoformat((iso or "").replace("Z", "+00:00"))
   622→    except Exception:
   623→        return iso or ""
   624→    return dt.strftime("%B %d, %Y %I:%M%p")

   626→def _merged_st_lines(path: list, char_name: str, user_name: str) -> list:
   627→    lines = []
   628→    for node, siblings, idx in path:
   629→        if node.get("message") is None:
   630→            continue
   631→        is_bot = bool(node.get("is_bot"))
   632→        entry = {
   633→            "name": char_name if is_bot else user_name,
   634→            "is_user": not is_bot,
   635→            "is_system": False,
   636→            "send_date": _fmt_chat_date(node.get("created_at", "")),
   637→            "mes": node.get("message") or "",
   638→            "extra": {},
   639→        }
   640→        if len(siblings) > 1:
   641→            entry["swipe_id"] = idx
   642→            entry["swipes"] = [s.get("message") or "" for s in siblings]
   643→        lines.append(entry)
   644→    return lines

   646→def _merged_chat_text(path: list, char_name: str, user_name: str, chat_fmt: str) -> str:
   647→    lines = _merged_st_lines(path, char_name, user_name)
   648→    if chat_fmt == "JSON":
   649→        header = {"user_name": user_name, "character_name": char_name,
   650→                  "create_date": _dt.datetime.now().strftime("%B %d, %Y %I:%M%p"),
   651→                  "chat_metadata": {}}
   652→        return json.dumps({"header": header, "messages": lines}, indent=2, ensure_ascii=False)
   653→    if chat_fmt == "TXT":
   654→        out = [f"Character: {char_name}", "─" * 40, ""]
   655→        for entry in lines:
   656→            out.append(f"{entry['name']}: {entry['mes']}")
   657→            if entry.get("swipes"):
   658→                for si, alt in enumerate(entry["swipes"]):
   659→                    if si == entry["swipe_id"]:
   660→                        continue
   661→                    out.append(f"  ↳ [swipe {si+1}/{len(entry['swipes'])}] {alt}")
   662→            out.append("")
   663→        return "\n".join(out)
   664→    # JSONL (SillyTavern chat format): header line + one message per line
   665→    header = {"user_name": user_name, "character_name": char_name,
   666→              "create_date": _dt.datetime.now().strftime("%B %d, %Y %I:%M%p"),
   667→              "chat_metadata": {}}
   668→    out_lines = [json.dumps(header, ensure_ascii=False)]
   669→    out_lines.extend(json.dumps(e, ensure_ascii=False) for e in lines)
   670→    return "\n".join(out_lines)

   672→def _req(sess, method: str, url: str, max_retries: int = 3, **kwargs):
   673→    for attempt in range(max_retries + 1):
   674→        r = getattr(sess, method)(url, **kwargs)
   675→        if r.status_code != 429:
   676→            return r
   677→        time.sleep(2 ** attempt)
   678→    return r

  1087→                    t = self._tiles[self._order[i]]
  1088→                    t.show()

  1090→            finally:
  1091→                self._container.setUpdatesEnabled(True)

  1499→        if count is not None:

  1504→            # Build with rich text so the count looks like a secondary chip.

  1507→            self._tab_btns[key].setText(name)

  1510→        self._active_tab = key

  1524→        is_chats = (key == "chats")
