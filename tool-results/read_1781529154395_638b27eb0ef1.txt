    95→    95→  const attributes = statsConfig?.attributes || [];
    96→    96→
    97→    97→  // State for new reply form
    98→    98→  const [newLabel, setNewLabel] = useState('');
    99→    99→  const [newResponse, setNewResponse] = useState('');
   100→   100→  const [newModifiers, setNewModifiers] = useState<QuickReplyAttributeModifier[]>([]);
   101→   101→
   102→   102→  // State for editing existing reply
   103→   103→  const [editingId, setEditingId] = useState<string | null>(null);
   104→   104→  const [editLabel, setEditLabel] = useState('');
   105→   105→  const [editResponse, setEditResponse] = useState('');
   106→   106→  const [editModifiers, setEditModifiers] = useState<QuickReplyAttributeModifier[]>([]);
   107→   107→
   108→   108→  // State for expanded modifiers section
   109→   109→  const [expandedModifiers, setExpandedModifiers] = useState<string | null>(null);
   110→   110→
   111→   111→  // State for sprite activation in new reply
   112→   112→  const [newSpriteActivation, setNewSpriteActivation] = useState<QuickReplySpriteActivation | undefined>(undefined);
   113→   113→
   114→   114→  // State for sprite activation in editing reply
   115→   115→  const [editSpriteActivation, setEditSpriteActivation] = useState<QuickReplySpriteActivation | undefined>(undefined);
   116→   116→
   117→   117→  // State for expanded sprite activation section
   118→   118→  const [expandedSpriteActivation, setExpandedSpriteActivation] = useState<string | null>(null);
   119→   119→
   120→   120→  // Whether sprite activation is available (has packs or collections)

   126→   126→    if (!isAdding) return;

   128→   128→      id: generateId(),
   129→   129→      label: newLabel.trim(),
   130→   130→      response: newResponse.trim(),
   131→   131→      modifiers: newModifiers.length > 0 ? newModifiers : undefined,
   132→   132→      spriteActivation: newSpriteActivation,
   133→   133→    };
   134→   134→    onChange([...replies, newReply]);
   135→   135→    setNewLabel('');
   136→   136→    setNewResponse('');
   137→   137→    setNewModifiers([]);
   138→   138→    setNewSpriteActivation(undefined);
   139→   139→  };
   140→   140→
   141→   141→  const handleDelete = (id: string) => {
   142→   142→    onChange(replies.filter((r) => r.id !== id));
   143→   143→  };
   144→   144→
   145→   145→  const handleStartEdit = (reply: CharacterQuickReply) => {
   146→   146→    setEditingId(reply.id);
   147→   147→    setEditLabel(reply.label);
   148→   148→    setEditResponse(reply.response);
   149→   149→    setEditModifiers(reply.modifiers ? [...reply.modifiers] : []);
   150→   150→    setEditSpriteActivation(reply.spriteActivation ? { ...reply.spriteActivation } : undefined);
   151→   151→  };
   152→   152→
   153→   153→  const handleSaveEdit = () => {
   154→   154→    if (!editingId || !editLabel.trim() || !editResponse.trim()) return;
   155→   155→    onChange(
   156→   156→      replies.map((r) =>
   157→   157→        r.id === editingId
   158→   158→          ? {
   159→   159→              ...r,
   160→   160→              label: editLabel.trim(),
   161→   161→              response: editResponse.trim(),
   162→   162→              modifiers: editModifiers.length > 0 ? editModifiers : undefined,
   163→   163→              spriteActivation: editSpriteActivation,
   164→   164→            }
   165→   165→          : r
   166→   166→      )
   167→   167→    );
   168→   168→    setEditingId(null);
   169→   169→    setEditLabel('');
   170→   170→    setEditResponse('');
   171→   171→    setEditModifiers([]);
   172→   172→    setEditSpriteActivation(undefined);
   173→   173→  };
   174→   174→
   175→   175→  const handleCancelEdit = () => {
   176→   176→    setEditingId(null);
   177→   177→    setEditLabel('');
   178→   178→    setEditResponse('');
   179→   179→    setEditModifiers([]);
   180→   180→    setEditSpriteActivation(undefined);
   181→   181→  };
   182→   182→
   183→   183→  // Add a modifier to the new reply form
   184→   184→  const addModifierToNew = () => {
   185→   185→    if (attributes.length === 0) return;
   186→   186→    setNewModifiers([
   187→   187→      ...newModifiers,
   188→   188→      { attributeKey: attributes[0].key, operation: 'add', value: 1 },
   189→   189→    ]);
   190→   190→  };
   191→   191→
   192→   192→  const updateNewModifier = (index: number, field: keyof QuickReplyAttributeModifier, value: string | number) => {
   193→   193→    const updated = [...newModifiers];
   194→   194→    updated[index] = { ...updated[index], [field]: value };
   195→   195→    setNewModifiers(updated);
   196→   196→  };
   197→   197→
   198→   198→  const removeNewModifier = (index: number) => {
   199→   199→    setNewModifiers(newModifiers.filter((_, i) => i !== index));
   200→   200→  };
   201→   201→
   202→   202→  // Add a modifier to the editing reply form
   203→   203→  const addModifierToEdit = () => {
   204→   204→    if (attributes.length === 0) return;
   205→   205→    setEditModifiers([
   206→   206→      ...editModifiers,
   207→   207→      { attributeKey: attributes[0].key, operation: 'add', value: 1 },
   208→   208→    ]);
   209→   209→  };
   210→   210→
   211→   211→  const updateEditModifier = (index: number, field: keyof QuickReplyAttributeModifier, value: string | number) => {
   212→   212→    const updated = [...editModifiers];
   213→   213→    updated[index] = { ...updated[index], [field]: value };
   214→   214→    setEditModifiers(updated);
   215→   215→  };
   216→   216→
   217→   217→  const removeEditModifier = (index: number) => {
   218→   218→    setEditModifiers(editModifiers.filter((_, i) => i !== index));
   219→   219→  };
   220→   220→
   221→   221→  // Render modifier row
   222→   222→  const renderModifierRow = (
   223→   223→    modifier: QuickReplyAttributeModifier,
   224→   224→    index: number,
   225→   225→    onUpdate: (index: number, field: keyof QuickReplyAttributeModifier, value: string | number) => void,
   226→   226→    onRemove: (index: number) => void
   227→   227→  ) => {
   228→   228→    const attr = attributes.find((a) => a.key === modifier.attributeKey);
   229→   229→    const isTextAttr = attr?.type === 'text' || attr?.type === 'keyword';
   230→   230→    const availableOps: QuickReplyModifierOperation[] = isTextAttr
   231→   231→      ? ['set']
   232→   232→      : ['set', 'add', 'subtract', 'multiply', 'divide'];

   234→   234→    return (
   235→   235→      <div key={index} className="flex items-center gap-1.5">
   236→   236→        {/* Attribute selector */}
   237→   237→        <Select
   238→   238→          value={modifier.attributeKey}
   239→   239→          onValueChange={(val) => onUpdate(index, 'attributeKey', val)}

   241→   241→          <SelectTrigger className="h-7 text-xs flex-1 min-w-[80px]">
   242→   242→            <SelectValue placeholder="Atributo" />
   243→   243→          </SelectTrigger>
   244→   244→          <SelectContent>
   245→   245→            {attributes.map((attr) => (
   246→   246→              <SelectItem key={attr.key} value={attr.key}>
   247→   247→                <span className="text-xs">{attr.icon ? `${attr.icon} ` : ''}{attr.name}</span>
   248→   248→              </SelectItem>
   249→   249→            ))}
   250→   250→          </SelectContent>
   251→   251→        </Select>
   252→   252→
   253→   253→        {/* Operation selector */}
   254→   254→        <Select
   255→   255→          value={modifier.operation}
   256→   256→          onValueChange={(val) => onUpdate(index, 'operation', val as QuickReplyModifierOperation)}
   257→   257→        >
   258→   258→          <SelectTrigger className="h-7 text-xs w-[90px]">
   259→   259→            <SelectValue />
   260→   260→          </SelectTrigger>
   261→   261→          <SelectContent>
   262→   262→            {availableOps.map((op) => (
   263→   263→              <SelectItem key={op} value={op}>
   264→   264→                <span className="text-xs">
   265→   265→                  {OPERATION_LABELS[op].symbol} {OPERATION_LABELS[op].label}
   266→   266→                </span>
   267→   267→              </SelectItem>
   268→   268→            ))}
   269→   269→          </SelectContent>
   270→   270→        </Select>
   271→   271→
   272→   272→        {/* Value input */}
   273→   273→        <Input
   274→   274→          type={isTextAttr ? 'text' : 'number'}
   275→   275→          value={String(modifier.value)}
   276→   276→          onChange={(e) => {
   277→   277→            const val = isTextAttr ? e.target.value : (parseFloat(e.target.value) || 0);
   278→   278→            onUpdate(index, 'value', val);
   279→   279→          }}
   280→   280→          className="h-7 text-xs w-[70px]"
   281→   281→          placeholder="Valor"
   282→   282→        />
   283→   283→
   284→   284→        {/* Remove button */}
   285→   285→        <Button
   286→   286→          variant="ghost"
   287→   287→          size="sm"
   288→   288→          className="h-7 w-7 p-0 text-muted-foreground hover:text-red-400 flex-shrink-0"
   289→   289→          onClick={() => onRemove(index)}
   290→   290→        >
   291→   291→          <X className="w-3 h-3" />
   292→   292→        </Button>
   293→   293→      </div>
   294→   294→    );
   295→   295→  };

   299→   299→    activation: QuickReplySpriteActivation | undefined,
   300→   300→    setActivation: (a: QuickReplySpriteActivation | undefined) => void,
   301→   301→    sectionKey: string,
   302→   302→  ) => {
   303→   303→    const isExpanded = expandedSpriteActivation === sectionKey;
   304→   304→    const isActive = !!activation;
   305→   305→
   306→   306→    return (
   307→   307→      <div className="space-y-2">
   308→   308→        <button
   309→   309→          type="button"
   310→   310→          onClick={() => {
   311→   311→            setExpandedSpriteActivation(isExpanded ? null : sectionKey);
   312→   312→            if (!isActive && !isExpanded) {
   313→   313→              // Pre-populate with defaults
   314→   314→              setActivation({
   315→   315→                mode: 'sprite_pack',
   316→   316→                targetId: '',
   317→   317→                fallbackMode: 'idle_collection',
   318→   318→                fallbackDelayMs: 3000,
   319→   319→              });
   320→   320→            }
   321→   321→          }}
   322→   322→          className="flex items-center gap-1.5 text-xs text-muted-foreground hover:text-foreground transition-colors"
   323→   323→        >
   324→   324→          <ImageIcon className="w-3.5 h-3.5 text-emerald-400" />
   325→   325→          <span>Activación de Sprite</span>

   327→   327→            <Badge variant="secondary" className="h-4 text-[10px] px-1 bg-emerald-500/20 text-emerald-400">
   328→   328→              {activation?.mode === 'trigger_collection' ? 'Trigger' : 'Pack'}
   329→   329→            </Badge>
   330→   330→          )}
   331→   331→          {isExpanded ? (
   332→   332→            <ChevronUp className="w-3 h-3" />
   333→   333→          ) : (
   334→   334→            <ChevronDown className="w-3 h-3" />
   335→   335→          )}
   336→   336→        </button>
   337→   337→
   338→   338→        {(isExpanded || isActive) && (
   339→   339→          <div className="ml-5 space-y-2 border-l-2 border-emerald-500/20 pl-3">
   340→   340→            {/* Enable/Disable toggle */}
   341→   341→            <div className="flex items-center gap-2">
   342→   342→              <label className="text-[10px] text-muted-foreground">Activar sprite:</label>
   343→   343→              <button
   344→   344→                type="button"
   345→   345→                onClick={() => {
   346→   346→                  if (isActive) {
   347→   347→                    setActivation(undefined);
   348→   348→                  } else {
   349→   349→                    setActivation({
   350→   350→                      mode: 'sprite_pack',
   351→   351→                      targetId: '',
   352→   352→                      fallbackMode: 'idle_collection',
   353→   353→                      fallbackDelayMs: 3000,
   354→   354→                    });
   355→   355→                  }
   356→   356→                }}
   357→   357→                className={cn(
   358→   358→                  "text-[10px] px-2 py-0.5 rounded-full transition-colors",
   359→   359→                  isActive ? "bg-emerald-500/20 text-emerald-400" : "bg-muted text-muted-foreground"
   360→   360→                )}
   361→   361→              >
   362→   362→                {isActive ? 'Activado' : 'Desactivado'}
   363→   363→              </button>
   364→   364→            </div>
   365→   365→
   366→   366→            {isActive && activation && (
   367→   367→              <>
   368→   368→                {/* Mode selector */}
   369→   369→                <div className="flex items-center gap-2">
   370→   370→                  <Label className="text-[10px] text-muted-foreground w-16 flex-shrink-0">Modo</Label>
   371→   371→                  <Select
   372→   372→                    value={activation.mode}
   373→   373→                    onValueChange={(val) => setActivation({ ...activation, mode: val as 'trigger_collection' | 'sprite_pack', targetId: '' })}
   374→   374→                  >
   375→   375→                    <SelectTrigger className="h-7 text-xs flex-1">
   376→   376→                      <SelectValue />
   377→   377→                    </SelectTrigger>
   378→   378→                    <SelectContent>
   379→   379→                      <SelectItem value="sprite_pack">
   380→   380→                        <span className="text-xs">📦 Sprite Pack (condicional)</span>
   381→   381→                      </SelectItem>
   382→   382→                      <SelectItem value="trigger_collection">
   383→   383→                        <span className="text-xs">🎯 Colección de Triggers</span>
   384→   384→                      </SelectItem>
   385→   385→                    </SelectContent>
   386→   386→                  </Select>
   387→   387→                </div>
   388→   388→
   389→   389→                {/* Target selector */}
   390→   390→                <div className="flex items-center gap-2">
   391→   391→                  <Label className="text-[10px] text-muted-foreground w-16 flex-shrink-0">Objetivo</Label>
   392→   392→                  <Select
   393→   393→                    value={activation.targetId}
   394→   394→                    onValueChange={(val) => setActivation({ ...activation, targetId: val })}
   395→   395→                  >
   396→   396→                    <SelectTrigger className="h-7 text-xs flex-1">
   397→   397→                      <SelectValue placeholder={activation.mode === 'trigger_collection' ? 'Seleccionar trigger...' : 'Seleccionar pack...'} />
   398→   398→                    </SelectTrigger>
   399→   399→                    <SelectContent>
   400→   400→                      {activation.mode === 'trigger_collection' ? (
   401→   401→                        (triggerCollections || []).map((tc) => (
   402→   402→                          <SelectItem key={tc.id} value={tc.id}>
   403→   403→                            <span className="text-xs">🎯 {tc.name} (Prioridad: {tc.priority})</span>
   404→   404→                          </SelectItem>
   405→   405→                        ))
   406→   406→                      ) : (
   407→   407→                        (spritePacksV2 || []).map((pack) => (
   408→   408→                          <SelectItem key={pack.id} value={pack.id}>
   409→   409→                            <span className="text-xs">📦 {pack.name} {pack.conditionalMode ? '(condicional)' : ''}</span>
   410→   410→                          </SelectItem>
   411→   411→                        ))
   412→   412→                      )}
   413→   413→                    </SelectContent>
   414→   414→                  </Select>
   415→   415→                </div>
   416→   416→
   417→   417→                {/* Fallback mode */}
   418→   418→                <div className="flex items-center gap-2">
   419→   419→                  <Label className="text-[10px] text-muted-foreground w-16 flex-shrink-0">Al terminar</Label>

   421→   421→                    value={activation.fallbackMode}
   422→   422→                    onValueChange={(val) => setActivation({ ...activation, fallbackMode: val as QuickReplySpriteFallbackMode })}
   423→   423→                  >
   424→   424→                    <SelectTrigger className="h-7 text-xs flex-1">
   425→   425→                      <SelectValue />
   426→   426→                    </SelectTrigger>
   427→   427→                    <SelectContent>
   428→   428→                      <SelectItem value="idle_collection">
   429→   429→                        <span className="text-xs">🔄 Volver al estado normal</span>
   430→   430→                      </SelectItem>
   431→   431→                      <SelectItem value="collection_default">

   434→   434→                      <SelectItem value="custom_sprite">
   435→   435→                        <span className="text-xs">✨ Sprite personalizado</span>
   436→   436→                      </SelectItem>
   437→   437→                    </SelectContent>
   438→   438→                  </Select>
   439→   439→                </div>

   441→   441→                {/* Fallback delay */}

   443→   443→                  <Label className="text-[10px] text-muted-foreground w-16 flex-shrink-0">

   452→   452→                    min={0}

   463→   463→                      value={activation.fallbackSpriteId || ''}
