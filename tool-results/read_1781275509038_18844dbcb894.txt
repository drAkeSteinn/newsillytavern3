    92→/**
    93→ * Execute detected tool calls, send SSE events, and return results.
    94→ * Returns { newContent: combined display messages, shouldContinue: true if tools were executed }
    95→ */
    96→async function executeToolCallsAndContinue(
    97→  toolCalls: NativeToolCall[],
    98→  availableTools: Array<{ id: string; name: string; label: string; icon: string }>,
    99→  currentRound: number,
   100→  maxRounds: number,
   101→  character: CharacterCard,
   102→  sessionId: string,
   103→  userName: string,
   104→  controller: { enqueue: (chunk: string) => void },
   105→  sessionQuests?: SessionQuestInstance[],
   106→  questTemplates?: QuestTemplate[],
   107→  statsConfig?: CharacterStatsConfig,
   108→  sessionStats?: SessionStats,
   109→  allCharacters?: CharacterCard[],
   110→  characterMemory?: CharacterMemory,
   111→  lorebooks?: Lorebook[],
   112→): Promise<{ newContent: string; shouldContinue: boolean; toolResults: Array<{ success: boolean; displayMessage: string }>; questActivations: QuestActivation[]; toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> }> {
   113→  if (toolCalls.length === 0 || currentRound >= maxRounds) {
   114→    return { newContent: '', shouldContinue: false, toolResults: [], questActivations: [], toolsUsed: [] };
   115→  }
   116→
   117→  const toolResults: Array<{ success: boolean; displayMessage: string }> = [];
   118→  const questActivations: QuestActivation[] = [];
   119→  const toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> = [];
   120→  let allDisplayMessages = '';
   121→
   122→  for (let i = 0; i < toolCalls.length; i++) {
   123→    const tc = toolCalls[i];
   124→    const toolDef = availableTools.find(t => t.name === tc.name);
   125→    const callId = `${Date.now()}_${i}`;
   126→
   127→    // Send tool_call_start event
   128→    controller.enqueue(createSSEJSON({
   129→      type: 'tool_call_start',
   130→      toolName: tc.name,
   131→      toolLabel: toolDef?.label || tc.name,
   132→      toolIcon: toolDef?.icon || 'Wrench',
   133→      params: tc.arguments,
   134→      callId,
   135→    }));
   136→
   137→    console.log(`[Tools] Executing tool call: ${tc.name}`, tc.arguments);
   138→
   139→    // Execute the tool
   140→    const toolResult = await executeTool(
   141→      tc.name,
   142→      tc.arguments,
   143→      {
   144→        characterId: character.id,
   145→        characterName: character.name,
   146→        sessionId,
   147→        userName,
   148→        sessionQuests,
   149→        questTemplates,
   150→        statsConfig: character.statsConfig,
   151→        sessionStats,
   152→        allCharacters,
   153→        characterMemory,
   154→        lorebooks,
   155→      },
   156→    );
   157→
   158→    // Send tool_call_result event
   159→    controller.enqueue(createSSEJSON({
   160→      type: 'tool_call_result',
   161→      toolName: tc.name,
   162→      success: toolResult.success,
   163→      displayMessage: toolResult.displayMessage,
   164→      duration: toolResult.duration || 0,
   165→      callId,
   166→    }));
   167→
   168→    // Check for quest activation and send SSE event
   169→    if (toolResult.questActivation) {
   170→      const activation = toolResult.questActivation;
   171→      console.log(`[Tools] Quest activation from ${tc.name}:`, activation);
   172→      
   173→      controller.enqueue(createSSEJSON({
   174→        type: 'quest_activation',
   175→        toolName: tc.name,
   176→        activationType: activation.type,
   177→        key: activation.key,
   178→        metadata: activation.metadata,
   179→      }));
   180→      
   181→      questActivations.push(activation);
   182→    }
   183→
   184→    // Check for action/skill activation and send SSE event
   185→    if (toolResult.actionActivation) {
   186→      const action = toolResult.actionActivation;
   187→      console.log(`[Tools] Action activation from ${tc.name}:`, action.skillName);
   188→      
   189→      controller.enqueue(createSSEJSON({
   190→        type: 'action_activation',
   191→        toolName: tc.name,
   192→        skillId: action.skillId,
   193→        skillName: action.skillName,
   194→        skillDescription: action.skillDescription,
   195→        skillCompletedDescription: action.skillCompletedDescription,
   196→        activationCosts: action.activationCosts,
   197→        activationRewards: action.activationRewards,
   198→        characterId: action.characterId,
   199→      }));
   200→    }
   201→
   202→    // Check for stat modification from modify_stat tool and send SSE event
   203→    if (toolResult.statActivation) {
   204→      const stat = toolResult.statActivation;
   205→      console.log(`[Tools] Stat activation from ${tc.name}:`, stat.attributeKey, stat.oldValue, '→', stat.newValue);
   206→      
   207→      controller.enqueue(createSSEJSON({
   208→        type: 'stat_activation',
   209→        toolName: tc.name,
   210→        characterId: stat.characterId,
   211→        attributeKey: stat.attributeKey,
   212→        attributeName: stat.attributeName,
   213→        attributeType: stat.attributeType,
   214→        oldValue: stat.oldValue,
   215→        newValue: stat.newValue,
   216→        reason: stat.reason,
   217→      }));
   218→    }
   219→
   220→    // Check for solicitud activation/completion and send SSE event
   221→    if (toolResult.solicitudActivation) {
   222→      const sol = toolResult.solicitudActivation;
   223→      console.log(`[Tools] Solicitud activation from ${tc.name}:`, sol.type, sol.solicitudKey);
   224→      
   225→      controller.enqueue(createSSEJSON({
   226→        type: 'solicitud_activation',
   227→        toolName: tc.name,
   228→        activationType: sol.type,
   229→        solicitudKey: sol.solicitudKey,
   230→        targetCharacterId: sol.targetCharacterId,
   231→        targetCharacterName: sol.targetCharacterName,
   232→        fromCharacterId: sol.fromCharacterId,
   233→        fromCharacterName: sol.fromCharacterName,
   234→        description: sol.description,
   235→        completionDescription: sol.completionDescription,
   236→        peticionKey: sol.peticionKey,
   237→      }));
   238→    }
   239→
   240→    // Check for memory activation (sync to client-side Character Memory)
   241→    if (toolResult.memoryActivation) {
   242→      const mem = toolResult.memoryActivation;
   243→      console.log(`[Tools] Memory activation from ${tc.name}:`, mem.type);
   244→      
   245→      controller.enqueue(createSSEJSON({
   246→        type: 'memory_activation',
   247→        toolName: tc.name,
   248→        activationType: mem.type,
   249→        characterId: mem.characterId,
   250→        eventData: mem.eventData,
   251→        relationshipData: mem.relationshipData,
   252→        noteContent: mem.noteContent,
   253→        deleteEventId: mem.deleteEventId,
   254→        deleteEmbeddingId: mem.deleteEmbeddingId,
   255→      }));
   256→    }
   257→
   258→    console.log(`[Tools] Tool ${tc.name}: success=${toolResult.success} duration=${toolResult.duration}ms`, toolResult.displayMessage);
   259→
   260→    toolResults.push({ success: toolResult.success, displayMessage: toolResult.displayMessage });
   261→    toolsUsed.push({
   262→      name: tc.name,
   263→      label: toolDef?.label || tc.name,
   264→      icon: toolDef?.icon || 'Wrench',
   265→      success: toolResult.success,
   266→    });
   267→    if (toolResult.displayMessage) {
   268→      allDisplayMessages += (allDisplayMessages ? '\n' : '') + toolResult.displayMessage;
   269→    }
   270→  }
   271→
   272→  // Return the display messages as content, tool results, and signal that a follow-up call is needed
   273→  return {
   274→    newContent: allDisplayMessages,
   275→    shouldContinue: true, // Always continue to get the LLM's natural response
   276→    toolResults,
   277→    questActivations,
   278→    toolsUsed,
   279→  };
   280→}

   282→/**
   283→ * POST /api/chat/proactive
   284→ *
   285→ * Generates a proactive message for a character using SSE streaming.
   286→ * Full feature parity with the stream route including tools, embeddings,
   287→ * key resolution, memory reinforcement, and extraction.
   288→ */
   289→export async function POST(request: NextRequest) {
   290→  try {
   291→    // Capture auth headers from Z.ai gateway for token resolution
   292→    const incomingXToken = request.headers.get('X-Token');
   293→    const fcSecurityToken = request.headers.get('x-fc-security-token');
   294→
   295→    const body = await request.json();
   296→
   297→    // Extract required fields from body
   298→    const {
   299→      character,
   300→      messages = [],
   301→      llmConfig,
   302→      userName = 'User',
   303→      persona,
   304→      proactiveConfig,
   305→      reason = 'timer_idle',
   306→      lastActivityAt,
   307→    } = body;
   308→
   309→    // Validate required fields
   310→    if (!character?.name || !llmConfig?.provider) {
   311→      return createErrorResponse('Missing required fields: character, llmConfig', 400);
   312→    }
   313→
   314→    if (!proactiveConfig?.enabled) {
   315→      return createErrorResponse('Proactive messages are disabled for this character', 400);
   316→    }
   317→
   318→    if (!llmConfig) {
   319→      return createErrorResponse('No LLM configuration provided', 400);
   320→    }
   321→
   322→    // Validate API key for providers that require one
   323→    const providersRequiringApiKey = ['openai', 'vllm', 'lm-studio', 'custom', 'anthropic', 'grok'];
   324→    if (providersRequiringApiKey.includes(llmConfig.provider) && !llmConfig.apiKey) {
   325→      return createErrorResponse(
   326→        `API Key is required for ${llmConfig.provider} provider. Please configure it in Settings.`,
   327→        400
   328→      );
   329→    }
   330→
   331→    // Validate endpoint for providers that require one
   332→    const providersRequiringEndpoint = ['openai', 'vllm', 'lm-studio', 'custom', 'anthropic', 'ollama', 'text-generation-webui', 'koboldcpp'];
   333→    if (providersRequiringEndpoint.includes(llmConfig.provider) && !llmConfig.endpoint) {
   334→      return createErrorResponse(
   335→        `Endpoint URL is required for ${llmConfig.provider} provider. Please configure it in Settings.`,
   336→        400
   337→      );
   338→    }
   339→
   340→    // If using Z.ai provider, collect gateway-forwarded auth headers
   341→    let zaiRuntimeToken: string | undefined;
   342→    if (llmConfig.provider === 'z-ai') {
   343→      const gatewayToken = incomingXToken || fcSecurityToken || undefined;
   344→      if (gatewayToken) {
   345→        zaiRuntimeToken = gatewayToken;
   346→        console.log(`[Proactive Route] Z.ai runtime token available (${gatewayToken.length} chars, source: ${incomingXToken ? 'X-Token' : 'fc-security-token'})`);
   347→      } else {
   348→        console.log(`[Proactive Route] Z.ai: no gateway token available, using config file only`);
   349→      }
   350→    }
   351→
   352→    console.log(`[Proactive] Generating proactive message for "${character.name}" (reason: ${reason})`);
   353→
   354→    // Extract all additional fields from body (matching stream route)
   355→    const lorebooks: Lorebook[] = body.lorebooks || [];
   356→    const sessionStats: SessionStats | undefined = body.sessionStats;
   357→    const allCharacters: CharacterCard[] = body.allCharacters || [character];
   358→
   359→    // Extract HUD context from body
   360→    const hudContext: HUDContextConfig | undefined = body.hudContext;
   361→
   362→    // Extract Quest data
   363→    const questTemplates: QuestTemplate[] = body.questTemplates || [];
   364→    const sessionQuests: SessionQuestInstance[] = body.sessionQuests || [];
   365→    const questSettings: QuestSettings = {
   366→      ...DEFAULT_QUEST_SETTINGS,
   367→      ...(body.questSettings || {})
   368→    };
   369→
   370→    // Extract Sound data for {{sonidos}} key resolution
   371→    const soundTriggers: SoundTrigger[] = body.soundTriggers || [];
   372→    const soundSettings = body.settings?.sound;
   373→
   374→    // Extract summary for memory/context compression
   375→    const summary: SessionSummary | undefined = body.summary;
   376→
   377→    // Extract embeddings chat settings
   378→    const embeddingsChat: Partial<EmbeddingsChatSettings> = body.embeddingsChat || {};
   379→    const characterMemory: CharacterMemory | undefined = body.characterMemory;
   380→    const sessionId: string | undefined = body.sessionId;
   381→    const characterId: string | undefined = body.characterId;
   382→    const inventoryData: InventoryPromptData | undefined = body.inventoryData;
   383→
   384→    // Extract tools settings
   385→    const toolsSettings: ToolsSettings = {
   386→      enabled: body.toolsSettings?.enabled ?? true,
   387→      maxToolCallsPerTurn: body.toolsSettings?.maxToolCallsPerTurn ?? 2,
   388→      characterConfigs: body.toolsSettings?.characterConfigs || [],
   389→      usePromptBasedFallback: body.toolsSettings?.usePromptBasedFallback ?? false,
   390→      disabledTools: body.toolsSettings?.disabledTools || [],
   391→    };
   392→
   393→    // Create default character if none provided
   394→    const effectiveCharacter: CharacterCard = character || DEFAULT_CHARACTER;
   395→
   396→    // Get effective user name from persona or use provided userName
   397→    const effectiveUserName = getEffectiveUserName(persona, userName);
   398→
   399→    // Build context configuration from request or use defaults
   400→    const contextConfig: Partial<ContextConfig> = body.contextConfig || {};
   401→
   402→    // Apply sliding window to messages
   403→    const contextWindow = selectContextMessages(messages, llmConfig, contextConfig);
   404→
   405→    // Log context stats (for debugging)
   406→    const stats = getContextStats(messages);
   407→
   408→    // Process lorebooks and get matched entries
   409→    const { plan: lorebookPlan, lorebookAttributeKeys, lorebookEntryKeyMap, lorebookDebugEntries } = buildLorebookSectionForPrompt(
   410→      messages,
   411→      lorebooks,
   412→      {
   413→        scanDepth: (contextConfig as any).scanDepth,
   414→        userName: effectiveUserName,
   415→        charName: effectiveCharacter?.name,
   416→      },
   417→      { sessionStats, characterId: effectiveCharacter?.id, characters: allCharacters }
   418→    );
   419→
   420→    // ========================================
   421→    // DEBUG: Build lorebook debug data for SSE event
   422→    // ========================================
   423→    const lorebookDebugData = {
   424→      lorebookAttributeKeys,
   425→      debugEntries: lorebookDebugEntries,
   426→      availableStats: sessionStats
   427→        ? Object.fromEntries(
   428→            Object.entries(sessionStats.characterStats || {}).map(([charId, cs]) => [
   429→              charId,
   430→              cs?.attributeValues ? { ...cs.attributeValues } : '(no values)',
   431→            ])
   432→          )
   433→        : '(no sessionStats)',
   434→    };
   435→
   436→    // ========================================
   437→    // Embeddings Context Retrieval
   438→    // ========================================
   439→    // For proactive messages, use recent chat context as the search query
   440→    // since there's no new user message
   441→    let enrichedSearchQuery = '';
   442→    const searchContextDepth = embeddingsChat.searchContextDepth || 0;
   443→    if (searchContextDepth > 0) {
   444→      const recentHistory = messages
   445→        .filter(m => !m.isDeleted)
   446→        .slice(-(searchContextDepth * 2))
   447→        .map(m => m.content?.trim())
   448→        .filter(Boolean);
   449→      
   450→      if (recentHistory.length > 0) {
   451→        enrichedSearchQuery = recentHistory.join(' ');
   452→      }
   453→    } else {
   454→      // Use last few messages as search context
   455→      const recentHistory = messages
   456→        .filter(m => !m.isDeleted)
   457→        .slice(-4)
   458→        .map(m => m.content?.trim())
   459→        .filter(Boolean);
   460→      
   461→      if (recentHistory.length > 0) {
   462→        enrichedSearchQuery = recentHistory.join(' ');
   463→      }
   464→    }
   465→
   466→    // Safety: smart truncation based on embedding model's context window.
   467→    try {
   468→      const { MODEL_CONTEXT_LENGTHS, DEFAULT_CONTEXT_LENGTH, CHARS_PER_TOKEN } = await import('@/lib/embeddings/types');
   469→      const { loadConfig } = await import('@/lib/embeddings/config-persistence');
   470→      const embConfig = loadConfig();
   471→      const modelKey = embConfig.model || 'bge-m3:567m';
   472→      const modelCtx = MODEL_CONTEXT_LENGTHS[modelKey]
   473→        || MODEL_CONTEXT_LENGTHS[modelKey.split(':')[0]]
   474→        || DEFAULT_CONTEXT_LENGTH;
   475→      const maxQueryChars = Math.floor(modelCtx * 0.75 * CHARS_PER_TOKEN);
   476→
   477→      if (enrichedSearchQuery.length > maxQueryChars) {
   478→        // For proactive: keep the most recent part (end of the string)
   479→        enrichedSearchQuery = enrichedSearchQuery.slice(-maxQueryChars);
   480→        console.warn(
   481→          `[Proactive Route] Search query trimmed to ${enrichedSearchQuery.length} chars ` +
   482→          `(model: ${modelKey}, context: ${modelCtx} tokens)`
   483→        );
   484→      }
   485→    } catch { /* fallback: Ollama client handles truncation */ }
   486→
   487→    // Retrieve relevant embeddings based on enriched query and settings
   488→    // Pass Character Memory events for deduplication (avoid duplicate memory in prompt)
   489→    const existingMemoryEvents = characterMemory?.events?.map(e => ({
   490→      content: e.content,
   491→      importance: e.importance,
   492→    }));
   493→
   494→    // Extract last assistant message for bidirectional search
   495→    const lastAssistantMsg = messages
   496→      .filter(m => !m.isDeleted && m.role === 'assistant')
   497→      .pop()?.content;
   498→
   499→    const embeddingsResult = await retrieveEmbeddingsContext(
   500→      enrichedSearchQuery,
   501→      characterId || effectiveCharacter.id,
   502→      sessionId,
   503→      embeddingsChat,
   504→      undefined, // groupId
   505→      existingMemoryEvents, // for deduplication
   506→      lastAssistantMsg, // bidirectional search with last assistant message
   507→    );
   508→    
   509→    if (embeddingsResult.found) {
   510→      console.log(`[Proactive Route] Retrieved ${embeddingsResult.count} embeddings from namespaces: ${embeddingsResult.searchedNamespaces.join(', ')}`);
   511→    }
   512→
   513→    // ========================================
   514→    // Build system prompt with unified key resolution
   515→    // ========================================
   516→    const { prompt: systemPrompt, sections: systemSections, lorebookChatInjections, exampleMessages } = buildSystemPrompt(
   517→      effectiveCharacter,
   518→      effectiveUserName,
   519→      persona,
   520→      lorebookPlan,
   521→      sessionStats,
   522→      allCharacters, // Pass all characters for peticiones/solicitudes resolution
   523→      soundTriggers,
   524→      soundSettings,
   525→      questTemplates,
   526→      sessionQuests,
   527→      questSettings,
   528→      lorebookAttributeKeys,
   529→      inventoryData,    // Pass inventory data for Inventory V2 section
   530→      lorebookEntryKeyMap // Pass lorebook entry key map for {{entryKey}} resolution
   531→    );
   532→
   533→    // Build key resolution context for HUD context and quest sections
   534→    let streamPersonaResolvedStats: import('@/types').ResolvedStats | null = null;
   535→    if (persona?.statsConfig?.enabled && sessionStats) {
   536→      streamPersonaResolvedStats = resolveStats({
   537→        characterId: '__user__',
   538→        statsConfig: persona.statsConfig,
   539→        sessionStats,
   540→      });
   541→    }
   542→    const resolvedStats = resolveStats({
   543→      characterId: effectiveCharacter.id,
   544→      statsConfig: effectiveCharacter.statsConfig,
   545→      sessionStats: sessionStats,
   546→      allCharacters,
   547→      userName: effectiveUserName,
   548→      characterName: effectiveCharacter.name,
   549→      questTemplates,
   550→      personaDescription: persona?.description,
   551→      personaResolvedStats: streamPersonaResolvedStats,
   552→    });
   553→
   554→    // Build outlet sections map from lorebook plan for {{outlet::name}} macro resolution
   555→    const outletSections: Record<string, string> = {};
   556→    if (lorebookPlan?.outletSections.length) {
   557→      for (const outletSection of lorebookPlan.outletSections) {
   558→        const match = outletSection.label.match(/^World Info \((.+)\)$/);
   559→        const outletName = match ? match[1] : outletSection.label;
   560→        outletSections[outletName] = outletSection.content;
   561→      }
   562→    }
   563→
   564→    const keyContext = buildKeyResolutionContext(
   565→      effectiveCharacter,
   566→      effectiveUserName,
   567→      persona,
   568→      resolvedStats,
   569→      sessionStats,
   570→      soundTriggers,
   571→      soundSettings,
   572→      streamPersonaResolvedStats,
   573→      questTemplates,
   574→      sessionQuests,
   575→      questSettings,
   576→      outletSections,
   577→      lorebookAttributeKeys,
   578→      inventoryData     // Pass inventory data for {{inventory}} and {{currency}} key resolution
   579→    );
   580→

  1700→
