    91→    91→// Operators recommended for numeric attributes
    92→    92→const NUMERIC_COMPARATORS: AttributeComparator[] = [
    93→    93→  '==', '!=', '<', '<=', '>', '>=',
    94→    94→];
    95→    95→
    96→    96→// Operators recommended for text/keyword attributes
    97→    97→const TEXT_COMPARATORS: AttributeComparator[] = [
    98→    98→  '==', '!=', 'contains', 'not_contains',
    99→    99→];
   100→   100→
   101→   101→/**
   102→   102→ * Returns the appropriate comparators based on attribute type.
   103→   103→ * Falls back to showing all if type is unknown.
   104→   104→ */
   105→   105→function getComparatorsForType(attrType?: string): AttributeComparator[] {
   106→   106→  if (attrType === 'number') return NUMERIC_COMPARATORS;
   107→   107→  if (attrType === 'keyword' || attrType === 'text') return TEXT_COMPARATORS;
   108→   108→  // Unknown type — show all
   109→   109→  return Object.keys(COMPARATOR_LABELS) as AttributeComparator[];
   110→   110→}

   639→   639→              <div>
   640→   640→                <Label>Descripción (opcional)</Label>

   678→   678→interface LorebookEntryEditorProps {
   679→   679→  entry: LorebookEntry;
   680→   680→  lorebookId: string;
   681→   681→  characters: Array<{ id: string; name: string; statsConfig?: { attributes?: Array<{ key: string; name: string }> } }>;
   682→   682→  activePersona?: { id: string; name: string; statsConfig?: { attributes?: Array<{ key: string; name: string }> } } | null;
   683→   683→  onUpdate: (updates: Partial<LorebookEntry>) => void;
   684→   684→  onDelete: () => void;
   685→   685→  onDuplicate: () => void;
   686→   686→}
   687→   687→
   688→   688→function LorebookEntryEditor({
   689→   689→  entry,
   690→   690→  characters,
   691→   691→  activePersona,
   692→   692→  onUpdate,
   693→   693→  onDelete,
   694→   694→  onDuplicate
   695→   695→}: LorebookEntryEditorProps) {
   696→   696→  const isAttribute = entry.entryType === 'attribute';
   697→   697→  const config = entry.attributeConfig;
   698→   698→
   699→   699→  const parseKeywords = (value: string): string[] => {
   700→   700→    return value
   701→   701→      .split(',')
   702→   702→      .map(k => k.trim())
   703→   703→      .filter(k => k.length > 0);
   704→   704→  };
   705→   705→
   706→   706→  const formatKeywords = (keywords: string[]): string => {
   707→   707→    return keywords.join(', ');
   708→   708→  };
   709→   709→
   710→   710→  // Get available attribute definitions based on selected character
   711→   711→  const availableAttributes = useMemo(() => {
   712→   712→    if (!isAttribute || !config) return [];
   713→   713→
   714→   714→    if (config.characterId === '__user__') {
   715→   715→      return activePersona?.statsConfig?.attributes || [];
   716→   716→    }
   717→   717→
   718→   718→    if (config.characterId === '__char__') {
   719→   719→      // Return empty — will be resolved at runtime
   720→   720→      return [];
   721→   721→    }
   722→   722→
   723→   723→    const char = characters.find(c => c.id === config.characterId);
   724→   724→    return char?.statsConfig?.attributes || [];
   725→   725→  }, [isAttribute, config, characters, activePersona]);
   726→   726→
   727→   727→  // Determine the type of the currently selected attribute and filter comparators
   728→   728→  const selectedAttributeType = useMemo(() => {
   729→   729→    if (!config?.attributeKey || availableAttributes.length === 0) return undefined;
   730→   730→    const attr = availableAttributes.find(a => a.key === config.attributeKey);
   731→   731→    return attr?.type;
   732→   732→  }, [config, availableAttributes]);
   733→   733→
   734→   734→  const filteredComparators = useMemo(() => {
   735→   735→    return getComparatorsForType(selectedAttributeType);
   736→   736→  }, [selectedAttributeType]);
   737→   737→
   738→   738→  // Auto-fix operator when switching attribute type (e.g., was using '<' on text attr)
   739→   739→  const sanitizeOperator = useCallback((operator: AttributeComparator): AttributeComparator => {
   740→   740→    if (filteredComparators.includes(operator)) return operator;
   741→   741→    // Reset to a safe default for the type
   742→   742→    return selectedAttributeType === 'number' ? '>' : '==';
   743→   743→  }, [filteredComparators, selectedAttributeType]);
   744→   744→
   745→   745→  // Handle entry type change
   746→   746→  const handleEntryTypeChange = (type: LorebookEntryType) => {
   747→   747→    if (type === 'attribute') {
   748→   748→      // Initialize default attribute config with injectionKey
   749→   749→      const newConfig: LorebookAttributeConfig = {
   750→   750→        characterId: '__user__',
   751→   751→        attributeKey: '',
   752→   752→        injectionKey: '',
   753→   753→        mode: 'static',
   754→   754→        staticCondition: { operator: '>', value: 0 },
   755→   755→        dynamicConditions: [],
   756→   756→      };
   757→   757→      onUpdate({
   758→   758→        entryType: 'attribute',
   759→   759→        attributeConfig: newConfig,
   760→   760→        constant: false,
   761→   761→        key: [],
   762→   762→        keysecondary: [],
   763→   763→      });
   764→   764→    } else {
   765→   765→      // Switch to traditional
   766→   766→      onUpdate({
   767→   767→        entryType: 'traditional',
   768→   768→        attributeConfig: undefined,
   769→   769→      });
   770→   770→    }
   771→   771→  };
   772→   772→
   773→   773→  // Handle attribute config changes
   774→   774→  const updateAttributeConfig = (partial: Partial<LorebookAttributeConfig>) => {
   775→   775→    const currentConfig = config || {
   776→   776→      characterId: '__user__',
   777→   777→      attributeKey: '',
   778→   778→      injectionKey: '',
   779→   779→      mode: 'static' as AttributeEntryMode,
   780→   780→      staticCondition: { operator: '>' as AttributeComparator, value: 0 },
   781→   781→    };
   782→   782→    onUpdate({ attributeConfig: { ...currentConfig, ...partial } });
   783→   783→  };
   784→   784→
   785→   785→  // Handle mode change
   786→   786→  const handleModeChange = (mode: AttributeEntryMode) => {
   787→   787→    // Use a type-appropriate default operator
   788→   788→    const defaultOp = selectedAttributeType === 'number' ? '>' : '==';
   789→   789→    if (mode === 'static') {
   790→   790→      updateAttributeConfig({
   791→   791→        mode: 'static',
   792→   792→        staticCondition: { operator: defaultOp, value: selectedAttributeType === 'number' ? 0 : '' },
   793→   793→        dynamicConditions: undefined,
   794→   794→        fallbackContent: undefined,
   795→   795→      });
   796→   796→    } else {
   797→   797→      updateAttributeConfig({
   798→   798→        mode: 'dynamic',
   799→   799→        staticCondition: undefined,
   800→   800→        dynamicConditions: [
   801→   801→          { id: simpleId(), operator: defaultOp, value: selectedAttributeType === 'number' ? 0 : '', content: '' }
   802→   802→        ],
   803→   803→        fallbackContent: '',
   804→   804→      });
   805→   805→    }
   806→   806→  };
   807→   807→
   808→   808→  // Handle static condition changes
   809→   809→  const updateStaticCondition = (operator: AttributeComparator, value: number | string) => {
   810→   810→    updateAttributeConfig({ staticCondition: { operator, value } });
   811→   811→  };
   812→   812→
   813→   813→  // Handle dynamic condition changes
   814→   814→  const updateDynamicCondition = (condId: string, partial: Partial<{ operator: AttributeComparator; value: number | string; content: string }>) => {
   815→   815→    const conds = config?.dynamicConditions || [];
   816→   816→    updateAttributeConfig({
   817→   817→      dynamicConditions: conds.map(c => c.id === condId ? { ...c, ...partial } : c),
   818→   818→    });
   819→   819→  };
   820→   820→
   821→   821→  const addDynamicCondition = () => {
   822→   822→    const conds = config?.dynamicConditions || [];
   823→   823→    updateAttributeConfig({
   824→   824→      dynamicConditions: [...conds, { id: simpleId(), operator: '>' as AttributeComparator, value: 0, content: '' }],
   825→   825→    });
   826→   826→  };
   827→   827→
   828→   828→  const removeDynamicCondition = (condId: string) => {
   829→   829→    const conds = (config?.dynamicConditions || []).filter(c => c.id !== condId);
   830→   830→    updateAttributeConfig({ dynamicConditions: conds.length > 0 ? conds : [{ id: simpleId(), operator: '>', value: 0, content: '' }] });
   831→   831→  };
   832→   832→
   833→   833→  const moveDynamicCondition = (condId: string, direction: 'up' | 'down') => {
   834→   834→    const conds = config?.dynamicConditions;
   835→   835→    if (!conds || conds.length < 2) return;
   836→   836→    const idx = conds.findIndex(c => c.id === condId);
   837→   837→    if (idx === -1) return;
   838→   838→    const newIdx = direction === 'up' ? idx - 1 : idx + 1;
   839→   839→    if (newIdx < 0 || newIdx >= conds.length) return;
   840→   840→    const swapped = [...conds];
   841→   841→    [swapped[idx], swapped[newIdx]] = [swapped[newIdx], swapped[idx]];
   842→   842→    updateAttributeConfig({ dynamicConditions: swapped });
   843→   843→  };
   844→   844→
   845→   845→  return (
   846→   846→    <div className="space-y-3 pt-1">
   847→   847→      {/* Row 1: Title, Entry Type, Active Toggle */}
   848→   848→      <div className="grid grid-cols-[1fr_auto_auto_auto] gap-3">
   849→   849→        <div>
   850→   850→          <Label className="text-xs">Título</Label>
   851→   851→          <Input
   852→   852→            value={entry.comment}
   853→   853→            onChange={(e) => onUpdate({ comment: e.target.value })}
   854→   854→            className="mt-1 h-7 text-sm"
   855→   855→          />
   856→   856→        </div>
   857→   857→        <div>
   858→   858→          <Label className="text-xs">Tipo</Label>
   859→   859→          <Select
   860→   860→            value={entry.entryType || 'traditional'}
   861→   861→            onValueChange={(v) => handleEntryTypeChange(v as LorebookEntryType)}
   862→   862→          >
   863→   863→            <SelectTrigger className="mt-1 h-7 text-xs w-[130px]">
   864→   864→              <SelectValue />
   865→   865→            </SelectTrigger>
   866→   866→            <SelectContent>
   867→   867→              <SelectItem value="traditional" className="text-xs">
   868→   868→                <span className="flex items-center gap-1.5">
   869→   869→                  <Key className="w-3 h-3" />
   870→   870→                  Tradicional
   871→   871→                </span>
   872→   872→              </SelectItem>
   873→   873→              <SelectItem value="attribute" className="text-xs">
   874→   874→                <span className="flex items-center gap-1.5">
   875→   875→                  <Sword className="w-3 h-3" />
   876→   876→                  Atributo
   877→   877→                </span>
   878→   878→              </SelectItem>
   879→   879→            </SelectContent>
   880→   880→          </Select>
   881→   881→        </div>
   882→   882→        <div className="flex items-end pb-1">
   883→   883→          <label className="flex items-center gap-1.5 text-xs cursor-pointer">
   884→   884→            <Switch
   885→   885→              checked={!entry.disable}
   886→   886→              onCheckedChange={(checked) => onUpdate({ disable: !checked })}
   887→   887→            />
   888→   888→            Activo
   889→   889→          </label>
   890→   890→        </div>
   891→   891→      </div>
   892→   892→
   893→   893→      {/* ============================================ */}
   894→   894→      {/* TRADITIONAL ENTRY FIELDS                   */}
   895→   895→      {/* ============================================ */}
   896→   896→      {!isAttribute && (
   897→   897→        <>
   898→   898→          {/* Primary Keys */}
   899→   899→          <div className="space-y-1.5">
   900→   900→            <div className="flex items-center gap-1.5">
   901→   901→              <Key className="w-3 h-3 text-primary" />
   902→   902→              <Label className="text-xs">Palabras Clave Primarias</Label>
   903→   903→            </div>
   904→   904→            <Input
   905→   905→              value={formatKeywords(entry.key)}
   906→   906→              onChange={(e) => onUpdate({ key: parseKeywords(e.target.value) })}
   907→   907→              placeholder="palabra1, palabra2... o /regex/i"
   908→   908→              className="h-7 text-sm"
   909→   909→            />
   910→   910→            <p className="text-[10px] text-muted-foreground">
   911→   911→              Soporta regex: /patrón/flags (ej: /(?:clima|lluvia)/i)
   912→   912→            </p>
   913→   913→          </div>
   914→   914→
   915→   915→          {/* Secondary Keys */}
   916→   916→          <div className="space-y-1.5">
   917→   917→            <div className="flex items-center gap-1.5">
   918→   918→              <Key className="w-3 h-3 text-muted-foreground" />
   919→   919→              <Label className="text-xs">Palabras Clave Secundarias</Label>
   920→   920→            </div>
   921→   921→            <Input
   922→   922→              value={formatKeywords(entry.keysecondary)}
   923→   923→              onChange={(e) => onUpdate({ keysecondary: parseKeywords(e.target.value) })}
   924→   924→              placeholder="adicional1, adicional2..."
   925→   925→              className="h-7 text-sm"
   926→   926→            />
   927→   927→          </div>
   928→   928→        </>
   929→   929→      )}
   930→   930→
   931→   931→      {/* ============================================ */}
   932→   932→      {/* ATTRIBUTE ENTRY FIELDS                     */}
