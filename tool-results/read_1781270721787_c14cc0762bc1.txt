   913→// ============================================
   914→// Attribute HUD Preview Component
   915→// ============================================
   916→
   917→interface AttributeHUDPreviewProps {
   918→  attribute: AttributeDefinition;
   919→}
   920→
   921→function AttributeHUDPreview({ attribute }: AttributeHUDPreviewProps) {
   922→  const value = attribute.defaultValue ?? (attribute.type === 'number' ? 0 : '');
   923→  const style = attribute.hudStyle || 'default';
   924→  const color = attribute.color || 'default';
   925→  const icon = attribute.icon;
   926→  const unit = attribute.hudUnit;
   927→  const min = attribute.min ?? 0;
   928→  const max = attribute.max ?? 100;
   929→  
   930→  // Color classes
   931→  const colorClasses: Record<string, string> = {
   932→    red: 'text-red-400',
   933→    green: 'text-green-400',
   934→    blue: 'text-blue-400',
   935→    yellow: 'text-yellow-400',
   936→    purple: 'text-purple-400',
   937→    orange: 'text-orange-400',
   938→    pink: 'text-pink-400',
   939→    cyan: 'text-cyan-400',
   940→    default: 'text-white/80',
   941→  };
   942→  
   943→  const bgColorClasses: Record<string, string> = {
   944→    red: 'bg-red-500/20 border-red-500/30',
   945→    green: 'bg-green-500/20 border-green-500/30',
   946→    blue: 'bg-blue-500/20 border-blue-500/30',
   947→    yellow: 'bg-yellow-500/20 border-yellow-500/30',
   948→    purple: 'bg-purple-500/20 border-purple-500/30',
   949→    orange: 'bg-orange-500/20 border-orange-500/30',
   950→    pink: 'bg-pink-500/20 border-pink-500/30',
   951→    cyan: 'bg-cyan-500/20 border-cyan-500/30',
   952→    default: 'bg-white/10 border-white/20',
   953→  };
   954→  
   955→  const progressColorClasses: Record<string, string> = {
   956→    red: 'bg-red-500',
   957→    green: 'bg-green-500',
   958→    blue: 'bg-blue-500',
   959→    yellow: 'bg-yellow-500',
   960→    purple: 'bg-purple-500',
   961→    orange: 'bg-orange-500',
   962→    pink: 'bg-pink-500',
   963→    cyan: 'bg-cyan-500',
   964→    default: 'bg-white/50',
   965→  };
   966→  
   967→  const textColor = colorClasses[color] || colorClasses.default;
   968→  const bgColor = bgColorClasses[color] || bgColorClasses.default;
   969→  const progressColor = progressColorClasses[color] || progressColorClasses.default;
   970→  
   971→  // Render based on style
   972→  switch (style) {
   973→    case 'progress': {
   974→      const percentage = Math.max(0, Math.min(100, ((Number(value) - min) / (max - min)) * 100));
   975→      return (
   976→        <div className="space-y-1">
   977→          <div className="flex items-center justify-between gap-2">
   978→            <div className="flex items-center gap-1.5">
   979→              {icon && <span className="text-sm">{icon}</span>}
   980→              <span className="text-xs text-white/50">{attribute.name}</span>
   981→            </div>
   982→            <span className="text-xs font-medium text-white/80">
   983→              {value}{unit && <span className="text-white/40 ml-0.5">{unit}</span>}
   984→            </span>
   985→          </div>
   986→          <div className="w-full bg-white/10 rounded-full h-2">
   987→            <div className={`h-full rounded-full transition-all ${progressColor}`} style={{ width: `${percentage}%` }} />
   988→          </div>
   989→        </div>
   990→      );
   991→    }
   992→    
   993→    case 'gauge': {
   994→      const percentage = Math.max(0, Math.min(100, ((Number(value) - min) / (max - min)) * 100));
   995→      const circumference = 2 * Math.PI * 28;
   996→      const offset = circumference - (percentage / 100) * circumference;
   997→      return (
   998→        <div className="flex items-center gap-2">
   999→          <div className="relative w-10 h-10">
  1000→            <svg className="w-10 h-10 transform -rotate-90">
  1001→              <circle cx="20" cy="20" r="16" stroke="rgba(255,255,255,0.1)" strokeWidth="4" fill="none" />
  1002→              <circle cx="20" cy="20" r="16" stroke="currentColor" strokeWidth="4" fill="none"
  1003→                strokeDasharray={circumference} strokeDashoffset={offset}
  1004→                className={textColor} style={{ transition: 'stroke-dashoffset 0.5s' }} />
  1005→            </svg>
  1006→            <div className="absolute inset-0 flex items-center justify-center">
  1007→              <span className="text-[10px] font-bold text-white">{value}</span>
  1008→            </div>
  1009→          </div>
  1010→          <div className="flex flex-col">
  1011→            <span className="text-xs text-white/50">{attribute.name}</span>
  1012→            {unit && <span className="text-[10px] text-white/30">{unit}</span>}
  1013→          </div>
  1014→        </div>
  1015→      );
  1016→    }
  1017→    
  1018→    case 'badge':
  1019→      return (
  1020→        <div className="flex items-center gap-2">
  1021→          {icon && <span className="text-sm">{icon}</span>}
  1022→          <span className="text-xs text-white/50">{attribute.name}:</span>
  1023→          <span className={`inline-flex items-center rounded-full border px-2 py-0.5 text-xs font-medium ${bgColor} ${textColor}`}>
  1024→            {value}
  1025→            {unit && <span className="ml-0.5 opacity-60">{unit}</span>}
  1026→          </span>
  1027→        </div>
  1028→      );
  1029→    
  1030→    case 'pill':
  1031→      return (

  1462→function ThresholdEffectDialog({ effect, open, onOpenChange, onSave, allAttributes, availableTargets, spritePacksV2, attributeKey }: ThresholdEffectDialogProps) {
  1463→  // Local state for editing
  1464→  const [localEffect, setLocalEffect] = useState<ThresholdEffect>({ ...effect });
  1465→
  1466→  // Sync when effect prop changes or dialog opens
  1467→  const prevOpenRef = useState(false);
  1468→  if (open && !prevOpenRef[0]) {
  1469→    setLocalEffect({ ...effect });
  1470→    prevOpenRef[1](true);
  1471→  }
  1472→  if (!open && prevOpenRef[0]) {
  1473→    prevOpenRef[1](false);
  1474→  }
  1475→
  1476→  const updateEffect = (updates: Partial<ThresholdEffect>) => {
  1477→    setLocalEffect(prev => ({ ...prev, ...updates }));
  1478→  };
  1479→
  1480→  // Condition management
  1481→  const addCondition = () => {
  1482→    const newCondition: StatRequirement = {
  1483→      attributeKey: attributeKey,  // Default to self attribute
  1484→      operator: '<=',
  1485→      value: 0,
  1486→    };
  1487→    updateEffect({ conditions: [...localEffect.conditions, newCondition] });
  1488→  };
  1489→
  1490→  const removeCondition = (idx: number) => {
  1491→    updateEffect({ conditions: localEffect.conditions.filter((_, i) => i !== idx) });
  1492→  };
  1493→
  1494→  const updateCondition = (idx: number, updates: Partial<StatRequirement>) => {
  1495→    const updated = [...localEffect.conditions];
  1496→    updated[idx] = { ...updated[idx], ...updates };
  1497→    updateEffect({ conditions: updated });
  1498→  };
  1499→
  1500→  // Reward management
  1501→  const addReward = (type: QuestReward['type']) => {
  1502→    let newReward: QuestReward;
  1503→    switch (type) {
  1504→      case 'trigger':
  1505→        newReward = createTriggerReward('sprite', '', 'self');
  1506→        break;
  1507→      case 'attribute':
  1508→        newReward = {
  1509→          id: `threshold-reward-${Date.now().toString(36)}`,
  1510→          type: 'attribute',
  1511→          attribute: { key: '', value: 0, action: 'set' }
  1512→        };
  1513→        break;
  1514→      case 'target_attribute':
  1515→        newReward = createTargetAttributeReward('', '', 0, 'set');
  1516→        break;
  1517→      case 'currency':
  1518→        newReward = createCurrencyReward(0);
  1519→        break;
  1520→      case 'activate_sprite_pack':
  1521→        newReward = createActivateSpritePackReward('', { targetMode: 'self' });
  1522→        break;
  1523→      case 'conditional_sprite_collection':
  1524→        newReward = {
  1525→          id: `threshold-reward-${Date.now().toString(36)}`,

  1528→            collectionId: '',
  1529→            targetMode: 'self',
  1530→          }
