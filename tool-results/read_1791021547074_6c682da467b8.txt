    40→export function normalizePath(uri: string): string {
    41→  if (uri.startsWith('mem://') || uri.startsWith('lancedb://')) return uri;
    42→  let normalized = path.normalize(uri);
    43→  if (isWindows()) normalized = normalized.replace(/\//g, '\\');
    44→  return normalized;
    45→}

    47→export function checkDirectoryPermissions(dirPath: string): {
    48→  exists: boolean; writable: boolean; error?: string;
    49→} {
    50→  try {
    51→    const normalizedPath = normalizePath(dirPath);
    52→    const parentDir = path.dirname(normalizedPath);
    53→
    54→    if (!fs.existsSync(parentDir)) {
    55→      return { exists: false, writable: false, error: 'Parent directory does not exist' };
    56→    }
    57→
    58→    if (fs.existsSync(normalizedPath)) {
    59→      try {
    60→        fs.accessSync(normalizedPath, fs.constants.W_OK);
    61→        return { exists: true, writable: true };
    62→      } catch {
    63→        return { exists: true, writable: false, error: 'No write permission' };
    64→      }
    65→    }
    66→
    67→    try {
    68→      fs.accessSync(parentDir, fs.constants.W_OK);
    69→      return { exists: false, writable: true };
    70→    } catch {
    71→      return { exists: false, writable: false, error: 'No write permission on parent directory' };
    72→    }
    73→  } catch (error: any) {
    74→    return { exists: false, writable: false, error: error.message };
    75→  }
    76→}

    78→export function ensureLanceDBDirectory(uri: string): { success: boolean; path: string; error?: string } {
    79→  try {
    80→    if (uri.startsWith('mem://') || uri.startsWith('lancedb://')) {
    81→      return { success: true, path: uri };
    82→    }
    83→    const normalizedPath = normalizePath(uri);
    84→    if (!fs.existsSync(normalizedPath)) {
    85→      fs.mkdirSync(normalizedPath, { recursive: true });
    86→    }
    87→    return { success: true, path: normalizedPath };
    88→  } catch (error: any) {
    89→    return { success: false, path: uri, error: error.message };
    90→  }
    91→}

   146→// ============ FASE 15: Platform Detection (arch + libc) ============
   147→
   148→export type Architecture = 'x64' | 'arm64' | 'ia32' | 'arm' | 'unknown';
   149→export type Libc = 'gnu' | 'musl' | 'unknown';
   150→
   151→/** Detect CPU architecture */
   152→export function getArchitecture(): Architecture {
   153→  if (typeof process !== 'undefined' && process.arch) {
   154→    const arch = process.arch;
   155→    if (arch === 'x64' || arch === 'arm64' || arch === 'ia32' || arch === 'arm') {
   156→      return arch;
   157→    }
   158→  }
   159→  return 'unknown';
   160→}
   161→
   162→/**
   163→ * Detect libc variant on Linux (glibc vs musl).
   164→ * Alpine Linux uses musl; most other distros use glibc.
   165→ * On non-Linux platforms, returns 'unknown'.
   166→ */
   167→export function detectLibc(): Libc {
   168→  if (!isLinux()) return 'unknown';
   169→
   170→  // Check if musl is in use by looking at the dynamic linker
   171→  // This is a heuristic — the napi-rs loader does something similar
   172→  try {
   173→    // Use dynamic import to avoid require() lint error
   174→    const { execSync } = require_child_process();
   175→    // Try ldd — musl systems report "musl" in the output
   176→    const lddOutput = execSync('ldd --version 2>&1', { encoding: 'utf-8', timeout: 2000 });
   177→    if (lddOutput.toLowerCase().includes('musl')) {
   178→      return 'musl';
   179→    }
   180→    return 'gnu';
   181→  } catch {
   182→    // Fallback: assume glibc (most common)
   183→    return 'gnu';
   184→  }
   185→}
   186→
   187→// Helper to import child_process without require() lint error
   188→function require_child_process() {
   189→  // eslint-disable-next-line @typescript-eslint/no-require-imports
   190→  return require('child_process');
   191→}

   193→/**
   194→ * Get the LanceDB platform package name for the current system.
   195→ * Returns null if the platform is not supported.
   196→ *
   197→ * Examples:
   198→ * - Windows x64 → @lancedb/lancedb-win32-x64-msvc
   199→ * - Linux x64 glibc → @lancedb/lancedb-linux-x64-gnu
   200→ * - Linux x64 musl (Alpine) → @lancedb/lancedb-linux-x64-musl
   201→ * - macOS arm64 (Apple Silicon) → @lancedb/lancedb-darwin-arm64
   202→ * - macOS x64 (Intel) → @lancedb/lancedb-darwin-x64 (may not exist)
   203→ */
   204→export function getLanceDBPlatformPackage(): string | null {
   205→  const platform = getPlatform();
   206→  const arch = getArchitecture();
   207→  const libc = detectLibc();
   208→
   209→  if (platform === 'win32') {
   210→    if (arch === 'x64') return '@lancedb/lancedb-win32-x64-msvc';
   211→    if (arch === 'arm64') return '@lancedb/lancedb-win32-arm64-msvc';
   212→  }
   213→
   214→  if (platform === 'linux') {
   215→    if (arch === 'x64') {
   216→      return libc === 'musl'
   217→        ? '@lancedb/lancedb-linux-x64-musl'
   218→        : '@lancedb/lancedb-linux-x64-gnu';
   219→    }
   220→    if (arch === 'arm64') {
   221→      return libc === 'musl'
   222→        ? '@lancedb/lancedb-linux-arm64-musl'
   223→        : '@lancedb/lancedb-linux-arm64-gnu';
   224→    }
   225→  }
   226→
   227→  if (platform === 'darwin') {
   228→    if (arch === 'arm64') return '@lancedb/lancedb-darwin-arm64';
   229→    if (arch === 'x64') return '@lancedb/lancedb-darwin-x64';
   230→  }
   231→
   232→  return null;
   233→}

   235→/**
   236→ * Get a human-readable description of the detected platform.
   237→ */
   238→export function getPlatformDescription(): string {
   239→  const platform = getPlatform();
   240→  const arch = getArchitecture();
   241→  const libc = detectLibc();
   242→
   243→  const platformName = platform === 'win32' ? 'Windows'
   244→    : platform === 'linux' ? 'Linux'
   245→    : platform === 'darwin' ? 'macOS'
   246→    : 'Unknown';
   247→
   248→  const archName = arch === 'x64' ? 'x64'
   249→    : arch === 'arm64' ? 'ARM64'
   250→    : arch === 'ia32' ? 'x86'
   251→    : arch === 'arm' ? 'ARM'
   252→    : 'Unknown';
   253→
   254→  let desc = `${platformName} ${archName}`;
   255→  if (platform === 'linux' && libc !== 'unknown') {
   256→    desc += ` (${libc})`;
   257→  }
   258→
   259→  return desc;
   260→}

   264→export class LanceDBError extends Error {
   265→  constructor(
   266→    message: string,
   267→    public code: string,
   268→    public platform?: Platform,
   269→    public details?: any
   270→  ) {
   271→    super(message);
   272→    this.name = 'LanceDBError';
   273→  }

   275→  getSuggestion(): string {
   276→    switch (this.code) {
   277→      case 'PERMISSION_DENIED':
   278→        return isWindows()
   279→          ? 'Run as administrator or change the database path.'
   280→          : 'Check directory permissions with chmod.';
   281→      case 'NATIVE_MODULE_ERROR':
   282→        return isWindows()
   283→          ? 'Install Visual C++ Redistributable. Reinstall @lancedb/lancedb.'
   284→          : 'Check native dependencies (glibc, etc.).';
   285→      case 'MODULE_NOT_AVAILABLE':
   286→        return 'LanceDB is not available on this system. Check platform compatibility.';
   287→      case 'TABLE_NOT_FOUND':
   288→        return 'Table does not exist. It will be created on first operation.';
   289→      default:
   290→        return 'Check the documentation or logs for more details.';
   291→    }
   292→  }

   297→async function loadLanceDBModule(): Promise<{ success: boolean; error?: string }> {

   306→  } catch (error: any) {
   307→    const errorMsg = error.message || String(error);
   308→    lancedbLoadError = errorMsg;
   309→    return { success: false, error: errorMsg };
   310→  }

   336→function normalizeVector(vector: number[]): number[] {
   337→  const magnitude = Math.sqrt(vector.reduce((sum, val) => sum + val * val, 0));
   338→  if (magnitude === 0) return vector;
   339→  return vector.map(val => val / magnitude);
   340→}
   341→
   342→/**
   343→ * Convert LanceDB's L2-squared distance to cosine similarity.
   344→ * LanceDB vectorSearch returns _distance = L2² (squared Euclidean distance).
   345→ * For unit-normalized vectors: L2² = 2 - 2·cos(θ), so cos(θ) = 1 - L2²/2.
   346→ */
   347→function l2SquaredToCosineSimilarity(l2SquaredDistance: number): number {
   348→  return 1 - l2SquaredDistance / 2;
   349→}

   359→// ============ Initialization ============
   360→
   361→// Track the dimension that the embeddings table was created with
   362→let tableDimension: number | null = null;
   363→
   364→async function initializeTables(): Promise<void> {
   365→  if (!db) throw new Error('Database not initialized');
   366→
   367→  // Get vector dimension from config
   368→  let vectorDimension = 1024;
   369→  let embeddingModel = 'bge-m3:567m';
   370→  try {
   371→    const { getConfig } = await import('./config-persistence');
   372→    const config = getConfig();
   373→    vectorDimension = config.dimension || 1024;
   374→    embeddingModel = config.model || 'bge-m3:567m';
   375→  } catch {
   376→    // Use defaults
   377→  }
   378→
   379→  const defaultEmbedding = {
   380→    id: 'placeholder',
   381→    content: 'placeholder',
   382→    vector: new Array(vectorDimension).fill(0),
   383→    metadata: '{}',
   384→    namespace: 'default',
   385→    source_type: 'system',
   386→    source_id: 'init',
   387→    model_name: embeddingModel,
   388→    created_at: new Date().toISOString(),
   389→    updated_at: new Date().toISOString(),
   390→  };
   391→
   392→  const defaultNamespace = {
   393→    id: 'placeholder',
   394→    namespace: 'default',
   395→    description: 'Default namespace',
   396→    metadata: '{}',
   397→    created_at: new Date().toISOString(),
   398→    updated_at: new Date().toISOString(),
   399→  };
   400→
   401→  // Try to open existing embeddings table
   402→  try {
   403→    embeddingsTable = await db.openTable(EMBEDDINGS_TABLE);
   404→
   405→    // Check if dimension matches by reading the first row's vector
   406→    // If the table was created with a different dimension, we must recreate it
   407→    const existingRows = await embeddingsTable.query().limit(1).toArray();
   408→    if (existingRows.length > 0 && existingRows[0].vector) {
   409→      const existingDim = existingRows[0].vector.length;
   410→      tableDimension = existingDim;
   411→
   412→      if (existingDim !== vectorDimension) {
   413→        console.warn(
   414→          `[LanceDB] Dimension mismatch detected! Table has ${existingDim}D vectors, but config requires ${vectorDimension}D. ` +
   415→          `Dropping and recreating embeddings table. All existing embeddings will be lost.`
   416→        );
   417→
   418→        // Drop the old table
   419→        try { await db.dropTable(EMBEDDINGS_TABLE); } catch { /* ignore */ }
   420→
   421→        // Also drop any namespace-specific tables that might have wrong dimensions
   422→        try {
   423→          const tables = await db.tableNames();
   424→          for (const tableName of tables) {
   425→            if (tableName !== EMBEDDINGS_TABLE && tableName !== NAMESPACES_TABLE) {
   426→              try { await db.dropTable(tableName); } catch { /* ignore */ }
   427→            }
   428→          }
   429→        } catch { /* ignore */ }
   430→
   431→        // Recreate with new dimension
   432→        await db.createTable(EMBEDDINGS_TABLE, [defaultEmbedding]);
   433→        embeddingsTable = await db.openTable(EMBEDDINGS_TABLE);
   434→        await embeddingsTable.delete("id = 'placeholder'");
   435→        tableDimension = vectorDimension;
   436→      }
   437→    } else if (existingRows.length === 0) {
   438→      // Table exists but is empty — recreate with new dimension to be safe
   439→      try { await db.dropTable(EMBEDDINGS_TABLE); } catch { /* ignore */ }
   440→      await db.createTable(EMBEDDINGS_TABLE, [defaultEmbedding]);
   441→      embeddingsTable = await db.openTable(EMBEDDINGS_TABLE);
   442→      await embeddingsTable.delete("id = 'placeholder'");
   443→      tableDimension = vectorDimension;
   444→    }
   445→  } catch {
   446→    // Table doesn't exist yet, create it
   447→    await db.createTable(EMBEDDINGS_TABLE, [defaultEmbedding]);
   448→    embeddingsTable = await db.openTable(EMBEDDINGS_TABLE);
   449→    await embeddingsTable.delete("id = 'placeholder'");
   450→    tableDimension = vectorDimension;
   451→  }
   452→
   453→  // Namespaces table (no vectors, no dimension concern)
   454→  try {
   455→    namespacesTable = await db.openTable(NAMESPACES_TABLE);
   456→  } catch {
   457→    await db.createTable(NAMESPACES_TABLE, [defaultNamespace]);
   458→    namespacesTable = await db.openTable(NAMESPACES_TABLE);
   459→    await namespacesTable.delete("id = 'placeholder'");
   460→  }
   461→}

   467→export async function initLanceDB(uri?: string, forceReinit: boolean = false): Promise<void> {
   468→  const dbUri = uri || process.env.LANCEDB_URI || getDefaultLanceDBPath();
   469→
   470→  if (isInitialized && db && currentUri === dbUri && !forceReinit) return;
   471→
   472→  // If permanently unavailable, skip silently (don't throw)
   473→  if (isPermanentlyUnavailable) {
   474→    return;
   475→  }
   476→
   477→  if (db) await closeLanceDB();
   478→
   479→  const { success, error } = await loadLanceDBModule();
   480→  if (!success) {
   481→    isPermanentlyUnavailable = true;
   482→    lancedbLoadError = error || 'Unknown error';
   483→    console.warn(
   484→      `[LanceDB] ⚠️ LanceDB native module not available for platform "${getPlatform()}". ` +
   485→      `Embeddings/vector search features will be disabled. Error: ${error}`
   486→    );
   487→    return;
   488→  }
   489→
   490→  isInitialized = true;
   491→  currentUri = dbUri;
   492→
   493→  try {
   494→    const dirResult = ensureLanceDBDirectory(dbUri);
   495→    if (!dirResult.success) {
   496→      throw new LanceDBError(
   497→        `Cannot create/verify directory: ${dirResult.error}`,
   498→        'DIRECTORY_ERROR',
   499→        getPlatform(),
   500→        { uri: dbUri }
   501→      );
   502→    }
   503→
   504→    db = await lancedbModule.connect(dirResult.path);
   505→    await initializeTables();
   506→  } catch (error: any) {
   507→    if (error instanceof LanceDBError) throw error;
   508→
   509→    isInitialized = false;
   510→    db = null;
   511→    currentUri = null;
   512→
   513→    const msg = error.message || String(error);
   514→    const lowerMsg = msg.toLowerCase();
   515→
   516→    let code = 'UNKNOWN_ERROR';
   517→    if (lowerMsg.includes('permission') || lowerMsg.includes('access denied')) code = 'PERMISSION_DENIED';
   518→    else if (lowerMsg.includes('native') || lowerMsg.includes('binding')) code = 'NATIVE_MODULE_ERROR';
   519→
   520→    throw new LanceDBError(msg, code, getPlatform(), { uri: dbUri, originalError: error });
   521→  }
   522→}

   597→  static async insertEmbedding(params: {
   598→    content: string;
   599→    vector: number[];
   600→    metadata?: Record<string, any>;
   601→    namespace?: string;
   602→    source_type?: string;
   603→    source_id?: string;
   604→    model_name?: string;
   605→  }): Promise<string> {
   606→    if (isPermanentlyUnavailable) {
   607→      console.warn('[LanceDB] insertEmbedding skipped: LanceDB is not available on this system.');
   608→      return '__unavailable__';
   609→    }
   610→    const { v4: uuidv4 } = await import('uuid');
   611→    const {
   612→      content, vector, metadata = {},

   622→    }
   623→
   624→    const table = await getEmbeddingsTable();
   625→    if (!table) {
   626→      console.warn('[LanceDB] insertEmbedding skipped: table not available.');
   627→      return '__unavailable__';
   628→    }
   629→    const normalizedVector = normalizeVector(vector);

   648→  static async searchSimilar(params: {
   649→    queryVector: number[];
   650→    namespace?: string;
   651→    limit?: number;
   652→    threshold?: number;
   653→  }): Promise<SearchResult[]> {
   654→    if (isPermanentlyUnavailable) return [];
   655→    const {
   656→      queryVector,
   657→      namespace,
   658→      limit = 10,
   659→      threshold = 0.3,
   660→    } = params;
   661→
   662→    const table = await getEmbeddingsTable();
   663→    if (!table) return [];
   664→    const normalizedQueryVector = normalizeVector(queryVector);
   665→
   666→    let results: any[];
   667→
   668→    // All embeddings are stored in the main table with a 'namespace' column.
   669→    // Always search the main table and filter by namespace in-memory.
   670→    const allResults = await table
   671→      .vectorSearch(normalizedQueryVector)
   672→      .limit(limit * 10)
   673→      .toArray();
   674→
   675→    results = allResults
   676→      .map((row: any) => ({
   677→        ...row,
   678→        similarity: l2SquaredToCosineSimilarity(row._distance || 0),
   679→      }))
   680→      .filter((r: any) => {
   681→        if (!namespace || namespace === 'all') return r.similarity >= threshold;
   682→        return (r.namespace || 'default') === namespace && r.similarity >= threshold;
   683→      })
   684→      .slice(0, limit);
   685→
   686→    return results.map((row: any) => ({
   687→      id: row.id,
   688→      content: row.content,
   689→      metadata: parseMetadata(row.metadata),
   690→      namespace: row.namespace || 'default',
   691→      source_type: row.source_type,
   692→      source_id: row.source_id,
   693→      similarity: row.similarity,
   694→    }));
   695→  }

   739→  static async upsertNamespace(params: {
   740→    namespace: string;
   741→    description?: string;
   742→    metadata?: Record<string, any>;
   743→  }): Promise<RecordNamespace> {
   744→    if (isPermanentlyUnavailable) {
   745→      // Return a stub namespace so callers don't crash
   746→      return {
   747→        id: '__unavailable__',
   748→        namespace: params.namespace,
   749→        description: params.description,
   750→        metadata: params.metadata || {},
   751→        created_at: new Date(),
   752→        updated_at: new Date(),
   753→      };
   754→    }
   755→    const { v4: uuidv4 } = await import('uuid');
   756→    const { namespace, description, metadata = {} } = params;
   757→    const table = await getNamespacesTable();
   758→    if (!table) {
   759→      return {
   760→        id: '__unavailable__',
   761→        namespace: params.namespace,
   762→        description: params.description,
   763→        metadata: params.metadata || {},
   764→        created_at: new Date(),
   765→        updated_at: new Date(),
   766→      };
   767→    }
   768→    const existing = await tableFilter(table, `namespace = '${escapeFilterValue(namespace)}'`);
   769→
   770→    const nsRecord = {
   771→      id: existing.length > 0 ? existing[0].id : uuidv4(),
   772→      namespace,
   773→      description,
   774→      metadata: JSON.stringify(metadata),
   775→      created_at: existing.length > 0 ? existing[0].created_at : new Date().toISOString(),
   776→      updated_at: new Date().toISOString(),
   777→    };
   778→
   779→    // Always delete all matching rows before inserting to prevent duplicates
   780→    // caused by race conditions (concurrent upsertNamespace calls).
   781→    if (existing.length > 0) {
   782→      await table.delete(`namespace = '${escapeFilterValue(namespace)}'`);
   783→    }
   784→    await table.add([nsRecord]);
   785→
   786→    // Safety net: if a race created extra duplicates, clean them up.
   787→    // Keep only the row we just inserted (by id).
   788→    const afterInsert = await tableFilter(table, `namespace = '${escapeFilterValue(namespace)}'`);
   789→    if (afterInsert.length > 1) {
   790→      const idsToDelete = afterInsert
   791→        .filter((row: any) => row.id !== nsRecord.id)
   792→        .map((row: any) => row.id);
   793→      if (idsToDelete.length > 0) {
   794→        const deleteFilter = idsToDelete
   795→          .map((id: string) => `id = '${escapeFilterValue(id)}'`)
   796→          .join(' OR ');
   797→        await table.delete(deleteFilter);
   798→        console.log(`[LanceDB] Cleaned up ${idsToDelete.length} duplicate namespace row(s) for "${namespace}"`);
   799→      }
   800→    }
   801→
   802→    return {
   803→      id: nsRecord.id,
   804→      namespace: nsRecord.namespace,
   805→      description: nsRecord.description,
   806→      metadata: JSON.parse(nsRecord.metadata),
   807→      created_at: new Date(nsRecord.created_at),
   808→      updated_at: new Date(nsRecord.updated_at),
   809→    };
   810→  }

   812→  static async getAllNamespaces(): Promise<RecordNamespace[]> {
   813→    if (isPermanentlyUnavailable) return [];
   814→    const table = await getNamespacesTable();
   815→    if (!table) return [];
   816→    const results = await tableToArray(table);
   817→    const mapped = results.map((row: any) => ({
   818→      id: row.id,
   819→      namespace: row.namespace,
   820→      description: row.description,
   821→      metadata: typeof row.metadata === 'string' ? JSON.parse(row.metadata) : row.metadata,
   822→      created_at: new Date(row.created_at),
   823→      updated_at: new Date(row.updated_at),
   824→    }));
   825→
   826→    // Deduplicate by namespace name — keep the most recently updated entry.
   827→    // This guards against duplicate rows that can appear due to a race condition
   828→    // in upsertNamespace (non-atomic delete-then-insert).
   829→    const deduped = new Map<string, RecordNamespace>();
   830→    for (const ns of mapped) {
   831→      const existing = deduped.get(ns.namespace);
   832→      if (!existing || ns.updated_at > existing.updated_at) {
   833→        deduped.set(ns.namespace, ns);
   834→      }
   835→    }
   836→    return Array.from(deduped.values());
   837→  }

   939→      // and manually strip the vector field

  1026→  static async getStats(): Promise<EmbeddingStats> {
  1027→    if (isPermanentlyUnavailable) {
  1028→      return {
  1029→        totalEmbeddings: 0,
  1030→        totalNamespaces: 0,
  1031→        embeddingsByNamespace: {},
  1032→        embeddingsBySourceType: {},
  1033→      };
  1034→    }
  1035→    const table = await getEmbeddingsTable();
  1036→    if (!table) {
  1037→      return {
  1038→        totalEmbeddings: 0,
  1039→        totalNamespaces: 0,
  1040→        embeddingsByNamespace: {},
  1041→        embeddingsBySourceType: {},
  1042→      };
  1043→    }
  1044→    const namespaces = await this.getAllNamespaces();
  1045→
  1046→    const embeddingsByNamespace: Record<string, number> = {};
  1047→    const embeddingsBySourceType: Record<string, number> = {};
  1048→    let totalEmbeddings = 0;
  1049→
  1050→    // Count per-namespace to avoid loading all vectors into memory at once
  1051→    for (const ns of namespaces) {
  1052→      try {
  1053→        const count = await this.countByNamespace(ns.namespace);
  1054→        if (count > 0) {
  1055→          embeddingsByNamespace[ns.namespace] = count;
  1056→          totalEmbeddings += count;
  1057→        }
  1058→      } catch { /* skip */ }
  1059→    }
  1060→
  1061→    // Also count 'default' namespace if not listed
  1062→    if (!embeddingsByNamespace['default']) {
  1063→      try {
  1064→        const defaultCount = await this.countByNamespace('default');
  1065→        if (defaultCount > 0) {
  1066→          embeddingsByNamespace['default'] = defaultCount;
  1067→          totalEmbeddings += defaultCount;
  1068→        }
  1069→      } catch { /* skip */ }
  1070→    }
  1071→
  1072→    // For source_type breakdown, we need to scan but we can use lightweight queries
  1073→    // Only load id + source_type (not full vectors) via filter
  1074→    const sourceTypes = ['character', 'world', 'lorebook', 'session', 'memory', 'custom', 'system'];
  1075→    for (const st of sourceTypes) {
  1076→      try {
  1077→        const matching = await tableFilter(table, `source_type = '${st}'`);
  1078→        if (matching.length > 0) {
  1079→          embeddingsBySourceType[st] = matching.length;
  1080→        }
  1081→      } catch { /* skip */ }
  1082→    }
  1083→
  1084→    return {
  1085→      totalEmbeddings,
  1086→      totalNamespaces: namespaces.length,
  1087→      embeddingsByNamespace,
  1088→      embeddingsBySourceType,
  1089→    };
  1090→  }

  1092→  static async resetAll(): Promise<{ deletedEmbeddings: number; deletedNamespaces: number }> {
  1093→    if (isPermanentlyUnavailable) return { deletedEmbeddings: 0, deletedNamespaces: 0 };
  1094→    // Count before dropping to report accurate numbers
  1095→    let deletedEmbeddings = 0;
  1096→    let deletedNamespaces = 0;
  1097→
  1098→    try {
  1099→      const allEmbeddings = await tableToArray(await getEmbeddingsTable());
  1100→      deletedEmbeddings = allEmbeddings.length;
  1101→    } catch { /* table may not exist */ }
  1102→
  1103→    try {
  1104→      const allNamespaces = await tableToArray(await getNamespacesTable());
  1105→      deletedNamespaces = allNamespaces.length;
  1106→    } catch { /* table may not exist */ }
  1107→
  1108→    // Drop and recreate the embeddings table (much faster than individual deletes)
  1109→    if (db) {
  1110→      try { await db.dropTable(EMBEDDINGS_TABLE); } catch { /* ignore */ }
  1111→      try { await db.dropTable(NAMESPACES_TABLE); } catch { /* ignore */ }
  1112→
  1113→      // Drop any extra tables
  1114→      try {
  1115→        const tables = await db.tableNames();
  1116→        for (const tableName of tables) {
  1117→          if (tableName !== EMBEDDINGS_TABLE && tableName !== NAMESPACES_TABLE) {
  1118→            try { await db.dropTable(tableName); } catch { /* ignore */ }
  1119→          }
  1120→        }
  1121→      } catch { /* ignore */ }
  1122→
  1123→      // Reinitialize with fresh tables
  1124→      await initializeTables();
  1125→    }
  1126→
  1127→    return { deletedEmbeddings, deletedNamespaces };
  1128→  }
