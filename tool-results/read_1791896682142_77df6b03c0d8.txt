     1→'use client';
     2→
     3→/**
     4→ * RewardEditor Component
     5→ * 
     6→ * Componente reutilizable para editar recompensas de quests.
     7→ * Soporta el sistema unificado de 2 tipos: attribute | trigger
     8→ * 
     9→ * Uso:
    10→ * <RewardEditor
    11→ *   reward={reward}
    12→ *   onChange={(updated) => ...}
    13→ *   onDelete={() => ...}
    14→ *   compact={true} // Para UI compacta en objetivos
    15→ * />
    16→ */
    17→
    18→import type {
    19→  QuestReward,
    20→  QuestRewardActivateSpritePack,
    21→  AttributeAction,
    22→  TriggerCategory,
    23→  TriggerTargetMode,
    24→  TriggerFallbackMode,
    25→} from '@/types';
    26→import {
    27→  normalizeReward,
    28→  describeReward,
    29→  createAttributeReward,
    30→  createTriggerReward,
    31→  createCurrencyReward,
    32→  createConditionalSpriteCollectionReward,
    33→  createActivateSpritePackReward,
    34→  getActionSymbol,
    35→} from '@/lib/quest/quest-reward-utils';
    36→import { cn } from '@/lib/utils';
    37→import { Button } from '@/components/ui/button';
    38→import { Input } from '@/components/ui/input';
    39→import { Label } from '@/components/ui/label';
    40→import { Badge } from '@/components/ui/badge';
    41→import {
    42→  Select,
    43→  SelectContent,
    44→  SelectItem,
    45→  SelectTrigger,
    46→  SelectValue,
    47→} from '@/components/ui/select';
    48→import {
    49→  X,
    50→  Hash,
    51→  Zap,
    52→  Image as ImageIcon,
    53→  Volume2,
    54→  Wallpaper,
    55→  Users,
    56→  User,
    57→  Crosshair,
    58→  ChevronDown,
    59→  Coins,
    60→  GitBranch,
    61→  Package,
    62→} from 'lucide-react';
    63→
    64→// ============================================
    65→// Types
    66→// ============================================
    67→
    68→export interface RewardEditorProps {
    69→  /** Recompensa a editar */
    70→  reward: QuestReward;
    71→  /** Callback cuando la recompensa cambia */
    72→  onChange: (reward: QuestReward) => void;
    73→  /** Callback cuando se solicita eliminar */
    74→  onDelete: () => void;
    75→  /** Lista de atributos disponibles para autocompletado */
    76→  availableAttributes?: string[];
    77→  /** Triggers disponibles por categoría */
    78→  availableTriggers?: {
    79→    sprites?: string[];
    80→    sounds?: string[];
    81→    backgrounds?: string[];
    82→  };
    83→  /** Si es chat grupal, mostrar opciones de targetMode */
    84→  isGroupChat?: boolean;
    85→  /** Modo compacto (para usar en objetivos) */
    86→  compact?: boolean;
    87→  /** Clase CSS adicional */
    88→  className?: string;
    89→  /** Mostrar campo de ID editable */
    90→  showIdField?: boolean;
    91→  /** Colecciones de triggers condicionales disponibles */
    92→  availableConditionalCollections?: Array<{ id: string; name: string }>;
    93→  /** Sprite packs V2 disponibles para activate_sprite_pack */
    94→  availableSpritePacks?: Array<{ id: string; name: string; conditionalMode?: boolean; spriteCount?: number }>;
    95→  /** Targets disponibles (otros personajes) para target mode en activate_sprite_pack */
    96→  availableTargets?: Array<{
    97→    id: string;
    98→    name: string;
    99→    spritePacks?: Array<{ id: string; name: string; conditionalMode?: boolean; spriteCount: number }>;
   100→  }>;
   101→}
   102→
   103→// ============================================
   104→// Constants
   105→// ============================================
   106→
   107→const ACTION_OPTIONS: { value: AttributeAction; label: string; symbol: string }[] = [
   108→  { value: 'set', label: 'Establecer', symbol: '=' },
   109→  { value: 'add', label: 'Sumar', symbol: '+' },
   110→  { value: 'subtract', label: 'Restar', symbol: '-' },
   111→  { value: 'multiply', label: 'Multiplicar', symbol: '×' },
   112→  { value: 'divide', label: 'Dividir', symbol: '÷' },
   113→  { value: 'percent', label: 'Porcentaje', symbol: '%' },
   114→];
   115→
   116→const TRIGGER_CATEGORIES: { value: TriggerCategory; label: string; icon: React.ReactNode }[] = [
   117→  { value: 'sprite', label: 'Sprite', icon: <ImageIcon className="w-3.5 h-3.5" /> },
   118→  { value: 'sound', label: 'Sonido', icon: <Volume2 className="w-3.5 h-3.5" /> },
   119→  { value: 'background', label: 'Fondo', icon: <Wallpaper className="w-3.5 h-3.5" /> },
   120→];
   121→
   122→const TARGET_MODES: { value: TriggerTargetMode; label: string; icon: React.ReactNode }[] = [
   123→  { value: 'self', label: 'Mismo personaje', icon: <User className="w-3.5 h-3.5" /> },
   124→  { value: 'all', label: 'Todos', icon: <Users className="w-3.5 h-3.5" /> },
   125→  { value: 'target', label: 'Objetivo', icon: <Crosshair className="w-3.5 h-3.5" /> },
   126→];
   127→
   128→// ============================================
   129→// Component
   130→// ============================================
   131→
   132→export function RewardEditor({
   133→  reward,
   134→  onChange,
   135→  onDelete,
   136→  availableAttributes,
   137→  availableTriggers,
   138→  availableConditionalCollections,
   139→  availableSpritePacks,
   140→  availableTargets,
   141→  isGroupChat = false,
   142→  compact = false,
   143→  className,
   144→  showIdField = false,
   145→}: RewardEditorProps) {
   146→  // Normalizar reward para obtener valores actuales
   147→  const normalized = normalizeReward(reward);
   148→  const isAttribute = normalized.type === 'attribute';
   149→  const isTrigger = normalized.type === 'trigger';
   150→  const isCurrency = normalized.type === 'currency';
   151→  const isConditionalSpriteCollection = normalized.type === 'conditional_sprite_collection';
   152→  const isActivateSpritePack = normalized.type === 'activate_sprite_pack';
   153→
   154→  // Handlers
   155→  const handleTypeChange = (newType: 'attribute' | 'trigger' | 'currency' | 'conditional_sprite_collection' | 'activate_sprite_pack') => {
   156→    if (newType === normalized.type) return;
   157→
   158→    let newReward: QuestReward;
   159→    if (newType === 'attribute') {
   160→      newReward = createAttributeReward(
   161→        normalized.attribute?.key || normalized.key || '',
   162→        normalized.attribute?.value ?? normalized.value ?? 0,
   163→        normalized.attribute?.action || 'add',
   164→        { id: reward.id }
   165→      );
   166→    } else if (newType === 'currency') {
   167→      newReward = createCurrencyReward(0, { id: reward.id });
   168→    } else if (newType === 'conditional_sprite_collection') {
   169→      newReward = createConditionalSpriteCollectionReward('', { id: reward.id });
   170→    } else if (newType === 'activate_sprite_pack') {
   171→      newReward = createActivateSpritePackReward('', { id: reward.id });
   172→    } else {
   173→      newReward = createTriggerReward(
   174→        normalized.trigger?.category || 'sprite',
   175→        normalized.trigger?.key || normalized.key || '',
   176→        normalized.trigger?.targetMode || 'self',
   177→        { id: reward.id }
   178→      );
   179→    }
   180→    onChange(newReward);
   181→  };
   182→
   183→  const handleAttributeChange = (updates: Partial<NonNullable<QuestReward['attribute']>>) => {
   184→    const currentAttr = normalized.attribute || { key: '', value: 0, action: 'add' as AttributeAction };
   185→    onChange({
   186→      ...reward,
   187→      type: 'attribute',
   188→      attribute: { ...currentAttr, ...updates },
   189→    });
   190→  };
   191→
   192→  const handleTriggerChange = (updates: Partial<NonNullable<QuestReward['trigger']>>) => {
   193→    const currentTrigger = normalized.trigger || { 
   194→      category: 'sprite' as TriggerCategory, 
   195→      key: '', 
   196→      targetMode: 'self' as TriggerTargetMode 
   197→    };
   198→    onChange({
   199→      ...reward,
   200→      type: 'trigger',
   201→      trigger: { ...currentTrigger, ...updates },
   202→    });
   203→  };
   204→
   205→  const handleCurrencyChange = (updates: Partial<NonNullable<QuestReward['currency']>>) => {
   206→    const currentCurrency = normalized.currency || { amount: 0 };
   207→    onChange({
   208→      ...reward,
   209→      type: 'currency',
   210→      currency: { ...currentCurrency, ...updates },
   211→    });
   212→  };
   213→
   214→  const handleConditionalSpriteCollectionChange = (updates: Partial<NonNullable<QuestReward['conditional_sprite_collection']>>) => {
   215→    const current = normalized.conditional_sprite_collection || {
   216→      collectionId: '',
   217→      targetMode: 'self' as TriggerTargetMode,
   218→      returnToIdleMs: 0,
   219→      fallbackMode: 'idle_collection' as TriggerFallbackMode,
   220→    };
   221→    onChange({
   222→      ...reward,
   223→      type: 'conditional_sprite_collection',
   224→      conditional_sprite_collection: { ...current, ...updates },
   225→    });
   226→  };
   227→
   228→  const handleActivateSpritePackChange = (updates: Partial<NonNullable<QuestReward['activate_sprite_pack']>>) => {
   229→    const current = normalized.activate_sprite_pack || {
   230→      packId: '',
   231→      targetMode: 'self' as TriggerTargetMode,
   232→      returnToIdleMs: 0,
   233→      fallbackMode: 'idle_collection' as TriggerFallbackMode,
   234→    };
   235→    onChange({
   236→      ...reward,
   237→      type: 'activate_sprite_pack',
   238→      activate_sprite_pack: { ...current, ...updates },
   239→    });
   240→  };
   241→
   242→  // Derived expiry mode for unified "Al expirar" dropdown
   243→  const getExpiryMode = (): string => {
   244→    const pack = normalized.activate_sprite_pack;
   245→    if (!pack) return 'idle_collection';
   246→    if (pack.fallbackPackId) return 'fallback_pack';
   247→    return pack.fallbackMode || 'idle_collection';
   248→  };
   249→
   250→  const handleExpiryModeChange = (value: string) => {
   251→    if (value === 'fallback_pack') {
   252→      handleActivateSpritePackChange({
   253→        fallbackMode: 'custom_sprite' as const,
   254→        fallbackPackId: normalized.activate_sprite_pack?.fallbackPackId || undefined,
   255→      });
   256→    } else {
   257→      handleActivateSpritePackChange({
   258→        fallbackMode: value as TriggerFallbackMode,
   259→        fallbackPackId: undefined,
   260→      });
   261→    }
   262→  };
   263→
   264→  // Compact mode
   265→  if (compact) {
   266→    return (
   267→      <div className={cn("p-2 rounded bg-muted/20 space-y-2", className)}>
   268→        {/* Tipo y preview */}
   269→        <div className="flex items-center gap-2">
   270→          <Select 
   271→            value={normalized.type} 
   272→            onValueChange={(v) => handleTypeChange(v as 'attribute' | 'trigger' | 'currency' | 'conditional_sprite_collection' | 'activate_sprite_pack')}
   273→          >
   274→            <SelectTrigger className="bg-background h-6 text-xs w-24">
   275→              <SelectValue />
   276→            </SelectTrigger>
   277→            <SelectContent>
   278→              <SelectItem value="attribute">📊 Atributo</SelectItem>
   279→              <SelectItem value="trigger">⚡ Trigger</SelectItem>
   280→              <SelectItem value="currency">💰 Divisa</SelectItem>
   281→              <SelectItem value="conditional_sprite_collection">🔀 Cond. Sprite</SelectItem>
   282→              <SelectItem value="activate_sprite_pack">🎨 Sprite Pack</SelectItem>
   283→            </SelectContent>
   284→          </Select>
   285→          <Badge variant="outline" className="text-[10px] flex-1">
   286→            {describeReward(normalized)}
   287→          </Badge>
   288→          <Button
   289→            variant="ghost"
   290→            size="sm"
   291→            className="h-6 w-6 p-0 text-red-500 hover:bg-red-500/10"
   292→            onClick={onDelete}
   293→          >
   294→            <X className="w-3 h-3" />
   295→          </Button>
   296→        </div>
   297→        
   298→        {/* Config attribute */}
   299→        {isAttribute && normalized.attribute && (
   300→          <div className="grid grid-cols-3 gap-2">
   301→            <Input
   302→              value={normalized.attribute.key}
   303→              onChange={(e) => handleAttributeChange({ key: e.target.value })}
   304→              placeholder="Key"
   305→              className="bg-background h-6 text-xs"
   306→              list={availableAttributes ? "available-attributes" : undefined}
   307→            />
   308→            {availableAttributes && (
   309→              <datalist id="available-attributes">
   310→                {availableAttributes.map(attr => (
   311→                  <option key={attr} value={attr} />
   312→                ))}
   313→              </datalist>
   314→            )}
   315→            <Input
   316→              type="number"
   317→              value={normalized.attribute.value}
   318→              onChange={(e) => handleAttributeChange({ value: Number(e.target.value) })}
   319→              placeholder="Valor"
   320→              className="bg-background h-6 text-xs"
   321→            />
   322→            <Select 
   323→              value={normalized.attribute.action} 
   324→              onValueChange={(v) => handleAttributeChange({ action: v as AttributeAction })}
   325→            >
   326→              <SelectTrigger className="bg-background h-6 text-xs">
   327→                <SelectValue />
   328→              </SelectTrigger>
   329→              <SelectContent>
   330→                {ACTION_OPTIONS.map(opt => (
   331→                  <SelectItem key={opt.value} value={opt.value}>
   332→                    {opt.symbol}
   333→                  </SelectItem>
   334→                ))}
   335→              </SelectContent>
   336→            </Select>
   337→          </div>
   338→        )}
   339→        
   340→        {/* Config trigger */}
   341→        {isTrigger && normalized.trigger && (
   342→          <div className="grid grid-cols-3 gap-2">
   343→            <Select 
   344→              value={normalized.trigger.category} 
   345→              onValueChange={(v) => handleTriggerChange({ category: v as TriggerCategory })}
   346→            >
   347→              <SelectTrigger className="bg-background h-6 text-xs">
   348→                <SelectValue />
   349→              </SelectTrigger>
   350→              <SelectContent>
   351→                {TRIGGER_CATEGORIES.map(cat => (
   352→                  <SelectItem key={cat.value} value={cat.value}>
   353→                    <span className="flex items-center gap-1">
   354→                      {cat.icon}
   355→                      {cat.label}
   356→                    </span>
   357→                  </SelectItem>
   358→                ))}
   359→              </SelectContent>
   360→            </Select>
   361→            <Input
   362→              value={normalized.trigger.key}
   363→              onChange={(e) => handleTriggerChange({ key: e.target.value })}
   364→              placeholder="Key"
   365→              className="bg-background h-6 text-xs"
   366→              list={getTriggerDatalistId(normalized.trigger.category, availableTriggers)}
   367→            />
   368→            {isGroupChat ? (
   369→              <Select 
   370→                value={normalized.trigger.targetMode} 
   371→                onValueChange={(v) => handleTriggerChange({ targetMode: v as TriggerTargetMode })}
   372→              >
   373→                <SelectTrigger className="bg-background h-6 text-xs">
   374→                  <SelectValue />
   375→                </SelectTrigger>
   376→                <SelectContent>
   377→                  {TARGET_MODES.map(mode => (
   378→                    <SelectItem key={mode.value} value={mode.value}>
   379→                      <span className="flex items-center gap-1">
   380→                        {mode.icon}
   381→                        {mode.label}
   382→                      </span>
   383→                    </SelectItem>
   384→                  ))}
   385→                </SelectContent>
   386→              </Select>
   387→            ) : (
   388→              <Input
   389→                type="number"
   390→                min={0}
   391→                value={normalized.trigger.returnToIdleMs || 0}
   392→                onChange={(e) => handleTriggerChange({ returnToIdleMs: Number(e.target.value) })}
   393→                placeholder="Idle ms"
   394→                className="bg-background h-6 text-xs"
   395→              />
   396→            )}
   397→          </div>
   398→        )}
   399→
   400→        {/* Config currency */}
   401→        {isCurrency && normalized.currency && (
   402→          <div className="flex items-center gap-2">
   403→            <Input
   404→              type="number"
   405→              value={normalized.currency.amount}
   406→              onChange={(e) => handleCurrencyChange({ amount: Number(e.target.value) })}
   407→              placeholder="Cantidad"
   408→              className="bg-background h-6 text-xs w-20"
   409→            />
   410→            <span className="text-[10px] text-muted-foreground">divisa para persona</span>
   411→          </div>
   412→        )}
   413→
   414→        {/* Config conditional_sprite_collection */}
   415→        {isConditionalSpriteCollection && normalized.conditional_sprite_collection && (
   416→          <div className="grid grid-cols-2 gap-2">
   417→            <Input
   418→              value={normalized.conditional_sprite_collection.collectionId}
   419→              onChange={(e) => handleConditionalSpriteCollectionChange({ collectionId: e.target.value })}
   420→              placeholder="ID Colección"
   421→              className="bg-background h-6 text-xs"
   422→            />
   423→            <Select
   424→              value={normalized.conditional_sprite_collection.targetMode || 'self'}
   425→              onValueChange={(v) => handleConditionalSpriteCollectionChange({ targetMode: v as TriggerTargetMode })}
   426→            >
   427→              <SelectTrigger className="bg-background h-6 text-xs">
   428→                <SelectValue />
   429→              </SelectTrigger>
   430→              <SelectContent>
   431→                {TARGET_MODES.map(mode => (
   432→                  <SelectItem key={mode.value} value={mode.value}>{mode.label}</SelectItem>
   433→                ))}
   434→              </SelectContent>
   435→            </Select>
   436→          </div>
   437→        )}
   438→
   439→        {/* Config activate_sprite_pack */}
   440→        {isActivateSpritePack && normalized.activate_sprite_pack && (
   441→          <div className="space-y-1.5">
   442→            {/* Target mode */}
   443→            <Select
   444→              value={normalized.activate_sprite_pack.targetMode || 'self'}
   445→              onValueChange={(v) => handleActivateSpritePackChange({ targetMode: v as TriggerTargetMode })}
   446→            >
   447→              <SelectTrigger className="bg-background h-6 text-xs">
   448→                <SelectValue />
   449→              </SelectTrigger>
   450→              <SelectContent>
   451→                {TARGET_MODES.map(mode => (
   452→                  <SelectItem key={mode.value} value={mode.value}>{mode.label}</SelectItem>
   453→                ))}
   454→              </SelectContent>
   455→            </Select>
   456→
   457→            {/* Pack selectors — varies by mode */}
   458→            {(normalized.activate_sprite_pack.targetMode || 'self') === 'self' ? (
   459→              <Select
   460→                value={normalized.activate_sprite_pack.packId}
   461→                onValueChange={(v) => handleActivateSpritePackChange({ packId: v })}
   462→              >
   463→                <SelectTrigger className="bg-background h-6 text-xs">
   464→                  <SelectValue placeholder="Sprite Pack..." />
   465→                </SelectTrigger>
   466→                <SelectContent>
   467→                  {availableSpritePacks?.map(p => (
   468→                    <SelectItem key={p.id} value={p.id}>
   469→                      <div className="flex items-center gap-1">
   470→                        <Package className="w-3 h-3" />
   471→                        {p.name}
   472→                        {p.conditionalMode && <span className="text-[9px] text-purple-500">cond.</span>}
   473→                      </div>
   474→                    </SelectItem>
   475→                  ))}
   476→                </SelectContent>
   477→              </Select>
   478→            ) : (normalized.activate_sprite_pack.targetMode) === 'target' ? (
   479→              <div className="grid grid-cols-2 gap-2">
   480→                <Select
   481→                  value={normalized.activate_sprite_pack.targetCharacterId || ''}
   482→                  onValueChange={(v) => handleActivateSpritePackChange({ targetCharacterId: v, targetPackId: undefined })}
   483→                >
   484→                  <SelectTrigger className="bg-background h-6 text-xs">
   485→                    <SelectValue placeholder="Personaje..." />
   486→                  </SelectTrigger>
   487→                  <SelectContent>
   488→                    {availableTargets?.map(t => (
   489→                      <SelectItem key={t.id} value={t.id}>
   490→                        {t.id === '__user__' ? '👤 ' : '🎭 '}{t.name}
   491→                      </SelectItem>
   492→                    ))}
   493→                  </SelectContent>
   494→                </Select>
   495→                <Select
   496→                  value={normalized.activate_sprite_pack.targetPackId || ''}
   497→                  onValueChange={(v) => handleActivateSpritePackChange({ targetPackId: v })}
   498→                >
   499→                  <SelectTrigger className="bg-background h-6 text-xs">
   500→                    <SelectValue placeholder="Pack objetivo..." />
   501→                  </SelectTrigger>
   502→                  <SelectContent>
   503→                    {(() => {
   504→                      const sel = availableTargets?.find(t => t.id === normalized.activate_sprite_pack!.targetCharacterId);
   505→                      return sel?.spritePacks?.map(p => (
   506→                        <SelectItem key={p.id} value={p.id}>
   507→                          {p.name}
   508→                        </SelectItem>
   509→                      )) || [];
   510→                    })()}
   511→                  </SelectContent>
   512→                </Select>
   513→              </div>
   514→            ) : (
   515→              <Select
   516→                value={normalized.activate_sprite_pack.packId}
   517→                onValueChange={(v) => handleActivateSpritePackChange({ packId: v })}
   518→              >
   519→                <SelectTrigger className="bg-background h-6 text-xs">
   520→                  <SelectValue placeholder="Sprite Pack..." />
   521→                </SelectTrigger>
   522→                <SelectContent>
   523→                  {availableSpritePacks?.map(p => (
   524→                    <SelectItem key={p.id} value={p.id}>
   525→                      {p.name}
   526→                    </SelectItem>
   527→                  ))}
   528→                </SelectContent>
   529→              </Select>
   530→            )}
   531→
   532→            {/* Duration + Al expirar */}
   533→            <div className="grid grid-cols-2 gap-2">
   534→              <Input
   535→                type="number"
   536→                min={0}
   537→                value={normalized.activate_sprite_pack.returnToIdleMs || 0}
   538→                onChange={(e) => handleActivateSpritePackChange({ returnToIdleMs: Number(e.target.value) })}
   539→                placeholder="Duración ms"
   540→                className="bg-background h-6 text-xs"
   541→              />
   542→              <Select
   543→                value={getExpiryMode()}
   544→                onValueChange={handleExpiryModeChange}
   545→              >
   546→                <SelectTrigger className="bg-background h-6 text-xs">
   547→                  <SelectValue />
   548→                </SelectTrigger>
   549→                <SelectContent>
   550→                  <SelectItem value="idle_collection">↩️ Idle</SelectItem>
   551→                  <SelectItem value="collection_default">⭐ Default</SelectItem>
   552→                  {(normalized.activate_sprite_pack.targetMode || 'self') !== 'target' && (
   553→                    <SelectItem value="fallback_pack">
   554→                      <div className="flex items-center gap-1">
   555→                        <Package className="w-3 h-3" /> Otro Pack
   556→                      </div>
   557→                    </SelectItem>
   558→                  )}
   559→                </SelectContent>
   560→              </Select>
   561→            </div>
   562→
   563→            {/* Fallback pack selector (only when Al expirar = fallback_pack, not target mode) */}
   564→            {getExpiryMode() === 'fallback_pack' && (normalized.activate_sprite_pack.targetMode || 'self') !== 'target' && (
   565→              <Select
   566→                value={normalized.activate_sprite_pack.fallbackPackId || ''}
   567→                onValueChange={(v) => handleActivateSpritePackChange({ fallbackPackId: v })}
   568→              >
   569→                <SelectTrigger className="bg-background h-6 text-xs">

  1094→                    </SelectContent>

  1184→      {/* ID field (collapsible) */}
  1185→      {showIdField && (

  1200→    </div>
  1201→  );
