    50→  try {
    51→    const normalizedPath = normalizePath(dirPath);
    52→    const parentDir = path.dirname(normalizedPath);
    53→
    54→    if (!fs.existsSync(parentDir)) {
    55→      return { exists: false, writable: false, error: 'Parent directory does not exist' };
    56→    }
    57→
    58→    if (fs.existsSync(normalizedPath)) {
    59→      try {
    60→        fs.accessSync(normalizedPath, fs.constants.W_OK);
    61→        return { exists: true, writable: true };
    62→      } catch {
    63→        return { exists: true, writable: false, error: 'No write permission' };
    64→      }
    65→    }
    66→
    67→    try {
    68→      fs.accessSync(parentDir, fs.constants.W_OK);
    69→      return { exists: false, writable: true };
    70→    } catch {
    71→      return { exists: false, writable: false, error: 'No write permission on parent directory' };
    72→    }
    73→  } catch (error: any) {
    74→    return { exists: false, writable: false, error: error.message };
    75→  }
    76→}

    78→export function ensureLanceDBDirectory(uri: string): { success: boolean; path: string; error?: string } {
    79→  try {
    80→    if (uri.startsWith('mem://') || uri.startsWith('lancedb://')) {
    81→      return { success: true, path: uri };
    82→    }
    83→    const normalizedPath = normalizePath(uri);
    84→    if (!fs.existsSync(normalizedPath)) {
    85→      fs.mkdirSync(normalizedPath, { recursive: true });
    86→    }
    87→    return { success: true, path: normalizedPath };
    88→  } catch (error: any) {
    89→    return { success: false, path: uri, error: error.message };
    90→  }
    91→}

   185→function parseMetadata(metadata: any): Record<string, any> {
   186→  if (!metadata) return {};
   187→  if (typeof metadata === 'string') {

   193→/**
   194→ * Escape a value for safe use in LanceDB filter expressions.
   195→ * Prevents injection by escaping single quotes which are used as string delimiters.
   196→ */
   197→function escapeFilterValue(value: string): string {
   198→  return value.replace(/\\/g, '\\\\').replace(/'/g, "\\'");
   199→}

   201→function normalizeVector(vector: number[]): number[] {
   202→  const magnitude = Math.sqrt(vector.reduce((sum, val) => sum + val * val, 0));
   203→  if (magnitude === 0) return vector;
   204→  return vector.map(val => val / magnitude);
   205→}
   206→
   207→/**
   208→ * Convert LanceDB's L2-squared distance to cosine similarity.
   209→ * LanceDB vectorSearch returns _distance = L2² (squared Euclidean distance).
   210→ * For unit-normalized vectors: L2² = 2 - 2·cos(θ), so cos(θ) = 1 - L2²/2.
   211→ */
   212→function l2SquaredToCosineSimilarity(l2SquaredDistance: number): number {
   213→  return 1 - l2SquaredDistance / 2;
   214→}

   224→// ============ Initialization ============
   225→
   226→// Track the dimension that the embeddings table was created with
   227→let tableDimension: number | null = null;

   229→async function initializeTables(): Promise<void> {
   230→  if (!db) throw new Error('Database not initialized');
   231→
   232→  // Get vector dimension from config
   233→  let vectorDimension = 1024;
   234→  let embeddingModel = 'bge-m3:567m';
   235→  try {
   236→    const { getConfig } = await import('./config-persistence');
   237→    const config = getConfig();
   238→    vectorDimension = config.dimension || 1024;
   239→    embeddingModel = config.model || 'bge-m3:567m';
   240→  } catch {
   241→    // Use defaults
   242→  }
   243→
   244→  const defaultEmbedding = {
   245→    id: 'placeholder',
   246→    content: 'placeholder',
   247→    vector: new Array(vectorDimension).fill(0),
   248→    metadata: '{}',
   249→    namespace: 'default',
   250→    source_type: 'system',
   251→    source_id: 'init',
   252→    model_name: embeddingModel,
   253→    created_at: new Date().toISOString(),
   254→    updated_at: new Date().toISOString(),
   255→  };
   256→
   257→  const defaultNamespace = {
   258→    id: 'placeholder',
   259→    namespace: 'default',
   260→    description: 'Default namespace',
   261→    metadata: '{}',
   262→    created_at: new Date().toISOString(),
   263→    updated_at: new Date().toISOString(),
   264→  };
   265→
   266→  // Try to open existing embeddings table
   267→  try {
   268→    embeddingsTable = await db.openTable(EMBEDDINGS_TABLE);
   269→
   270→    // Check if dimension matches by reading the first row's vector
   271→    // If the table was created with a different dimension, we must recreate it
   272→    const existingRows = await embeddingsTable.query().limit(1).toArray();
   273→    if (existingRows.length > 0 && existingRows[0].vector) {
   274→      const existingDim = existingRows[0].vector.length;
   275→      tableDimension = existingDim;
   276→
   277→      if (existingDim !== vectorDimension) {
   278→        console.warn(
   279→          `[LanceDB] Dimension mismatch detected! Table has ${existingDim}D vectors, but config requires ${vectorDimension}D. ` +
   280→          `Dropping and recreating embeddings table. All existing embeddings will be lost.`
   281→        );
   282→
   283→        // Drop the old table
   284→        try { await db.dropTable(EMBEDDINGS_TABLE); } catch { /* ignore */ }
   285→
   286→        // Also drop any namespace-specific tables that might have wrong dimensions
   287→        try {
   288→          const tables = await db.tableNames();
   289→          for (const tableName of tables) {
   290→            if (tableName !== EMBEDDINGS_TABLE && tableName !== NAMESPACES_TABLE) {
   291→              try { await db.dropTable(tableName); } catch { /* ignore */ }
   292→            }
   293→          }
   294→        } catch { /* ignore */ }
   295→
   296→        // Recreate with new dimension
   297→        await db.createTable(EMBEDDINGS_TABLE, [defaultEmbedding]);
   298→        embeddingsTable = await db.openTable(EMBEDDINGS_TABLE);
   299→        await embeddingsTable.delete("id = 'placeholder'");
   300→        tableDimension = vectorDimension;
   301→      }
   302→    } else if (existingRows.length === 0) {
   303→      // Table exists but is empty — recreate with new dimension to be safe
   304→      try { await db.dropTable(EMBEDDINGS_TABLE); } catch { /* ignore */ }
   305→      await db.createTable(EMBEDDINGS_TABLE, [defaultEmbedding]);
   306→      embeddingsTable = await db.openTable(EMBEDDINGS_TABLE);
   307→      await embeddingsTable.delete("id = 'placeholder'");
   308→      tableDimension = vectorDimension;
   309→    }
   310→  } catch {
   311→    // Table doesn't exist yet, create it
   312→    await db.createTable(EMBEDDINGS_TABLE, [defaultEmbedding]);
   313→    embeddingsTable = await db.openTable(EMBEDDINGS_TABLE);
   314→    await embeddingsTable.delete("id = 'placeholder'");
   315→    tableDimension = vectorDimension;
   316→  }
   317→
   318→  // Namespaces table (no vectors, no dimension concern)
   319→  try {
   320→    namespacesTable = await db.openTable(NAMESPACES_TABLE);
   321→  } catch {
   322→    await db.createTable(NAMESPACES_TABLE, [defaultNamespace]);
   323→    namespacesTable = await db.openTable(NAMESPACES_TABLE);
   324→    await namespacesTable.delete("id = 'placeholder'");
   325→  }
   326→}

   332→export async function initLanceDB(uri?: string, forceReinit: boolean = false): Promise<void> {
   333→  const dbUri = uri || process.env.LANCEDB_URI || getDefaultLanceDBPath();
   334→
   335→  if (isInitialized && db && currentUri === dbUri && !forceReinit) return;
   336→
   337→  // If permanently unavailable, skip silently (don't throw)
   338→  if (isPermanentlyUnavailable) {
   339→    return;
   340→  }
   341→
   342→  if (db) await closeLanceDB();
   343→
   344→  const { success, error } = await loadLanceDBModule();
   345→  if (!success) {
   346→    isPermanentlyUnavailable = true;
   347→    lancedbLoadError = error || 'Unknown error';
   348→    console.warn(
   349→      `[LanceDB] ⚠️ LanceDB native module not available for platform "${getPlatform()}". ` +
   350→      `Embeddings/vector search features will be disabled. Error: ${error}`
   351→    );
   352→    return;
   353→  }
   354→
   355→  isInitialized = true;
   356→  currentUri = dbUri;
   357→
   358→  try {
   359→    const dirResult = ensureLanceDBDirectory(dbUri);
   360→    if (!dirResult.success) {
   361→      throw new LanceDBError(
   362→        `Cannot create/verify directory: ${dirResult.error}`,
   363→        'DIRECTORY_ERROR',
   364→        getPlatform(),
   365→        { uri: dbUri }
   366→      );
   367→    }
   368→
   369→    db = await lancedbModule.connect(dirResult.path);
   370→    await initializeTables();
   371→  } catch (error: any) {
   372→    if (error instanceof LanceDBError) throw error;
   373→
   374→    isInitialized = false;
   375→    db = null;
   376→    currentUri = null;
   377→
   378→    const msg = error.message || String(error);
   379→    const lowerMsg = msg.toLowerCase();
   380→
   381→    let code = 'UNKNOWN_ERROR';
   382→    if (lowerMsg.includes('permission') || lowerMsg.includes('access denied')) code = 'PERMISSION_DENIED';
   383→    else if (lowerMsg.includes('native') || lowerMsg.includes('binding')) code = 'NATIVE_MODULE_ERROR';
   384→
   385→    throw new LanceDBError(msg, code, getPlatform(), { uri: dbUri, originalError: error });
   386→  }
   387→}

   509→  static async searchSimilar(params: {
   510→    queryVector: number[];
   511→    namespace?: string;
   512→    limit?: number;
   513→    threshold?: number;
   514→  }): Promise<SearchResult[]> {
   515→    if (isPermanentlyUnavailable) return [];
   516→    const {
   517→      queryVector,
   518→      namespace,
   519→      limit = 10,
   520→      threshold = 0.3,
   521→    } = params;
   522→
   523→    const table = await getEmbeddingsTable();
   524→    if (!table) return [];
   525→    const normalizedQueryVector = normalizeVector(queryVector);
   526→
   527→    let results: any[];
   528→
   529→    // All embeddings are stored in the main table with a 'namespace' column.
   530→    // Always search the main table and filter by namespace in-memory.
   531→    const allResults = await table
   532→      .vectorSearch(normalizedQueryVector)
   533→      .limit(limit * 10)
   534→      .toArray();
   535→
   536→    results = allResults
   537→      .map((row: any) => ({
   538→        ...row,
   539→        similarity: l2SquaredToCosineSimilarity(row._distance || 0),
   540→      }))
   541→      .filter((r: any) => {
   542→        if (!namespace || namespace === 'all') return r.similarity >= threshold;
   543→        return (r.namespace || 'default') === namespace && r.similarity >= threshold;
   544→      })
   545→      .slice(0, limit);
   546→
   547→    return results.map((row: any) => ({
   548→      id: row.id,
   549→      content: row.content,
   550→      metadata: parseMetadata(row.metadata),
   551→      namespace: row.namespace || 'default',
   552→      source_type: row.source_type,
   553→      source_id: row.source_id,
   554→      similarity: row.similarity,
   555→    }));
   556→  }

   600→  static async upsertNamespace(params: {
   601→    namespace: string;
   602→    description?: string;
   603→    metadata?: Record<string, any>;
   604→  }): Promise<RecordNamespace> {
   605→    if (isPermanentlyUnavailable) {
   606→      // Return a stub namespace so callers don't crash
   607→      return {
   608→        id: '__unavailable__',
   609→        namespace: params.namespace,
   610→        description: params.description,
   611→        metadata: params.metadata || {},
   612→        created_at: new Date(),
   613→        updated_at: new Date(),
   614→      };
   615→    }
   616→    const { v4: uuidv4 } = await import('uuid');
   617→    const { namespace, description, metadata = {} } = params;
   618→    const table = await getNamespacesTable();
   619→    if (!table) {
   620→      return {
   621→        id: '__unavailable__',
   622→        namespace: params.namespace,
   623→        description: params.description,
   624→        metadata: params.metadata || {},
   625→        created_at: new Date(),
   626→        updated_at: new Date(),
   627→      };
   628→    }
   629→    const existing = await tableFilter(table, `namespace = '${escapeFilterValue(namespace)}'`);
   630→
   631→    const nsRecord = {
   632→      id: existing.length > 0 ? existing[0].id : uuidv4(),
   633→      namespace,
   634→      description,
   635→      metadata: JSON.stringify(metadata),
   636→      created_at: existing.length > 0 ? existing[0].created_at : new Date().toISOString(),
   637→      updated_at: new Date().toISOString(),
   638→    };
   639→
   640→    // Always delete all matching rows before inserting to prevent duplicates
   641→    // caused by race conditions (concurrent upsertNamespace calls).
   642→    if (existing.length > 0) {
   643→      await table.delete(`namespace = '${escapeFilterValue(namespace)}'`);
   644→    }
   645→    await table.add([nsRecord]);
   646→
   647→    // Safety net: if a race created extra duplicates, clean them up.
   648→    // Keep only the row we just inserted (by id).
   649→    const afterInsert = await tableFilter(table, `namespace = '${escapeFilterValue(namespace)}'`);
   650→    if (afterInsert.length > 1) {
   651→      const idsToDelete = afterInsert
   652→        .filter((row: any) => row.id !== nsRecord.id)
   653→        .map((row: any) => row.id);
   654→      if (idsToDelete.length > 0) {
   655→        const deleteFilter = idsToDelete
   656→          .map((id: string) => `id = '${escapeFilterValue(id)}'`)
   657→          .join(' OR ');
   658→        await table.delete(deleteFilter);
   659→        console.log(`[LanceDB] Cleaned up ${idsToDelete.length} duplicate namespace row(s) for "${namespace}"`);
   660→      }
   661→    }
   662→
   663→    return {
   664→      id: nsRecord.id,

   673→  static async getAllNamespaces(): Promise<RecordNamespace[]> {
   674→    if (isPermanentlyUnavailable) return [];
   675→    const table = await getNamespacesTable();
   676→    if (!table) return [];
   677→    const results = await tableToArray(table);
   678→    const mapped = results.map((row: any) => ({
   679→      id: row.id,
   680→      namespace: row.namespace,
   681→      description: row.description,
   682→      metadata: typeof row.metadata === 'string' ? JSON.parse(row.metadata) : row.metadata,
   683→      created_at: new Date(row.created_at),
   684→      updated_at: new Date(row.updated_at),
   685→    }));
   686→
   687→    // Deduplicate by namespace name — keep the most recently updated entry.
   688→    // This guards against duplicate rows that can appear due to a race condition
   689→    // in upsertNamespace (non-atomic delete-then-insert).
   690→    const deduped = new Map<string, RecordNamespace>();
   691→    for (const ns of mapped) {
   692→      const existing = deduped.get(ns.namespace);
   693→      if (!existing || ns.updated_at > existing.updated_at) {
   694→        deduped.set(ns.namespace, ns);
   695→      }
   696→    }
   697→    return Array.from(deduped.values());
   698→  }

   754→   * Lightweight version of getNamespaceEmbeddings that does NOT load vector data.

   758→  static async getNamespaceEmbeddingsMetadata(
   759→    namespace: string,
   760→    options?: {
   761→      limit?: number;
   762→      sourceType?: string;
   763→    }
   764→  ): Promise<Array<Omit<Embedding, 'vector'> & { vector?: never }>> {
   765→    if (isPermanentlyUnavailable) return [];
   766→    const table = await getEmbeddingsTable();
   767→    if (!table) return [];
   768→    const limit = options?.limit ?? 1000;
   769→    const sourceType = options?.sourceType;
   770→
   771→    // Build the filter expression
   772→    let filter = `namespace = '${escapeFilterValue(namespace)}'`;
   773→    if (sourceType) {
   774→      filter += ` AND source_type = '${escapeFilterValue(sourceType)}'`;
   775→    }
   776→
   777→    // Use select() to exclude the vector column for memory efficiency
   778→    // Each vector can be 4-16KB, so skipping it saves 40-160MB for 10K embeddings
   779→    try {
   780→      const results = await table
   781→        .query()
   782→        .where(filter)
   783→        .select(['id', 'content', 'metadata', 'namespace', 'source_type', 'source_id', 'model_name', 'created_at', 'updated_at'])
   784→        .limit(limit)
   785→        .toArray();
   786→
   787→      return results.map((row: any) => ({
   788→        id: row.id,
   789→        content: row.content,
   790→        metadata: parseMetadata(row.metadata),
   791→        namespace: row.namespace || 'default',
   792→        source_type: row.source_type,
   793→        source_id: row.source_id,
   794→        model_name: row.model_name,
   795→        created_at: new Date(row.created_at),
   796→        updated_at: new Date(row.updated_at),
   797→      }));
   798→    } catch {
   799→      // Fallback: if select() fails (older LanceDB), use the filter approach
   800→      // and manually strip the vector field
   801→      const results = await tableFilter(table, filter);
   802→      return results.slice(0, limit).map((row: any) => ({
   803→        id: row.id,
   804→        content: row.content,
   805→        metadata: parseMetadata(row.metadata),
   806→        namespace: row.namespace || 'default',
   807→        source_type: row.source_type,
   808→        source_id: row.source_id,
   809→        model_name: row.model_name,
   810→        created_at: new Date(row.created_at),
   811→        updated_at: new Date(row.updated_at),
   812→      }));
   813→    }
   814→  }

   837→  static async searchInNamespace(params: {
   838→    namespace: string;
   839→    queryVector: number[];
   840→    limit?: number;
   841→    threshold?: number;
   842→  }): Promise<SearchResult[]> {
   843→    if (isPermanentlyUnavailable) return [];
   844→    const { namespace, queryVector, limit = 10, threshold = 0.3 } = params;
   845→    const table = await getEmbeddingsTable();
   846→    if (!table) return [];
   847→    const normalizedQueryVector = normalizeVector(queryVector);
   848→
   849→    const results = await table
   850→      .vectorSearch(normalizedQueryVector)
   851→      .limit(limit * 5)
   852→      .toArray();
   853→
   854→    return results
   855→      .filter((row: any) => (row.namespace || 'default') === namespace)
   856→      .map((row: any) => ({
   857→        id: row.id,
   858→        content: row.content,
   859→        metadata: parseMetadata(row.metadata),
   860→        namespace: row.namespace || 'default',
   861→        source_type: row.source_type,
   862→        source_id: row.source_id,
   863→        similarity: l2SquaredToCosineSimilarity(row._distance || 0),
   864→      }))
   865→      .filter((r: any) => r.similarity >= threshold)
   866→      .slice(0, limit);
   867→  }

   887→  static async getStats(): Promise<EmbeddingStats> {
   888→    if (isPermanentlyUnavailable) {
   889→      return {
   890→        totalEmbeddings: 0,
   891→        totalNamespaces: 0,
   892→        embeddingsByNamespace: {},
   893→        embeddingsBySourceType: {},
   894→      };
   895→    }
   896→    const table = await getEmbeddingsTable();
   897→    if (!table) {
   898→      return {
   899→        totalEmbeddings: 0,
   900→        totalNamespaces: 0,
   901→        embeddingsByNamespace: {},
   902→        embeddingsBySourceType: {},
   903→      };
   904→    }
   905→    const namespaces = await this.getAllNamespaces();
   906→
   907→    const embeddingsByNamespace: Record<string, number> = {};
   908→    const embeddingsBySourceType: Record<string, number> = {};
   909→    let totalEmbeddings = 0;
   910→
   911→    // Count per-namespace to avoid loading all vectors into memory at once
   912→    for (const ns of namespaces) {
   913→      try {
   914→        const count = await this.countByNamespace(ns.namespace);
   915→        if (count > 0) {
   916→          embeddingsByNamespace[ns.namespace] = count;
   917→          totalEmbeddings += count;
   918→        }
   919→      } catch { /* skip */ }
   920→    }
   921→
   922→    // Also count 'default' namespace if not listed
   923→    if (!embeddingsByNamespace['default']) {
   924→      try {
   925→        const defaultCount = await this.countByNamespace('default');
   926→        if (defaultCount > 0) {
   927→          embeddingsByNamespace['default'] = defaultCount;
   928→          totalEmbeddings += defaultCount;
   929→        }
   930→      } catch { /* skip */ }
   931→    }
   932→
   933→    // For source_type breakdown, we need to scan but we can use lightweight queries
   934→    // Only load id + source_type (not full vectors) via filter
   935→    const sourceTypes = ['character', 'world', 'lorebook', 'session', 'memory', 'custom', 'system'];
   936→    for (const st of sourceTypes) {
   937→      try {
   938→        const matching = await tableFilter(table, `source_type = '${st}'`);
   939→        if (matching.length > 0) {
   940→          embeddingsBySourceType[st] = matching.length;
   941→        }
   942→      } catch { /* skip */ }
   943→    }
   944→
   945→    return {
   946→      totalEmbeddings,
   947→      totalNamespaces: namespaces.length,
   948→      embeddingsByNamespace,
   949→      embeddingsBySourceType,
   950→    };
   951→  }

   953→  static async resetAll(): Promise<{ deletedEmbeddings: number; deletedNamespaces: number }> {
   954→    if (isPermanentlyUnavailable) return { deletedEmbeddings: 0, deletedNamespaces: 0 };
   955→    // Count before dropping to report accurate numbers
   956→    let deletedEmbeddings = 0;
   957→    let deletedNamespaces = 0;
   958→
   959→    try {
   960→      const allEmbeddings = await tableToArray(await getEmbeddingsTable());
   961→      deletedEmbeddings = allEmbeddings.length;
   962→    } catch { /* table may not exist */ }
   963→
   964→    try {
   965→      const allNamespaces = await tableToArray(await getNamespacesTable());
   966→      deletedNamespaces = allNamespaces.length;
   967→    } catch { /* table may not exist */ }
   968→
   969→    // Drop and recreate the embeddings table (much faster than individual deletes)
   970→    if (db) {
   971→      try { await db.dropTable(EMBEDDINGS_TABLE); } catch { /* ignore */ }
   972→      try { await db.dropTable(NAMESPACES_TABLE); } catch { /* ignore */ }
   973→
   974→      // Drop any extra tables
   975→      try {
   976→        const tables = await db.tableNames();
   977→        for (const tableName of tables) {
   978→          if (tableName !== EMBEDDINGS_TABLE && tableName !== NAMESPACES_TABLE) {
   979→            try { await db.dropTable(tableName); } catch { /* ignore */ }
   980→          }
   981→        }
   982→      } catch { /* ignore */ }
   983→
   984→      // Reinitialize with fresh tables
   985→      await initializeTables();
   986→    }
   987→
   988→    return { deletedEmbeddings, deletedNamespaces };
   989→  }
