    65→/**
    66→ * Recompute the `relacion`/`relacion_etapa` mirror for one character from
    67→ * the FULL bonds record. A character may be in several bonds (group chats):
    68→ * the mirror prefers the bond with the user; otherwise the highest bond.
    69→ * If no bonds remain, the mirror keys are removed.
    70→ */
    71→function recomputeRelationshipMirror(
    72→  characterStats: SessionStats['characterStats'],
    73→  id: string,
    74→  relationships: Record<string, import('@/types').SessionRelationship>,
    75→  now: number
    76→): void {
    77→  const existing = characterStats[id] || { attributeValues: {}, lastUpdated: {} };
    78→  const attributeValues = { ...existing.attributeValues };
    79→  const lastUpdated = { ...existing.lastUpdated };
    80→  const bonds = Object.values(relationships).filter(b => b.aId === id || b.bId === id);
    81→  if (bonds.length === 0) {
    82→    delete attributeValues[RELATIONSHIP_MIRROR_KEY];
    83→    delete attributeValues[RELATIONSHIP_STAGE_KEY];
    84→    delete lastUpdated[RELATIONSHIP_MIRROR_KEY];
    85→    delete lastUpdated[RELATIONSHIP_STAGE_KEY];
    86→  } else {
    87→    const withUser = bonds.find(b => b.aId === '__user__' || b.bId === '__user__');
    88→    const chosen = withUser || bonds.reduce((m, b) => (b.points > m.points ? b : m), bonds[0]);
    89→    const stage = computeRelationshipStage(chosen.points);
    90→    attributeValues[RELATIONSHIP_MIRROR_KEY] = chosen.points;
    91→    attributeValues[RELATIONSHIP_STAGE_KEY] = stage.label;
    92→    lastUpdated[RELATIONSHIP_MIRROR_KEY] = now;
    93→    lastUpdated[RELATIONSHIP_STAGE_KEY] = now;
    94→  }
    95→  characterStats[id] = { ...existing, attributeValues, lastUpdated };
    96→}

    98→// ============================================
    99→// Threshold Reward Execution (runs inside updateCharacterStat)
   100→// ============================================
   101→//
   102→// Rewards from matched ThresholdEffects are executed HERE (store side) so that
   103→// EVERY path that changes an attribute executes them: LLM tool activations,
   104→// post-LLM key detection, skills, timers, manual edits and quest rewards.
   105→// Previously only the skill path executed them — rewards silently did nothing
   106→// for the other paths.
   107→//
   108→// 'message' rewards are skipped here: they are DEFERRED to the end of the turn
   109→// (collected by the chat panel and sent as quick replies with edge-triggering).
   110→
   111→let thresholdRewardDepth = 0;
   112→const THRESHOLD_REWARD_MAX_DEPTH = 3;
   113→
   114→function executeThresholdRewardsInternal(
   115→  getFn: () => any,
   116→  sessionId: string,
   117→  characterId: string,
   118→  thresholdsReached: ThresholdReachedInfo[]
   119→): void {
   120→  if (!thresholdsReached || thresholdsReached.length === 0) return;
   121→
   122→  if (thresholdRewardDepth >= THRESHOLD_REWARD_MAX_DEPTH) {
   123→    console.warn(`[StatsSlice] Threshold reward chain too deep (${thresholdRewardDepth}), skipping to prevent loops`);
   124→    return;
   125→  }
   126→
   127→  thresholdRewardDepth++;
   128→  try {
   129→    const state = getFn();
   130→    const session = state.sessions?.find((s: any) => s.id === sessionId);
   131→    const sessionStats = session?.sessionStats;
   132→    if (!sessionStats) return;
   133→
   134→    const activePersona = state.personas?.find((p: any) => p.id === state.activePersonaId);
   135→
   136→    // Resolve the character (or persona pseudo-character for __user__)
   137→    let character: any = state.characters?.find((c: any) => c.id === characterId);
   138→    if (!character && characterId === '__user__' && activePersona) {
   139→      character = { id: '__user__', name: activePersona.name || 'User', statsConfig: activePersona.statsConfig };
   140→    }
   141→
   142→    const allCharacters = [
   143→      ...(state.characters || []),
   144→      ...(activePersona?.statsConfig?.enabled
   145→        ? [{ id: '__user__', name: activePersona.name || 'User', statsConfig: activePersona.statsConfig }]
   146→        : []),
   147→    ];
   148→
   149→    const settings = state.settings || {};
   150→    const storeActions = buildRewardStoreActions(state);
   151→
   152→    for (const threshold of thresholdsReached) {
   153→      for (const reward of threshold.rewards || []) {
   154→        // Message rewards are deferred to end-of-turn (chat panel)
   155→        if (reward.type === 'message') continue;
   156→        try {
   157→          const result = executeReward(reward, {
   158→            sessionId,
   159→            characterId,
   160→            character,
   161→            allCharacters,
   162→            sessionStats,
   163→            timestamp: Date.now(),
   164→            soundCollections: state.soundCollections,
   165→            soundTriggers: state.soundTriggers,
   166→            backgroundPacks: state.backgroundTriggerPacks,
   167→            soundSettings: {
   168→              enabled: settings.sound?.enabled ?? false,
   169→              globalVolume: settings.sound?.globalVolume ?? 0.85,
   170→            },
   171→            backgroundSettings: {
   172→              transitionDuration: settings.backgroundTriggers?.transitionDuration ?? 500,
   173→              defaultTransitionType: settings.backgroundTriggers?.defaultTransitionType ?? 'fade',
   174→            },
   175→          }, storeActions);
   176→          if (!result.success) {
   177→            console.warn(`[StatsSlice] Threshold reward failed (${reward.type}):`, result.error || result.message);
   178→          }
   179→        } catch (err) {
   180→          console.error('[StatsSlice] Failed to execute threshold reward:', err);
   181→        }
   182→      }
   183→    }
   184→  } finally {
   185→    thresholdRewardDepth--;
   186→  }
   187→}

   392→/**
   393→ * Clamp a numeric value to the attribute's min/max bounds
   394→ * Returns the clamped value, or the original value if not numeric or no bounds defined
   395→ */
   396→function clampAttributeValue(
   397→  value: number | string,
   398→  attributeDef: AttributeDefinition | undefined
   399→): number | string {
   400→  // Only clamp numeric values
   401→  if (typeof value !== 'number') {
   402→    return value;
   403→  }
   404→
   405→  // If no attribute definition, return as-is
   406→  if (!attributeDef) {
   407→    return value;
   408→  }
   409→
   410→  let clampedValue = value;
   411→
   412→  // Apply min bound - handle both number and string min values
   413→  if (attributeDef.min !== undefined) {
   414→    const minVal = typeof attributeDef.min === 'number'
   415→      ? attributeDef.min
   416→      : parseFloat(String(attributeDef.min));
   417→
   418→    if (!isNaN(minVal)) {
   419→      clampedValue = Math.max(clampedValue, minVal);
   420→    }
   421→  }
   422→
   423→  // Apply max bound - handle both number and string max values
   424→  if (attributeDef.max !== undefined) {
   425→    const maxVal = typeof attributeDef.max === 'number'
   426→      ? attributeDef.max
   427→      : parseFloat(String(attributeDef.max));
   428→
   429→    if (!isNaN(maxVal)) {
   430→      clampedValue = Math.min(clampedValue, maxVal);
   431→    }
   432→  }
   433→
   434→  return clampedValue;
   435→}

   437→/**
   438→ * Create default character stats from config
   439→ */
   440→function createDefaultCharacterStats(
   441→  statsConfig?: CharacterStatsConfig
   442→): CharacterSessionStats {
   443→  const attributeValues: Record<string, number | string> = {};
   444→  const lastUpdated: Record<string, number> = {};
   445→  const now = Date.now();
   446→  
   447→  if (statsConfig?.attributes) {
   448→    for (const attr of statsConfig.attributes) {
   449→      attributeValues[attr.key] = attr.defaultValue;
   450→      lastUpdated[attr.key] = now;
   451→    }
   452→  }
   453→  
   454→  return {
   455→    attributeValues,
   456→    lastUpdated,
   457→    changeLog: [],
   458→  };
   459→}

   461→/**
   462→ * Add entry to change log
   463→ */
   464→function addChangeLogEntry(
   465→  stats: CharacterSessionStats,
   466→  attribute: AttributeDefinition | undefined,
   467→  attributeKey: string,
   468→  oldValue: number | string | undefined,
   469→  newValue: number | string,
   470→  reason: StatChangeLogEntry['reason']
   471→): void {
   472→  if (!stats.changeLog) {
   473→    stats.changeLog = [];
   474→  }
   475→  
   476→  stats.changeLog.push({
   477→    attributeId: attribute?.id || attributeKey,
   478→    attributeKey,
   479→    attributeName: attribute?.name || attributeKey,
   480→    oldValue: oldValue ?? '',
   481→    newValue,
   482→    reason,
   483→    timestamp: Date.now(),
   484→  });
   485→  
   486→  // Keep only last 100 entries
   487→  if (stats.changeLog.length > 100) {
   488→    stats.changeLog = stats.changeLog.slice(-100);
   489→  }
   490→}

   495→
   496→export const createStatsSlice = (set: any, get: any): StatsSlice => ({
   497→  // Initial State
   498→  sessionStats: null,
   499→
   500→  // ============================================
   501→  // Session Stats Actions
   502→  // ============================================
   503→
   504→  initializeSessionStats: (sessionId, characters) => {
   505→    const state = get();
   506→    const sessions = state.sessions as Array<{ id: string; sessionStats?: SessionStats }>;
   507→    const session = sessions.find(s => s.id === sessionId);
   508→    
   509→    if (!session) return;
   510→    
   511→    // Check if already initialized
   512→    if (session.sessionStats?.initialized) return;
   513→    
   514→    const now = Date.now();
   515→    const characterStats: Record<string, CharacterSessionStats> = {};
   516→    
   517→    // Initialize stats for each character
   518→    for (const char of characters) {
   519→      const stats = createDefaultCharacterStats(char.statsConfig);
   520→      // FASE 5: Initialize emotional state if emotional config is present
   521→      if (char.emotionalConfig?.enabled && char.emotionalConfig.initialState) {
   522→        stats.emotionalState = char.emotionalConfig.initialState;
   523→        stats.emotionalStateLastEval = now;
   524→        stats.emotionalStateTurnCount = 0;
   525→        // Also set as attribute value for {{emocion}} key resolution
   526→        stats.attributeValues['emocion'] = char.emotionalConfig.initialState;
   527→      }
   528→      characterStats[char.id] = stats;
   529→    }
   530→    
   531→    const newSessionStats: SessionStats = {
   532→      characterStats,
   533→      solicitudes: {
   534→        characterSolicitudes: {},
   535→        lastModified: now,
   536→      },
   537→      initialized: true,
   538→      lastModified: now,
   539→      // Initialize timer state
   540→      lastTimerUpdate: now,
   541→      keywordCycleIndex: {},
   542→    };
   543→    
   544→    // Update session with new stats
   545→    set((state: any) => ({
   546→      sessions: state.sessions.map((s: any) =>
   547→        s.id === sessionId 
   548→          ? { ...s, sessionStats: newSessionStats, updatedAt: new Date().toISOString() }
   549→          : s
   550→      ),
   551→    }));
   552→  },
   553→
   554→  updateCharacterStat: (sessionId, characterId, attributeKey, value, reason = 'manual') => {
   555→    // Default result
   556→    const defaultResult: UpdateCharacterStatResult = {
   557→      oldValue: undefined,
   558→      newValue: typeof value === 'number' ? value : value,
   559→      clamped: false,
   560→      thresholdsReached: [],
   561→    };
   562→
   563→    // Read current state
   564→    const state = get();
   565→    const sessions = state.sessions as Array<{ 
   566→      id: string; 
   567→      sessionStats?: SessionStats;
   568→      characterId?: string;
   569→      groupId?: string;
   570→    }>;
   571→    const sessionIndex = sessions.findIndex(s => s.id === sessionId);
   572→    
   573→    if (sessionIndex === -1) return defaultResult;
   574→    
   575→    const session = sessions[sessionIndex];
   576→    let sessionStats = session.sessionStats;
   577→    
   578→    // Auto-initialize sessionStats if missing
   579→    if (!sessionStats) {
   580→      sessionStats = {
   581→        characterStats: {},
   582→        solicitudes: {
   583→          characterSolicitudes: {},
   584→          lastModified: Date.now(),
   585→        },
   586→        ultimo_objetivo_completado: undefined,
   587→        ultima_solicitud_completada: undefined,
   588→        ultima_solicitud_realizada: undefined,
   589→        ultima_accion_realizada: undefined,
   590→        initialized: true,
   591→        lastModified: Date.now(),
   592→      };
   593→    }
   594→    
   595→    // Auto-initialize character stats if missing
   596→    if (!sessionStats.characterStats[characterId]) {
   597→      let statsConfig: CharacterStatsConfig | undefined;
   598→      if (characterId === '__user__') {
   599→        // Look up active persona's statsConfig for __user__
   600→        const activePersonaId = (state as any).activePersonaId;
   601→        const personas: any[] = (state as any).personas || [];

  1630→    
  1631→    const now = Date.now();

  1662→        }

  1664→        return solicitud;
  1665→      });

  1667→      updatedCharacterSolicitudes[charId] = updatedSolicitudes;

  1670→    if (!hasChanges) return [];

  1674→      solicitudes: {
  1675→        characterSolicitudes: updatedCharacterSolicitudes,

  1677→      },
  1678→      lastModified: now,

  1685→              ...s, 
  1686→              sessionStats: newSessionStats,
  1687→              updatedAt: new Date().toISOString()

  1734→      
  1735→      // Update the specific event field

  1737→        ...sessionStats,

  1739→        lastModified: Date.now(),

  1741→
  1742→      // Push to the event log ring buffer ({{eventos}} history)
  1743→      try {

  1751→            type: 'solicitud_created', description, turn,
  1752→          });

  1754→          newSessionStats = appendEventLogEntry(newSessionStats, {
  1755→            type: 'solicitud_completed', description, turn,

  1758→          // Arrives right after 'ultima_accion_realizada' — build the full entry here
  1759→          const actionDesc = sessionStats.ultima_accion_realizada || newSessionStats.ultima_accion_realizada;
  1760→          if (actionDesc) {
  1761→            newSessionStats = appendEventLogEntry(newSessionStats, {
  1762→              type: 'action',
  1763→              description: actionDesc,
  1764→              characterName: description, // description param carries the author name here
  1765→              turn,
  1766→            });
  1767→          }
  1768→        }
  1769→      } catch (e) {
