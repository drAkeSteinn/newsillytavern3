   222→// ============================================
   223→// CONDITIONAL STATE VARIANT
   224→// ============================================
   225→
   226→/**
   227→ * @deprecated Conditions are now defined at the SpritePack level instead of State Collections.
   228→ * This interface is kept for backward compatibility but should not be used in new code.
   229→ * Use SpritePackV2.conditionalMode + SpritePackEntryV2.conditions instead.
   230→ *
   231→ * Conditional State Variant - Attribute-driven sprite variant for a state collection.
   232→ *
   233→ * When a character has attributes (e.g., "exhaustion", "vida"), these variants
   234→ * allow the idle/talk/thinking sprite to change dynamically based on attribute values.
   235→ *
   236→ * Priority system (like lorebook dynamic conditions):
   237→ * - Higher priority wins when multiple variants match
   238→ * - If no variant matches, the default state collection pack is used
   239→ *
   240→ * Example:
   241→ *   Variant "Exhausted" → Priority 2, Condition: exhaustion >= 80 → pack_exhausted
   242→ *   Variant "Tired"     → Priority 1, Condition: exhaustion >= 50 → pack_tired
   243→ *   Variant "Normal"    → (no conditions, default pack)
   244→ *
   245→ * When exhaustion = 90: Both match, but "Exhausted" (P2) wins
   246→ * When exhaustion = 60: "Tired" (P1) wins
   247→ * When exhaustion = 30: Default pack is used
   248→ */
   249→export interface ConditionalStateVariant {
   250→  id: string;
   251→  name: string;                        // Display name: "Exhausted Idle", "Injured Idle"
   252→  enabled: boolean;                    // Enable/disable this variant
   253→
   254→  // Priority - Higher number = higher priority (wins when multiple match)
   255→  priority: number;                    // Default: 0
   256→
   257→  // Conditions - evaluated with AND or OR logic (controlled by conditionOperator)
   258→  // Reuses StatRequirement which supports:
   259→  //   - Operators: <, <=, >, >=, ==, !=, between, contains, not_contains
   260→  //   - Cross-character: targetCharacterId for other characters' or persona's attributes
   261→  conditions: StatRequirement[];
   262→  conditionOperator?: 'AND' | 'OR'; // Logic operator for conditions (default: 'AND')
   263→
   264→  // Which pack to use when this variant is active
   265→  packId: string;                      // References SpritePackV2.id
   266→
   267→  // Behavior when selecting sprite from the pack
   268→  behavior: 'principal' | 'random' | 'list';
   269→  principalSpriteId?: string;          // For 'principal' mode
   270→  spriteOrder?: string[];              // For 'list' mode custom ordering
   271→  excludedSpriteIds?: string[];        // Sprites to exclude
   272→}

   355→/**
   356→ * Conditional Sprite Entry - Attribute-driven sprite within a trigger collection.
   357→ *
   358→ * Used when a Trigger Collection is in conditional mode. Instead of matching keys
   359→ * to select a sprite, it evaluates attribute conditions with priority.
   360→ *
   361→ * When a Skill/Action reward activates a conditional sprite collection,
   362→ * the entries are evaluated by priority (highest first). The first entry
   363→ * whose conditions all match determines the sprite to display.
   364→ * If no entry matches, the default sprite (principalSpriteId) is used.
   365→ *
   366→ * Example (Trigger Collection "Attack Faces"):
   367→ *   Entry "Desperate"  → Priority 2, Condition: vida <= 20  → attack_desperate.webp
   368→ *   Entry "Empowered"  → Priority 1, Condition: mana >= 50  → attack_empowered.webp
   369→ *   Default             → attack_normal.webp
   370→ */
   371→export interface ConditionalSpriteEntry {
   372→  id: string;
   373→  name: string;                        // Display name: "Desperate Attack", "Low HP Face"
   374→  enabled: boolean;                    // Enable/disable this entry
   375→
   376→  // Priority - Higher number = higher priority (wins when multiple match)
   377→  priority: number;                    // Default: 0
   378→
   379→  // Conditions - evaluated with AND or OR logic (controlled by conditionOperator)
   380→  // Reuses StatRequirement for cross-character and all operator support
   381→  conditions: StatRequirement[];
   382→  conditionOperator?: 'AND' | 'OR'; // Logic operator for conditions (default: 'AND')
   383→
   384→  // Which sprite from the pack to show when this entry matches
   385→  spriteId: string;                    // References SpritePackEntryV2.id
   386→}
