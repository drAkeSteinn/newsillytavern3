    54→interface AttributeEntryEditorProps {

    58→
    59→export function AttributeEntryEditor({ entry, onUpdate }: AttributeEntryEditorProps) {
    60→  const config = useMemo((): AttributeEntryConfig => {
    61→    const existing = entry.extensions?.attributeConfig as AttributeEntryConfig | undefined;
    62→    return existing || getDefaultConfig();
    63→  }, [entry.extensions]);
    64→
    65→  const updateConfig = (newConfig: Partial<AttributeEntryConfig>) => {
    66→    const merged: AttributeEntryConfig = { ...config, ...newConfig };
    67→    onUpdate({
    68→      extensions: { ...entry.extensions, attributeConfig: merged },
    69→    });
    70→  };
    71→
    72→  const contentType = config.content.type;
    73→  const setContentType = (type: AttributeContentType) => {
    74→    if (type === 'static') {
    75→      updateConfig({
    76→        content: {
    77→          type: 'static',
    78→          content: (config.content as DynamicContentConfig)?.defaultContent || '',
    79→        } as StaticContentConfig,
    80→      });
    81→    } else {
    82→      updateConfig({
    83→        content: {
    84→          type: 'dynamic',
    85→          rules: [],
    86→          defaultContent: (config.content as StaticContentConfig)?.content || '',
    87→        } as DynamicContentConfig,
    88→      });
    89→    }
    90→  };
    91→
    92→  return (
    93→    <div className="space-y-4 p-3 border rounded-lg bg-muted/20">
    94→      {/* Template Key */}
    95→      <div className="space-y-1.5">
    96→        <div className="flex items-center gap-1.5">
    97→          <Link2 className="w-3.5 h-3.5 text-primary" />
    98→          <Label className="text-xs font-medium">Key de la entrada</Label>
    99→        </div>
   100→        <div className="flex items-center gap-2">
   101→          <span className="text-sm text-muted-foreground flex-shrink-0">{'{{'}</span>
   102→          <Input
   103→            value={config.templateKey}
   104→            onChange={(e) => updateConfig({ templateKey: e.target.value.replace(/\s/g, '') })}
   105→            placeholder="miKey"
   106→            className="h-7 text-sm font-mono"
   107→          />
   108→          <span className="text-sm text-muted-foreground flex-shrink-0">{'}'}</span>
   109→        </div>
   110→        <p className="text-[10px] text-muted-foreground">
   111→          Coloca {'{{' + (config.templateKey || 'miKey') + '}'}  en cualquier sección del personaje para inyectar el contenido.
   112→        </p>
   113→      </div>
   114→
   115→      <Separator />
   116→
   117→      {/* Requirement Logic */}
   118→      <div className="flex items-center gap-3">
   119→        <Label className="text-xs font-medium">Requisitos</Label>
   120→        <Select
   121→          value={config.requirementLogic || 'AND'}
   122→          onValueChange={(v) => updateConfig({ requirementLogic: v as AttributeRequirementLogic })}
   123→        >
   124→          <SelectTrigger className="h-6 w-20 text-xs">
   125→            <SelectValue />
   126→          </SelectTrigger>
   127→          <SelectContent>
   128→            <SelectItem value="AND" className="text-xs">AND (Todos)</SelectItem>
   129→            <SelectItem value="OR" className="text-xs">OR (Cualquiera)</SelectItem>
   130→          </SelectContent>
   131→        </Select>
   132→        <label className="flex items-center gap-1.5 text-xs cursor-pointer ml-auto">
   133→          <Switch
   134→            checked={config.isFallback ?? false}
   135→            onCheckedChange={(checked) => updateConfig({ isFallback: checked })}
   136→          />
   137→          Fallback
   138→        </label>
   139→      </div>
   140→
   141→      {/* Requirements List */}
   142→      <div className="space-y-2">
   143→        {config.requirements.map((req, index) => (
   144→          <RequirementRow
   145→            key={req.id}
   146→            requirement={req}
   147→            index={index}
   148→            onUpdate={(updated) => {
   149→              const newReqs = [...config.requirements];
   150→              newReqs[index] = updated;
   151→              updateConfig({ requirements: newReqs });
   152→            }}
   153→            onRemove={() => {
   154→              updateConfig({
   155→                requirements: config.requirements.filter((_, i) => i !== index),
   156→              });
   157→            }}
   158→          />
   159→        ))}
   160→
   161→        {!config.isFallback && (
   162→          <Button
   163→            size="sm"
   164→            variant="outline"
   165→            className="h-7 text-xs w-full border-dashed"
   166→            onClick={() => {
   167→              const newReq: AttributeRequirement = {
   168→                id: crypto.randomUUID(),
   169→                targetId: '',
   170→                targetName: '',
   171→                targetType: 'character',
   172→                attributeKey: '',
   173→                attributeName: '',
   174→                attributeType: 'text',
   175→                operator: 'equals',
   176→                value: '',
   177→              };
   178→              updateConfig({
   179→                requirements: [...config.requirements, newReq],
   180→              });
   181→            }}
   182→          >
   183→            <Plus className="w-3 h-3 mr-1" />
   184→            Agregar requisito
   185→          </Button>
   186→        )}
   187→      </div>
   188→
   189→      {config.isFallback && (
   190→        <div className="flex items-center gap-2 p-2 bg-amber-50 dark:bg-amber-900/20 rounded border border-amber-200 dark:border-amber-800">
   191→          <AlertTriangle className="w-3.5 h-3.5 text-amber-600" />
   192→          <span className="text-[11px] text-amber-700 dark:text-amber-400">
   193→            Modo fallback: esta entrada se activará siempre que ninguna otra entrada con la misma key cumpla sus requisitos.

   195→        </div>
   196→      )}
   197→
   198→      <Separator />
   199→
   200→      {/* Content Type */}
   201→      <div className="flex items-center gap-3">
   202→        <Label className="text-xs font-medium">Tipo de contenido</Label>
   203→        <Select value={contentType} onValueChange={(v) => setContentType(v as AttributeContentType)}>
   204→          <SelectTrigger className="h-6 w-28 text-xs">
   205→            <SelectValue />
   206→          </SelectTrigger>
   207→          <SelectContent>
   208→            <SelectItem value="static" className="text-xs">Estático</SelectItem>
   209→            <SelectItem value="dynamic" className="text-xs">Dinámico</SelectItem>
   210→          </SelectContent>
   211→        </Select>
   212→      </div>
   213→
   214→      {/* Content */}
   215→      {contentType === 'static' ? (
   216→        <div className="space-y-1.5">
   217→          <Label className="text-xs">Contenido estático</Label>
   218→          <Textarea
   219→            value={(config.content as StaticContentConfig).content}
   220→            onChange={(e) =>
   221→              updateConfig({
   222→                content: { type: 'static', content: e.target.value } as StaticContentConfig,
   223→              })
   224→            }
   225→            placeholder="Contenido que se inyectará cuando los requisitos se cumplan..."
   226→            rows={3}
   227→            className="font-mono text-sm"
   228→          />
   229→        </div>
   230→      ) : (
   231→        <DynamicContentEditor
   232→          config={config.content as DynamicContentConfig}
   233→          requirements={config.requirements}
   234→          onUpdate={(newContent) => updateConfig({ content: newContent })}
   235→        />
   236→      )}
   237→
   238→      {/* Priority */}
   239→      <div className="flex items-center gap-3">
   240→        <Label className="text-xs">Prioridad</Label>
   241→        <Input
   242→          type="number"
   243→          min={0}
   244→          value={config.priority ?? 0}
   245→          onChange={(e) => updateConfig({ priority: parseInt(e.target.value) || 0 })}
   246→          className="h-6 w-20 text-xs"
   247→        />
   248→        <p className="text-[10px] text-muted-foreground">
   249→          Mayor = se evalúa primero (útil cuando múltiples entradas usan la misma key)
   250→        </p>
   251→      </div>
   252→    </div>
   253→  );

   260→interface RequirementRowProps {
   261→  requirement: AttributeRequirement;
   262→  index: number;
   263→  onUpdate: (updated: AttributeRequirement) => void;
   264→  onRemove: () => void;
   265→}
   266→
   267→function RequirementRow({ requirement, index, onUpdate, onRemove }: RequirementRowProps) {
   268→  const { characters, personas } = useTavernStore();
   269→
   270→  const [expanded, setExpanded] = useState(true);
   271→
   272→  // Get available targets
   273→  const characterOptions = useMemo(() =>
   274→    characters.map(c => ({
   275→      id: c.id,
   276→      name: c.name,
   277→      type: 'character' as AttributeTargetType,
   278→    })),
   279→    [characters]
   280→  );
   281→
   282→  const personaOptions = useMemo(() =>
   283→    personas.map(p => ({
   284→      id: p.id,
   285→      name: p.name,
   286→      type: 'persona' as AttributeTargetType,
   287→    })),
   288→    [personas]
   289→  );
   290→
   291→  // Find the selected target to get attributes
   292→  const selectedTarget = useMemo(() => {
   293→    if (!requirement.targetId) return null;
   294→    if (requirement.targetType === 'persona') {
   295→      return personas.find(p => p.id === requirement.targetId);
   296→    }
   297→    return characters.find(c => c.id === requirement.targetId);
   298→  }, [requirement.targetId, requirement.targetType, characters, personas]);
   299→
   300→  // Get attribute definitions for selected target
   301→  const attributeDefs = useMemo(() => {
   302→    if (!selectedTarget?.statsConfig?.attributes) return [];
   303→    return selectedTarget.statsConfig.attributes;
   304→  }, [selectedTarget]);
   305→
   306→  // Get available operators based on attribute type
   307→  const availableOperators = useMemo(() => {
   308→    const typeKey = requirement.attributeType || 'text';
   309→    return OPERATORS_BY_ATTR_TYPE[typeKey] || OPERATORS_BY_ATTR_TYPE.text || [];
   310→  }, [requirement.attributeType]);
   311→
   312→  return (
   313→    <div className="border rounded-lg overflow-hidden">
   314→      {/* Header */}
   315→      <div
   316→        className="flex items-center gap-2 px-2.5 py-1.5 bg-muted/50 cursor-pointer hover:bg-muted/80 transition-colors"
   317→        onClick={() => setExpanded(!expanded)}
   318→      >
   319→        {expanded ? (
   320→          <ChevronDown className="w-3.5 h-3.5 text-muted-foreground" />
   321→        ) : (
   322→          <ChevronRight className="w-3.5 h-3.5 text-muted-foreground" />
   323→        )}
   324→        <Badge variant="outline" className="text-[9px] px-1 py-0 h-4">
   325→          #{index + 1}
   326→        </Badge>
   327→        {requirement.targetName && (
   328→          <span className="text-xs font-medium truncate flex-1">
   329→            {requirement.targetName}
   330→            {requirement.attributeName ? ` → ${requirement.attributeName}` : ''}
   331→          </span>
   332→        )}
   333→        {!requirement.targetName && (
   334→          <span className="text-xs text-muted-foreground italic">Sin configurar</span>
   335→        )}
   336→        <Button
   337→          size="icon"
   338→          variant="ghost"
   339→          className="h-5 w-5 text-destructive hover:text-destructive"
   340→          onClick={(e) => { e.stopPropagation(); onRemove(); }}
   341→        >
   342→          <Trash2 className="w-3 h-3" />
   343→        </Button>
   344→      </div>
   345→
   346→      {/* Expanded Content */}
   347→      {expanded && (
   348→        <div className="p-2.5 space-y-2 border-t">
   349→          {/* Row 1: Target selector */}
   350→          <div className="grid grid-cols-2 gap-2">
   351→            <div className="space-y-1">
   352→              <Label className="text-[10px] text-muted-foreground">Objetivo</Label>
   353→              <Select
   354→                value={requirement.targetId || '__none__'}
   355→                onValueChange={(v) => {
   356→                  if (v === '__none__') {
   357→                    onUpdate({ ...requirement, targetId: '', targetName: '', attributeKey: '', attributeName: '' });
   358→                    return;
   359→                  }
   360→                  const isPersona = personaOptions.some(p => p.id === v);
   361→                  const isChar = characterOptions.some(c => c.id === v);
   362→                  const target = isChar
   363→                    ? characterOptions.find(c => c.id === v)
   364→                    : personaOptions.find(p => p.id === v);
   365→                  if (target) {
   366→                    onUpdate({
   367→                      ...requirement,
   368→                      targetId: target.id,
   369→                      targetName: target.name,
   370→                      targetType: target.type,
   371→                      attributeKey: '',
   372→                      attributeName: '',
   373→                    });
   374→                  }
   375→                }}
   376→              >
   377→                <SelectTrigger className="h-7 text-xs">
   378→                  <SelectValue placeholder="Seleccionar..." />
   379→                </SelectTrigger>
   380→                <SelectContent>
   381→                  {characterOptions.length > 0 && (
   382→                    <>
   383→                      <div className="px-2 py-1 text-[10px] text-muted-foreground font-medium">Personajes</div>
   384→                      {characterOptions.map(c => (
   385→                        <SelectItem key={c.id} value={c.id} className="text-xs">
   386→                          {c.name}
   387→                        </SelectItem>
   388→                      ))}
   389→                    </>
   390→                  )}
   391→                  {personaOptions.length > 0 && (
   392→                    <>
   393→                      <div className="px-2 py-1 text-[10px] text-muted-foreground font-medium mt-1">Personas</div>
   394→                      {personaOptions.map(p => (
   395→                        <SelectItem key={p.id} value={p.id} className="text-xs">
   396→                          {p.name}
   397→                        </SelectItem>
   398→                      ))}
   399→                    </>
   400→                  )}
   401→                </SelectContent>
   402→              </Select>
   403→            </div>
   404→
   405→            {/* Attribute selector */}
   406→            <div className="space-y-1">
   407→              <Label className="text-[10px] text-muted-foreground">Atributo</Label>
   408→              <Select
   409→                value={requirement.attributeKey || '__none__'}
   410→                onValueChange={(v) => {
   411→                  if (v === '__none__') {
   412→                    onUpdate({ ...requirement, attributeKey: '', attributeName: '', attributeType: 'text' });
   413→                    return;
   414→                  }
   415→                  const attrDef = attributeDefs.find(a => a.key === v);
   416→                  if (attrDef) {
   417→                    onUpdate({
   418→                      ...requirement,
   419→                      attributeKey: attrDef.key,
   420→                      attributeName: attrDef.name,
   421→                      attributeType: attrDef.type as any,
   422→                    });
   423→                  }
   424→                }}
   425→                disabled={!requirement.targetId}
   426→              >
   427→                <SelectTrigger className="h-7 text-xs">
   428→                  <SelectValue placeholder="Seleccionar..." />
   429→                </SelectTrigger>
   430→                <SelectContent>
   431→                  {attributeDefs.length === 0 && (
   432→                    <div className="px-2 py-1 text-[10px] text-muted-foreground italic">
   433→                      Sin atributos definidos
   434→                    </div>
   435→                  )}
   436→                  {attributeDefs.map(attr => (
   437→                    <SelectItem key={attr.key} value={attr.key} className="text-xs">
   438→                      {attr.name}
   439→                      <Badge variant="outline" className="ml-1 text-[8px] px-0.5 h-3">
   440→                        {attr.type}
   441→                      </Badge>
   442→                    </SelectItem>
   443→                  ))}
   444→                </SelectContent>
   445→              </Select>
   446→            </div>
   447→          </div>
   448→
   449→          {/* Row 2: Operator and values */}
   450→          <div className="grid grid-cols-[1fr_1fr_1fr_auto] gap-2">
   451→            {/* Operator */}
   452→            <div className="space-y-1">
   453→              <Label className="text-[10px] text-muted-foreground">Operador</Label>
   454→              <Select
   455→                value={requirement.operator}
   456→                onValueChange={(v) => onUpdate({ ...requirement, operator: v as AttributeOperator })}
   457→              >
   458→                <SelectTrigger className="h-7 text-xs">
   459→                  <SelectValue />
   460→                </SelectTrigger>
   461→                <SelectContent>
   462→                  {availableOperators.map(op => (
   463→                    <SelectItem key={op} value={op} className="text-xs">
   464→                      {ATTRIBUTE_OPERATOR_LABELS[op]}
   465→                    </SelectItem>
   466→                  ))}
   467→                </SelectContent>
   468→              </Select>
   469→            </div>
   470→
   471→            {/* Value */}
   472→            {OPERATOR_REQUIRES_VALUE[requirement.operator] && (
   473→              <div className="space-y-1">
   474→                <Label className="text-[10px] text-muted-foreground">Valor</Label>
   475→                <Input
   476→                  value={requirement.value}
   477→                  onChange={(e) => onUpdate({ ...requirement, value: e.target.value })}
   478→                  placeholder="..."
   479→                  className="h-7 text-xs"
   480→                />
   481→              </div>
   482→            )}
   483→
   484→            {/* Secondary Value (for "between", "one_of", "none_of") */}
   485→            {OPERATOR_REQUIRES_SECONDARY_VALUE[requirement.operator] && (
   486→              <div className="space-y-1">
   487→                <Label className="text-[10px] text-muted-foreground">
   488→                  {requirement.operator === 'between' ? 'Valor máx.' : 'Lista (CSV)'}
   489→                </Label>
   490→                <Input
   491→                  value={requirement.valueSecondary || ''}
   492→                  onChange={(e) => onUpdate({ ...requirement, valueSecondary: e.target.value })}
   493→                  placeholder={requirement.operator === 'between' ? '10' : 'a, b, c'}
   494→                  className="h-7 text-xs"
   495→                />
   496→              </div>
   497→            )}
   498→
   499→            {/* Spacer when no values needed */}
   500→            {!OPERATOR_REQUIRES_VALUE[requirement.operator] && (
   501→              <div />
   502→            )}
   503→          </div>
   504→        </div>
   505→      )}
   506→    </div>
   507→  );
   508→}

   510→// ============================================
   511→// Dynamic Content Editor
   512→// ============================================
   513→
   514→interface DynamicContentEditorProps {
   515→  config: DynamicContentConfig;
   516→  requirements: AttributeRequirement[];
   517→  onUpdate: (content: DynamicContentConfig) => void;
   518→}
   519→
   520→function DynamicContentEditor({ config, requirements, onUpdate }: DynamicContentEditorProps) {
   521→  const addRule = () => {
   522→    const newRule: DynamicContentRule = {
   523→      id: crypto.randomUUID(),
   524→      conditions: [],
   525→      content: '',
   526→    };
   527→    onUpdate({ ...config, rules: [...config.rules, newRule] });
   528→  };
   529→
   530→  const updateRule = (index: number, updated: DynamicContentRule) => {
   531→    const newRules = [...config.rules];
   532→    newRules[index] = updated;
   533→    onUpdate({ ...config, rules: newRules });
   534→  };
   535→
   536→  const removeRule = (index: number) => {
   537→    onUpdate({ ...config, rules: config.rules.filter((_, i) => i !== index) });
   538→  };
   539→
   540→  return (
   541→    <div className="space-y-2">
   542→      {/* Rules */}
   543→      {config.rules.map((rule, index) => (
   544→        <DynamicRuleEditor
   545→          key={rule.id}
   546→          rule={rule}
   547→          ruleIndex={index}
   548→          requirements={requirements}
   549→          onUpdate={(updated) => updateRule(index, updated)}
   550→          onRemove={() => removeRule(index)}
   551→        />
   552→      ))}
   553→
   554→      <Button
   555→        size="sm"
   556→        variant="outline"
   557→        className="h-7 text-xs w-full border-dashed"
   558→        onClick={addRule}
   559→      >
   560→        <Plus className="w-3 h-3 mr-1" />
   561→        Agregar regla condicional
   562→      </Button>
   563→
   564→      {/* Default Content */}
   565→      <div className="space-y-1.5">
   566→        <Label className="text-xs">Contenido por defecto (si ninguna regla coincide)</Label>
   567→        <Textarea
   568→          value={config.defaultContent || ''}
   569→          onChange={(e) => onUpdate({ ...config, defaultContent: e.target.value })}
   570→          placeholder="Contenido por defecto..."
   571→          rows={2}
   572→          className="font-mono text-sm"
   573→        />
   574→      </div>
   575→    </div>
   576→  );
   577→}

   579→// ============================================
   580→// Dynamic Rule Editor
   581→// ============================================
   582→
   583→interface DynamicRuleEditorProps {
   584→  rule: DynamicContentRule;
   585→  ruleIndex: number;
   586→  requirements: AttributeRequirement[];
   587→  onUpdate: (updated: DynamicContentRule) => void;
   588→  onRemove: () => void;
   589→}
   590→
   591→function DynamicRuleEditor({ rule, ruleIndex, requirements, onUpdate, onRemove }: DynamicRuleEditorProps) {
   592→  const [expanded, setExpanded] = useState(true);
   593→
   594→  const addCondition = () => {
   595→    const newCondition = {
   596→      requirementIndex: 0,
   597→      operator: 'equals' as AttributeOperator,
   598→      value: '',
   599→    };
   600→    onUpdate({ ...rule, conditions: [...rule.conditions, newCondition] });
   601→  };
   602→
   603→  const updateCondition = (condIndex: number, updated: typeof rule.conditions[0]) => {
   604→    const newConditions = [...rule.conditions];
   605→    newConditions[condIndex] = updated;
   606→    onUpdate({ ...rule, conditions: newConditions });
   607→  };
   608→
   609→  const removeCondition = (condIndex: number) => {
   610→    onUpdate({ ...rule, conditions: rule.conditions.filter((_, i) => i !== condIndex) });
   611→  };
   612→
   613→  // Get available operators for the referenced requirement's attribute type
   614→  const getAvailableOperators = (reqIndex: number): AttributeOperator[] => {
   615→    const req = requirements[reqIndex];
   616→    if (!req) return OPERATORS_BY_ATTR_TYPE.text || [];
   617→    const typeKey = req.attributeType || 'text';
   618→    return OPERATORS_BY_ATTR_TYPE[typeKey] || OPERATORS_BY_ATTR_TYPE.text || [];
   619→  };
   620→
   621→  return (
   622→    <div className="border rounded-lg overflow-hidden">
   623→      {/* Header */}
   624→      <div
   625→        className="flex items-center gap-2 px-2.5 py-1.5 bg-emerald-50 dark:bg-emerald-900/20 cursor-pointer hover:bg-emerald-100 dark:hover:bg-emerald-900/30 transition-colors"
   626→        onClick={() => setExpanded(!expanded)}
   627→      >
   628→        {expanded ? (
   629→          <ChevronDown className="w-3.5 h-3.5" />
   630→        ) : (
   631→          <ChevronRight className="w-3.5 h-3.5" />
   632→        )}
   633→        <Badge className="text-[9px] px-1 py-0 h-4 bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400 border-emerald-200 dark:border-emerald-800">
   634→          Regla {ruleIndex + 1}
   635→        </Badge>
   636→        <span className="text-xs text-muted-foreground truncate flex-1">
   637→          {rule.conditions.length} condición(es)
   638→        </span>
   639→        <Button
   640→          size="icon"
   641→          variant="ghost"
   642→          className="h-5 w-5 text-destructive hover:text-destructive"
   643→          onClick={(e) => { e.stopPropagation(); onRemove(); }}
   644→        >
   645→          <Trash2 className="w-3 h-3" />
   646→        </Button>
   647→      </div>
   648→
   649→      {expanded && (
   650→        <div className="p-2.5 space-y-2 border-t">
   651→          {/* Conditions */}
   652→          {rule.conditions.map((cond, condIndex) => {
   653→            const refReq = requirements[cond.requirementIndex];
   654→            const ops = getAvailableOperators(cond.requirementIndex);
   655→
   656→            return (
   657→              <div key={condIndex} className="grid grid-cols-[1fr_1fr_1fr_auto] gap-2 items-end">
   658→                {/* Requirement reference */}
   659→                <div className="space-y-0.5">
   660→                  <Label className="text-[10px] text-muted-foreground">Atributo</Label>
   661→                  <Select
   662→                    value={String(cond.requirementIndex)}
   663→                    onValueChange={(v) =>
   664→                      updateCondition(condIndex, { ...cond, requirementIndex: parseInt(v) })
   665→                    }
   666→                  >
   667→                    <SelectTrigger className="h-6 text-[11px]">
   668→                      <SelectValue />
   669→                    </SelectTrigger>
   670→                    <SelectContent>
   671→                      {requirements.map((req, ri) => (
   672→                        <SelectItem key={ri} value={String(ri)} className="text-[11px]">
   673→                          #{ri + 1}: {req.targetName} → {req.attributeName || '?'}
   674→                        </SelectItem>
   675→                      ))}
   676→                    </SelectContent>
   677→                  </Select>
   678→                </div>
   679→
   680→                {/* Operator */}
   681→                <div className="space-y-0.5">
   682→                  <Label className="text-[10px] text-muted-foreground">Condición</Label>
   683→                  <Select
   684→                    value={cond.operator}
   685→                    onValueChange={(v) =>
   686→                      updateCondition(condIndex, { ...cond, operator: v as AttributeOperator })
   687→                    }
   688→                  >
   689→                    <SelectTrigger className="h-6 text-[11px]">
   690→                      <SelectValue />
   691→                    </SelectTrigger>
   692→                    <SelectContent>
   693→                      {ops.map(op => (
   694→                        <SelectItem key={op} value={op} className="text-[11px]">
   695→                          {ATTRIBUTE_OPERATOR_LABELS[op]}
   696→                        </SelectItem>
   697→                      ))}
   698→                    </SelectContent>
   699→                  </Select>
   700→                </div>
   701→
   702→                {/* Value */}
   703→                {OPERATOR_REQUIRES_VALUE[cond.operator] && (
   704→                  <div className="space-y-0.5">
   705→                    <Label className="text-[10px] text-muted-foreground">Valor</Label>
   706→                    <Input
   707→                      value={cond.value}
   708→                      onChange={(e) =>
   709→                        updateCondition(condIndex, { ...cond, value: e.target.value })
   710→                      }
   711→                      placeholder="..."
   712→                      className="h-6 text-[11px]"
   713→                    />
   714→                  </div>
   715→                )}
   716→
   717→                {!OPERATOR_REQUIRES_VALUE[cond.operator] && <div />}
   718→
   719→                <Button
   720→                  size="icon"
   721→                  variant="ghost"
   722→                  className="h-6 w-6 text-destructive hover:text-destructive"
   723→                  onClick={() => removeCondition(condIndex)}
   724→                >
   725→                  <Trash2 className="w-3 h-3" />
   726→                </Button>
   727→              </div>
   728→            );
   729→          })}
   730→
   731→          <Button
   732→            size="sm"
   733→            variant="ghost"
   734→            className="h-6 text-[11px] w-full border-dashed border"
   735→            onClick={addCondition}
   736→          >
   737→            <Plus className="w-3 h-3 mr-1" />
   738→            Agregar condición
   739→          </Button>
   740→
   741→          {/* Rule Content */}
   742→          <div className="space-y-1">
   743→            <Label className="text-[11px]">Contenido si se cumple</Label>
   744→            <Textarea
   745→              value={rule.content}
   746→              onChange={(e) => onUpdate({ ...rule, content: e.target.value })}
   747→              placeholder="Texto que aparecerá cuando se cumplan todas las condiciones..."
   748→              rows={2}
   749→              className="font-mono text-[11px]"
   750→            />
   751→          </div>
   752→        </div>
   753→      )}
   754→    </div>
   755→  );
   756→}

   777→// Live Preview Component
   778→// ============================================
   779→
   780→interface AttributeEntryPreviewProps {
   781→  entry: LorebookEntry;
   782→}
   783→
   784→export function AttributeEntryPreview({ entry }: AttributeEntryPreviewProps) {
   785→  const { characters, personas } = useTavernStore();
   786→  const { sessionStats } = useTavernStore();
   787→
   788→  const config = useMemo((): AttributeEntryConfig | null => {
   789→    if (entry.entryType !== 'attribute') return null;
   790→    return (entry.extensions?.attributeConfig as AttributeEntryConfig) || null;
   791→  }, [entry]);
   792→
   793→  if (!config) return null;
   794→
   795→  // Get current attribute values
   796→  const getAttrValue = (req: AttributeRequirement) => {
   797→    let values: Record<string, number | string> = {};
   798→    if (req.targetType === 'character') {
   799→      values = sessionStats?.characterStats?.[req.targetId]?.attributeValues || {};
   800→    } else {
   801→      values = sessionStats?.characterStats?.['__user__']?.attributeValues || {};
   802→    }
   803→    return values[req.attributeKey];
   804→  };
   805→
   806→  return (
   807→    <div className="p-3 border rounded-lg bg-muted/10 space-y-2">
   808→      <div className="flex items-center gap-1.5">
   809→        <Eye className="w-3.5 h-3.5 text-primary" />
   810→        <span className="text-xs font-medium">Preview (valores actuales)</span>
   811→      </div>
   812→
   813→      {/* Requirements evaluation */}
   814→      {config.requirements.length > 0 && (
   815→        <div className="space-y-1">
   816→          {config.requirements.map((req, i) => {
   817→            const val = getAttrValue(req);
   818→            const hasVal = val !== undefined && val !== null;
   819→
   820→            return (
   821→              <div key={req.id} className="flex items-center gap-2 text-[11px]">
   822→                {hasVal ? (
   823→                  <CheckCircle2 className="w-3 h-3 text-green-500" />
   824→                ) : (
   825→                  <XCircle className="w-3 h-3 text-muted-foreground" />
   826→                )}
   827→                <span className="text-muted-foreground">
   828→                  {req.targetName} → {req.attributeName}:
   829→                </span>
   830→                <span className="font-medium">{hasVal ? String(val) : 'sin valor'}</span>
   831→              </div>
   832→            );
   833→          })}
   834→        </div>
   835→      )}
   836→
   837→      {/* Key */}
   838→      {config.templateKey && (
   839→        <div className="text-[11px]">
   840→          <span className="text-muted-foreground">Key: </span>
   841→          <code className="bg-muted px-1 py-0.5 rounded font-mono text-primary">
   842→            {'{{' + config.templateKey + '}}'}
   843→          </code>
   844→        </div>
   845→      )}
   846→
   847→      {/* Content preview */}
   848→      <div className="text-[11px] text-muted-foreground">
   849→        <span>Tipo de contenido: </span>
   850→        <Badge variant="outline" className="text-[9px] px-1 py-0 h-3">
   851→          {config.content.type === 'static' ? 'Estático' : `Dinámico (${config.content.rules?.length || 0} reglas)`}
   852→        </Badge>
   853→      </div>
   854→    </div>
   855→  );
   856→}
