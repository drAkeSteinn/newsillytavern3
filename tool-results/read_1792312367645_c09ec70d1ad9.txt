    40→export function normalizePath(uri: string): string {
    41→  if (uri.startsWith('mem://') || uri.startsWith('lancedb://')) return uri;
    42→  let normalized = path.normalize(uri);
    43→  if (isWindows()) normalized = normalized.replace(/\//g, '\\');
    44→  return normalized;
    45→}

    47→export function checkDirectoryPermissions(dirPath: string): {
    48→  exists: boolean; writable: boolean; error?: string;
    49→} {
    50→  try {
    51→    const normalizedPath = normalizePath(dirPath);
    52→    const parentDir = path.dirname(normalizedPath);
    53→
    54→    if (!fs.existsSync(parentDir)) {
    55→      return { exists: false, writable: false, error: 'Parent directory does not exist' };
    56→    }
    57→
    58→    if (fs.existsSync(normalizedPath)) {
    59→      try {
    60→        fs.accessSync(normalizedPath, fs.constants.W_OK);
    61→        return { exists: true, writable: true };
    62→      } catch {
    63→        return { exists: true, writable: false, error: 'No write permission' };
    64→      }
    65→    }
    66→
    67→    try {
    68→      fs.accessSync(parentDir, fs.constants.W_OK);
    69→      return { exists: false, writable: true };
    70→    } catch {
    71→      return { exists: false, writable: false, error: 'No write permission on parent directory' };
    72→    }
    73→  } catch (error: any) {
    74→    return { exists: false, writable: false, error: error.message };
    75→  }
    76→}

    78→export function ensureLanceDBDirectory(uri: string): { success: boolean; path: string; error?: string } {
    79→  try {
    80→    if (uri.startsWith('mem://') || uri.startsWith('lancedb://')) {
    81→      return { success: true, path: uri };
    82→    }
    83→    const normalizedPath = normalizePath(uri);
    84→    if (!fs.existsSync(normalizedPath)) {
    85→      fs.mkdirSync(normalizedPath, { recursive: true });
    86→    }
    87→    return { success: true, path: normalizedPath };
    88→  } catch (error: any) {
    89→    return { success: false, path: uri, error: error.message };
    90→  }
    91→}

   146→// ============ FASE 15: Platform Detection (arch + libc) ============

   148→export type Architecture = 'x64' | 'arm64' | 'ia32' | 'arm' | 'unknown';
   149→export type Libc = 'gnu' | 'musl' | 'unknown';
   150→
   151→/** Detect CPU architecture */
   152→export function getArchitecture(): Architecture {
   153→  if (typeof process !== 'undefined' && process.arch) {
   154→    const arch = process.arch;
   155→    if (arch === 'x64' || arch === 'arm64' || arch === 'ia32' || arch === 'arm') {
   156→      return arch;
   157→    }
   158→  }
   159→  return 'unknown';
   160→}
   161→
   162→/**
   163→ * Detect libc variant on Linux (glibc vs musl).
   164→ * Alpine Linux uses musl; most other distros use glibc.
   165→ * On non-Linux platforms, returns 'unknown'.
   166→ */
   167→export function detectLibc(): Libc {
   168→  if (!isLinux()) return 'unknown';
   169→
   170→  // Check if musl is in use by looking at the dynamic linker
   171→  // This is a heuristic — the napi-rs loader does something similar
   172→  try {
   173→    // Use dynamic import to avoid require() lint error
   174→    const { execSync } = require_child_process();
   175→    // Try ldd — musl systems report "musl" in the output
   176→    const lddOutput = execSync('ldd --version 2>&1', { encoding: 'utf-8', timeout: 2000 });
   177→    if (lddOutput.toLowerCase().includes('musl')) {
   178→      return 'musl';
   179→    }
   180→    return 'gnu';
   181→  } catch {
   182→    // Fallback: assume glibc (most common)
   183→    return 'gnu';
   184→  }
   185→}
   186→
   187→// Helper to import child_process without require() lint error
   188→function require_child_process() {
   189→  // eslint-disable-next-line @typescript-eslint/no-require-imports
   190→  return require('child_process');
   191→}

   193→/**
   194→ * Get the LanceDB platform package name for the current system.
   195→ * Returns null if the platform is not supported.
   196→ *
   197→ * Examples:
   198→ * - Windows x64 → @lancedb/lancedb-win32-x64-msvc
   199→ * - Linux x64 glibc → @lancedb/lancedb-linux-x64-gnu
   200→ * - Linux x64 musl (Alpine) → @lancedb/lancedb-linux-x64-musl
   201→ * - macOS arm64 (Apple Silicon) → @lancedb/lancedb-darwin-arm64
   202→ * - macOS x64 (Intel) → @lancedb/lancedb-darwin-x64 (may not exist)
   203→ */
   204→export function getLanceDBPlatformPackage(): string | null {
   205→  const platform = getPlatform();
   206→  const arch = getArchitecture();
   207→  const libc = detectLibc();
   208→
   209→  if (platform === 'win32') {
   210→    if (arch === 'x64') return '@lancedb/lancedb-win32-x64-msvc';
   211→    if (arch === 'arm64') return '@lancedb/lancedb-win32-arm64-msvc';
   212→  }
   213→
   214→  if (platform === 'linux') {
   215→    if (arch === 'x64') {
   216→      return libc === 'musl'
   217→        ? '@lancedb/lancedb-linux-x64-musl'
   218→        : '@lancedb/lancedb-linux-x64-gnu';
   219→    }
   220→    if (arch === 'arm64') {
   221→      return libc === 'musl'
   222→        ? '@lancedb/lancedb-linux-arm64-musl'
   223→        : '@lancedb/lancedb-linux-arm64-gnu';
   224→    }
   225→  }
   226→
   227→  if (platform === 'darwin') {
   228→    if (arch === 'arm64') return '@lancedb/lancedb-darwin-arm64';
   229→    if (arch === 'x64') return '@lancedb/lancedb-darwin-x64';
   230→  }
   231→
   232→  return null;
   233→}

   235→/**
   236→ * Get a human-readable description of the detected platform.
   237→ */
   238→export function getPlatformDescription(): string {
   239→  const platform = getPlatform();
   240→  const arch = getArchitecture();
   241→  const libc = detectLibc();
   242→
   243→  const platformName = platform === 'win32' ? 'Windows'
   244→    : platform === 'linux' ? 'Linux'
   245→    : platform === 'darwin' ? 'macOS'
   246→    : 'Unknown';
   247→
   248→  const archName = arch === 'x64' ? 'x64'
   249→    : arch === 'arm64' ? 'ARM64'
   250→    : arch === 'ia32' ? 'x86'
   251→    : arch === 'arm' ? 'ARM'
   252→    : 'Unknown';
   253→
   254→  let desc = `${platformName} ${archName}`;
   255→  if (platform === 'linux' && libc !== 'unknown') {
   256→    desc += ` (${libc})`;
   257→  }
   258→
   259→  return desc;
   260→}

   264→export class LanceDBError extends Error {
   265→  constructor(
   266→    message: string,
   267→    public code: string,
   268→    public platform?: Platform,
   269→    public details?: any
   270→  ) {
   271→    super(message);
   272→    this.name = 'LanceDBError';
   273→  }
   274→
   275→  getSuggestion(): string {
   276→    switch (this.code) {
   277→      case 'PERMISSION_DENIED':
   278→        return isWindows()
   279→          ? 'Run as administrator or change the database path.'
   280→          : 'Check directory permissions with chmod.';
   281→      case 'NATIVE_MODULE_ERROR':
   282→        return isWindows()
   283→          ? 'Install Visual C++ Redistributable. Reinstall @lancedb/lancedb.'
   284→          : 'Check native dependencies (glibc, etc.).';
   285→      case 'MODULE_NOT_AVAILABLE':
   286→        return 'LanceDB is not available on this system. Check platform compatibility.';
   287→      case 'TABLE_NOT_FOUND':
   288→        return 'Table does not exist. It will be created on first operation.';
   289→      default:
   290→        return 'Check the documentation or logs for more details.';
   291→    }
   292→  }

   297→async function loadLanceDBModule(): Promise<{ success: boolean; error?: string }> {
   298→  if (isModuleLoadAttempted) {
   299→    return { success: lancedbModule !== null, error: lancedbLoadError || undefined };
   300→  }
   301→  isModuleLoadAttempted = true;
   302→
   303→  try {
   304→    lancedbModule = await import('@lancedb/lancedb');
   305→    return { success: true };
   306→  } catch (error: any) {
   307→    const errorMsg = error.message || String(error);
   308→    lancedbLoadError = errorMsg;
   309→    return { success: false, error: errorMsg };
   310→  }
   311→}

   336→function normalizeVector(vector: number[]): number[] {
   337→  const magnitude = Math.sqrt(vector.reduce((sum, val) => sum + val * val, 0));
   338→  if (magnitude === 0) return vector;
   339→  return vector.map(val => val / magnitude);
   340→}
   341→
   342→/**
   343→ * Convert LanceDB's L2-squared distance to cosine similarity.
   344→ * LanceDB vectorSearch returns _distance = L2² (squared Euclidean distance).
   345→ * For unit-normalized vectors: L2² = 2 - 2·cos(θ), so cos(θ) = 1 - L2²/2.
   346→ */
   347→function l2SquaredToCosineSimilarity(l2SquaredDistance: number): number {
   348→  return 1 - l2SquaredDistance / 2;
   349→}

   359→// ============ Initialization ============
   360→
   361→// Track the dimension that the embeddings table was created with
   362→let tableDimension: number | null = null;
   363→
   364→async function initializeTables(): Promise<void> {
   365→  if (!db) throw new Error('Database not initialized');
   366→
   367→  // Get vector dimension from config
   368→  let vectorDimension = 1024;
   369→  let embeddingModel = 'bge-m3:567m';
   370→  try {
   371→    const { getConfig } = await import('./config-persistence');
   372→    const config = getConfig();
   373→    vectorDimension = config.dimension || 1024;
   374→    embeddingModel = config.model || 'bge-m3:567m';
   375→  } catch {
   376→    // Use defaults
   377→  }
   378→
   379→  const defaultEmbedding = {
   380→    id: 'placeholder',
   381→    content: 'placeholder',
   382→    vector: new Array(vectorDimension).fill(0),
   383→    metadata: '{}',
   384→    namespace: 'default',
   385→    source_type: 'system',
   386→    source_id: 'init',
   387→    model_name: embeddingModel,
   388→    created_at: new Date().toISOString(),
   389→    updated_at: new Date().toISOString(),
   390→  };
   391→
   392→  const defaultNamespace = {
   393→    id: 'placeholder',
   394→    namespace: 'default',
   395→    description: 'Default namespace',
   396→    metadata: '{}',
   397→    created_at: new Date().toISOString(),
   398→    updated_at: new Date().toISOString(),
   399→  };
   400→
   401→  // Try to open existing embeddings table
   402→  try {
   403→    embeddingsTable = await db.openTable(EMBEDDINGS_TABLE);
   404→
   405→    // Check if dimension matches by reading the first row's vector
   406→    // If the table was created with a different dimension, we must recreate it
   407→    const existingRows = await embeddingsTable.query().limit(1).toArray();
   408→    if (existingRows.length > 0 && existingRows[0].vector) {
   409→      const existingDim = existingRows[0].vector.length;
   410→      tableDimension = existingDim;
   411→
   412→      if (existingDim !== vectorDimension) {
   413→        console.warn(
   414→          `[LanceDB] Dimension mismatch detected! Table has ${existingDim}D vectors, but config requires ${vectorDimension}D. ` +
   415→          `Dropping and recreating embeddings table. All existing embeddings will be lost.`
   416→        );
   417→
   418→        // Drop the old table
   419→        try { await db.dropTable(EMBEDDINGS_TABLE); } catch { /* ignore */ }
   420→
   421→        // Also drop any namespace-specific tables that might have wrong dimensions
   422→        try {
   423→          const tables = await db.tableNames();
   424→          for (const tableName of tables) {
   425→            if (tableName !== EMBEDDINGS_TABLE && tableName !== NAMESPACES_TABLE) {
   426→              try { await db.dropTable(tableName); } catch { /* ignore */ }
   427→            }
   428→          }
   429→        } catch { /* ignore */ }
   430→
   431→        // Recreate with new dimension
   432→        await db.createTable(EMBEDDINGS_TABLE, [defaultEmbedding]);
   433→        embeddingsTable = await db.openTable(EMBEDDINGS_TABLE);
   434→        await embeddingsTable.delete("id = 'placeholder'");
   435→        tableDimension = vectorDimension;
   436→      }
   437→    } else if (existingRows.length === 0) {
   438→      // Table exists but is empty — recreate with new dimension to be safe
   439→      try { await db.dropTable(EMBEDDINGS_TABLE); } catch { /* ignore */ }
   440→      await db.createTable(EMBEDDINGS_TABLE, [defaultEmbedding]);
   441→      embeddingsTable = await db.openTable(EMBEDDINGS_TABLE);
   442→      await embeddingsTable.delete("id = 'placeholder'");
   443→      tableDimension = vectorDimension;
   444→    }
   445→  } catch {
   446→    // Table doesn't exist yet, create it
   447→    await db.createTable(EMBEDDINGS_TABLE, [defaultEmbedding]);
   448→    embeddingsTable = await db.openTable(EMBEDDINGS_TABLE);
   449→    await embeddingsTable.delete("id = 'placeholder'");
   450→    tableDimension = vectorDimension;
   451→  }
   452→
   453→  // Namespaces table (no vectors, no dimension concern)
   454→  try {
   455→    namespacesTable = await db.openTable(NAMESPACES_TABLE);
   456→  } catch {
   457→    await db.createTable(NAMESPACES_TABLE, [defaultNamespace]);
   458→    namespacesTable = await db.openTable(NAMESPACES_TABLE);
   459→    await namespacesTable.delete("id = 'placeholder'");
   460→  }
   461→}

   467→/**
   468→ * Memory V2: access the shared LanceDB instance so additional tables
   469→ * (e.g. `memories_v2`) can live in the same database directory.
   470→ * Returns null when the native module is unavailable or init failed —
   471→ * callers (memory V2 store) must fall back to their JSON backend.
   472→ */
   473→export async function getLanceDBInstance(): Promise<any | null> {
   474→  if (isPermanentlyUnavailable) return null;
   475→  if (!db || !isInitialized) {
   476→    try {
   477→      await initLanceDB();
   478→    } catch {
   479→      return null;
   480→    }
   481→  }
   482→  return db && isInitialized ? db : null;
   483→}
   484→
   485→export async function initLanceDB(uri?: string, forceReinit: boolean = false): Promise<void> {
   486→  const dbUri = uri || process.env.LANCEDB_URI || getDefaultLanceDBPath();
   487→
   488→  if (isInitialized && db && currentUri === dbUri && !forceReinit) return;
   489→
   490→  // If permanently unavailable, skip silently (don't throw)
   491→  if (isPermanentlyUnavailable) {
   492→    return;
   493→  }
   494→
   495→  if (db) await closeLanceDB();
   496→
   497→  const { success, error } = await loadLanceDBModule();
   498→  if (!success) {
   499→    isPermanentlyUnavailable = true;
   500→    lancedbLoadError = error || 'Unknown error';
   501→    console.warn(
   502→      `[LanceDB] ⚠️ LanceDB native module not available for platform "${getPlatform()}". ` +
   503→      `Embeddings/vector search features will be disabled. Error: ${error}`
   504→    );
   505→    return;
   506→  }
   507→
   508→  isInitialized = true;
   509→  currentUri = dbUri;
   510→
   511→  try {
   512→    const dirResult = ensureLanceDBDirectory(dbUri);
   513→    if (!dirResult.success) {
   514→      throw new LanceDBError(
   515→        `Cannot create/verify directory: ${dirResult.error}`,
   516→        'DIRECTORY_ERROR',
   517→        getPlatform(),
   518→        { uri: dbUri }
   519→      );
   520→    }
   521→
   522→    db = await lancedbModule.connect(dirResult.path);
   523→    await initializeTables();
   524→  } catch (error: any) {
   525→    if (error instanceof LanceDBError) throw error;
   526→
   527→    isInitialized = false;
   528→    db = null;
   529→    currentUri = null;
   530→
   531→    const msg = error.message || String(error);
   532→    const lowerMsg = msg.toLowerCase();
   533→
   534→    let code = 'UNKNOWN_ERROR';
   535→    if (lowerMsg.includes('permission') || lowerMsg.includes('access denied')) code = 'PERMISSION_DENIED';
   536→    else if (lowerMsg.includes('native') || lowerMsg.includes('binding')) code = 'NATIVE_MODULE_ERROR';
   537→
   538→    throw new LanceDBError(msg, code, getPlatform(), { uri: dbUri, originalError: error });
   539→  }
   540→}

   666→  static async searchSimilar(params: {
   667→    queryVector: number[];
   668→    namespace?: string;
   669→    limit?: number;
   670→    threshold?: number;
   671→  }): Promise<SearchResult[]> {
   672→    if (isPermanentlyUnavailable) return [];
   673→    const {
   674→      queryVector,
   675→      namespace,
   676→      limit = 10,
   677→      threshold = 0.3,
   678→    } = params;
   679→
   680→    const table = await getEmbeddingsTable();
   681→    if (!table) return [];
   682→    const normalizedQueryVector = normalizeVector(queryVector);
   683→
   684→    let results: any[];
   685→
   686→    // All embeddings are stored in the main table with a 'namespace' column.
   687→    // Always search the main table and filter by namespace in-memory.
   688→    const allResults = await table
   689→      .vectorSearch(normalizedQueryVector)
   690→      .limit(limit * 10)
   691→      .toArray();
   692→
   693→    results = allResults
   694→      .map((row: any) => ({
   695→        ...row,
   696→        similarity: l2SquaredToCosineSimilarity(row._distance || 0),
   697→      }))
   698→      .filter((r: any) => {
   699→        if (!namespace || namespace === 'all') return r.similarity >= threshold;
   700→        return (r.namespace || 'default') === namespace && r.similarity >= threshold;
   701→      })
   702→      .slice(0, limit);
   703→
   704→    return results.map((row: any) => ({
   705→      id: row.id,
   706→      content: row.content,
   707→      metadata: parseMetadata(row.metadata),
   708→      namespace: row.namespace || 'default',
   709→      source_type: row.source_type,
   710→      source_id: row.source_id,
   711→      similarity: row.similarity,
   712→    }));
   713→  }

   757→  static async upsertNamespace(params: {
   758→    namespace: string;
   759→    description?: string;
   760→    metadata?: Record<string, any>;
   761→  }): Promise<RecordNamespace> {
   762→    if (isPermanentlyUnavailable) {
   763→      // Return a stub namespace so callers don't crash
   764→      return {
   765→        id: '__unavailable__',
   766→        namespace: params.namespace,
   767→        description: params.description,
   768→        metadata: params.metadata || {},
   769→        created_at: new Date(),
   770→        updated_at: new Date(),
   771→      };
   772→    }
   773→    const { v4: uuidv4 } = await import('uuid');
   774→    const { namespace, description, metadata = {} } = params;
   775→    const table = await getNamespacesTable();
   776→    if (!table) {
   777→      return {
   778→        id: '__unavailable__',
   779→        namespace: params.namespace,
   780→        description: params.description,
   781→        metadata: params.metadata || {},
   782→        created_at: new Date(),
   783→        updated_at: new Date(),
   784→      };
   785→    }
   786→    const existing = await tableFilter(table, `namespace = '${escapeFilterValue(namespace)}'`);
   787→
   788→    const nsRecord = {
   789→      id: existing.length > 0 ? existing[0].id : uuidv4(),
   790→      namespace,
   791→      description,
   792→      metadata: JSON.stringify(metadata),
   793→      created_at: existing.length > 0 ? existing[0].created_at : new Date().toISOString(),
   794→      updated_at: new Date().toISOString(),
   795→    };
   796→
   797→    // Always delete all matching rows before inserting to prevent duplicates
   798→    // caused by race conditions (concurrent upsertNamespace calls).
   799→    if (existing.length > 0) {
   800→      await table.delete(`namespace = '${escapeFilterValue(namespace)}'`);
   801→    }
   802→    await table.add([nsRecord]);
   803→
   804→    // Safety net: if a race created extra duplicates, clean them up.
   805→    // Keep only the row we just inserted (by id).
   806→    const afterInsert = await tableFilter(table, `namespace = '${escapeFilterValue(namespace)}'`);
   807→    if (afterInsert.length > 1) {
   808→      const idsToDelete = afterInsert
   809→        .filter((row: any) => row.id !== nsRecord.id)
   810→        .map((row: any) => row.id);
   811→      if (idsToDelete.length > 0) {
   812→        const deleteFilter = idsToDelete
   813→          .map((id: string) => `id = '${escapeFilterValue(id)}'`)
   814→          .join(' OR ');
   815→        await table.delete(deleteFilter);
   816→        console.log(`[LanceDB] Cleaned up ${idsToDelete.length} duplicate namespace row(s) for "${namespace}"`);
   817→      }
   818→    }
   819→
   820→    return {
   821→      id: nsRecord.id,
   822→      namespace: nsRecord.namespace,
   823→      description: nsRecord.description,
   824→      metadata: JSON.parse(nsRecord.metadata),
   825→      created_at: new Date(nsRecord.created_at),
   826→      updated_at: new Date(nsRecord.updated_at),
   827→    };
   828→  }

   830→  static async getAllNamespaces(): Promise<RecordNamespace[]> {
   831→    if (isPermanentlyUnavailable) return [];
   832→    const table = await getNamespacesTable();
   833→    if (!table) return [];
   834→    const results = await tableToArray(table);
   835→    const mapped = results.map((row: any) => ({
   836→      id: row.id,
   837→      namespace: row.namespace,
   838→      description: row.description,
   839→      metadata: typeof row.metadata === 'string' ? JSON.parse(row.metadata) : row.metadata,
   840→      created_at: new Date(row.created_at),
   841→      updated_at: new Date(row.updated_at),
   842→    }));
   843→
   844→    // Deduplicate by namespace name — keep the most recently updated entry.
   845→    // This guards against duplicate rows that can appear due to a race condition
   846→    // in upsertNamespace (non-atomic delete-then-insert).
   847→    const deduped = new Map<string, RecordNamespace>();
   848→    for (const ns of mapped) {
   849→      const existing = deduped.get(ns.namespace);
   850→      if (!existing || ns.updated_at > existing.updated_at) {
   851→        deduped.set(ns.namespace, ns);
   852→      }
   853→    }
   854→    return Array.from(deduped.values());
   855→  }

  1047→        totalEmbeddings: 0,
  1048→        totalNamespaces: 0,
  1049→        embeddingsByNamespace: {},
  1050→        embeddingsBySourceType: {},

  1052→    }
  1053→    const table = await getEmbeddingsTable();

  1055→      return {
  1056→        totalEmbeddings: 0,
  1057→        totalNamespaces: 0,

  1059→        embeddingsBySourceType: {},
  1060→      };
  1061→    }

  1063→

  1065→    const embeddingsBySourceType: Record<string, number> = {};
  1066→    let totalEmbeddings = 0;
  1067→
  1068→    // Count per-namespace to avoid loading all vectors into memory at once
  1069→    for (const ns of namespaces) {
  1070→      try {
  1071→        const count = await this.countByNamespace(ns.namespace);
  1072→        if (count > 0) {
  1073→          embeddingsByNamespace[ns.namespace] = count;
  1074→          totalEmbeddings += count;
  1075→        }
  1076→      } catch { /* skip */ }
  1077→    }
  1078→
  1079→    // Also count 'default' namespace if not listed
  1080→    if (!embeddingsByNamespace['default']) {
  1081→      try {
  1082→        const defaultCount = await this.countByNamespace('default');
  1083→        if (defaultCount > 0) {
  1084→          embeddingsByNamespace['default'] = defaultCount;
  1085→          totalEmbeddings += defaultCount;
  1086→        }
  1087→      } catch { /* skip */ }
  1088→    }
  1089→
  1090→    // For source_type breakdown, we need to scan but we can use lightweight queries
  1091→    // Only load id + source_type (not full vectors) via filter
  1092→    const sourceTypes = ['character', 'world', 'lorebook', 'session', 'memory', 'custom', 'system'];
  1093→    for (const st of sourceTypes) {
  1094→      try {
  1095→        const matching = await tableFilter(table, `source_type = '${st}'`);
  1096→        if (matching.length > 0) {
  1097→          embeddingsBySourceType[st] = matching.length;
  1098→        }
  1099→      } catch { /* skip */ }
  1100→    }
  1101→
  1102→    return {
  1103→      totalEmbeddings,
  1104→      totalNamespaces: namespaces.length,
  1105→      embeddingsByNamespace,
  1106→      embeddingsBySourceType,
  1107→    };
  1108→  }

  1111→    if (isPermanentlyUnavailable) return { deletedEmbeddings: 0, deletedNamespaces: 0 };
  1112→    // Count before dropping to report accurate numbers

  1114→    let deletedNamespaces = 0;

  1118→      deletedEmbeddings = allEmbeddings.length;

  1120→
  1121→    try {
  1122→      const allNamespaces = await tableToArray(await getNamespacesTable());
  1123→      deletedNamespaces = allNamespaces.length;
  1124→    } catch { /* table may not exist */ }
  1125→
  1126→    // Drop and recreate the embeddings table (much faster than individual deletes)
  1127→    if (db) {
  1128→      try { await db.dropTable(EMBEDDINGS_TABLE); } catch { /* ignore */ }
  1129→      try { await db.dropTable(NAMESPACES_TABLE); } catch { /* ignore */ }
  1130→
  1131→      // Drop any extra tables
  1132→      try {

  1134→        for (const tableName of tables) {
  1135→          if (tableName !== EMBEDDINGS_TABLE && tableName !== NAMESPACES_TABLE) {
  1136→            try { await db.dropTable(tableName); } catch { /* ignore */ }
  1137→          }
  1138→        }
  1139→      } catch { /* ignore */ }
  1140→
  1141→      // Reinitialize with fresh tables
  1142→      await initializeTables();
