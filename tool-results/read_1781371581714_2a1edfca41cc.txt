    10→// PNG tEXt chunk header for character cards
    11→const PNG_CHUNK_HEADER = 'tEXt';
    12→const TAVERN_CHAR_KEY = 'chara';
    13→
    14→/**
    15→ * Parse a PNG file to extract embedded character card data
    16→ * Character cards are stored in tEXt chunks with key "chara"
    17→ * The value is Base64 encoded JSON
    18→ */
    19→export async function parseCharacterCardFromPng(file: File): Promise<{
    20→  character: Partial<CharacterCard>;
    21→  avatar: string;
    22→} | null> {
    23→  try {
    24→    // Read the file as ArrayBuffer
    25→    const buffer = await file.arrayBuffer();
    26→    const view = new DataView(buffer);
    27→    
    28→    // Verify PNG signature
    29→    const pngSignature = [0x89, 0x50, 0x4E, 0x47, 0x0D, 0x0A, 0x1A, 0x0A];
    30→    for (let i = 0; i < 8; i++) {
    31→      if (view.getUint8(i) !== pngSignature[i]) {
    32→        console.error('Invalid PNG file');
    33→        return null;
    34→      }
    35→    }
    36→    
    37→    // Parse PNG chunks
    38→    let offset = 8; // Skip PNG signature
    39→    let characterData: string | null = null;
    40→    
    41→    while (offset < buffer.byteLength) {
    42→      const chunkLength = view.getUint32(offset);
    43→      offset += 4;
    44→      
    45→      const chunkType = String.fromCharCode(
    46→        view.getUint8(offset),
    47→        view.getUint8(offset + 1),
    48→        view.getUint8(offset + 2),
    49→        view.getUint8(offset + 3)
    50→      );
    51→      offset += 4;
    52→      
    53→      if (chunkType === PNG_CHUNK_HEADER) {
    54→        // Read chunk data
    55→        const chunkData = new Uint8Array(buffer, offset, chunkLength);
    56→        
    57→        // Find the null separator between key and value
    58→        let separatorIndex = -1;
    59→        for (let i = 0; i < chunkData.length; i++) {
    60→          if (chunkData[i] === 0) {
    61→            separatorIndex = i;
    62→            break;
    63→          }
    64→        }
    65→        
    66→        if (separatorIndex > 0) {
    67→          const key = new TextDecoder().decode(chunkData.slice(0, separatorIndex));
    68→          
    69→          if (key === TAVERN_CHAR_KEY) {
    70→            const value = new TextDecoder().decode(chunkData.slice(separatorIndex + 1));
    71→            characterData = value;
    72→          }
    73→        }
    74→      }
    75→      
    76→      // Skip chunk data and CRC
    77→      offset += chunkLength + 4;
    78→      
    79→      // Stop at IEND chunk
    80→      if (chunkType === 'IEND') break;
    81→    }
    82→    
    83→    if (!characterData) {
    84→      console.log('No character data found in PNG');
    85→      return null;
    86→    }
    87→    
    88→    // Decode Base64 and parse JSON (UTF-8 safe)
    89→    const binaryStr = atob(characterData);
    90→    const bytes = Uint8Array.from(binaryStr, c => c.charCodeAt(0));
    91→    const jsonStr = new TextDecoder().decode(bytes);
    92→    const jsonData = JSON.parse(jsonStr);
    93→    
    94→    // Parse the character data
    95→    const character = parseCharacterData(jsonData);
    96→    
    97→    // Create avatar data URL from the PNG
    98→    const avatar = await fileToDataUrl(file);
    99→    
   100→    return { character, avatar };
   101→  } catch (error) {
   102→    console.error('Error parsing character card:', error);
   103→    return null;
   104→  }
   105→}

   222→/**
   223→ * Convert File to Data URL (for importing)
   224→ */
   225→async function fileToDataUrl(file: File): Promise<string> {
   226→  return new Promise((resolve, reject) => {
   227→    const reader = new FileReader();
   228→    reader.onload = () => resolve(reader.result as string);
   229→    reader.onerror = reject;
   230→    reader.readAsDataURL(file);
   231→  });
   232→}

   234→/**
   235→ * Upload an image to the server and return the URL
   236→ */
   237→async function uploadImage(file: File, type: string = 'avatar'): Promise<string> {
   238→  const formData = new FormData();
   239→  formData.append('file', file);
   240→  formData.append('type', type);
   241→
   242→  const response = await fetch('/api/upload', {
   243→    method: 'POST',
   244→    body: formData
   245→  });
   246→
   247→  if (!response.ok) {
   248→    const error = await response.json();
   249→    throw new Error(error.error || 'Failed to upload image');
   250→  }
   251→
   252→  const { url } = await response.json();
   253→  return url;
   254→}

   256→/**
   257→ * Upload a data URL to the server and return the URL
   258→ */
   259→async function uploadDataUrl(dataUrl: string, type: string = 'avatar'): Promise<string> {
   260→  // Convert data URL to Blob
   261→  const response = await fetch(dataUrl);
   262→  const blob = await response.blob();
   263→  
   264→  // Create a File object
   265→  const file = new File([blob], `imported-${Date.now()}.png`, { type: blob.type || 'image/png' });
   266→  
   267→  return uploadImage(file, type);
   268→}

   273→export async function parseCharacterCardFromJson(file: File): Promise<{

   295→/**
   296→ * Import a character card from file (PNG or JSON)
   297→ * 
   298→ * After import, automatically migrates legacy sprite data to V2 if:
   299→ * - The character has legacy sprite data (sprites[], spriteConfig, spriteTriggers)
   300→ * - No V2 data exists yet (spritePacksV2, stateCollectionsV2, triggerCollections)
   301→ */
   302→export async function importCharacterCard(file: File): Promise<{
   303→  character: Partial<CharacterCard>;
   304→  avatar: string;
   305→} | null> {
   306→  const extension = file.name.split('.').pop()?.toLowerCase();
   307→  
   308→  try {
   309→    let importResult: { character: Partial<CharacterCard>; avatar: string } | null = null;
   310→
   311→    if (extension === 'png' || file.type === 'image/png') {
   312→      const result = await parseCharacterCardFromPng(file);
   313→      if (result) {
   314→        // Upload the avatar to the server instead of storing base64
   315→        const avatarUrl = await uploadDataUrl(result.avatar, 'avatar');
   316→        importResult = {
   317→          character: result.character,
   318→          avatar: avatarUrl
   319→        };
   320→      }
   321→    } else if (extension === 'json' || file.type === 'application/json') {
   322→      const result = await parseCharacterCardFromJson(file);
   323→      if (result) {
   324→        let avatarUrl = '';
   325→        if (result.avatar) {
   326→          // If avatar is a data URL, upload it
   327→          if (result.avatar.startsWith('data:')) {
   328→            avatarUrl = await uploadDataUrl(result.avatar, 'avatar');
   329→          } else {
   330→            avatarUrl = result.avatar;
   331→          }
   332→        }
   333→        importResult = {
   334→          character: result.character,
   335→          avatar: avatarUrl
   336→        };
   337→      }
   338→    }
   339→
   340→    // Auto-migrate legacy sprite data after import
   341→    if (importResult?.character) {
   342→      importResult.character = autoMigrateOnImport(importResult.character);
   343→    }
   344→
   345→    return importResult;
   346→  } catch (error) {
   347→    console.error('Error importing character card:', error);
   348→    return null;
   349→  }
   350→}

   352→/**
   353→ * Auto-migrate legacy sprite data to V2 on import.
   354→ * Only runs if legacy data exists and V2 data doesn't.
   355→ * Preserves all existing V2 data.
   356→ * 
   357→ * @deprecated This auto-migration is temporary during the deprecation period.
   358→ * Eventually, legacy fields will be removed and this won't be needed.
   359→ */
   360→function autoMigrateOnImport(character: Partial<CharacterCard>): Partial<CharacterCard> {
   361→  // We need enough data to check for migration
   362→  const minimalCard = character as CharacterCard;
   363→  
   364→  if (!needsMigration(minimalCard)) {
   365→    return character;
   366→  }
   367→
   368→  try {
   369→    const result = migrateCharacterSprites(minimalCard, {
   370→      defaultPackName: `${character.name || 'Character'} - Migrated`,
   371→      createDefaultStateCollections: true,
   372→      skipIfV2Exists: true, // Don't overwrite any V2 data that might have been imported
   373→    });
   374→
   375→    if (result.success && (result.report.packsCreated > 0 || result.report.triggerCollectionsCreated > 0 || result.report.stateCollectionsCreated > 0)) {
   376→      const migrationUpdates = applyMigrationResult(minimalCard, result);
   377→      return {
   378→        ...character,
   379→        ...migrationUpdates,
   380→      };
   381→    }
   382→  } catch (error) {
   383→    // Silently fail - migration is best-effort on import
   384→    console.warn('[TavernFlow] Auto-migration on import failed:', error);
   385→  }
   386→
   387→  return character;
   388→}

   472→/**
   473→ * Create a PNG blob with embedded character data
   474→ */
   475→async function createPngWithCharacterData(
   476→  data: CharacterCardV2,
   477→  imageBlob: Blob | null
   478→): Promise<Blob> {
   479→  // If we have an image, inject the character data
   480→  if (imageBlob) {
   481→    const buffer = await imageBlob.arrayBuffer();
   482→    return injectCharacterDataIntoPng(buffer, data);
   483→  }
   484→  
   485→  // Create a minimal 1x1 PNG with the character data
   486→  const width = 1;
   487→  const height = 1;
   488→  
   489→  // Create PNG chunks
   490→  const signature = new Uint8Array([0x89, 0x50, 0x4E, 0x47, 0x0D, 0x0A, 0x1A, 0x0A]);
   491→  
   492→  // IHDR chunk
   493→  const ihdrData = new Uint8Array(13);
   494→  const ihdrView = new DataView(ihdrData.buffer);
   495→  ihdrView.setUint32(0, width);
   496→  ihdrView.setUint32(4, height);
   497→  ihdrView.setUint8(8, 8);  // bit depth
   498→  ihdrView.setUint8(9, 6);  // color type (RGBA)
   499→  ihdrView.setUint8(10, 0); // compression
   500→  ihdrView.setUint8(11, 0); // filter
   501→  ihdrView.setUint8(12, 0); // interlace
   502→  const ihdrChunk = createPngChunk('IHDR', ihdrData);
   503→  
   504→  // tEXt chunk with character data (UTF-8 safe Base64)
   505→  const jsonStr = JSON.stringify(data);
   506→  const jsonBytes = new TextEncoder().encode(jsonStr);
   507→  const binaryStr = String.fromCharCode(...jsonBytes);
   508→  const base64Str = btoa(binaryStr);
   509→  const textData = new TextEncoder().encode(`${TAVERN_CHAR_KEY}\x00${base64Str}`);
   510→  const textChunk = createPngChunk('tEXt', textData);
   511→  
   512→  // IDAT chunk (minimal compressed RGBA data)
   513→  const idatData = new Uint8Array([0x78, 0x9C, 0x62, 0x60, 0x60, 0x60, 0x00, 0x00, 0x00, 0x04, 0x00, 0x01]);
   514→  const idatChunk = createPngChunk('IDAT', idatData);
   515→  
   516→  // IEND chunk
   517→  const iendChunk = createPngChunk('IEND', new Uint8Array(0));
   518→  
   519→  // Combine all chunks
   520→  const totalLength = signature.length + ihdrChunk.length + textChunk.length + idatChunk.length + iendChunk.length;
   521→  const pngBuffer = new Uint8Array(totalLength);
   522→  let offset = 0;
   523→  
   524→  pngBuffer.set(signature, offset);
   525→  offset += signature.length;
   526→  pngBuffer.set(ihdrChunk, offset);
   527→  offset += ihdrChunk.length;
   528→  pngBuffer.set(textChunk, offset);
   529→  offset += textChunk.length;
   530→  pngBuffer.set(idatChunk, offset);
   531→  offset += idatChunk.length;
   532→  pngBuffer.set(iendChunk, offset);
   533→  
   534→  return new Blob([pngBuffer], { type: 'image/png' });
   535→}

   537→/**
   538→ * Inject character data into an existing PNG
   539→ */
   540→async function injectCharacterDataIntoPng(
   541→  buffer: ArrayBuffer,
   542→  data: CharacterCardV2
   543→): Promise<Blob> {
   544→  const view = new DataView(buffer);
   545→  const pngChunks: { type: string; data: Uint8Array }[] = [];
   546→  
   547→  // Parse existing chunks
   548→  let offset = 8; // Skip signature
   549→  
   550→  while (offset < buffer.byteLength) {
   551→    const chunkLength = view.getUint32(offset);
   552→    const chunkType = String.fromCharCode(
   553→      view.getUint8(offset + 4),
   554→      view.getUint8(offset + 5),
   555→      view.getUint8(offset + 6),
   556→      view.getUint8(offset + 7)
   557→    );
   558→    
   559→    const chunkData = new Uint8Array(buffer, offset + 8, chunkLength);
   560→    pngChunks.push({ type: chunkType, data: chunkData });
   561→    
   562→    offset += 12 + chunkLength; // length + type + data + CRC
   563→    
   564→    if (chunkType === 'IEND') break;
   565→  }
   566→  
   567→  // Remove existing tEXt chara chunks
   568→  const filteredChunks = pngChunks.filter(c => 
   569→    !(c.type === 'tEXt' && isCharaChunk(c.data))
   570→  );
   571→  
   572→  // Create new tEXt chunk with character data (UTF-8 safe Base64)
   573→  const jsonStr = JSON.stringify(data);
   574→  const jsonBytes = new TextEncoder().encode(jsonStr);
   575→  const binaryStr = String.fromCharCode(...jsonBytes);
   576→  const base64Str = btoa(binaryStr);
   577→  const textData = new TextEncoder().encode(`${TAVERN_CHAR_KEY}\x00${base64Str}`);
   578→  
   579→  // Insert before IEND
   580→  const iendIndex = filteredChunks.findIndex(c => c.type === 'IEND');
   581→  const newChunks = [
   582→    ...filteredChunks.slice(0, iendIndex),
   583→    { type: 'tEXt', data: textData },
   584→    ...filteredChunks.slice(iendIndex)
   585→  ];
   586→  
   587→  // Rebuild PNG
   588→  const signature = new Uint8Array([0x89, 0x50, 0x4E, 0x47, 0x0D, 0x0A, 0x1A, 0x0A]);
   589→  const chunksData = newChunks.map(c => createPngChunk(c.type, c.data));
   590→  
   591→  const totalLength = signature.length + chunksData.reduce((sum, c) => sum + c.length, 0);
   592→  const pngBuffer = new Uint8Array(totalLength);
   593→  
   594→  pngBuffer.set(signature, 0);
   595→  let pos = signature.length;
   596→  for (const chunk of chunksData) {
   597→    pngBuffer.set(chunk, pos);
   598→    pos += chunk.length;
   599→  }
   600→  
   601→  return new Blob([pngBuffer], { type: 'image/png' });
   602→}
   603→
   604→/**
   605→ * Check if a tEXt chunk is a chara chunk
   606→ */
   607→function isCharaChunk(data: Uint8Array): boolean {
   608→  for (let i = 0; i < data.length; i++) {
   609→    if (data[i] === 0) {
   610→      const key = new TextDecoder().decode(data.slice(0, i));
   611→      return key === TAVERN_CHAR_KEY;
   612→    }
   613→  }
   614→  return false;
   615→}
   616→
   617→/**
   618→ * Create a PNG chunk with CRC
   619→ */
   620→function createPngChunk(type: string, data: Uint8Array): Uint8Array {
   621→  const typeBytes = new TextEncoder().encode(type);
   622→  const chunk = new Uint8Array(12 + data.length);
   623→  const view = new DataView(chunk.buffer);
   624→  
   625→  // Length (4 bytes, big-endian)
   626→  view.setUint32(0, data.length);
   627→  
   628→  // Type (4 bytes)
   629→  chunk.set(typeBytes, 4);
   630→  
   631→  // Data
   632→  chunk.set(data, 8);
   633→  
   634→  // CRC (4 bytes) - calculated over type + data
   635→  const crc = crc32(new Uint8Array([...typeBytes, ...data]));
   636→  view.setUint32(8 + data.length, crc);
   637→  
   638→  return chunk;
   639→}
   640→
   641→/**
   642→ * CRC32 calculation for PNG chunks
   643→ */
   644→function crc32(data: Uint8Array): number {
   645→  let crc = 0xFFFFFFFF;
   646→  const table = getCrc32Table();
   647→  
   648→  for (let i = 0; i < data.length; i++) {
   649→    crc = (crc >>> 8) ^ table[(crc ^ data[i]) & 0xFF];
   650→  }
   651→  
   652→  return (crc ^ 0xFFFFFFFF) >>> 0;
   653→}
   654→
   655→let crc32Table: Uint32Array | null = null;
   656→
   657→function getCrc32Table(): Uint32Array {
   658→  if (crc32Table) return crc32Table;
   659→  
   660→  crc32Table = new Uint32Array(256);
   661→  for (let i = 0; i < 256; i++) {
   662→    let c = i;
   663→    for (let j = 0; j < 8; j++) {
   664→      c = (c & 1) ? (0xEDB88320 ^ (c >>> 1)) : (c >>> 1);
   665→    }
   666→    crc32Table[i] = c;
   667→  }
   668→  
   669→  return crc32Table;
   670→}
