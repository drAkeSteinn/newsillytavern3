   127→// Format memory date to relative time
   128→function formatMemoryDate(date: Date): string {
   129→  const now = new Date();
   130→  const diff = now.getTime() - date.getTime();
   131→  const minutes = Math.floor(diff / 60000);
   132→  if (minutes < 1) return 'Ahora';
   133→  if (minutes < 60) return `Hace ${minutes}m`;
   134→  const hours = Math.floor(minutes / 60);
   135→  if (hours < 24) return `Hace ${hours}h`;
   136→  const days = Math.floor(hours / 24);
   137→  if (days < 7) return `Hace ${days}d`;
   138→  return date.toLocaleDateString('es-ES', { day: 'numeric', month: 'short' });
   139→}

   164→      {/* Type indicator bar */}

   167→      <div className="flex-1 min-w-0">

   179→              Consolidada
   180→            </span>
   181→          )}

   183→            {createdDate ? formatMemoryDate(createdDate) : ''}
   184→          </span>
   185→        </div>
   186→
   187→        {/* Memory content */}
   188→        <p className="text-xs leading-relaxed text-foreground/90 line-clamp-3">
   189→          {memory.content}
   190→        </p>
   191→      </div>
   192→
   193→      {/* Delete button (appears on hover) */}

   195→        onClick={() => onDelete(memory.id)}
