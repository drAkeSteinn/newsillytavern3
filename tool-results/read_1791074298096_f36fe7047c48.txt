    89→/**
    90→ * Build a HUD context section for prompt injection
    91→ * 
    92→ * Now resolves ALL keys including:
    93→ * - Template variables: {{user}}, {{char}}, {{userpersona}}
    94→ * - Stats keys: {{resistencia}}, {{habilidades}}, etc.
    95→ */
    96→export function buildHUDContextSection(
    97→  contextConfig: HUDContextConfig,
    98→  keyContext?: KeyResolutionContext
    99→): PromptSection | null {
   100→  if (!contextConfig.enabled || !contextConfig.content.trim()) {
   101→    return null;
   102→  }
   103→
   104→  // Resolve all keys in the context content
   105→  let resolvedContent = contextConfig.content;
   106→  if (keyContext) {
   107→    resolvedContent = resolveAllKeys(contextConfig.content, keyContext);
   108→  }
   109→
   110→  return {
   111→    type: 'hud_context',
   112→    label: 'HUD Context',
   113→    content: resolvedContent,
   114→    color: SECTION_COLORS.hud_context
   115→  };
   116→}
   117→
   118→/**
   119→ * Inject HUD context at the specified position
   120→ *
   121→ * Positions:
   122→ * 0 = After system prompt
   123→ * 1 = After user message (after last user message)
   124→ * 2 = Before user message (before last user message)
   125→ * 3 = After assistant message (after last assistant message)
   126→ * 4 = Before assistant message (before last assistant message)
   127→ * 5 = At top of chat (before chat history)
   128→ * 6 = At bottom of chat (after all messages)
   129→ * 7 = After lorebook / Author's Note position
   130→ */
   131→export function injectHUDContextIntoMessages(
   132→  messages: ChatApiMessage[],
   133→  contextSection: PromptSection,
   134→  position: number
   135→): ChatApiMessage[] {
   136→  if (!contextSection.content.trim()) {
   137→    return messages;
   138→  }
   139→
   140→  const contextContent = `[${contextSection.label}]\n${contextSection.content}`;
   141→  const result: ChatApiMessage[] = [...messages];
   142→
   143→  switch (position) {
   144→    case 0: // After system prompt
   145→      // Find the first system/assistant message (the system prompt)
   146→      const sysIdx = result.findIndex(m => m.role === 'assistant' || m.role === 'system');
   147→      if (sysIdx >= 0) {
   148→        result[sysIdx] = {
   149→          ...result[sysIdx],
   150→          content: result[sysIdx].content + '\n\n' + contextContent
   151→        };
   152→      }
   153→      break;
   154→
   155→    case 1: // After user message
   156→      // Find the last user message
   157→      const lastUserIdx = result.map((m, i) => m.role === 'user' ? i : -1).filter(i => i >= 0).pop();
   158→      if (lastUserIdx !== undefined && lastUserIdx >= 0) {
   159→        result[lastUserIdx] = {
   160→          ...result[lastUserIdx],
   161→          content: result[lastUserIdx].content + '\n\n' + contextContent
   162→        };
   163→      }
   164→      break;
   165→
   166→    case 2: // Before user message
   167→      const lastUserIdx2 = result.map((m, i) => m.role === 'user' ? i : -1).filter(i => i >= 0).pop();
   168→      if (lastUserIdx2 !== undefined && lastUserIdx2 >= 0) {
   169→        result[lastUserIdx2] = {
   170→          ...result[lastUserIdx2],
   171→          content: contextContent + '\n\n' + result[lastUserIdx2].content
   172→        };
   173→      }
   174→      break;
   175→
   176→    case 3: // After assistant message
   177→      const lastAsstIdx = result.map((m, i) => m.role === 'assistant' ? i : -1).filter(i => i >= 0).pop();
   178→      if (lastAsstIdx !== undefined && lastAsstIdx >= 0) {
   179→        result[lastAsstIdx] = {
   180→          ...result[lastAsstIdx],
   181→          content: result[lastAsstIdx].content + '\n\n' + contextContent
   182→        };
   183→      }
   184→      break;
   185→
   186→    case 4: // Before assistant message
   187→      const lastAsstIdx4 = result.map((m, i) => m.role === 'assistant' ? i : -1).filter(i => i >= 0).pop();
   188→      if (lastAsstIdx4 !== undefined && lastAsstIdx4 >= 0) {
   189→        result[lastAsstIdx4] = {
   190→          ...result[lastAsstIdx4],
   191→          content: contextContent + '\n\n' + result[lastAsstIdx4].content
   192→        };
   193→      }
   194→      break;
   195→
   196→    case 5: // At top of chat (after system, before first message)
   197→      // Merge into the first system message to avoid breaking alternation
   198→      {
   199→        const sysMsgIdx5 = result.findIndex(m => m.role === 'system');
   200→        if (sysMsgIdx5 >= 0) {
   201→          result[sysMsgIdx5] = {
   202→            ...result[sysMsgIdx5],
   203→            content: result[sysMsgIdx5].content + '\n\n' + contextContent
   204→          };
   205→        }
   206→      }
   207→      break;
   208→
   209→    case 6: // At bottom of chat (after all messages)
   210→      // Merge into the first system message to avoid breaking alternation
   211→      {
   212→        const sysMsgIdx6 = result.findIndex(m => m.role === 'system');
   213→        if (sysMsgIdx6 >= 0) {
   214→          result[sysMsgIdx6] = {
   215→            ...result[sysMsgIdx6],
   216→            content: result[sysMsgIdx6].content + '\n\n' + contextContent
   217→          };
   218→        }
   219→      }
   220→      break;
   221→
   222→    case 7: // After lorebook / Author's Note position
   223→      // Merge into the first system message to avoid breaking alternation
   224→      {
   225→        const sysMsgIdx7 = result.findIndex(m => m.role === 'system' || m.role === 'assistant');
   226→        if (sysMsgIdx7 >= 0) {
   227→          result[sysMsgIdx7] = {
   228→            ...result[sysMsgIdx7],
   229→            content: result[sysMsgIdx7].content + '\n\n' + contextContent
   230→          };
   231→        }
   232→      }
   233→      break;
   234→
   235→    default:
   236→      // Merge into the first system message to avoid breaking alternation
   237→      {
   238→        const sysMsgIdxDef = result.findIndex(m => m.role === 'system');
   239→        if (sysMsgIdxDef >= 0) {
   240→          result[sysMsgIdxDef] = {
   241→            ...result[sysMsgIdxDef],
   242→            content: result[sysMsgIdxDef].content + '\n\n' + contextContent
   243→          };
   244→        } else {
   245→          result.push({
   246→            role: 'system',
   247→            content: contextContent
   248→          });
   249→        }
   250→      }
   251→      break;
   252→  }
   253→
   254→  return result;
   255→}

   257→/**
   258→ * Inject HUD context into prompt sections
   259→ * This is for the prompt viewer display
   260→ *
   261→ * Positions:
   262→ * 0 = After system prompt
   263→ * 1 = After user message (not applicable to sections, uses chat_history)
   264→ * 2 = Before user message (not applicable to sections, uses chat_history)
   265→ * 3 = After assistant message (not applicable to sections, uses chat_history)
   266→ * 4 = Before assistant message (not applicable to sections, uses chat_history)
   267→ * 5 = At top of chat (before chat history)
   268→ * 6 = At bottom of chat (after all sections)
   269→ * 7 = After lorebook
   270→ */
   271→export function injectHUDContextIntoSections(
   272→  sections: PromptSection[],
   273→  contextSection: PromptSection,
   274→  position: number
   275→): PromptSection[] {
   276→  if (!contextSection.content.trim()) {
   277→    return sections;
   278→  }
   279→
   280→  const result: PromptSection[] = [...sections];
   281→
   282→  switch (position) {
   283→    case 0: // After system prompt
   284→      const sysSectionIdx = result.findIndex(s => s.type === 'system');
   285→      if (sysSectionIdx >= 0) {
   286→        result.splice(sysSectionIdx + 1, 0, contextSection);
   287→      } else {
   288→        result.unshift(contextSection);
   289→      }
   290→      break;
   291→
   292→    case 1: // After user message - inject before chat history
   293→    case 2: // Before user message - inject before chat history
   294→    case 3: // After assistant message - inject before chat history
   295→    case 4: // Before assistant message - inject before chat history
   296→      // These positions apply to messages, for sections we inject before chat history
   297→      const chatHistoryIdx = result.findIndex(s => s.type === 'chat_history');
   298→      if (chatHistoryIdx >= 0) {
   299→        result.splice(chatHistoryIdx, 0, contextSection);
   300→      } else {
   301→        result.push(contextSection);
   302→      }
   303→      break;
   304→
   305→    case 5: // At top of chat (before chat history)
   306→      const chatIdx = result.findIndex(s => s.type === 'chat_history');
   307→      if (chatIdx >= 0) {
   308→        result.splice(chatIdx, 0, contextSection);
   309→      } else {
   310→        result.push(contextSection);
   311→      }
   312→      break;
   313→
   314→    case 7: // After lorebook
   315→      const lorebookIdx = result.findIndex(s => s.type === 'lorebook');
   316→      if (lorebookIdx >= 0) {
   317→        result.splice(lorebookIdx + 1, 0, contextSection);
   318→      } else {
   319→        // If no lorebook, insert after system prompt
   320→        const sysIdx = result.findIndex(s => s.type === 'system');
   321→        if (sysIdx >= 0) {
   322→          result.splice(sysIdx + 1, 0, contextSection);
   323→        } else {
   324→          result.unshift(contextSection);
   325→        }
   326→      }
   327→      break;
   328→
   329→    case 6: // At bottom of chat
   330→    default:
   331→      result.push(contextSection);
   332→      break;
   333→  }
   334→
   335→  return result;
   336→}

   342→/**
   343→ * Build an inventory section for prompt injection.
   344→ *
   345→ * Shows:
   346→ * - Current currency
   347→ * - Equipped items and their effects
   348→ * - Active consumable effects with remaining turns
   349→ * - Available (non-equipped) items in inventory
   350→ *
   351→ * Uses the `promptTemplate` from `inventorySettings` for formatting.
   352→ * If no template is provided or the template is empty, a default format is used.
   353→ *
   354→ * Returns null if inventory data is empty or invalid.
   355→ */
   356→export function buildInventorySection(
   357→  inventoryData: InventoryPromptData,
   358→  keyContext?: KeyResolutionContext
   359→): PromptSection | null {
   360→  const {
   361→    personaItems,
   362→    equippedItems,
   363→    sessionEquipment,
   364→    activeEffects,
   365→    currency,
   366→    currencyName,
   367→    currencyIcon,
   368→    inventorySettings,
   369→  } = inventoryData;
   370→
   371→  // If no items and no effects and no currency, skip section
   372→  if (personaItems.length === 0 && activeEffects.length === 0 && currency === 0) {
   373→    return null;
   374→  }
   375→
   376→  // Use the template from inventory settings, or fall back to default
   377→  const template = inventorySettings.promptTemplate?.trim() || `[Inventario Activo]
   378→{{activeItems}}
   379→
   380→[Efectos Activos]
   381→{{activeEffects}}
   382→
   383→[Divisa]
   384→{{currency}}`;
   385→
   386→  // Build items list (all items in inventory)
   387→  const itemLines = personaItems.map(({ entry, item }) => {
   388→    const qty = entry.quantity > 1 ? ` x${entry.quantity}` : '';
   389→    const eq = entry.equipped ? ' [Equipado]' : '';
   390→    const effects = (item.attributeEffects && item.attributeEffects.length > 0)
   391→      ? ` (${item.attributeEffects.map(e => `${e.operator}${e.value} ${e.attributeKey}`).join(', ')})`
   392→      : '';
   393→    return `- ${item.icon || ''} ${item.name}${qty}${eq}${effects}`;
   394→  }).join('\n');
   395→
   396→  // Build active effects list
   397→  const effectLines = activeEffects.map(e => {
   398→    const turnsLeft = e.remainingTurns > 0 ? ` (${e.remainingTurns}/${e.totalTurns} turnos)` : '';
   399→    // Use consumableEffect free text if available, otherwise fall back to attribute effects
   400→    let effectDesc: string;
   401→    if (e.consumableEffect) {
   402→      effectDesc = e.consumableEffect;
   403→    } else {
   404→      effectDesc = e.effects.map(ef =>
   405→        `${ef.operator}${ef.value} ${ef.attributeKey}${ef.targetId !== '__user__' ? ` → ${ef.targetName || ef.targetId}` : ''}`
   406→      ).join(', ');
   407→    }
   408→    return `- ${e.itemName}: ${effectDesc}${turnsLeft}`;
   409→  }).join('\n');
   410→
   411→  // Build equipped items list - prefer sessionEquipment (per-session), fallback to equippedItems (legacy)
   412→  let equipLines = '';
   413→  if (sessionEquipment && sessionEquipment.length > 0) {
   414→    // Use session equipment data
   415→    const items = personaItems.map(({ item }) => item);
   416→    equipLines = sessionEquipment.map(eq => {
   417→      const item = items.find(i => i.id === eq.itemId);
   418→      const slotDef = inventorySettings.equipmentSlots?.find(s => s.id === eq.equippedSlotId);
   419→      const slotLabel = slotDef?.name || eq.equippedSlotId;
   420→      const effectText = eq.slotEffectText
   421→        || item?.slotEffects?.find(se => se.slotId === eq.equippedSlotId)?.effectText
   422→        || '';
   423→      return `- ${item?.icon || ''} ${item?.name || '???'} [${slotLabel}]${effectText ? ` → ${effectText}` : ''}`;
   424→    }).join('\n');
   425→  } else if (equippedItems && equippedItems.length > 0) {
   426→    // Fallback to legacy equipped items
   427→    equipLines = equippedItems.map(({ item }) => {
   428→      // Prefer slotEffects (V3) over attributeEffects (V2)
   429→      let effects = '';
   430→      if (item.slotEffects && item.slotEffects.length > 0) {
   431→        effects = item.slotEffects.map(se => se.effectText).filter(Boolean).join('; ');
   432→      } else if (item.attributeEffects && item.attributeEffects.length > 0) {
   433→        effects = item.attributeEffects.map(e => `${e.operator}${e.value} ${e.attributeKey}`).join(', ');
   434→      }
   435→      const slotDef = inventorySettings.equipmentSlots?.find(s => s.id === item.slot);
   436→      const slotLabel = slotDef?.name || item.slot || '';
   437→      return `- ${item.icon || ''} ${item.name}${slotLabel ? ` [${slotLabel}]` : ''}${effects ? ` → ${effects}` : ''}`;
   438→    }).join('\n');
   439→  }
   440→
   441→  // Build currency line
   442→  const currencyLine = `${currencyIcon || '💰'} ${currencyName || 'Divisa'}: ${currency}`;
   443→
   444→  // Apply template replacements
   445→  let content = template
   446→    .replace('{{activeItems}}', itemLines || 'Vacío')
   447→    .replace('{{activeEffects}}', effectLines || 'Ninguno')
   448→    .replace('{{equippedItems}}', equipLines || 'Ninguno')
   449→    .replace('{{currency}}', currencyLine);
   450→
   451→  // Resolve all keys in the inventory content if keyContext is available
   452→  if (keyContext) {
   453→    content = resolveAllKeys(content, keyContext);
   454→  }
   455→
   456→  return {
   457→    type: 'inventory',
   458→    label: 'Inventory',
   459→    content,
   460→    color: SECTION_COLORS.inventory
   461→  };
   462→}

   499→/**
   500→ * Build the system prompt from character data (SillyTavern style)
   501→ *
   502→ * Uses unified key resolution for ALL sections:
   503→ * - Template variables: {{user}}, {{char}}, {{userpersona}}, etc.
   504→ * - Stats keys: {{resistencia}}, {{habilidades}}, etc.
   505→ * - Sound keys: {{sonidos}}
   506→ */
   507→export function buildSystemPrompt(
   508→  character: CharacterCard,
   509→  userName: string = 'User',
   510→  persona?: Persona,
   511→  lorebookPlan?: LorebookInjectionPlan | null,
   512→  sessionStats?: SessionStats,
   513→  allCharacters?: CharacterCard[],
   514→  soundTriggers?: SoundTrigger[],
   515→  soundSettings?: AppSettings['sound'],
   516→  questTemplates?: QuestTemplate[],
   517→  sessionQuests?: SessionQuestInstance[],
   518→  questSettings?: QuestSettings,
   519→  lorebookAttributeKeys?: Record<string, string>,
   520→  inventoryData?: InventoryPromptData,
   521→  lorebookEntryKeyMap?: Record<string, string>
   522→): { prompt: string; sections: PromptSection[]; lorebookChatInjections: LorebookChatInjection[]; exampleMessages: ChatApiMessage[] } {
   523→  const sections: PromptSection[] = [];
   524→
   525→  // Resolve stats for the persona FIRST (user attributes like {{resistencia}})
   526→  // This must be done before character stats so personaResolvedStats is available
   527→  let personaResolvedStats: ResolvedStats | null = null;
   528→  if (persona?.statsConfig?.enabled && sessionStats) {
   529→    personaResolvedStats = resolveStats({
   530→      characterId: '__user__',
   531→      statsConfig: persona.statsConfig,
   532→      sessionStats,
   533→      lorebookEntryKeys: lorebookEntryKeyMap,
   534→    });
   535→  }
   536→
   537→  // Resolve stats for this character (includes skills block with full key resolution)
   538→  const resolvedStats = resolveStats({
   539→    characterId: character.id,
   540→    statsConfig: character.statsConfig,
   541→    sessionStats,
   542→    allCharacters,
   543→    userName,
   544→    characterName: character.name,
   545→    questTemplates,
   546→    personaDescription: persona?.description,
   547→    personaResolvedStats,
   548→    lorebookEntryKeys: lorebookEntryKeyMap,
   549→  });
   550→
   551→  // Build outlet sections map from lorebook plan for {{outlet::name}} macro resolution
   552→  const outletSections: Record<string, string> = {};
   553→  if (lorebookPlan?.outletSections.length) {
   554→    for (const outletSection of lorebookPlan.outletSections) {
   555→      // Extract outlet name from label like "World Info (myOutlet)" → "myOutlet"
   556→      const match = outletSection.label.match(/^World Info \((.+)\)$/);
   557→      const outletName = match ? match[1] : outletSection.label;

  1000→  const sections: PromptSection[] = [];

  1003→

  1010→  if (historyParts.length > 0) {

  1014→      content: historyParts.join('\n\n'),

  1016→    });
  1017→  }
  1018→
  1019→  return sections;

  1028→ * 2 = Before last user message
  1029→ * 3 = After last assistant message
  1030→ * 4 = Before last assistant message
  1031→ */
  1032→function applyChatInjections(
  1033→  messages: ChatApiMessage[],
  1034→  injections: LorebookChatInjection[]
  1035→): void {
  1036→  for (const injection of injections) {
  1037→    const content = `\n\n[${injection.label}]\n${injection.content}`;
  1038→
  1039→    switch (injection.position) {
  1040→      case 1: { // After last user message
  1041→        const lastUserIdx = messages.map((m, i) => m.role === 'user' ? i : -1).filter(i => i >= 0).pop();
  1042→        if (lastUserIdx !== undefined && lastUserIdx >= 0) {
  1043→          messages[lastUserIdx] = {
  1044→            ...messages[lastUserIdx],
  1045→            content: messages[lastUserIdx].content + content
  1046→          };
  1047→        }
  1048→        break;
  1049→      }
  1050→      case 2: { // Before last user message
  1051→        const lastUserIdx2 = messages.map((m, i) => m.role === 'user' ? i : -1).filter(i => i >= 0).pop();
  1052→        if (lastUserIdx2 !== undefined && lastUserIdx2 >= 0) {
  1053→          messages[lastUserIdx2] = {
  1054→            ...messages[lastUserIdx2],
  1055→            content: content + messages[lastUserIdx2].content
  1056→          };
  1057→        }
  1058→        break;
  1059→      }
  1060→      case 3: { // After last assistant message
  1061→        const lastAsstIdx = messages.map((m, i) => m.role === 'assistant' ? i : -1).filter(i => i >= 0).pop();
  1062→        if (lastAsstIdx !== undefined && lastAsstIdx >= 0) {
  1063→          messages[lastAsstIdx] = {
  1064→            ...messages[lastAsstIdx],
  1065→            content: messages[lastAsstIdx].content + content
  1066→          };
  1067→        }
  1068→        break;
  1069→      }
  1070→      case 4: { // Before last assistant message
  1071→        const lastAsstIdx4 = messages.map((m, i) => m.role === 'assistant' ? i : -1).filter(i => i >= 0).pop();
  1072→        if (lastAsstIdx4 !== undefined && lastAsstIdx4 >= 0) {
  1073→          messages[lastAsstIdx4] = {
  1074→            ...messages[lastAsstIdx4],
  1075→            content: content + messages[lastAsstIdx4].content
  1076→          };
  1077→        }
  1078→        break;
  1079→      }
  1080→    }
  1081→  }
  1082→}

  1084→/**
  1085→ * Build messages array for chat models
  1086→ * 
  1087→ * Order (SillyTavern style):
  1088→ * 1. System message (system prompt)
  1089→ * 2. Chat history
  1090→ * 3. Author's Note (injected AFTER chat history, as system message)
  1091→ * 4. Post-History Instructions (injected AFTER Author's Note, as system message)
  1092→ */
  1093→export function buildChatMessages(
  1094→  systemPrompt: string,
  1095→  messages: ChatMessage[],
  1096→  character: CharacterCard,
  1097→  userName: string = 'User',
  1098→  postHistoryInstructions?: string,
  1099→  authorNote?: string,
  1100→  useSystemRole: boolean = false,
  1101→  embeddingsContext?: string,  // embeddings injected before chat history
  1102→  lorebookChatInjections?: LorebookChatInjection[],  // positions 1-4: inject into specific messages
  1103→  exampleMessages?: ChatApiMessage[]  // SillyTavern-style example dialogue as chat messages
  1104→): ChatApiMessage[] {
  1105→  // =============================================
  1106→  // Step 1: Build all system content as ONE message
  1107→  // =============================================
  1108→  // Consolidate system prompt + embeddings + author note + post-history
  1109→  // into a single system message to avoid consecutive system messages
  1110→  // which break Jinja templates in models like LM Studio.
  1111→  const systemParts: string[] = [];
  1112→
  1113→  // Main system prompt
  1114→  if (systemPrompt.trim()) {
  1115→    systemParts.push(systemPrompt);
  1116→  }
  1117→
  1118→  // Embeddings Context
  1119→  if (embeddingsContext?.trim()) {
  1120→    systemParts.push(embeddingsContext);
  1121→  }
  1122→
  1123→  // Author's Note
  1124→  if (authorNote?.trim()) {
  1125→    systemParts.push(`[Author's Note]\n${authorNote}`);
  1126→  }
  1127→
  1128→  // Post-History Instructions
  1129→  if (postHistoryInstructions?.trim()) {
  1130→    systemParts.push(postHistoryInstructions);
  1131→  }
  1132→
  1133→  const chatMessages: ChatApiMessage[] = [];
  1134→
  1135→  // Single system message (or assistant if useSystemRole is false)
  1136→  if (systemParts.length > 0) {
  1137→    chatMessages.push({
  1138→      role: useSystemRole ? 'system' : 'assistant',
  1139→      content: systemParts.join('\n\n---\n\n')
  1140→    });
  1141→  }
  1142→
  1143→  // =============================================
  1144→  // Step 2: Inject example dialogue as chat messages (SillyTavern style)
  1145→  // =============================================
  1146→  // These go between the system message and the actual chat history
  1147→  // They serve as few-shot examples for the LLM
  1148→  if (exampleMessages && exampleMessages.length > 0) {
  1149→    chatMessages.push(...exampleMessages);
  1150→  }
  1151→
  1152→  // =============================================
  1153→  // Step 3: Build chat history with proper alternation
  1154→  // =============================================
  1155→  // Merge consecutive same-role messages and ensure
  1156→  // strict user/assistant alternation for OpenAI-compatible APIs.
  1157→  const visibleMessages = messages.filter(m => !m.isDeleted && !m.isNarratorMessage);
  1158→
  1159→  // First pass: merge consecutive same-role messages
  1160→  const mergedMessages: ChatApiMessage[] = [];
  1161→  for (const msg of visibleMessages) {
  1162→    const role = msg.role === 'user' ? 'user' : 'assistant';
  1163→    const last = mergedMessages[mergedMessages.length - 1];
  1164→    if (last && last.role === role) {
  1165→      // Merge into previous message
  1166→      last.content += '\n' + msg.content;
  1167→    } else {
  1168→      mergedMessages.push({ role, content: msg.content });
  1169→    }
  1170→  }
  1171→
  1172→  // Second pass: enforce user/assistant alternation
  1173→  // If the first chat message is 'assistant', prepend a synthetic user message
  1174→  if (mergedMessages.length > 0 && mergedMessages[0].role === 'assistant') {
  1175→    mergedMessages.unshift({ role: 'user', content: '*continúa*' });
  1176→  }
  1177→
  1178→  // If two same-role messages are still adjacent (shouldn't happen after merge,
  1179→  // but as safety), insert a bridging message
  1180→  const finalHistory: ChatApiMessage[] = [];
  1181→  for (const msg of mergedMessages) {
  1182→    const last = finalHistory[finalHistory.length - 1];
  1183→    if (last && last.role === msg.role) {
  1184→      // Insert bridging message
  1185→      const bridgeRole = msg.role === 'user' ? 'assistant' : 'user';
  1186→      finalHistory.push({ role: bridgeRole, content: '*continúa*' });
  1187→    }
  1188→    finalHistory.push(msg);
  1189→  }
  1190→
  1191→  // Bridge between example messages and chat history if needed
  1192→  // If the last example message has the same role as the first chat message,
  1193→  // insert a bridging message to maintain proper user/assistant alternation
  1194→  if (finalHistory.length > 0 && chatMessages.length > 0) {
  1195→    const lastExistingMsg = chatMessages[chatMessages.length - 1];
  1196→    const firstChatMsg = finalHistory[0];
  1197→    if (lastExistingMsg.role === firstChatMsg.role) {
  1198→      const bridgeRole = firstChatMsg.role === 'user' ? 'assistant' : 'user';
  1199→      chatMessages.push({ role: bridgeRole, content: '*continúa*' });
  1200→    }
  1201→  }
  1202→
  1203→  chatMessages.push(...finalHistory);
  1204→
  1205→  // =============================================
  1206→  // Step 4: Inject lorebook chat-level content (positions 1-4)
  1207→  // =============================================
  1208→  if (lorebookChatInjections?.length) {
  1209→    applyChatInjections(chatMessages, lorebookChatInjections);
  1210→  }
  1211→
  1212→  return chatMessages;
  1213→}

  1215→/**
  1216→ * Build prompt for completion-style APIs (Ollama, KoboldCPP, etc.)
  1217→ * 
  1218→ * Order (SillyTavern style):
  1219→ * 1. System prompt
  1220→ * 2. Chat history
  1221→ * 3. Author's Note
  1222→ * 4. Post-History Instructions
  1223→ * 5. Assistant prefix
  1224→ */
  1225→export function buildCompletionPrompt(config: CompletionPromptConfig): string {
  1226→  const { systemPrompt, messages, character, userName, postHistoryInstructions, authorNote, embeddingsContext, exampleMessages, allCharacters } = config;
  1227→  const parts: string[] = [];
  1228→
  1229→  parts.push(systemPrompt);
  1230→  parts.push('\n---\n');
  1231→
  1232→  // Embeddings Context - injected before chat history for recency primacy
  1233→  if (embeddingsContext?.trim()) {
  1234→    parts.push(embeddingsContext);
  1235→    parts.push('\n---\n');
  1236→  }
  1237→
  1238→  // Example dialogue is now included in the systemPrompt string (as [EJEMPLO] sections).
  1239→  // The exampleMessages parameter is kept for backward compatibility but will always be empty.
  1240→  // No separate injection needed here.
  1241→
  1242→  // Exclude narrator messages from prompt
  1243→  const visibleMessages = messages.filter(m => !m.isDeleted && !m.isNarratorMessage);
  1244→
  1245→  for (const msg of visibleMessages) {
  1246→    if (msg.role === 'user') {
  1247→      parts.push(`${userName}: ${msg.content}`);
  1248→    } else if (msg.role === 'assistant') {
  1249→      // Use actual speaker name from characterId (important for group chats where
  1250→      // multiple characters speak - each should be attributed to their own name)
  1251→      const speakerName = msg.characterId
  1252→        ? (allCharacters?.find(c => c.id === msg.characterId)?.name || character.name)
  1253→        : character.name;
  1254→      parts.push(`${speakerName}: ${msg.content}`);
  1255→    }
  1256→  }
  1257→
  1258→  // Author's Note - injected AFTER chat history, BEFORE post-history instructions
  1259→  if (authorNote?.trim()) {
  1260→    parts.push(`\n[Author's Note]\n${authorNote}`);
  1261→  }
  1262→
  1263→  // Post-History Instructions - injected AFTER Author's Note
  1264→  if (postHistoryInstructions) {
  1265→    parts.push(`\n${postHistoryInstructions}`);
  1266→  }
  1267→
  1268→  parts.push(`\n${character.name}:`);
  1269→
  1270→  return parts.join('\n');
  1271→}

  1697→/**
  1698→ * Build messages array for group chat
  1699→ *
  1700→ * Order (SillyTavern style):
  1701→ * 1. System message (system prompt)
  1702→ * 2. Chat history (all visible messages)
  1703→ * 3. Previous responses from this turn
  1704→ * 4. Author's Note (injected AFTER chat history, as system message)
  1705→ * 5. Post-History Instructions (injected AFTER Author's Note, as system message)
  1706→ *
  1707→ * @param isForNarrator - If true, narrator messages ARE included (narrator sees all)
  1708→ */
  1709→export function buildGroupChatMessages(
  1710→  systemPrompt: string,
  1711→  messages: ChatMessage[],
  1712→  character: CharacterCard,
  1713→  allCharacters: CharacterCard[],
  1714→  userName: string = 'User',
  1715→  previousResponses?: Array<{ characterName: string; content: string }>,
  1716→  postHistoryInstructions?: string,
  1717→  authorNote?: string,
  1718→  isForNarrator: boolean = false,
  1719→  embeddingsContext?: string,  // embeddings injected before chat history
  1720→  lorebookChatInjections?: LorebookChatInjection[],  // positions 1-4: inject into specific messages
  1721→  exampleMessages?: ChatApiMessage[],  // SillyTavern-style example dialogue as chat messages
  1722→  turnDirective?: string  // per-responder directive appended to the last user message (anti impersonation)
  1723→): GroupPromptBuildResult {
  1724→  // =============================================
  1725→  // Step 1: Build all system content as ONE message
  1726→  // =============================================
  1727→  const systemParts: string[] = [];
  1728→
  1729→  if (systemPrompt.trim()) {
  1730→    systemParts.push(systemPrompt);
  1731→  }
  1732→  if (embeddingsContext?.trim()) {
  1733→    systemParts.push(embeddingsContext);
  1734→  }

  1736→    systemParts.push(`[Author's Note]\n${authorNote}`);

  1743→
  1744→  // Single system message
  1745→  // role 'system' (matches 1-to-1 buildChatMessages with useSystemRole=true):
  1746→  // sending the system prompt as 'assistant' weakens instruction-following and
  1747→  // encourages the model to keep narrating the whole scene by itself.
  1748→  if (systemParts.length > 0) {
  1749→    chatMessages.push({
  1750→      role: 'system',
  1751→      content: systemParts.join('\n\n---\n\n')
  1752→    });
  1753→  }
  1754→
  1755→  // =============================================
  1756→  // Step 2: Inject example dialogue as chat messages (SillyTavern style)

  1761→    chatMessages.push(...exampleMessages);

  1763→

  1765→  // Step 3: Build chat history with proper alternation
  1766→  // =============================================
  1767→  const visibleMessages = isForNarrator
  1768→    ? messages.filter(m => !m.isDeleted)

  1775→    const speaker = msg.role === 'user' ? userName :

  1777→    historyLines.push(`${speaker}: ${msg.content}`);
  1778→  }
  1779→

  1781→  // IMPORTANT: In group chats, multiple characters speak as 'assistant' role.
  1782→  // When merging consecutive assistant messages, we MUST include speaker names
  1783→  // so the LLM can distinguish which character said what.
  1784→  const mergedMessages: ChatApiMessage[] = [];
  1785→  for (const msg of visibleMessages) {
  1786→    const role = msg.role === 'user' ? 'user' : 'assistant';
  1787→    // Include speaker name for assistant messages so different characters' lines

  1789→    const speakerName = msg.role === 'user' ? userName :

  1792→    const last = mergedMessages[mergedMessages.length - 1];
  1793→    if (last && last.role === role) {
  1794→      last.content += '\n' + contentWithSpeaker;

  1837→    }
  1838→  }
  1839→
  1840→  chatMessages.push(...finalHistory);

  1875→    systemPrompt,
  1876→    sections: [],
  1877→    chatMessages,
  1878→    chatHistorySection
  1879→  };
  1880→}
