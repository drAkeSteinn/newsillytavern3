   128→   128→// Format memory date to relative time
   129→   129→function formatMemoryDate(date: Date): string {
   130→   130→  const now = new Date();
   131→   131→  const diff = now.getTime() - date.getTime();
   132→   132→  const minutes = Math.floor(diff / 60000);
   133→   133→  if (minutes < 1) return 'Ahora';
   134→   134→  if (minutes < 60) return `Hace ${minutes}m`;
   135→   135→  const hours = Math.floor(minutes / 60);
   136→   136→  if (hours < 24) return `Hace ${hours}h`;
   137→   137→  const days = Math.floor(hours / 24);
   138→   138→  if (days < 7) return `Hace ${days}d`;
   139→   139→  return date.toLocaleDateString('es-ES', { day: 'numeric', month: 'short' });
   140→   140→}
