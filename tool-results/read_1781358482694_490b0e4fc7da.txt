   147→
   148→export function TriggerCollectionEditor({ 
   149→  character, 
   150→  onChange 
   151→}: TriggerCollectionEditorProps) {
   152→  // Get collections and packs
   153→  const triggerCollections: TriggerCollection[] = useMemo(() => {
   154→    return character.triggerCollections || [];
   155→  }, [character.triggerCollections]);
   156→
   157→  const spritePacksV2: SpritePackV2[] = useMemo(() => {
   158→    return character.spritePacksV2 || [];
   159→  }, [character.spritePacksV2]);
   160→
   161→  // Store actions for testing
   162→  const applyTriggerForCharacter = useTavernStore((state) => state.applyTriggerForCharacter);
   163→  const scheduleReturnToIdleForCharacter = useTavernStore((state) => state.scheduleReturnToIdleForCharacter);
   164→  const cancelReturnToIdleForCharacter = useTavernStore((state) => state.cancelReturnToIdleForCharacter);
   165→
   166→  // State
   167→  const [showEditor, setShowEditor] = useState(false);
   168→  const [editingCollection, setEditingCollection] = useState<TriggerCollection | null>(null);
   169→  const [testingCollectionId, setTestingCollectionId] = useState<string | null>(null);
   170→  const [expandedSections, setExpandedSections] = useState<Record<string, boolean>>({
   171→    keys: true,
   172→    behavior: true,
   173→    fallback: true,
   174→    chains: false,
   175→    sprites: false,
   176→  });
   177→
   178→  // Update collections
   179→  const updateCollections = (newCollections: TriggerCollection[]) => {
   180→    onChange({
   181→      triggerCollections: newCollections,
   182→    });
   183→  };
   184→
   185→  // Test trigger - Apply the trigger sprite to the character
   186→  const handleTestTrigger = (collection: TriggerCollection) => {
   187→    const pack = spritePacksV2.find(p => p.id === collection.packId);
   188→    if (!pack || pack.sprites.length === 0) {
   189→      logger.warn('Cannot test: no sprites in pack', { collectionId: collection.id });
   190→      return;
   191→    }
   192→
   193→    // Get the sprite to show based on behavior
   194→    let sprite;
   195→    switch (collection.collectionBehavior) {
   196→      case 'random':
   197→        const randomIndex = Math.floor(Math.random() * pack.sprites.length);
   198→        sprite = pack.sprites[randomIndex];
   199→        break;
   200→      case 'principal':
   201→      case 'list':
   202→      default:
   203→        sprite = collection.principalSpriteId 
   204→          ? pack.sprites.find(s => s.id === collection.principalSpriteId)
   205→          : pack.sprites[0];
   206→        break;
   207→    }
   208→
   209→    if (!sprite) {
   210→      logger.warn('Cannot test: no sprite found', { collectionId: collection.id });
   211→      return;
   212→    }
   213→
   214→    // Mark as testing
   215→    setTestingCollectionId(collection.id);
   216→    logger.info('Testing trigger', { 
   217→      collectionId: collection.id, 
   218→      spriteUrl: sprite.url,
   219→      spriteLabel: sprite.label 
   220→    });
   221→
   222→    // Apply the trigger sprite
   223→    applyTriggerForCharacter(character.id, {
   224→      spriteUrl: sprite.url,
   225→      spriteLabel: sprite.label,
   226→      returnToIdleMs: collection.fallbackDelayMs,
   227→      packId: collection.packId,
   228→      collectionId: collection.id,
   229→      useTimelineSounds: collection.useTimelineSounds ?? false,
   230→    });
   231→
   232→    // Note: Timeline sounds are now handled by useTimelineSpriteSounds hook
   233→    // which watches for triggerSpriteUrl changes and checks useTimelineSounds
   234→
   235→    // Schedule fallback if configured
   236→    if (collection.fallbackDelayMs > 0) {
   237→      // Get fallback sprite based on mode
   238→      let fallbackSpriteUrl: string | null = null;
   239→      let returnToMode: 'idle' | 'talk' | 'thinking' | 'clear' = 'idle';
   240→
   241→      if (collection.fallbackMode === 'custom_sprite' && collection.fallbackSpriteId) {
   242→        const fallbackSprite = pack.sprites.find(s => s.id === collection.fallbackSpriteId);
   243→        fallbackSpriteUrl = fallbackSprite?.url || null;
   244→        returnToMode = 'idle'; // Apply the custom sprite
   245→      } else if (collection.fallbackMode === 'collection_default') {
   246→        const principalSprite = collection.principalSpriteId
   247→          ? pack.sprites.find(s => s.id === collection.principalSpriteId)
   248→          : pack.sprites[0];
   249→        fallbackSpriteUrl = principalSprite?.url || null;
   250→        returnToMode = 'idle'; // Apply the collection default sprite
   251→      } else if (collection.fallbackMode === 'idle_collection') {
   252→        // For 'idle_collection', clear the trigger and let the normal state logic
   253→        // (idle state from State Collections V2) determine what to show
   254→        returnToMode = 'clear';
   255→        // fallbackSpriteUrl can be empty for 'clear' mode
   256→        fallbackSpriteUrl = '';
   257→      }
   258→
   259→      // Always schedule fallback when delay > 0
   260→      // For 'clear' mode, the empty string is fine (won't be used)
   261→      scheduleReturnToIdleForCharacter(
   262→        character.id,
   263→        sprite.url,
   264→        returnToMode,
   265→        fallbackSpriteUrl || '',
   266→        sprite.label,
   267→        collection.fallbackDelayMs
   268→      );
   269→
   270→      logger.info('Test trigger: fallback scheduled', {
   271→        collectionId: collection.id,
   272→        fallbackMode: collection.fallbackMode,
   273→        returnToMode,
   274→        fallbackDelayMs: collection.fallbackDelayMs,
   275→      });
   276→    }
   277→
   278→    // Clear testing indicator after a delay
   279→    setTimeout(() => {
   280→      setTestingCollectionId(null);
   281→    }, 1000);
   282→  };
   283→
   284→  // Create new collection
   285→  const handleCreateCollection = () => {
   286→    const now = new Date().toISOString();
   287→    const newCollection: TriggerCollection = {
   288→      id: uuidv4(),
   289→      name: 'Nueva Colección de Triggers',
   290→      active: true,
   291→      priority: 1,
   292→      packId: spritePacksV2[0]?.id || '',
   293→      collectionKey: '',
   294→      collectionKeys: [],
   295→      collectionKeyRequirePipes: true,
   296→      collectionKeyCaseSensitive: false,
   297→      collectionBehavior: 'principal',
   298→      principalSpriteId: undefined,
   299→      fallbackMode: 'idle_collection',
   300→      fallbackSpriteId: undefined,
   301→      fallbackDelayMs: 3000,
   302→      spriteChain: undefined,
   303→      useTimelineSounds: false,
   304→      cooldownMs: 1000,
   305→      spriteConfigs: {},
   306→      conditionalMode: false,
   307→      conditionalEntries: [],
   308→      defaultSpriteId: undefined,
   309→      createdAt: now,
   310→      updatedAt: now,
   311→    };
   312→
   313→    setEditingCollection(newCollection);
   314→    setShowEditor(true);
   315→  };
   316→
   317→  // Edit collection
   318→  const handleEditCollection = (collection: TriggerCollection) => {
   319→    setEditingCollection({ ...collection });
   320→    setShowEditor(true);
   321→  };
   322→
   323→  // Delete collection
   324→  const handleDeleteCollection = (id: string) => {
   325→    const collection = triggerCollections.find(c => c.id === id);
   326→    if (!collection) return;
   327→
   328→    if (!confirm(`¿Eliminar la colección "${collection.name}"?`)) return;
   329→
   330→    updateCollections(triggerCollections.filter(c => c.id !== id));
   331→    logger.info('Deleted trigger collection', { collectionId: id });
   332→  };
   333→
   334→  // Duplicate collection
   335→  const handleDuplicateCollection = (collection: TriggerCollection) => {
   336→    const now = new Date().toISOString();
   337→    const newCollection: TriggerCollection = {
   338→      ...collection,
   339→      id: uuidv4(),
   340→      name: `${collection.name} (copia)`,
   341→      createdAt: now,
   342→      updatedAt: now,
   343→    };
   344→
   345→    updateCollections([...triggerCollections, newCollection]);
   346→    logger.info('Duplicated trigger collection', { 
   347→      originalId: collection.id, 
   348→      newId: newCollection.id 
   349→    });
   350→  };
   351→
   352→  // Save collection
   353→  const handleSaveCollection = () => {
   354→    if (!editingCollection) return;
   355→
   356→    const now = new Date().toISOString();
   357→    const updatedCollection: TriggerCollection = {
   358→      ...editingCollection,
   359→      updatedAt: now,
   360→    };
   361→
   362→    const existingIndex = triggerCollections.findIndex(c => c.id === updatedCollection.id);
   363→    
   364→    if (existingIndex >= 0) {
   365→      // Update existing
   366→      const newCollections = [...triggerCollections];
   367→      newCollections[existingIndex] = updatedCollection;
   368→      updateCollections(newCollections);
   369→    } else {
   370→      // Add new
   371→      updatedCollection.createdAt = now;
   372→      updateCollections([...triggerCollections, updatedCollection]);
   373→    }
   374→
   375→    setEditingCollection(null);
   376→    setShowEditor(false);
   377→    logger.info('Saved trigger collection', { collectionId: updatedCollection.id });
   378→  };
   379→
   380→  // Toggle collection active
   381→  const handleToggleActive = (id: string, active: boolean) => {
   382→    updateCollections(triggerCollections.map(c =>
   383→      c.id === id ? { ...c, active, updatedAt: new Date().toISOString() } : c
   384→    ));
   385→  };
   386→
   387→  // Get pack name
   388→  const getPackName = (packId: string) => {
   389→    const pack = spritePacksV2.find(p => p.id === packId);
   390→    return pack?.name || 'Pack no encontrado';
   391→  };
   392→
   393→  // Get pack sprites
   394→  const getPackSprites = (packId: string) => {
   395→    const pack = spritePacksV2.find(p => p.id === packId);
   396→    return pack?.sprites || [];
   397→  };
   398→
   399→  // Format time
   400→  const formatTime = (ms: number) => {
   401→    if (ms <= 0) return 'Nunca';
   402→    if (ms < 1000) return `${ms}ms`;
   403→    const seconds = Math.floor(ms / 1000);
   404→    if (seconds < 60) return `${seconds}s`;
   405→    const minutes = Math.floor(seconds / 60);
   406→    const remainingSeconds = seconds % 60;
   407→    return remainingSeconds > 0 ? `${minutes}m ${remainingSeconds}s` : `${minutes}m`;
   408→  };
   409→
   410→  // Toggle section
   411→  const toggleSection = (section: string) => {
   412→    setExpandedSections(prev => ({
   413→      ...prev,
   414→      [section]: !prev[section],
   415→    }));
   416→  };
   417→
   418→  return (
   419→    <div className="space-y-4">
   420→      {/* Header */}
   421→      <div className="flex items-center justify-between">
   422→        <div>
   423→          <h4 className="text-sm font-medium flex items-center gap-2">
   424→            <Zap className="w-4 h-4 text-amber-500" />
   425→            Trigger Collections
   426→          </h4>
   427→          <p className="text-xs text-muted-foreground">
   428→            Colecciones de triggers con sistema de prioridades y cadenas.
   429→          </p>
   430→        </div>
   431→        <Button
   432→          size="sm"
   433→          className="h-8"
   434→          onClick={handleCreateCollection}
   435→          disabled={spritePacksV2.length === 0}
   436→        >
   437→          <Plus className="w-4 h-4 mr-1" />
   438→          Nueva Colección
   439→        </Button>
   440→      </div>
   441→
   442→      {/* Info Banner */}
   443→      <div className="text-xs bg-amber-500/10 border border-amber-500/20 rounded-lg p-3 space-y-2">
   444→        <div className="flex items-center gap-2 font-medium text-amber-600">
   445→          <Zap className="w-4 h-4" />
   446→          Sistema de Trigger Collections
   447→        </div>
   448→        <p className="text-muted-foreground">
   449→          Las colecciones de triggers usan <strong>prioridades</strong> para determinar qué trigger 
   450→          se ejecuta. Puedes activar por <strong>key de colección</strong> (general) o por 
   451→          <strong> keys individuales</strong> por sprite.
   452→        </p>
   453→        <div className="grid grid-cols-2 gap-2 mt-2">
   454→          <div className="p-2 bg-background/50 rounded border">
   455→            <div className="flex items-center gap-1 text-xs font-medium text-amber-600">
   456→              <Crown className="w-3 h-3" />
   457→              Prioridad
   458→            </div>
   459→            <p className="text-[10px] text-muted-foreground mt-0.5">
   460→              Mayor número = mayor prioridad
   461→            </p>
   462→          </div>
   463→          <div className="p-2 bg-background/50 rounded border">
   464→            <div className="flex items-center gap-1 text-xs font-medium text-blue-600">
   465→              <Play className="w-3 h-3" />
   466→              Cadenas
   467→            </div>
   468→            <p className="text-[10px] text-muted-foreground mt-0.5">
   469→              Secuencias de sprites/sonidos
   470→            </p>
   471→          </div>
   472→        </div>
   473→      </div>
   474→
   475→      {/* No packs warning */}
   476→      {spritePacksV2.length === 0 && (
   477→        <div className="text-center py-4 text-amber-600 bg-amber-500/5 border border-amber-500/20 rounded-lg">
   478→          <Package className="w-6 h-6 mx-auto mb-2" />
   479→          <p className="text-sm font-medium">No hay Sprite Packs creados</p>
   480→          <p className="text-xs mt-1">
   481→            Crea un Sprite Pack en la pestaña "Sprite Packs" antes de crear triggers.
   482→          </p>
   483→        </div>
   484→      )}
   485→
   486→      {/* Collections List */}
   487→      {triggerCollections.length > 0 ? (
   488→        <Accordion type="multiple" className="w-full space-y-2">
   489→          {triggerCollections
   490→            .sort((a, b) => b.priority - a.priority)
   491→            .map(collection => {
   492→              const pack = spritePacksV2.find(p => p.id === collection.packId);
   493→              const sprites = pack?.sprites || [];
   494→
   495→              return (
   496→                <AccordionItem
   497→                  key={collection.id}
   498→                  value={collection.id}
   499→                  className="border rounded-lg px-0"
   500→                >
   501→                  <AccordionTrigger className="px-4 py-3 hover:no-underline">
   502→                    <div className="flex items-center gap-3 w-full">
   503→                      <div className={cn(
   504→                        "p-1.5 rounded",
   505→                        collection.active ? "bg-amber-500/10" : "bg-muted"
   506→                      )}>
   507→                        <Zap className={cn(
   508→                          "w-4 h-4",
   509→                          collection.active ? "text-amber-500" : "text-muted-foreground"
   510→                        )} />
   511→                      </div>
   512→                      <div className="flex-1 text-left">
   513→                        <div className="flex items-center gap-2">
   514→                          <span className="text-sm font-medium">{collection.name}</span>
   515→                          <Badge variant="outline" className="text-xs">
   516→                            Prioridad: {collection.priority}
   517→                          </Badge>
   518→                        </div>
   519→                        <div className="flex items-center gap-2 text-xs text-muted-foreground">
   520→                          <Package className="w-3 h-3" />
   521→                          <span>{getPackName(collection.packId)}</span>
   522→                          <span>•</span>
   523→                          <span>{sprites.length} sprites</span>
   524→                          {collection.collectionKey && (
   525→                            <>
   526→                              <span>•</span>
   527→                              <code className="bg-muted px-1 rounded text-[10px]">
   528→                                {collection.collectionKeyRequirePipes ? `|${collection.collectionKey}|` : collection.collectionKey}
   529→                              </code>
   530→                            </>
   531→                          )}
   532→                        </div>
   533→                      </div>
   534→                      {!collection.active && (
   535→                        <Badge variant="secondary" className="text-xs">Inactivo</Badge>
   536→                      )}
   537→                      {collection.conditionalMode && (
   538→                        <Badge variant="secondary" className="text-xs bg-teal-500/10 text-teal-600">
   539→                          <GitBranch className="w-3 h-3 mr-1" />
   540→                          Condicional
   541→                        </Badge>
   542→                      )}
   543→                      {collection.conditionalMode && (collection.conditionalEntries?.length || 0) > 0 && (
   544→                        <Badge variant="outline" className="text-xs">
   545→                          {collection.conditionalEntries?.length} entradas
   546→                        </Badge>
   547→                      )}
   548→                    </div>
   549→                  </AccordionTrigger>
   550→                  <AccordionContent className="px-4 pb-4">
   551→                    <div className="space-y-3">
   552→                      {/* Quick Info */}
   553→                      <div className="grid grid-cols-3 gap-2 text-xs">
   554→                        <div className="p-2 bg-muted/50 rounded border">
   555→                          <div className="text-muted-foreground">Comportamiento</div>
   556→                          <div className="font-medium mt-0.5 flex items-center gap-1">
   557→                            {BEHAVIOR_OPTIONS.find(b => b.value === collection.collectionBehavior)?.icon}
   558→                            {BEHAVIOR_OPTIONS.find(b => b.value === collection.collectionBehavior)?.label}
   559→                          </div>
   560→                        </div>
   561→                        <div className="p-2 bg-muted/50 rounded border">
   562→                          <div className="text-muted-foreground">Fallback</div>
   563→                          <div className="font-medium mt-0.5">
   564→                            {formatTime(collection.fallbackDelayMs)}
   565→                          </div>
   566→                        </div>
   567→                        <div className="p-2 bg-muted/50 rounded border">
   568→                          <div className="text-muted-foreground">Cooldown</div>
   569→                          <div className="font-medium mt-0.5">
   570→                            {formatTime(collection.cooldownMs)}
   571→                          </div>
   572→                        </div>
   573→                      </div>
   574→
   575→                      {/* Chain indicators */}
   576→                      {(collection.spriteChain?.enabled || collection.useTimelineSounds || collection.conditionalMode) && (
   577→                        <div className="flex gap-2 flex-wrap">
   578→                          {collection.spriteChain?.enabled && (
   579→                            <Badge variant="secondary" className="text-xs bg-purple-500/10 text-purple-600">
   580→                              <Play className="w-3 h-3 mr-1" />
   581→                              Sprite Chain ({collection.spriteChain.steps.length} steps)
   582→                            </Badge>
   583→                          )}
   584→                          {collection.useTimelineSounds && (
   585→                            <Badge variant="secondary" className="text-xs bg-green-500/10 text-green-600">
   586→                              <Volume2 className="w-3 h-3 mr-1" />

   740→interface ConditionalSimulatorProps {

   746→function ConditionalSimulator({
   747→  collection,
   748→  character,
   749→  packSprites,
   750→}: ConditionalSimulatorProps) {
   751→  const [simValues, setSimValues] = useState<Record<string, number>>({});
   752→  const [showSimulator, setShowSimulator] = useState(false);
   753→
   754→  // Get unique attribute keys from all conditional entries
   755→  const uniqueAttributes = useMemo(() => {
   756→    const keys = new Set<string>();
   757→    (collection.conditionalEntries || []).forEach(entry => {
   758→      entry.conditions.forEach(cond => {
   759→        if (cond.attributeKey) keys.add(cond.attributeKey);
   760→      });
   761→    });
   762→    return Array.from(keys);
   763→  }, [collection.conditionalEntries]);
   764→
   765→  // Get attribute definitions from character stats config
   766→  const attributeDefs = useMemo(() => {
   767→    return character.statsConfig?.attributes || [];
   768→  }, [character.statsConfig?.attributes]);
   769→
   770→  // Get attribute display info
   771→  const getAttrInfo = (key: string) => {
   772→    const def = attributeDefs.find(a => a.key === key);
   773→    return {
   774→      name: def?.name || key,
   775→      min: def?.min ?? 0,
   776→      max: def?.max ?? 100,
   777→    };
   778→  };
   779→
   780→  // Get session stats from store
   781→  const sessionStats = useTavernStore((state) => {
   782→    const activeSessionId = state.activeSessionId;
   783→    if (!activeSessionId) return null;
   784→    const session = state.sessions.find(s => s.id === activeSessionId);
   785→    return session?.sessionStats ?? null;
   786→  });
   787→
   788→  // Build simulated session stats
   789→  const simulatedStats: SessionStats | null = useMemo(() => {
   790→    if (!sessionStats) return null;
   791→    // Override the character's attribute values with slider values
   792→    const charStats = sessionStats.characterStats?.[character.id];
   793→    if (!charStats) return sessionStats;
   794→
   795→    return {
   796→      ...sessionStats,
   797→      characterStats: {
   798→        ...sessionStats.characterStats,
   799→        [character.id]: {
   800→          ...charStats,
   801→          attributeValues: {
   802→            ...charStats.attributeValues,
   803→            ...simValues,
   804→          },
   805→        },
   806→      },
   807→    };
   808→  }, [sessionStats, character.id, simValues]);
   809→
   810→  // Evaluate which entry wins
   811→  const winner = useMemo(() => {
   812→    if (!simulatedStats || !collection.conditionalEntries?.length) return null;
   813→    return evaluateConditionalEntries(collection.conditionalEntries, simulatedStats, character.id);
   814→  }, [collection.conditionalEntries, simulatedStats, character.id]);
   815→
   816→  // Get session value for an attribute
   817→  const getSessionValue = (key: string): number | string | null => {
   818→    if (!sessionStats) return null;
   819→    return getAttributeValueFromStats(character.id, key, sessionStats);
   820→  };
   821→
   822→  if (uniqueAttributes.length === 0) {
   823→    return null;
   824→  }
   825→
   826→  return (
   827→    <div className="space-y-3 p-4 border rounded-lg bg-slate-500/5 border-slate-500/20">
   828→      <div
   829→        className="flex items-center justify-between cursor-pointer"
   830→        onClick={() => setShowSimulator(!showSimulator)}
   831→      >
   832→        <div>
   833→          <h4 className="text-sm font-medium flex items-center gap-2 text-slate-600">
   834→            <SlidersHorizontal className="w-4 h-4" />
   835→            Simulador de Condiciones
   836→          </h4>
   837→          <p className="text-xs text-muted-foreground mt-1">
   838→            Ajusta los valores para ver qué sprite se mostraría.
   839→          </p>
   840→        </div>
   841→        <Button variant="ghost" size="sm" className="h-7 w-7 p-0">
   842→          {showSimulator ? <ChevronUp className="w-4 h-4" /> : <ChevronDown className="w-4 h-4" />}
   843→        </Button>
   844→      </div>
   845→
   846→      {showSimulator && (
   847→        <div className="space-y-3">
   848→          {/* Attribute Sliders */}
   849→          {uniqueAttributes.map(key => {
   850→            const info = getAttrInfo(key);
   851→            const currentValue = simValues[key] ?? (getSessionValue(key) as number ?? info.min);
   852→            const sessionVal = getSessionValue(key);
   853→            return (
   854→              <div key={key} className="space-y-1">
   855→                <div className="flex items-center justify-between">
   856→                  <Label className="text-xs font-medium">{info.name}</Label>
   857→                  <div className="flex items-center gap-2">
   858→                    <Input
   859→                      type="number"
   860→                      value={currentValue}
   861→                      onChange={(e) => {
   862→                        setSimValues(prev => ({
   863→                          ...prev,
   864→                          [key]: parseFloat(e.target.value) || 0,
   865→                        }));
   866→                      }}
   867→                      className="h-6 w-16 text-xs text-center"
   868→                    />
   869→                    {sessionVal !== null && (
   870→                      <span className="text-[10px] text-muted-foreground">
   871→                        (sesión: {String(sessionVal)})
   872→                      </span>
   873→                    )}
   874→                  </div>
   875→                </div>
   876→                <input
   877→                  type="range"
   878→                  min={info.min}
   879→                  max={info.max}
   880→                  value={currentValue}
   881→                  onChange={(e) => {
   882→                    setSimValues(prev => ({
   883→                      ...prev,
   884→                      [key]: parseFloat(e.target.value),
   885→                    }));
   886→                  }}
   887→                  className="w-full h-1.5 bg-muted rounded-lg appearance-none cursor-pointer accent-teal-500"
   888→                />
   889→              </div>
   890→            );
   891→          })}
   892→
   893→          {/* Result */}
   894→          <div className="p-3 bg-background rounded-lg border border-teal-500/20">
   895→            {winner ? (
   896→              <div className="space-y-2">
   897→                <div className="flex items-center gap-2 text-sm font-medium text-teal-600">
   898→                  <CheckCircle className="w-4 h-4" />
   899→                  <span>Entrada activa: &quot;{winner.name}&quot; (P{winner.priority})</span>
   900→                </div>
   901→                <div className="flex items-center gap-2">
   902→                  {(() => {
   903→                    const sprite = packSprites.find(s => s.id === winner.spriteId);
   904→                    return sprite ? (
   905→                      <>
   906→                        <div className="w-10 h-10 rounded overflow-hidden bg-muted/50">
   907→                          <SpritePreview
   908→                            src={sprite.url}
   909→                            alt={sprite.label}
   910→                            className="w-full h-full"
   911→                            objectFit="contain"
   912→                          />
   913→                        </div>
   914→                        <span className="text-xs">{sprite.label}</span>
   915→                      </>
   916→                    ) : (
   917→                      <span className="text-xs text-muted-foreground">Sprite no encontrado</span>
   918→                    );
   919→                  })()}
   920→                </div>
   921→                {/* Show condition evaluation details */}
   922→                {winner.conditions.map((cond, i) => {
   923→                  const attrVal = simValues[cond.attributeKey] ?? getSessionValue(cond.attributeKey);
   924→                  const passes = simulatedStats
   925→                    ? evaluateStatConditions([cond], simulatedStats, character.id)
   926→                    : false;
   927→                  const operatorSymbol = cond.operator === '<=' ? '≤' : cond.operator === '>=' ? '≥' : cond.operator;
   928→                  return (
   929→                    <div key={i} className={cn(
   930→                      "text-xs flex items-center gap-1",
   931→                      passes ? "text-green-600" : "text-red-500"
   932→                    )}>
   933→                      {passes ? '✅' : '❌'}
   934→                      <span>{cond.attributeKey} {operatorSymbol} {String(cond.value)}</span>
   935→                      {cond.operator === 'between' && <span> a {cond.valueMax}</span>}
   936→                      <span className="text-muted-foreground">→ {String(attrVal ?? '?')} {operatorSymbol} {String(cond.value)}</span>
   937→                      <span>{passes ? '✓' : '✗'}</span>
   938→                    </div>
   939→                  );
   940→                })}
   941→              </div>
   942→            ) : (
   943→              <div className="space-y-1">
   944→                <div className="text-sm font-medium text-muted-foreground flex items-center gap-2">
   945→                  <Info className="w-4 h-4" />
   946→                  Ninguna condición coincide
   947→                </div>
   948→                {collection.defaultSpriteId ? (
   949→                  <div className="flex items-center gap-2">
   950→                    {(() => {
   951→                      const sprite = packSprites.find(s => s.id === collection.defaultSpriteId);
   952→                      return sprite ? (
   953→                        <>
   954→                          <div className="w-8 h-8 rounded overflow-hidden bg-muted/50">
   955→                            <SpritePreview
   956→                              src={sprite.url}
   957→                              alt={sprite.label}
   958→                              className="w-full h-full"
   959→                              objectFit="contain"
   960→                            />
   961→                          </div>
   962→                          <span className="text-xs">Usando sprite por defecto: {sprite.label}</span>
   963→                        </>
   964→                      ) : (
   965→                        <span className="text-xs text-muted-foreground">Sprite por defecto no encontrado</span>
   966→                      );
   967→                    })()}
   968→                  </div>
   969→                ) : (
   970→                  <p className="text-xs text-muted-foreground">Se usará el comportamiento normal de la colección</p>
   971→                )}
   972→              </div>
   973→            )}
   974→          </div>
   975→        </div>
   976→      )}
   977→    </div>
   978→  );
   979→}
