   255→
   256→export function useTriggerSystem(config: TriggerSystemConfig = {}): TriggerSystemResult {
   257→  const store = useTavernStore();
   258→  const settings = store.settings;
   259→  
   260→  // Handler states (created once)
   261→  const soundHandlerState = useMemo(() => createSoundHandlerState(), []);
   262→  const spriteHandlerState = useMemo(() => createSpriteHandlerState(), []);
   263→  const hudHandlerState = useMemo(() => createHUDHandlerState(), []);
   264→  const backgroundHandlerState = useMemo(() => createBackgroundHandlerState(), []);
   265→  const questHandlerState = useMemo(() => createQuestHandlerState(), []);
   266→  const itemHandlerState = useMemo(() => createItemHandlerState(), []);
   267→  const statsHandlerState = useMemo(() => createStatsHandlerState(), []);
   268→  const skillActivationHandlerState = useMemo(() => createSkillActivationHandlerState(), []);
   269→  const solicitudHandlerState = useMemo(() => createSolicitudHandlerState(), []);
   270→
   271→  // Key handlers (created once to preserve deduplication state)
   272→  // IMPORTANT: These must be created once and reused, otherwise the position tracking
   273→  // for deduplication is reset on every call, causing double activations
   274→  const soundHandler = useMemo(() => createSoundKeyHandler(config.maxSoundsPerMessage ?? 10), [config.maxSoundsPerMessage]);
   275→  const spriteHandler = useMemo(() => createSpriteKeyHandler(), []);
   276→  const skillHandler = useMemo(() => createSkillKeyHandler(), []);
   277→  const solicitudHandler = useMemo(() => createSolicitudKeyHandler(), []);
   278→  const backgroundHandler = useMemo(() => createBackgroundKeyHandler(), []);
   279→  const hudHandler = useMemo(() => createHUDKeyHandler(), []);
   280→  const questHandler = useMemo(() => createQuestKeyHandler(), []);
   281→  const statsHandler = useMemo(() => createStatsKeyHandler(), []);
   282→  const itemHandler = useMemo(() => createItemKeyHandler(), []);
   283→
   284→  // Track last processed content per message
   285→  const lastProcessedRef = useRef<Map<string, string>>(new Map());
   286→  
   287→  // Initialize trigger bus and detector
   288→  useEffect(() => {
   289→    const bus = getTriggerBus();
   290→    const detector = getTokenDetector(config.tokenDetector);
   291→    
   292→    if (config.debug) {
   293→      bus.setDebug(true);
   294→    }
   295→    
   296→    // Cleanup on unmount
   297→    return () => {
   298→      soundHandlerState.triggeredPositions.clear();
   299→      soundHandlerState.soundCountPerMessage.clear();
   300→      spriteHandlerState.triggeredPositions.clear();
   301→      spriteHandlerState.lastPackMatches.clear();
   302→      hudHandlerState.updatedFields.clear();
   303→      backgroundHandlerState.triggeredPositions.clear();
   304→      backgroundHandlerState.lastTriggeredBackground.clear();
   305→      backgroundHandlerState.lastTriggerTime.clear();
   306→      backgroundHandlerState.currentActivePack.clear();
   307→      questHandlerState.processedQuests.clear();
   308→      questHandlerState.triggeredPositions.clear();
   309→      itemHandlerState.processedItems.clear();
   310→      itemHandlerState.triggeredPositions.clear();
   311→      statsHandlerState.detectionStates.clear();
   312→      statsHandlerState.processedMessages.clear();
   313→    };
   314→  }, [config.debug, config.tokenDetector, soundHandlerState, spriteHandlerState, hudHandlerState, backgroundHandlerState, questHandlerState, itemHandlerState, statsHandlerState]);
   315→  
   316→  // Check for return to default background periodically
   317→  useEffect(() => {
   318→    if (config.backgroundEnabled === false) return;
   319→    
   320→    const checkInterval = setInterval(() => {
   321→      const bgSettings = settings.backgroundTriggers;
   322→      if (!bgSettings?.enabled || !bgSettings?.returnToDefaultEnabled) return;
   323→      
   324→      const context: BackgroundTriggerContext = {
   325→        character: null,
   326→        characters: undefined,
   327→        fullText: '',
   328→        messageKey: 'return-check',
   329→        isStreaming: false,
   330→        timestamp: Date.now(),
   331→        backgroundPacks: store.backgroundTriggerPacks ?? [],
   332→        backgroundCollections: store.backgroundCollections ?? [],
   333→        backgroundSettings: {
   334→          enabled: bgSettings.enabled,
   335→          globalCooldown: bgSettings.globalCooldown ?? 0,
   336→          transitionDuration: bgSettings.transitionDuration ?? 500,
   337→          defaultTransitionType: (bgSettings.defaultTransitionType as BackgroundTransitionType) ?? 'fade',
   338→          returnToDefaultEnabled: bgSettings.returnToDefaultEnabled ?? false,
   339→          returnToDefaultAfter: bgSettings.returnToDefaultAfter ?? 300000,
   340→          defaultBackgroundUrl: bgSettings.defaultBackgroundUrl ?? '',
   341→          globalOverlays: (bgSettings.globalOverlays as BackgroundOverlay[]) ?? [],
   342→        },
   343→        cooldownContextKey: 'default',
   344→      };
   345→      
   346→      const returnResult = checkReturnToDefault(context, backgroundHandlerState);
   347→      
   348→      if (returnResult?.shouldReturn && returnResult.defaultUrl) {
   349→        console.log('[TriggerSystem] Returning to default background');
   350→        store.setBackground(returnResult.defaultUrl);
   351→        if (returnResult.overlays) {
   352→          store.setActiveOverlays(returnResult.overlays);
   353→        }
   354→      }
   355→    }, 10000); // Check every 10 seconds
   356→    
   357→    return () => clearInterval(checkInterval);
   358→  }, [settings.backgroundTriggers, store, backgroundHandlerState, config.backgroundEnabled]);
   359→  
   360→  /**
   361→   * Get the active HUD template based on character or group
   362→   */
   363→  const getActiveHUDTemplate = useCallback((): HUDTemplate | null => {
   364→    const hudSessionState = store.hudSessionState;
   365→    
   366→    // If there's an active template in session, use it
   367→    if (hudSessionState.activeTemplateId) {
   368→      return store.hudTemplates.find(t => t.id === hudSessionState.activeTemplateId) || null;
   369→    }
   370→    
   371→    return null;
   372→  }, [store.hudTemplates, store.hudSessionState]);
   373→  
   374→  /**
   375→   * Process streaming content incrementally
   376→   */
   377→  const processStreamingContent = useCallback((
   378→    content: string,
   379→    character: CharacterCard | null,
   380→    messageKey: string,
   381→    characters?: CharacterCard[]
   382→  ) => {
   383→    // DEBUG: Log all processStreamingContent calls
   384→    console.log(`[TriggerSystem] processStreamingContent called`, {
   385→      messageKey,
   386→      characterName: character?.name,
   387→      characterId: character?.id,
   388→      contentLength: content.length,
   389→      contentPreview: content.slice(0, 100),
   390→      charactersCount: characters?.length,
   391→      hasPersona: characters?.some(c => c.id === '__user__'),
   392→    });
   393→
   394→    // Check if already processed this exact content
   395→    // Use characterMessageKey for cache to avoid cross-character conflicts in group chat
   396→    // But we need to compute characterMessageKey first
   397→    let characterMessageKey: string;
   398→    if (character?.id) {
   399→      // Check if messageKey already ends with the characterId (group chat case)
   400→      if (messageKey.endsWith(`_${character.id}`)) {
   401→        characterMessageKey = messageKey;
   402→      } else {
   403→        characterMessageKey = `${messageKey}_${character.id}`;
   404→      }
   405→    } else {
   406→      characterMessageKey = messageKey;
   407→    }
   408→    
   409→    const lastProcessed = lastProcessedRef.current.get(characterMessageKey);
   410→    if (lastProcessed === content) {
   411→      console.log(`[TriggerSystem] Skipping - same content as last processed for ${characterMessageKey}`);
   412→      return;
   413→    }
   414→    lastProcessedRef.current.set(characterMessageKey, content);
   415→    
   416→    console.log(`[TriggerSystem] Using characterMessageKey: ${characterMessageKey}`);
   417→    
   418→    // Create context (needed for both token-based triggers AND solicitud processing)
   419→    const context: TriggerContext = {
   420→      character,
   421→      characters,
   422→      fullText: content,
   423→      messageKey: characterMessageKey,
   424→      isStreaming: true,
   425→      timestamp: Date.now(),
   426→    };
   427→    
   428→    // ============================================
   429→    // UNIFIED KEY DETECTION (NEW SYSTEM)
   430→    // Detect ALL key formats in one pass: [key], |key|, Peticion:key, key:value, etc.
   431→    // ============================================
   432→    const keyDetector = getKeyDetector();
   433→    const registry = getHandlerRegistry();
   434→    
   435→    // Detect formatted keys incrementally (only NEW keys since last call)
   436→    // Use characterMessageKey for independent tracking per character
   437→    const newKeys = keyDetector.detectKeys(content, characterMessageKey);
   438→    
   439→    // Also detect plain word keys for registered sound triggers
   440→    // This is needed because "glohg" and "gluck" are plain words without special format
   441→    const soundKeywords = store.soundTriggers
   442→      ?.filter(t => t.active)
   443→      ?.flatMap(t => t.keywords) || [];
   444→    
   445→    // Also detect skill activation keys as plain words
   446→    // This is needed because "hab1" can appear as a plain word
   447→    const skillActivationKeys = character?.statsConfig?.skills
   448→      ?.filter(s => s.activationKey || (s.activationKeys && s.activationKeys.length > 0))
   449→      ?.flatMap(s => [
   450→        s.activationKey,
   451→        ...(s.activationKeys || [])
   452→      ].filter(Boolean)) || [];
   453→    
   454→    // Also detect background trigger keys as plain words
   455→    // This is needed because "sofa", "cama", etc. are plain words without special format
   456→    const backgroundKeywords = store.backgroundTriggerPacks
   457→      ?.filter(p => p.active)
   458→      ?.flatMap(p => p.items.filter(i => i.enabled).flatMap(i => i.triggerKeys)) || [];
   459→    
   460→    // Also detect quest activation and objective completion keys as plain words
   461→    // Quest activation keys from available quests
   462→    const activeSession = store.getActiveSession?.();
   463→    const sessionQuests = activeSession?.sessionQuests || [];
   464→    
   465→    // Get quest templates for looking up activation/completion keys
   466→    const questTemplates = store.questTemplates || [];
   467→    
   468→    // Activation keys from available quests
   469→    const questActivationKeys = sessionQuests
   470→      .filter(q => q.status === 'available')
   471→      .flatMap(q => {
   472→        const template = questTemplates.find(t => t.id === q.templateId);
   473→        if (template?.activation?.method === 'keyword') {
   474→          return [
   475→            template.activation.key,
   476→            ...(template.activation.keys || [])
   477→          ].filter(Boolean);
   478→        }
   479→        return [];
   480→      });
   481→    
   482→    // Objective completion keys from active quests
   483→    const questObjectiveKeys = sessionQuests
   484→      .filter(q => q.status === 'active')
   485→      .flatMap(q => {
   486→        const template = questTemplates.find(t => t.id === q.templateId);
   487→        if (!template) return [];
   488→        return template.objectives.flatMap(obj => {
   489→          const sessionObj = q.objectives.find(o => o.templateId === obj.id);
   490→          if (sessionObj?.isCompleted) return [];
   491→          return [
   492→            obj.completion?.key,
   493→            ...(obj.completion?.keys || [])
   494→          ].filter(Boolean);
   495→        });
   496→      });
   497→    
   498→    // Quest completion keys from active quests
   499→    const questCompletionKeys = sessionQuests
   500→      .filter(q => q.status === 'active')
   501→      .flatMap(q => {
   502→        const template = questTemplates.find(t => t.id === q.templateId);
   503→        if (!template) return [];
   504→        return [
   505→          template.completion?.key,
   506→          ...(template.completion?.keys || [])
   507→        ].filter(Boolean);
   508→      });
   509→    
   510→    // Combine all keywords to detect as plain words
   511→    const allKeywords = [
   512→      ...soundKeywords,
   513→      ...skillActivationKeys,
   514→      ...backgroundKeywords,
   515→      ...questActivationKeys,
   516→      ...questObjectiveKeys,
   517→      ...questCompletionKeys,
   518→      // Item trigger keywords from inventory V2
   519→      ...(store.inventorySettings?.enabled
   520→        ? (store.items ?? []).flatMap(item => item.triggerKeywords ?? [])
   521→        : []),
   522→    ];
   523→    
   524→    // DEBUG: Log keyword detection setup
   525→    console.log(`[TriggerSystem] Keyword detection setup:`, {
   526→      soundKeywordsCount: soundKeywords.length,
   527→      skillActivationKeysCount: skillActivationKeys.length,
   528→      backgroundKeywordsCount: backgroundKeywords.length,
   529→      questActivationKeysCount: questActivationKeys.length,
   530→      questObjectiveKeysCount: questObjectiveKeys.length,
   531→      questCompletionKeysCount: questCompletionKeys.length,
   532→      allKeywordsCount: allKeywords.length,
   533→      // Show actual quest keywords for debugging
   534→      questActivationKeys,
   535→      questObjectiveKeys,
   536→      questCompletionKeys,
   537→      sessionQuestsCount: sessionQuests.length,
   538→      sessionQuestsStatuses: sessionQuests.map(q => q.status),
   539→    });
   540→    
   541→    // Use characterMessageKey for independent tracking per character
   542→    const wordKeys = keyDetector.detectWordKeys(content, characterMessageKey, allKeywords);
   543→    
   544→    // Combine all detected keys (formatted + plain words)
   545→    const allDetectedKeys = [...newKeys, ...wordKeys];
   546→    
   547→    console.log(`[TriggerSystem] Detection summary: ${newKeys.length} formatted + ${wordKeys.length} word = ${allDetectedKeys.length} total`);
   548→    
   549→    if (allDetectedKeys.length > 0) {
   550→      console.log(`[TriggerSystem] Detected ${allDetectedKeys.length} keys:`, 
   551→        allDetectedKeys.map(k => ({ key: k.key, format: k.format, position: k.position })));
   552→      
   553→      // Build contexts for each handler type
   554→      const activeSession = store.getActiveSession?.();
   555→      const sessionId = store.activeSessionId || '';
   556→      
   557→      // Include persona as pseudo-character for peticiones targeting __user__
   558→      const allCharactersWithPersona = [
   559→        ...(characters || []),
   560→        ...(config.activePersona?.statsConfig?.enabled ? [{
   561→          id: '__user__',
   562→          name: config.activePersona.name || 'User',
   563→          statsConfig: config.activePersona.statsConfig,
   564→        }] as CharacterCard[] : []),
   565→      ];
   566→      
   567→      // ============================================
   568→      // UNIFIED HANDLER REGISTRY PROCESSING
   569→      // Register all handlers and process keys through registry
   570→      // This replaces the manual loop with a unified handler system
   571→      // ============================================
   572→      
   573→      // Build context for handlers
   574→      const soundKeyHandlerContext: SoundKeyHandlerContext = {
   575→        ...context,
   576→        sessionId,
   577→        characterId: character?.id,
   578→        soundTriggers: store.soundTriggers,
   579→        soundCollections: store.soundCollections,
   580→        soundSequenceTriggers: store.soundSequenceTriggers,
   581→        soundSettings: {
   582→          enabled: settings.sound?.enabled ?? false,
   583→          globalVolume: settings.sound?.globalVolume ?? 0.85,
   584→          globalCooldown: settings.sound?.globalCooldown ?? 0,
   585→        },
   586→        cooldownContextKey: character?.id || 'default',
   587→        playSound: store.playSound?.bind(store),
   588→      };
   589→      
   590→      const spriteKeyHandlerContext: SpriteKeyHandlerContext = {
   591→        ...context,
   592→        sessionId,
   593→        characterId: character?.id,
   594→        character,
   595→        allCharacters: characters,
   596→        triggerCollections: character?.triggerCollections ?? [],
   597→        spritePacksV2: character?.spritePacksV2 ?? store.spritePacksV2 ?? [],
   598→        stateCollectionsV2: character?.stateCollectionsV2,
   599→        spritePacks: character?.spritePacks ?? store.spritePacks ?? [],
   600→        spriteTriggers: character?.spriteTriggers ?? [],
   601→        spriteIndex: store.spriteIndex,

  1111→

  1113→          executeHUDTrigger(hudResult.trigger, context, {
  1114→            updateHUDFieldValue: store.updateHUDFieldValue.bind(store),
  1115→          });
  1116→        }
  1117→      }
  1118→    }
  1119→    
  1120→    // Process Background triggers - Use Multi version for multiple background changes
  1121→    if (config.backgroundEnabled !== false && settings.backgroundTriggers?.enabled) {
  1122→      const bgContext: BackgroundTriggerContext = {
  1123→        ...context,
  1124→        backgroundPacks: store.backgroundTriggerPacks ?? [],
  1125→        backgroundCollections: store.backgroundCollections ?? [],
  1126→        backgroundSettings: {
  1127→          enabled: settings.backgroundTriggers?.enabled ?? false,
  1128→          globalCooldown: settings.backgroundTriggers?.globalCooldown ?? 0,
  1129→          transitionDuration: settings.backgroundTriggers?.transitionDuration ?? 500,
  1130→          defaultTransitionType: (settings.backgroundTriggers?.defaultTransitionType as BackgroundTransitionType) ?? 'fade',
  1131→          returnToDefaultEnabled: settings.backgroundTriggers?.returnToDefaultEnabled ?? false,
  1132→          returnToDefaultAfter: settings.backgroundTriggers?.returnToDefaultAfter ?? 300000,
  1133→          defaultBackgroundUrl: settings.backgroundTriggers?.defaultBackgroundUrl ?? '',
  1134→          globalOverlays: (settings.backgroundTriggers?.globalOverlays as BackgroundOverlay[]) ?? [],
  1135→        },
  1136→        cooldownContextKey: character?.id || 'default',
  1137→      };
  1138→      
  1139→      // Use Multi version to detect ALL background matches in order of appearance
  1140→      const bgResult = checkBackgroundTriggersMulti(
  1141→        newTokens,
  1142→        bgContext,
  1143→        backgroundHandlerState
  1144→      );
  1145→      
  1146→      if (bgResult?.matched && bgResult.triggers.length > 0) {
  1147→        executeAllBackgroundTriggers(bgResult, context, {
  1148→          setBackground: store.setBackground.bind(store),
  1149→          setOverlays: store.setActiveOverlays.bind(store),
  1150→        });
  1151→      }
  1152→    }
  1153→    
  1154→    // Process Quest triggers (New Template/Instance System)
  1155→    if (config.questEnabled !== false && store.questSettings?.enabled) {
  1156→      const activeSession = store.getActiveSession?.();
  1157→      const sessionId = store.activeSessionId || '';
  1158→      
  1159→      // Get templates and session quest instances
  1160→      const templates = store.questTemplates || [];
  1161→      const sessionQuests = activeSession?.sessionQuests || [];
  1162→      const turnCount = activeSession?.turnCount || 0;
  1163→      
  1164→      // Only process if we have quests in this session
  1165→      if (templates.length > 0 || sessionQuests.length > 0) {
  1166→        const questContext: QuestTriggerContext = {
  1167→          ...context,
  1168→          templates,
  1169→          sessionQuests,
  1170→          questSettings: store.questSettings,
  1171→          sessionId,
  1172→          turnCount,
  1173→        };
  1174→        
  1175→        const questResult = checkQuestTriggers(
  1176→          newTokens,
  1177→          content,
  1178→          questContext,
  1179→          questHandlerState
  1180→        );
  1181→        
  1182→        if (questResult.hits.length > 0) {
  1183→          // Process quest trigger hits
  1184→          for (const hit of questResult.hits) {
  1185→            console.log(`[TriggerSystem] Quest trigger: ${hit.message}`);
  1186→            
  1187→            // Handle different quest actions
  1188→            switch (hit.action) {
  1189→              case 'activate':
  1190→                // Activate a quest in the session
  1191→                store.activateQuest(sessionId, hit.questId);
  1192→                break;
  1193→                
  1194→              case 'progress':
  1195→                // Progress an objective
  1196→                // Note: store.progressQuestObjective now handles reward execution internally
  1197→                if (hit.questId && hit.objectiveId) {
  1198→                  store.progressQuestObjective(
  1199→                    sessionId,
  1200→                    hit.questId,
  1201→                    hit.objectiveId,
  1202→                    hit.progress || 1,
  1203→                    character?.id
  1204→                  );
  1205→                  // Rewards (objective + quest auto-completion) are now executed by the store.
  1206→                  // No need for duplicate reward execution here.
  1207→                }
  1208→                break;
  1209→                
  1210→              case 'complete':
  1211→                if (hit.questId) {
  1212→                  // Complete the quest
  1213→                  // Note: store.completeQuest now handles reward execution internally
  1214→                  store.completeQuest(sessionId, hit.questId, character?.id);
  1215→                  // Rewards are now executed by the store. No duplicate here.
  1216→                }
  1217→                break;
  1218→                
  1219→              case 'fail':
  1220→                if (hit.questId) {
  1221→                  store.failQuest(sessionId, hit.questId);
  1222→                }
  1223→                break;
  1224→            }
  1225→            
  1226→            // Add notification for quest events (except complete and progress with rewards which are handled separately)
  1227→            // Only add notification if:
  1228→            // - action is 'activate' or 'fail'
  1229→            // - action is 'progress' but doesn't complete objective (no rewards)
  1230→            if (store.questSettings.showNotifications) {
  1231→              if (hit.action === 'activate') {
  1232→                store.addQuestNotification({
  1233→                  questId: hit.questId,
  1234→                  questName: hit.template?.name || 'Quest',
  1235→                  type: 'quest_activated',
  1236→                  message: hit.message,
  1237→                });
  1238→              } else if (hit.action === 'fail') {
  1239→                store.addQuestNotification({
  1240→                  questId: hit.questId,
  1241→                  questName: hit.template?.name || 'Quest',
  1242→                  type: 'quest_failed',
  1243→                  message: hit.message,
  1244→                });
  1245→              } else if (hit.action === 'progress' && !hit.completesObjective) {
  1246→                // Only notify for progress that doesn't complete objective
  1247→                // Progress that completes objective already notified with rewards above
  1248→                store.addQuestNotification({
  1249→                  questId: hit.questId,
  1250→                  questName: hit.template?.name || 'Quest',
  1251→                  type: 'objective_complete',
  1252→                  message: hit.message,
  1253→                });
  1254→              }
  1255→            }
  1256→          }
  1257→        }
  1258→      }
  1259→    }
  1260→    
  1261→    // Process Item triggers (V2 — uses persona-based inventory)
  1262→    if (config.inventoryEnabled !== false && store.inventorySettings?.enabled) {
  1263→      const activePersonaId = store.activePersonaId ?? config.activePersona?.id;
  1264→      
  1265→      const itemContext: ItemTriggerContext = {
  1266→        ...context,
  1267→        items: store.items,
  1268→        inventoryEntries: [],  // Legacy V1 — not used in V2
  1269→        inventorySettings: store.inventorySettings as any,
  1270→        defaultContainerId: '',
  1271→      };
  1272→      
  1273→      const itemResult = checkItemTriggers(
  1274→        newTokens,
  1275→        content,
  1276→        itemContext,
  1277→        itemHandlerState
  1278→      );
  1279→      
  1280→      if (itemResult.hits.length > 0) {
  1281→        // Process item trigger hits using V2 store methods
  1282→        for (const hit of itemResult.hits) {
  1283→          console.log(`[TriggerSystem] Item trigger: ${hit.message}`);
  1284→          
  1285→          if (!activePersonaId) continue;
  1286→          
  1287→          switch (hit.type) {
  1288→            case 'add':
  1289→              store.addToPersona?.(activePersonaId, hit.itemId, hit.quantity);
  1290→              break;
  1291→            case 'remove':
  1292→              store.removeFromPersona?.(activePersonaId, hit.itemId, hit.quantity);
  1293→              break;
  1294→            case 'equip':
  1295→              store.equipItem?.(activePersonaId, hit.itemId);
  1296→              break;
  1297→          }
  1298→          
  1299→          // Add notification if enabled
  1300→          if (store.inventorySettings.showNotifications) {
  1301→            store.addInventoryNotification({
  1302→              type: hit.type,
  1303→              itemName: hit.item?.name || 'Item',
  1304→              quantity: hit.quantity,
  1305→              message: hit.message,
  1306→            });
  1307→          }
  1308→        }
  1309→      }
  1310→    }
  1311→    
  1312→    // Process Stats triggers (Character Attributes)
  1313→    // IMPORTANT: Only process stats for the SPEAKING character (the one in `character` parameter)
  1314→    // This prevents conflicts when multiple characters have the same attribute key
  1315→    if (config.statsEnabled !== false) {
  1316→      const activeSession = store.getActiveSession?.();
  1317→      const sessionId = store.activeSessionId || '';
  1318→
  1319→      // Only process stats for the speaking character
  1320→      // In group chats, `character` is the current speaker, not all characters
  1321→      if (character?.statsConfig?.enabled) {
  1322→        const statsContext: StatsTriggerContext = {
  1323→          ...context,
  1324→          characterId: character.id,
  1325→          statsConfig: character.statsConfig,
  1326→          sessionStats: activeSession?.sessionStats,
  1327→        };
  1328→
  1329→        const statsResult = checkStatsTriggersInText(
  1330→          content,
  1331→          statsContext,
  1332→          statsHandlerState
  1333→        );
  1334→
  1335→        if (statsResult.matched && statsResult.trigger) {
  1336→          const hits = executeStatsTrigger(statsResult.trigger, context, {
  1337→            updateCharacterStat: store.updateCharacterStat.bind(store),
  1338→            activeSessionId: sessionId,
  1339→          });
  1340→
  1341→          if (hits.length > 0) {
  1342→            console.log(`[TriggerSystem] Stats updated for ${character.name} (${character.id}): ${hits.map(h => `${h.attributeName}=${h.newValue}`).join(', ')}`);
  1343→          }
  1344→        }
  1345→      }
  1346→
  1347→      // Also process stats for the user persona (if persona has stats enabled)
  1348→      if (config.activePersona?.statsConfig?.enabled) {
  1349→        const personaStatsContext: StatsTriggerContext = {
  1350→          ...context,
  1351→          characterId: '__user__',
  1352→          characterName: config.activePersona.name || 'User',
  1353→          statsConfig: config.activePersona.statsConfig,
  1354→          sessionStats: activeSession?.sessionStats,
  1355→        };
  1356→
  1357→        const personaStatsResult = checkStatsTriggersInText(
  1358→          content,
  1359→          personaStatsContext,
  1360→          statsHandlerState
  1361→        );
  1362→
  1363→        if (personaStatsResult.matched && personaStatsResult.trigger) {
  1364→          const hits = executeStatsTrigger(personaStatsResult.trigger, personaStatsContext, {
  1365→            updateCharacterStat: store.updateCharacterStat.bind(store),
  1366→            activeSessionId: sessionId,
  1367→          });
  1368→
  1369→          if (hits.length > 0) {
  1370→            console.log(`[TriggerSystem] Persona stats updated: ${hits.map(h => `${h.attributeName}=${h.newValue}`).join(', ')}`);
  1371→          }
  1372→        }
  1373→      }
  1374→    }
  1375→    
  1376→    // ============================================
  1377→    // LEGACY: Skill Activation triggers - DISABLED
  1378→    // Now handled by SkillKeyHandler in the unified system above
  1379→    // ============================================
  1380→    // if (config.statsEnabled !== false) {
  1381→    //   ... legacy skill activation code removed ...
  1382→    // }
  1383→  }, [config, settings, store, soundHandlerState, spriteHandlerState, hudHandlerState, backgroundHandlerState, questHandlerState, itemHandlerState, statsHandlerState, skillActivationHandlerState, solicitudHandlerState, getActiveHUDTemplate, soundHandler, spriteHandler, skillHandler, solicitudHandler, backgroundHandler, hudHandler, questHandler, statsHandler, itemHandler]);
  1384→  
  1385→  /**
  1386→   * Process full content at once (non-streaming)

  1403→  
  1404→  /**
  1405→   * Complete any pending partial matches and process them
  1406→   * This should be called when streaming ends to capture key:value pairs at end of text
  1407→   */
  1408→  const completePartialMatches = useCallback((
  1409→    messageKey: string, 
  1410→    character: CharacterCard | null,
  1411→    characters?: CharacterCard[]
  1412→  ): DetectedKey | null => {
  1413→    const keyDetector = getKeyDetector();
  1414→    
  1415→    // Create characterMessageKey with same logic as processStreamingContent
  1416→    let characterMessageKey: string;
  1417→    if (character?.id) {
  1418→      // Check if messageKey already ends with the characterId (group chat case)
  1419→      if (messageKey.endsWith(`_${character.id}`)) {
  1420→        characterMessageKey = messageKey;
  1421→      } else {
  1422→        characterMessageKey = `${messageKey}_${character.id}`;
  1423→      }
  1424→    } else {
  1425→      characterMessageKey = messageKey;
  1426→    }
  1427→    
  1428→    // Complete any pending partial match
  1429→    const completedKey = keyDetector.completePartialMatch(characterMessageKey);
  1430→    
  1431→    if (completedKey) {
  1432→      console.log(`[TriggerSystem] Completed partial match:`, {
  1433→        key: completedKey.key,
  1434→        value: completedKey.value,
  1435→        format: completedKey.format,
  1436→        characterId: character?.id,
  1437→      });
  1438→      
  1439→      // Process the completed key through handlers
  1440→      const sessionId = store.activeSessionId || '';
  1441→      const activeSession = store.getActiveSession?.();
  1442→      
  1443→      // Include persona as pseudo-character for peticiones targeting __user__
  1444→      const allCharactersWithPersona = [
  1445→        ...(characters || []),
  1446→        ...(config.activePersona?.statsConfig?.enabled ? [{
  1447→          id: '__user__',
  1448→          name: config.activePersona.name || 'User',
  1449→          statsConfig: config.activePersona.statsConfig,
  1450→        }] as CharacterCard[] : []),
  1451→      ];
  1452→      
  1453→      // Build context for sprite handler (most common case for key:value)
  1454→      const spriteKeyHandlerContext: SpriteKeyHandlerContext = {
  1455→        character,
  1456→        characters,
  1457→        characterId: character?.id,
  1458→        messageKey,
  1459→        fullText: '',
  1460→        isStreaming: false,
  1461→        timestamp: Date.now(),
  1462→        sessionId,
  1463→        allCharacters: characters,
  1464→        triggerCollections: character?.triggerCollections ?? [],
  1465→        spritePacksV2: character?.spritePacksV2 ?? store.spritePacksV2 ?? [],
  1466→        stateCollectionsV2: character?.stateCollectionsV2,
  1467→        spritePacks: character?.spritePacks ?? store.spritePacks ?? [],
  1468→        spriteTriggers: character?.spriteTriggers ?? [],
  1469→        spriteIndex: store.spriteIndex,

  1471→        applyTriggerForCharacter: store.applyTriggerForCharacter?.bind(store),
  1472→        scheduleReturnToIdleForCharacter: store.scheduleReturnToIdleForCharacter?.bind(store),

  1474→        startSpriteChain: store.startSpriteChain?.bind(store),
  1475→        startSoundChain: store.startSoundChain?.bind(store),
  1476→      };
  1477→      
  1478→      const soundKeyHandlerContext: SoundKeyHandlerContext = {
  1479→        character,
  1480→        characters,
  1481→        characterId: character?.id,
  1482→        messageKey,
  1483→        fullText: '',
  1484→        isStreaming: false,
  1485→        timestamp: Date.now(),
  1486→        sessionId,
  1487→        soundTriggers: store.soundTriggers,
  1488→        soundCollections: store.soundCollections,
  1489→        soundSequenceTriggers: store.soundSequenceTriggers,
  1490→        soundSettings: {

  1492→          globalVolume: settings.sound?.globalVolume ?? 0.85,
  1493→          globalCooldown: settings.sound?.globalCooldown ?? 0,
  1494→        },
  1495→        cooldownContextKey: character?.id || 'default',
  1496→        playSound: store.playSound?.bind(store),
  1497→      };
  1498→      
  1499→      // Process through handlers
  1500→      const spriteHandler = createSpriteKeyHandler();
  1501→      const soundHandler = createSoundKeyHandler(config.maxSoundsPerMessage ?? 10);

  1519→            console.log(`[TriggerSystem] Completed partial match - Sound key matched: ${completedKey.key}=${completedKey.value}`);

  1523→      }
  1524→    }
  1525→

  1527→  }, [config, store, settings]);

  1529→  /**
  1530→   * Reset for new message
  1531→   */
  1532→  const resetForNewMessage = useCallback((messageKey: string, character: CharacterCard | null, characters?: CharacterCard[]) => {
  1533→    // First, complete any pending partial matches
  1534→    completePartialMatches(messageKey, character, characters);
  1535→    
  1536→    // Create characterMessageKey for resetting with same logic as processStreamingContent
  1537→    let characterMessageKey: string;
  1538→    if (character?.id) {
  1539→      // Check if messageKey already ends with the characterId (group chat case)
  1540→      if (messageKey.endsWith(`_${character.id}`)) {
  1541→        characterMessageKey = messageKey;
  1542→      } else {
  1543→        characterMessageKey = `${messageKey}_${character.id}`;
  1544→      }
  1545→    } else {
  1546→      characterMessageKey = messageKey;
  1547→    }
  1548→    
  1549→    console.log(`[TriggerSystem] resetForNewMessage using characterMessageKey: ${characterMessageKey}`);
  1550→    
  1551→    // Reset unified KeyDetector
  1552→    const keyDetector = getKeyDetector();
  1553→    keyDetector.reset(characterMessageKey);
  1554→    
  1555→    // Reset legacy TokenDetector
  1556→    const tokenDetector = getTokenDetector();
  1557→    tokenDetector.reset(messageKey);
  1558→    
  1559→    // Get sessionId for quest handler reset
  1560→    const sessionId = store.activeSessionId || undefined;
  1561→    
  1562→    // Reset handlers
  1563→    resetSoundHandlerState(soundHandlerState, messageKey);
  1564→    resetSpriteHandlerState(spriteHandlerState, messageKey);
  1565→    resetHUDHandlerState(hudHandlerState, messageKey);
  1566→    resetBackgroundHandlerState(backgroundHandlerState, messageKey);
  1567→    resetQuestHandlerState(questHandlerState, messageKey, sessionId);

  1570→    resetSkillActivationState(skillActivationHandlerState, messageKey);
  1571→    resetSolicitudHandlerState(solicitudHandlerState, character?.id || '', messageKey);
  1572→    
  1573→    // Clear last processed
  1574→    lastProcessedRef.current.delete(messageKey);
  1575→    
  1576→    // Emit message end event
  1577→    const bus = getTriggerBus();
  1578→    bus.emit(createMessageEndEvent(messageKey, character, ''));
  1579→  }, [soundHandlerState, spriteHandlerState, hudHandlerState, backgroundHandlerState, questHandlerState, itemHandlerState, statsHandlerState, skillActivationHandlerState, solicitudHandlerState, store.activeSessionId, completePartialMatches]);
  1580→  
  1581→  /**
  1582→   * Clear ALL trigger state - call this when chat is reset
  1583→   * This clears all detection states so quests can be re-activated

  1589→    // Clear all handler states
  1590→    clearSoundHandlerState(soundHandlerState);
  1591→    clearSpriteHandlerState(spriteHandlerState);
  1592→    clearHUDHandlerState(hudHandlerState);
  1593→    clearBackgroundHandlerState(backgroundHandlerState);
  1594→    clearQuestHandlerState(questHandlerState);
  1595→    clearItemHandlerState(itemHandlerState);
  1596→    clearStatsHandlerState(statsHandlerState);
  1597→    clearSkillActivationHandlerState(skillActivationHandlerState);
  1598→    clearSolicitudHandlerState(solicitudHandlerState);
  1599→    
  1600→    // Clear processed tracking
  1601→    lastProcessedRef.current.clear();
  1602→    
  1603→    console.log('[TriggerSystem] All state cleared');
  1604→  }, [soundHandlerState, spriteHandlerState, hudHandlerState, backgroundHandlerState, questHandlerState, itemHandlerState, statsHandlerState, skillActivationHandlerState, solicitudHandlerState]);
  1605→  
  1606→  return {
  1607→    processStreamingContent,
  1608→    processFullContent,
  1609→    completePartialMatches,
  1610→    resetForNewMessage,
  1611→    clearAllState,
  1612→    isEnabled: true,
  1613→  };
  1614→}
