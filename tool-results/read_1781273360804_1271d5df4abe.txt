   148→/**
   149→ * Clamp a numeric value to the attribute's min/max bounds
   150→ * Returns the clamped value, or the original value if not numeric or no bounds defined
   151→ */
   152→function clampAttributeValue(
   153→  value: number | string,
   154→  attributeDef: AttributeDefinition | undefined
   155→): number | string {
   156→  // Only clamp numeric values
   157→  if (typeof value !== 'number') {
   158→    return value;
   159→  }
   160→
   161→  // If no attribute definition, return as-is
   162→  if (!attributeDef) {
   163→    return value;
   164→  }
   165→
   166→  let clampedValue = value;
   167→
   168→  // Apply min bound - handle both number and string min values
   169→  if (attributeDef.min !== undefined) {
   170→    const minVal = typeof attributeDef.min === 'number'
   171→      ? attributeDef.min
   172→      : parseFloat(String(attributeDef.min));
   173→
   174→    if (!isNaN(minVal)) {
   175→      clampedValue = Math.max(clampedValue, minVal);
   176→    }
   177→  }
   178→
   179→  // Apply max bound - handle both number and string max values
   180→  if (attributeDef.max !== undefined) {
   181→    const maxVal = typeof attributeDef.max === 'number'
   182→      ? attributeDef.max
   183→      : parseFloat(String(attributeDef.max));
   184→
   185→    if (!isNaN(maxVal)) {
   186→      clampedValue = Math.min(clampedValue, maxVal);
   187→    }
   188→  }
   189→
   190→  return clampedValue;
   191→}

   193→/**
   194→ * Create default character stats from config
   195→ */
   196→function createDefaultCharacterStats(
   197→  statsConfig?: CharacterStatsConfig
   198→): CharacterSessionStats {
   199→  const attributeValues: Record<string, number | string> = {};
   200→  const lastUpdated: Record<string, number> = {};
   201→  const now = Date.now();
   202→  
   203→  if (statsConfig?.attributes) {
   204→    for (const attr of statsConfig.attributes) {
   205→      attributeValues[attr.key] = attr.defaultValue;
   206→      lastUpdated[attr.key] = now;
   207→    }
   208→  }
   209→  
   210→  return {
   211→    attributeValues,
   212→    lastUpdated,
   213→    changeLog: [],
   214→  };
   215→}

   217→/**
   218→ * Add entry to change log
   219→ */
   220→function addChangeLogEntry(
   221→  stats: CharacterSessionStats,
   222→  attribute: AttributeDefinition | undefined,
   223→  attributeKey: string,
   224→  oldValue: number | string | undefined,
   225→  newValue: number | string,
   226→  reason: StatChangeLogEntry['reason']
   227→): void {
   228→  if (!stats.changeLog) {
   229→    stats.changeLog = [];
   230→  }
   231→  
   232→  stats.changeLog.push({
   233→    attributeId: attribute?.id || attributeKey,
   234→    attributeKey,
   235→    attributeName: attribute?.name || attributeKey,
   236→    oldValue: oldValue ?? '',
   237→    newValue,
   238→    reason,
   239→    timestamp: Date.now(),
   240→  });
   241→  
   242→  // Keep only last 100 entries
   243→  if (stats.changeLog.length > 100) {
   244→    stats.changeLog = stats.changeLog.slice(-100);
   245→  }
   246→}

   252→export const createStatsSlice = (set: any, get: any): StatsSlice => ({
   253→  // Initial State
   254→  sessionStats: null,
   255→
   256→  // ============================================
   257→  // Session Stats Actions

   259→
   260→  initializeSessionStats: (sessionId, characters) => {
   261→    const state = get();
   262→    const sessions = state.sessions as Array<{ id: string; sessionStats?: SessionStats }>;
   263→    const session = sessions.find(s => s.id === sessionId);
   264→    
   265→    if (!session) return;
   266→    
   267→    // Check if already initialized
   268→    if (session.sessionStats?.initialized) return;
   269→    
   270→    const now = Date.now();
   271→    const characterStats: Record<string, CharacterSessionStats> = {};
   272→    
   273→    // Initialize stats for each character
   274→    for (const char of characters) {
   275→      characterStats[char.id] = createDefaultCharacterStats(char.statsConfig);
   276→    }
   277→    
   278→    const newSessionStats: SessionStats = {
   279→      characterStats,
   280→      solicitudes: {
   281→        characterSolicitudes: {},
   282→        lastModified: now,
   283→      },
   284→      initialized: true,
   285→      lastModified: now,
   286→      // Initialize timer state
   287→      lastTimerUpdate: now,
   288→      keywordCycleIndex: {},
   289→    };
   290→    
   291→    // Update session with new stats
   292→    set((state: any) => ({
   293→      sessions: state.sessions.map((s: any) =>
   294→        s.id === sessionId 
   295→          ? { ...s, sessionStats: newSessionStats, updatedAt: new Date().toISOString() }
   296→          : s
   297→      ),
   298→    }));
   299→  },
   300→
   301→  updateCharacterStat: (sessionId, characterId, attributeKey, value, reason = 'manual') => {
   302→    // Default result
   303→    const defaultResult: UpdateCharacterStatResult = {
   304→      oldValue: undefined,
   305→      newValue: typeof value === 'number' ? value : value,
   306→      clamped: false,
   307→      thresholdsReached: [],
   308→    };
   309→
   310→    // Read current state
   311→    const state = get();
   312→    const sessions = state.sessions as Array<{ 
   313→      id: string; 
   314→      sessionStats?: SessionStats;
   315→      characterId?: string;
   316→      groupId?: string;
   317→    }>;
   318→    const sessionIndex = sessions.findIndex(s => s.id === sessionId);
   319→    
   320→    if (sessionIndex === -1) return defaultResult;
   321→    
   322→    const session = sessions[sessionIndex];
   323→    let sessionStats = session.sessionStats;
   324→    
   325→    // Auto-initialize sessionStats if missing
   326→    if (!sessionStats) {
   327→      sessionStats = {
   328→        characterStats: {},
   329→        solicitudes: {
   330→          characterSolicitudes: {},
   331→          lastModified: Date.now(),
   332→        },
   333→        ultimo_objetivo_completado: undefined,
   334→        ultima_solicitud_completada: undefined,
   335→        ultima_solicitud_realizada: undefined,
   336→        ultima_accion_realizada: undefined,
   337→        initialized: true,
   338→        lastModified: Date.now(),
   339→      };
   340→    }
   341→    
   342→    // Auto-initialize character stats if missing
   343→    if (!sessionStats.characterStats[characterId]) {
   344→      let statsConfig: CharacterStatsConfig | undefined;
   345→      if (characterId === '__user__') {
   346→        // Look up active persona's statsConfig for __user__
   347→        const activePersonaId = (state as any).activePersonaId;
   348→        const personas: any[] = (state as any).personas || [];
   349→        const activePersona = personas.find((p: any) => p.id === activePersonaId);
   350→        statsConfig = activePersona?.statsConfig;
   351→      } else {
   352→        const character = state.characters.find((c: any) => c.id === characterId);
   353→        statsConfig = character?.statsConfig;
   354→      }
   355→      let newStats = createDefaultCharacterStats(statsConfig);
   356→      // For persona (__user__), also initialize equipment slot values
   357→      if (characterId === '__user__') {
   358→        const equipmentSlots = (state as any).inventorySettings?.equipmentSlots;
   359→        if (equipmentSlots && equipmentSlots.length > 0) {
   360→          for (const slot of equipmentSlots) {
   361→            if (!(slot.key in newStats.attributeValues)) {
   362→              newStats.attributeValues[slot.key] = '';
   363→              newStats.lastUpdated[slot.key] = Date.now();
   364→            }
   365→          }
   366→        }
   367→      }
   368→      sessionStats = {
   369→        ...sessionStats,
   370→        characterStats: {
   371→          ...sessionStats.characterStats,
   372→          [characterId]: newStats,
   373→        },
   374→      };
   375→    }
   376→    
   377→    const stats = sessionStats.characterStats[characterId];
   378→    if (!stats) return defaultResult;
   379→
   380→    const oldValue = stats.attributeValues[attributeKey];
   381→
   382→    // Find attribute definition for logging and clamping
   383→    let attributeDef: AttributeDefinition | undefined;
   384→    if (characterId === '__user__') {
   385→      // Look up active persona's statsConfig for __user__
   386→      const activePersonaId = (state as any).activePersonaId;
   387→      const personas: any[] = (state as any).personas || [];
   388→      const activePersona = personas.find((p: any) => p.id === activePersonaId);
   389→      attributeDef = activePersona?.statsConfig?.attributes?.find(
   390→        (a: AttributeDefinition) => a.key === attributeKey
   391→      );
   392→    } else {
   393→      const character = state.characters.find((c: any) => c.id === characterId);
   394→      attributeDef = character?.statsConfig?.attributes?.find(
   395→        (a: AttributeDefinition) => a.key === attributeKey
   396→      );
   397→    }
   398→
   399→    // Clamp value to min/max bounds
   400→    const clampedValue = clampAttributeValue(value, attributeDef);
   401→    const clamped = clampedValue !== value;
   402→
   403→    // Log if clamping occurred
   404→    if (clamped) {
   405→      console.log(`[StatsSlice] Clamped ${attributeKey}: ${value} → ${clampedValue} (min: ${attributeDef?.min}, max: ${attributeDef?.max})`);
   406→    }
   407→
   408→    // Detect threshold reached
   409→    const thresholdsReached: ThresholdReachedInfo[] = [];
   410→    
   411→    if (attributeDef) {
   412→      // V2: Evaluate new thresholdEffects (flexible conditions with priority)
   413→      if (attributeDef.thresholdEffects && attributeDef.thresholdEffects.length > 0) {
   414→        // Build a temporary session stats with the new value for evaluation
   415→        const tempSessionStats: SessionStats = {
   416→          ...sessionStats,
   417→          characterStats: {
   418→            ...sessionStats.characterStats,
   419→            [characterId]: {
   420→              ...stats,
   421→              attributeValues: {
   422→                ...stats.attributeValues,
   423→                [attributeKey]: clampedValue,
   424→              },
   425→            },
   426→          },
   427→        };
   428→        
   429→        const matchingEffects = evaluateThresholdEffects(
   430→          attributeDef.thresholdEffects,
   431→          tempSessionStats,
   432→          characterId
   433→        );
   434→        
   435→        for (const effect of matchingEffects) {
   436→          thresholdsReached.push({
   437→            attributeKey: attributeDef.key,
   438→            attributeName: attributeDef.name,
   439→            thresholdType: 'custom',
   440→            effectName: effect.name,
   441→            effectId: effect.id,
   442→            priority: effect.priority,
   443→            rewards: effect.rewards,
   444→          });
   445→          console.log(`[StatsSlice] Threshold effect "${effect.name}" triggered for ${attributeDef.name} (priority: ${effect.priority})`);
   446→        }
   447→      }
   448→      
   449→      // Legacy: Check old onMinReached/onMaxReached (only if no thresholdEffects defined)
   450→      if (!attributeDef.thresholdEffects || attributeDef.thresholdEffects.length === 0) {
   451→        if (typeof clampedValue === 'number') {
   452→          // Check if reached minimum
   453→          if (attributeDef.min !== undefined && clampedValue === attributeDef.min) {
   454→            if (attributeDef.onMinReached?.enabled && attributeDef.onMinReached.rewards.length > 0) {
   455→              thresholdsReached.push({
   456→                attributeKey: attributeDef.key,
   457→                attributeName: attributeDef.name,
   458→                thresholdType: 'min',
   459→                thresholdValue: attributeDef.min,
   460→                rewards: attributeDef.onMinReached.rewards,
   461→              });
   462→              console.log(`[StatsSlice] Threshold reached: ${attributeDef.name} hit minimum (${attributeDef.min})`);
   463→            }
   464→          }
   465→          
   466→          // Check if reached maximum
   467→          if (attributeDef.max !== undefined && clampedValue === attributeDef.max) {
   468→            if (attributeDef.onMaxReached?.enabled && attributeDef.onMaxReached.rewards.length > 0) {
   469→              thresholdsReached.push({
   470→                attributeKey: attributeDef.key,
   471→                attributeName: attributeDef.name,
   472→                thresholdType: 'max',
   473→                thresholdValue: attributeDef.max,
   474→                rewards: attributeDef.onMaxReached.rewards,
   475→              });
   476→              console.log(`[StatsSlice] Threshold reached: ${attributeDef.name} hit maximum (${attributeDef.max})`);
   477→            }
   478→          }
   479→        }
   480→      }
   481→    }
   482→
   483→    // Build the updated state
   484→    const updatedCharacterStats = {
   485→      ...sessionStats.characterStats,
   486→      [characterId]: {
   487→        ...stats,
   488→        attributeValues: {
   489→          ...stats.attributeValues,
   490→          [attributeKey]: clampedValue,
   491→        },
   492→        lastUpdated: {
   493→          ...stats.lastUpdated,
   494→          [attributeKey]: Date.now(),
   495→        },
   496→      },
   497→    };
   498→
   499→    // Add to change log
   500→    addChangeLogEntry(
   501→      updatedCharacterStats[characterId],
   502→      attributeDef,
   503→      attributeKey,
   504→      oldValue,
   505→      clampedValue,
   506→      reason
   507→    );
   508→    
   509→    const newSessionStats: SessionStats = {
   510→      ...sessionStats,
   511→      characterStats: updatedCharacterStats,
   512→      lastModified: Date.now(),
   513→    };
   514→    
   515→    // Update the store
   516→    set({
   517→      sessions: state.sessions.map((s: any) =>
   518→        s.id === sessionId
   519→          ? { 
   520→              ...s, 
   521→              sessionStats: newSessionStats,
   522→              updatedAt: new Date().toISOString() 
   523→            }
   524→          : s
   525→      ),
   526→    });
   527→
   528→    // Sync currency changes to persona (Inventory V2)
   529→    // When currency is updated via quest rewards/triggers on __user__,
   530→    // also update persona.currency so the InventoryPanel/HUD stay in sync
   531→    if (characterId === '__user__' && attributeKey === 'currency' && typeof clampedValue === 'number') {
   532→      try {
   533→        const activePersonaId = (get() as any).activePersonaId;
   534→        if (activePersonaId) {
   535→          (get() as any).updatePersona?.(activePersonaId, { currency: clampedValue });
   536→        }
   537→      } catch { /* non-critical sync */ }
   538→    }
   539→
   540→    // Refresh objective visibility since attribute conditions may have changed
   541→    try {
   542→      (get() as any).refreshAllObjectiveVisibility?.(sessionId);
   543→    } catch { /* non-critical */ }
   544→
   545→    // Refresh activation conditions since attribute conditions may have changed
   546→    try {
   547→      (get() as any).refreshAllActivationConditions?.(sessionId);
   548→    } catch { /* non-critical */ }
   549→
   550→    // Return result with threshold info
   551→    return {
   552→      oldValue,
   553→      newValue: clampedValue,
   554→      clamped,
   555→      thresholdsReached,
   556→    };
   557→  },
   558→
   559→  batchUpdateCharacterStats: (sessionId, characterId, updates, reason = 'llm_detection') => {
   560→    set((state: any) => {
   561→      const sessions = state.sessions as Array<{
   562→        id: string; 
   563→        sessionStats?: SessionStats;
   564→      }>;
   565→      const sessionIndex = sessions.findIndex(s => s.id === sessionId);
   566→      
   567→      if (sessionIndex === -1) return state;
   568→      
   569→      const session = sessions[sessionIndex];
   570→      let sessionStats = session.sessionStats;
   571→      
   572→      // Auto-initialize sessionStats if missing (includes event fields reset)
   573→      if (!sessionStats) {
   574→        sessionStats = {
   575→          characterStats: {},
   576→          solicitudes: {
   577→            characterSolicitudes: {},
   578→            lastModified: Date.now(),
   579→          },
   580→          ultimo_objetivo_completado: undefined,
   581→          ultima_solicitud_completada: undefined,
   582→          ultima_solicitud_realizada: undefined,
   583→          ultima_accion_realizada: undefined,
   584→          initialized: true,
   585→          lastModified: Date.now(),
   586→        };
   587→      }
   588→      
   589→      // Auto-initialize character stats if missing
   590→      if (!sessionStats.characterStats[characterId]) {
   591→        let statsConfig: CharacterStatsConfig | undefined;
   592→        if (characterId === '__user__') {
   593→          // Look up active persona's statsConfig for __user__
   594→          const activePersonaId = (state as any).activePersonaId;
   595→          const personas: any[] = (state as any).personas || [];
   596→          const activePersona = personas.find((p: any) => p.id === activePersonaId);
   597→          statsConfig = activePersona?.statsConfig;
   598→        } else {
   599→          const character = state.characters.find((c: any) => c.id === characterId);
   600→          statsConfig = character?.statsConfig;
   601→        }
   602→        let autoStats = createDefaultCharacterStats(statsConfig);
   603→        // For persona (__user__), also initialize equipment slot values
   604→        if (characterId === '__user__') {
   605→          const equipmentSlots = (state as any).inventorySettings?.equipmentSlots;
   606→          if (equipmentSlots && equipmentSlots.length > 0) {
   607→            for (const slot of equipmentSlots) {
   608→              if (!(slot.key in autoStats.attributeValues)) {
   609→                autoStats.attributeValues[slot.key] = '';
   610→                autoStats.lastUpdated[slot.key] = Date.now();
   611→              }
   612→            }
   613→          }
   614→        }
   615→        sessionStats = {
   616→          ...sessionStats,
   617→          characterStats: {
   618→            ...sessionStats.characterStats,
   619→            [characterId]: autoStats,
   620→          },
   621→        };
   622→      }
   623→      
   624→      const stats = sessionStats.characterStats[characterId];
   625→      if (!stats) return state;
   626→      
   627→      // Look up statsConfig for the character (or persona for __user__)
   628→      let statsConfig: CharacterStatsConfig | undefined;
   629→      if (characterId === '__user__') {
   630→        const activePersonaId = (state as any).activePersonaId;
   631→        const personas: any[] = (state as any).personas || [];
   632→        const activePersona = personas.find((p: any) => p.id === activePersonaId);
   633→        statsConfig = activePersona?.statsConfig;
   634→      } else {
   635→        const character = state.characters.find((c: any) => c.id === characterId);
   636→        statsConfig = character?.statsConfig;
   637→      }
   638→      const now = Date.now();
   639→      
   640→      // Apply all updates
   641→      const newAttributeValues = { ...stats.attributeValues };
   642→      const newLastUpdated = { ...stats.lastUpdated };
   643→      const newChangeLog = [...(stats.changeLog || [])];
   644→
   645→      for (const update of updates) {
   646→        const oldValue = newAttributeValues[update.attributeKey];
   647→        const attributeDef = statsConfig?.attributes?.find(
   648→          (a: AttributeDefinition) => a.key === update.attributeKey
   649→        );
   650→
   651→        // Clamp value to min/max bounds
   652→        const clampedValue = clampAttributeValue(update.value, attributeDef);
   653→
   654→        // Log if clamping occurred
   655→        if (clampedValue !== update.value) {
   656→          console.log(`[StatsSlice] Clamped ${update.attributeKey}: ${update.value} → ${clampedValue} (min: ${attributeDef?.min}, max: ${attributeDef?.max})`);
   657→        }
   658→
   659→        newAttributeValues[update.attributeKey] = clampedValue;
   660→        newLastUpdated[update.attributeKey] = now;
   661→
   662→        newChangeLog.push({
   663→          attributeId: attributeDef?.id || update.attributeKey,
   664→          attributeKey: update.attributeKey,
   665→          attributeName: attributeDef?.name || update.attributeKey,
   666→          oldValue: oldValue ?? '',
   667→          newValue: clampedValue,
   668→          reason,
   669→          timestamp: now,
   670→        });
   671→      }
   672→      
   673→      // Keep only last 100 entries
   674→      const trimmedChangeLog = newChangeLog.slice(-100);
   675→      
   676→      const updatedCharacterStats = {
   677→        ...sessionStats.characterStats,
   678→        [characterId]: {
   679→          ...stats,
   680→          attributeValues: newAttributeValues,
   681→          lastUpdated: newLastUpdated,
   682→          changeLog: trimmedChangeLog,
   683→        },
   684→      };
   685→      
   686→      const newSessionStats: SessionStats = {
   687→        ...sessionStats,
   688→        characterStats: updatedCharacterStats,
   689→        lastModified: now,
   690→      };
   691→      
   692→      return {
   693→        sessions: state.sessions.map((s: any) =>
   694→          s.id === sessionId
   695→            ? { 
   696→                ...s, 
   697→                sessionStats: newSessionStats,
   698→                updatedAt: new Date().toISOString() 
   699→              }
   700→            : s
   701→        ),
   702→      };
   703→    });
   704→
   705→    // Refresh objective visibility since attribute conditions may have changed
   706→    try {
   707→      (get() as any).refreshAllObjectiveVisibility?.(sessionId);
   708→    } catch { /* non-critical */ }
   709→
   710→    // Refresh activation conditions since attribute conditions may have changed
   711→    try {
   712→      (get() as any).refreshAllActivationConditions?.(sessionId);
   713→    } catch { /* non-critical */ }
   714→  },

  1334→  processTimerTicks: (sessionId, characterId, statsConfig) => {
  1335→    const state = get();
  1336→    const sessions = state.sessions as Array<{
  1337→      id: string;
  1338→      sessionStats?: SessionStats;
  1339→    }>;
  1340→    const session = sessions.find(s => s.id === sessionId);
  1341→
  1342→    if (!session?.sessionStats) return null;
  1343→
  1344→    // Check if timer is enabled
  1345→    if (!statsConfig.timerEnabled) return null;
  1346→
  1347→    // Evaluate timer ticks
  1348→    const result = evaluateTimerTicks(statsConfig, session.sessionStats, characterId);
  1349→
  1350→    if (result.updates.length === 0) {
  1351→      // No updates needed — preserve lastTimerUpdate for fractional progress
  1352→      // Only update if timer state changed (e.g., newLastTimerUpdate advanced from consumed ticks)
  1353→      const currentLastTimerUpdate = session.sessionStats.lastTimerUpdate;
  1354→      if (result.newLastTimerUpdate !== currentLastTimerUpdate) {
  1355→        set((state: any) => ({
  1356→          sessions: state.sessions.map((s: any) =>
  1357→            s.id === sessionId
  1358→              ? {
  1359→                  ...s,
  1360→                  sessionStats: {
  1361→                    ...s.sessionStats,
  1362→                    lastTimerUpdate: result.newLastTimerUpdate,
  1363→                    keywordCycleIndex: {
  1364→                      ...s.sessionStats?.keywordCycleIndex,
  1365→                      ...result.newCycleIndex,
  1366→                    },
  1367→                  },
  1368→                  updatedAt: new Date().toISOString(),
  1369→                }
  1370→              : s
  1371→          ),
  1372→        }));
  1373→      }
  1374→      return result;
  1375→    }
  1376→
  1377→    // Apply all updates via batchUpdateCharacterStats
  1378→    const storeActions = {
  1379→      updateCharacterStat: get().updateCharacterStat,
  1380→    };
  1381→
  1382→    // Apply each update individually to get proper clamping, logging, and threshold detection
  1383→    for (const update of result.updates) {
  1384→      storeActions.updateCharacterStat(
  1385→        sessionId,
  1386→        characterId,
  1387→        update.attributeKey,
  1388→        update.value,
  1389→        'timer'
  1390→      );
  1391→    }
  1392→
  1393→    // Update timer state (lastTimerUpdate + keywordCycleIndex)
  1394→    set((state: any) => {
  1395→      const sessions = state.sessions as Array<{
  1396→        id: string;
  1397→        sessionStats?: SessionStats;
  1398→      }>;
  1399→      const session = sessions.find(s => s.id === sessionId);
  1400→      if (!session?.sessionStats) return state;
  1401→
  1402→      const newSessionStats: SessionStats = {
  1403→        ...session.sessionStats,
  1404→        lastTimerUpdate: result.newLastTimerUpdate,
  1405→        keywordCycleIndex: {
  1406→          ...session.sessionStats.keywordCycleIndex,
  1407→          ...result.newCycleIndex,
  1408→        },
  1409→        lastModified: Date.now(),
  1410→      };
  1411→
  1412→      return {
  1413→        sessions: state.sessions.map((s: any) =>
  1414→          s.id === sessionId
  1415→            ? {
  1416→                ...s,
  1417→                sessionStats: newSessionStats,
  1418→                updatedAt: new Date().toISOString(),
  1419→              }
  1420→            : s
  1421→        ),
  1422→      };
  1423→    });
  1424→
  1425→    // Log timer results
  1426→    if (result.details.length > 0) {
  1427→      console.log(`[Timer] Processed ${result.details.length} timer updates for character ${characterId}:`);
  1428→      for (const detail of result.details) {
  1429→        console.log(`  ${detail.attributeName}: ${detail.oldValue} → ${detail.newValue} (${detail.operation})`);
  1430→      }
  1431→    }
  1432→
  1433→    return result;
  1434→  },

  1436→  startSessionTimer: (sessionId, characterId, statsConfig) => {
  1437→    // First, process any accumulated ticks from offline time for this character
  1438→    get().processTimerTicks(sessionId, characterId, statsConfig);
  1439→
  1440→    // Check if timer should run
  1441→    if (!hasActiveTimers(statsConfig)) {
  1442→      console.log(`[Timer] No active timers for character ${characterId}, not starting periodic timer`);
  1443→      return;
  1444→    }
  1445→
  1446→    // Get tick interval from config (default 60 seconds)
  1447→    const tickSeconds = statsConfig.timerTickSeconds || 60;
  1448→
  1449→    // Store timer reference
  1450→    const timerKey = `timer_${sessionId}`;
  1451→    const existingInterval = (get() as any)._timerIntervals?.[timerKey];
  1452→    if (existingInterval) {
  1453→      // Timer already running for this session - check if we need a faster interval
  1454→      const existingTickSeconds = (get() as any)._timerTickSeconds?.[timerKey] || 60;
  1455→      if (tickSeconds < existingTickSeconds) {
  1456→        // Restart with the faster interval to honor this character's requirement
  1457→        clearInterval(existingInterval);
  1458→        console.log(`[Timer] Restarting timer for session ${sessionId} with faster interval: ${existingTickSeconds}s → ${tickSeconds}s`);
  1459→      } else {
  1460→        // Existing interval is fast enough, no need to restart
  1461→        console.log(`[Timer] Timer already running for session ${sessionId} at ${existingTickSeconds}s, requested ${tickSeconds}s is slower or equal`);
  1462→        return;
  1463→      }
  1464→    }
  1465→
  1466→    const intervalId = setInterval(() => {
  1467→      // Process timer ticks for ALL characters in this session that have timerEnabled
  1468→      const currentState = get();
  1469→      const currentSession = (currentState.sessions as any[])?.find((s: any) => s.id === sessionId);
  1470→      if (!currentSession) {
  1471→        get().stopSessionTimer(sessionId);
  1472→        return;
  1473→      }
  1474→
  1475→      let anyTimerActive = false;
  1476→
  1477→      // Collect all character IDs that have stats in this session
  1478→      const charIds = Object.keys(currentSession.sessionStats?.characterStats || {});
  1479→      for (const charId of charIds) {
  1480→        if (charId === '__user__') continue; // Skip persona
  1481→        const charStatsConfig = currentState.characters?.find((c: any) => c.id === charId)?.statsConfig;
  1482→        if (charStatsConfig?.timerEnabled && hasActiveTimers(charStatsConfig)) {
  1483→          get().processTimerTicks(sessionId, charId, charStatsConfig);
  1484→          anyTimerActive = true;
  1485→        }
  1486→      }
  1487→
  1488→      // If no timers are active anymore, stop the interval
  1489→      if (!anyTimerActive) {
  1490→        console.log(`[Timer] No active timers remaining for session ${sessionId}, stopping`);
  1491→        get().stopSessionTimer(sessionId);
  1492→      }
  1493→    }, tickSeconds * 1000);
  1494→
  1495→    // Store interval reference
  1496→    set((state: any) => ({
  1497→      ...state,
  1498→      _timerIntervals: {
  1499→        ...(state._timerIntervals || {}),
  1500→        [timerKey]: intervalId,
  1501→      },
  1502→      _timerTickSeconds: {
  1503→        ...(state._timerTickSeconds || {}),
  1504→        [timerKey]: tickSeconds,
  1505→      },
  1506→      _timerRunning: {
  1507→        ...(state._timerRunning || {}),
  1508→        [sessionId]: true,
  1509→      },
  1510→    }));
  1511→
  1512→    console.log(`[Timer] Started session timer for ${sessionId} (tick: ${tickSeconds}s)`);
  1513→  },
  1514→
  1515→  stopSessionTimer: (sessionId) => {
  1516→    const timerKey = `timer_${sessionId}`;
  1517→    const state = get();
  1518→    const intervalId = (state as any)._timerIntervals?.[timerKey];
  1519→
  1520→    if (intervalId) {
  1521→      clearInterval(intervalId);
  1522→      set((state: any) => {
  1523→        const newIntervals = { ...(state._timerIntervals || {}) };
  1524→        delete newIntervals[timerKey];
  1525→        const newTickSeconds = { ...(state._timerTickSeconds || {}) };
  1526→        delete newTickSeconds[timerKey];
  1527→        const newRunning = { ...(state._timerRunning || {}) };
  1528→        delete newRunning[sessionId];
  1529→        return {
  1530→          _timerIntervals: newIntervals,
  1531→          _timerTickSeconds: newTickSeconds,
  1532→          _timerRunning: newRunning,
  1533→        };
  1534→      });
  1535→      console.log(`[Timer] Stopped session timer for ${sessionId}`);
  1536→    }
  1537→  },
  1538→
  1539→  getTimerRunning: (sessionId) => {

  1549→/**
  1550→ * Evaluate a single requirement against current stats
  1551→ * Supports target requirements (checking attributes of other characters or persona)
  1552→ */
  1553→export function evaluateRequirement(
  1554→  requirement: StatRequirement,
  1555→  attributeValues: Record<string, number | string>,
  1556→  sessionStats?: SessionStats | null
  1557→): boolean {
  1558→  let currentValue: number | string | undefined;
  1559→
  1560→  if (requirement.targetCharacterId) {
  1561→    // Target mode - look up attribute from another character or persona
  1562→    currentValue = sessionStats?.characterStats?.[requirement.targetCharacterId]?.attributeValues?.[requirement.attributeKey];
  1563→  } else {
  1564→    // Self mode - use own attribute values
  1565→    currentValue = attributeValues[requirement.attributeKey];
  1566→  }
  1567→
  1568→  if (currentValue === undefined) return false;
  1569→
  1570→  const currentNum = typeof currentValue === 'number' ? currentValue : parseFloat(currentValue);
  1571→  const valueNum = typeof requirement.value === 'number' ? requirement.value : parseFloat(requirement.value);
  1572→
  1573→  // String comparison for text operators
  1574→  const isTextOperator = requirement.operator === 'contains' || requirement.operator === 'not_contains';
  1575→  if (isTextOperator || isNaN(currentNum) || isNaN(valueNum)) {
  1576→    const currentStr = String(currentValue).toLowerCase();
  1577→    const valueStr = String(requirement.value).toLowerCase();
  1578→
  1579→    switch (requirement.operator) {
  1580→      case '==': return currentStr === valueStr;
  1581→      case '!=': return currentStr !== valueStr;
  1582→      case 'contains': return currentStr.includes(valueStr);
  1583→      case 'not_contains': return !currentStr.includes(valueStr);
  1584→      default: return false;
  1585→    }
  1586→  }
  1587→
  1588→  switch (requirement.operator) {
  1589→    case '<': return currentNum < valueNum;
  1590→    case '<=': return currentNum <= valueNum;
  1591→    case '>': return currentNum > valueNum;
  1592→    case '>=': return currentNum >= valueNum;
  1593→    case '==': return currentNum === valueNum;
  1594→    case '!=': return currentNum !== valueNum;
  1595→    case 'between': {
  1596→      const maxNum = typeof requirement.valueMax === 'number'
  1597→        ? requirement.valueMax
  1598→        : parseFloat(requirement.valueMax?.toString() || '0');
  1599→      return currentNum >= valueNum && currentNum <= maxNum;
  1600→    }
  1601→    default: return false;
  1602→  }
  1603→}
  1604→
  1605→/**
  1606→ * Evaluate all requirements (AND logic)
  1607→ */
  1608→export function evaluateRequirements(
  1609→  requirements: StatRequirement[],
  1610→  attributeValues: Record<string, number | string>,
  1611→  sessionStats?: SessionStats | null
  1612→): boolean {
  1613→  if (!requirements || requirements.length === 0) return true;
  1614→  return requirements.every(req => evaluateRequirement(req, attributeValues, sessionStats));
  1615→}
