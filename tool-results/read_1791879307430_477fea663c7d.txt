    61→    61→
    62→    62→const bgLogger = getLogger('background');
    63→    63→
    64→    64→interface CollectionManagerProps {
    65→    65→  open: boolean;
    66→    66→  onOpenChange: (open: boolean) => void;
    67→    67→}
    68→    68→
    69→    69→export function BackgroundCollectionManager({ open, onOpenChange }: CollectionManagerProps) {
    70→    70→  const [collections, setCollections] = useState<BackgroundCollection[]>([]);
    71→    71→  const [isLoading, setIsLoading] = useState(false);
    72→    72→  const [selectedCollectionName, setSelectedCollectionName] = useState<string | null>(null);
    73→    73→  const [localCollection, setLocalCollection] = useState<BackgroundCollection | null>(null);
    74→    74→  const localCollectionRef = useRef<BackgroundCollection | null>(null);
    75→    75→  
    76→    76→  // Create collection dialog
    77→    77→  const [showCreateDialog, setShowCreateDialog] = useState(false);
    78→    78→  const [newCollectionName, setNewCollectionName] = useState('');
    79→    79→  const [newCollectionDescription, setNewCollectionDescription] = useState('');
    80→    80→  
    81→    81→  // Delete confirmation
    82→    82→  const [showDeleteDialog, setShowDeleteDialog] = useState(false);
    83→    83→  const [deleteTarget, setDeleteTarget] = useState<{ type: 'collection' | 'file'; name: string; filename?: string } | null>(null);
    84→    84→  
    85→    85→  // Upload state
    86→    86→  const [isUploading, setIsUploading] = useState(false);
    87→    87→  const fileInputRef = useRef<HTMLInputElement>(null);
    88→    88→  
    89→    89→  // Save state
    90→    90→  const [isSaving, setIsSaving] = useState(false);
    91→    91→  const [hasChanges, setHasChanges] = useState(false);
    92→    92→  
    93→    93→  // Track initial load
    94→    94→  const hasLoadedRef = useRef(false);
    95→    95→
    96→    96→  // Keep ref in sync with state
    97→    97→  useEffect(() => {
    98→    98→    localCollectionRef.current = localCollection;
    99→    99→  }, [localCollection]);
   100→   100→
   101→   101→  // Fetch collections - only depends on open
   102→   102→  const fetchCollections = useCallback(async () => {
   103→   103→    setIsLoading(true);
   104→   104→    try {
   105→   105→      const response = await fetch('/api/backgrounds/collections');
   106→   106→      if (!response.ok) {
   107→   107→        bgLogger.error('Error fetching collections', { status: response.status });
   108→   108→        return [];
   109→   109→      }
   110→   110→      const data = await response.json();
   111→   111→      setCollections(data.collections || []);
   112→   112→      return data.collections || [];
   113→   113→    } catch (error) {
   114→   114→      bgLogger.error('Error fetching collections', { error });
   115→   115→      return [];
   116→   116→    } finally {
   117→   117→      setIsLoading(false);
   118→   118→    }
   119→   119→  }, []);
   120→   120→
   121→   121→  // Initial fetch when dialog opens
   122→   122→  useEffect(() => {
   123→   123→    if (open && !hasLoadedRef.current) {
   124→   124→      fetchCollections();
   125→   125→      hasLoadedRef.current = true;
   126→   126→    }
   127→   127→    if (!open) {
   128→   128→      hasLoadedRef.current = false;
   129→   129→    }
   130→   130→  }, [open, fetchCollections]);
   131→   131→
   132→   132→  // Select a collection
   133→   133→  const handleSelectCollection = useCallback((collection: BackgroundCollection) => {
   134→   134→    setSelectedCollectionName(collection.name);
   135→   135→    setLocalCollection(collection);
   136→   136→    setHasChanges(false);
   137→   137→  }, []);
   138→   138→
   139→   139→  // Create new collection
   140→   140→  const handleCreateCollection = async () => {
   141→   141→    if (!newCollectionName.trim()) return;
   142→   142→    
   143→   143→    setIsLoading(true);
   144→   144→    try {
   145→   145→      const response = await fetch('/api/backgrounds/collections', {
   146→   146→        method: 'POST',
   147→   147→        headers: { 'Content-Type': 'application/json' },
   148→   148→        body: JSON.stringify({
   149→   149→          action: 'create',
   150→   150→          name: newCollectionName.trim(),
   151→   151→          description: newCollectionDescription.trim()
   152→   152→        })
   153→   153→      });
   154→   154→      
   155→   155→      if (!response.ok) {
   156→   156→        alert(`Error del servidor (${response.status})`);
   157→   157→        return;
   158→   158→      }
   159→   159→      
   160→   160→      const data = await response.json();
   161→   161→      
   162→   162→      if (data.success) {
   163→   163→        const newCollections = await fetchCollections();
   164→   164→        setShowCreateDialog(false);
   165→   165→        setNewCollectionName('');
   166→   166→        setNewCollectionDescription('');
   167→   167→        if (data.collection) {
   168→   168→          handleSelectCollection(data.collection);
   169→   169→        }
   170→   170→      } else {
   171→   171→        alert(data.error || 'Failed to create collection');
   172→   172→      }
   173→   173→    } catch (error) {
   174→   174→      bgLogger.error('Error creating collection', { error });
   175→   175→    } finally {
   176→   176→      setIsLoading(false);
   177→   177→    }
   178→   178→  };
   179→   179→
   180→   180→  // Delete collection
   181→   181→  const handleDeleteCollection = async () => {
   182→   182→    if (!deleteTarget || deleteTarget.type !== 'collection') return;
   183→   183→    
   184→   184→    setIsLoading(true);
   185→   185→    try {
   186→   186→      const folderName = collections.find(c => c.name === deleteTarget.name)?.path.split('/').pop();
   187→   187→      
   188→   188→      const response = await fetch(
   189→   189→        `/api/backgrounds/collections?folderName=${encodeURIComponent(folderName || deleteTarget.name)}`,
   190→   190→        { method: 'DELETE' }
   191→   191→      );
   192→   192→      
   193→   193→      if (!response.ok) {
   194→   194→        alert(`Error del servidor (${response.status})`);
   195→   195→        return;
   196→   196→      }
   197→   197→      
   198→   198→      const data = await response.json();
   199→   199→      
   200→   200→      if (data.success) {
   201→   201→        setSelectedCollectionName(null);
   202→   202→        setLocalCollection(null);
   203→   203→        await fetchCollections();
   204→   204→      } else {
   205→   205→        alert(data.error || 'Failed to delete collection');
   206→   206→      }
   207→   207→    } catch (error) {
   208→   208→      bgLogger.error('Error deleting collection', { error });
   209→   209→    } finally {
   210→   210→      setIsLoading(false);
   211→   211→      setShowDeleteDialog(false);
   212→   212→      setDeleteTarget(null);
   213→   213→    }
   214→   214→  };
   215→   215→
   216→   216→  // Delete file from collection
   217→   217→  const handleDeleteFile = async () => {
   218→   218→    if (!deleteTarget || deleteTarget.type !== 'file' || !localCollection) return;
   219→   219→    
   220→   220→    setIsLoading(true);
   221→   221→    try {
   222→   222→      const folderName = localCollection.path.split('/').pop();
   223→   223→      
   224→   224→      const response = await fetch(
   225→   225→        `/api/backgrounds/collections?folderName=${encodeURIComponent(folderName || '')}&filename=${encodeURIComponent(deleteTarget.filename || '')}`,
   226→   226→        { method: 'DELETE' }
   227→   227→      );
   228→   228→      
   229→   229→      if (!response.ok) {
   230→   230→        alert(`Error del servidor (${response.status})`);
   231→   231→        return;
   232→   232→      }
   233→   233→      
   234→   234→      const data = await response.json();
   235→   235→      
   236→   236→      if (data.success) {
   237→   237→        const newCollections = await fetchCollections();
   238→   238→        const updated = newCollections.find((c: BackgroundCollection) => c.name === selectedCollectionName);
   239→   239→        if (updated) {
   240→   240→          setLocalCollection(updated);
   241→   241→        }
   242→   242→      } else {
   243→   243→        alert(data.error || 'Failed to delete file');
   244→   244→      }
   245→   245→    } catch (error) {
   246→   246→      bgLogger.error('Error deleting file', { error });
   247→   247→    } finally {
   248→   248→      setIsLoading(false);
   249→   249→      setShowDeleteDialog(false);
   250→   250→      setDeleteTarget(null);
   251→   251→    }
   252→   252→  };
   253→   253→
   254→   254→  // Upload file to collection
   255→   255→  const handleUploadFile = async (event: React.ChangeEvent<HTMLInputElement>) => {
   256→   256→    const file = event.target.files?.[0];
   257→   257→    if (!file || !localCollection) return;
   258→   258→    
   259→   259→    setIsUploading(true);
   260→   260→    try {
   261→   261→      const formData = new FormData();
   262→   262→      formData.append('file', file);
   263→   263→      formData.append('type', 'background');
   264→   264→      formData.append('collection', localCollection.path.split('/').pop() || '');
   265→   265→      
   266→   266→      const response = await fetch('/api/upload', {
   267→   267→        method: 'POST',
   268→   268→        body: formData
   269→   269→      });
   270→   270→      
   271→   271→      const data = await response.json();
   272→   272→      
   273→   273→      if (data.success) {
   274→   274→        const newCollections = await fetchCollections();
   275→   275→        const updated = newCollections.find((c: BackgroundCollection) => c.name === selectedCollectionName);
   276→   276→        if (updated) {
   277→   277→          setLocalCollection(updated);
   278→   278→        }
   279→   279→      } else {
   280→   280→        alert(data.error || 'Failed to upload file');
   281→   281→      }
   282→   282→    } catch (error) {
   283→   283→      bgLogger.error('Error uploading file', { error });
   284→   284→    } finally {
   285→   285→      setIsUploading(false);
   286→   286→      if (fileInputRef.current) {
   287→   287→        fileInputRef.current.value = '';
   288→   288→      }
   289→   289→    }
   290→   290→  };
   291→   291→
   292→   292→  // Save collection metadata
   293→   293→  const handleSaveCollection = async () => {
   294→   294→    if (!localCollection) return;
   295→   295→    
   296→   296→    setIsSaving(true);
   297→   297→    try {
   298→   298→      const folderName = localCollection.path.split('/').pop();
   299→   299→      
   300→   300→      const response = await fetch('/api/backgrounds/collections', {
   301→   301→        method: 'PUT',
   302→   302→        headers: { 'Content-Type': 'application/json' },
   303→   303→        body: JSON.stringify({
   304→   304→          folderName,
   305→   305→          name: localCollection.name,
   306→   306→          description: localCollection.description,
   307→   307→          entries: localCollection.entries,
   308→   308→          transitionDuration: localCollection.transitionDuration
   309→   309→        })
   310→   310→      });
   311→   311→      
   312→   312→      if (!response.ok) {
   313→   313→        alert(`Error del servidor (${response.status})`);
   314→   314→        return;
   315→   315→      }
   316→   316→      
   317→   317→      const data = await response.json();
   318→   318→      
   319→   319→      if (data.success) {
   320→   320→        setHasChanges(false);
   321→   321→        await fetchCollections();
   322→   322→      } else {
   323→   323→        alert(data.error || 'Failed to save collection');
   324→   324→      }
   325→   325→    } catch (error) {
   326→   326→      bgLogger.error('Error saving collection', { error });
   327→   327→    } finally {
   328→   328→      setIsSaving(false);
   329→   329→    }
   330→   330→  };
   331→   331→
   332→   332→  // Update local collection state
   333→   333→  const updateLocalCollection = (updates: Partial<BackgroundCollection>) => {
   334→   334→    if (!localCollection) return;
   335→   335→    setLocalCollection({ ...localCollection, ...updates });
   336→   336→    setHasChanges(true);
   337→   337→  };
   338→   338→
   339→   339→  // Update entry in local collection
   340→   340→  const updateEntry = (entryId: string, updates: Partial<BackgroundCollectionEntry>) => {
   341→   341→    if (!localCollection) return;
   342→   342→    
   343→   343→    setLocalCollection(prev => {
   344→   344→      if (!prev) return prev;
   345→   345→      return {
   346→   346→        ...prev,
   347→   347→        entries: prev.entries.map(e => 
   348→   348→          e.id === entryId ? { ...e, ...updates } : e
   349→   349→        )
   350→   350→      };
   351→   351→    });
   352→   352→    setHasChanges(true);
   353→   353→  };
   354→   354→
   355→   355→  // Sync entries with files
   356→   356→  const syncEntriesWithFiles = () => {
   357→   357→    if (!localCollection) return;
   358→   358→    
   359→   359→    const existingUrls = new Set(localCollection.entries.map(e => e.url));
   360→   360→    const newEntries: BackgroundCollectionEntry[] = [...localCollection.entries];
   361→   361→    
   362→   362→    for (const file of localCollection.files) {
   363→   363→      if (!existingUrls.has(file.url)) {
   364→   364→        newEntries.push({
   365→   365→          id: `bg_${Date.now()}_${Math.random().toString(36).substring(2, 8)}`,
   366→   366→          name: file.name.replace(/\.[^.]+$/, ''),
   367→   367→          url: file.url,
   368→   368→          triggerKeys: [],
   369→   369→          contextKeys: [],
   370→   370→          tags: file.type === 'video' ? ['video'] : []
   371→   371→        });
   372→   372→      }
   373→   373→    }
   374→   374→    
   375→   375→    setLocalCollection(prev => {
   376→   376→      if (!prev) return prev;
   377→   377→      return { ...prev, entries: newEntries };
   378→   378→    });
   379→   379→    setHasChanges(true);
   380→   380→  };
   381→   381→
   382→   382→  // Get entry for file
   383→   383→  const getEntryForFile = (file: BackgroundFile): BackgroundCollectionEntry | undefined => {
   384→   384→    return localCollection?.entries.find(e => e.url === file.url);
   385→   385→  };
   386→   386→
   387→   387→  return (
   388→   388→    <AnimatePresence>
   389→   389→      {open && (
   390→   390→        <motion.div
   391→   391→          key="collection-manager-overlay"
   392→   392→          initial={{ opacity: 0 }}
   393→   393→          animate={{ opacity: 1 }}
   394→   394→          exit={{ opacity: 0 }}
   395→   395→          transition={{ duration: 0.2, ease: 'easeInOut' }}
   396→   396→          className="fixed inset-0 z-[55] bg-background"
   397→   397→        >
   398→   398→          <div className="h-full flex flex-col">
   399→   399→            {/* Header */}
   400→   400→            <motion.div
   401→   401→              initial={{ y: -8, opacity: 0 }}
   402→   402→              animate={{ y: 0, opacity: 1 }}
   403→   403→              transition={{ duration: 0.2, delay: 0.05, ease: 'easeOut' }}
   404→   404→              className="h-14 border-b bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/60 flex items-center justify-between px-4 flex-shrink-0"
   405→   405→            >
   406→   406→              <div className="flex items-center gap-2">
   407→   407→                <FolderOpen className="w-5 h-5 text-muted-foreground" />
   408→   408→                <h2 className="font-semibold text-sm">Gestionar Colecciones</h2>
   409→   409→                {hasChanges && (
   410→   410→                  <Badge variant="outline" className="text-violet-500 border-violet-500 text-xs ml-2">
   411→   411→                    Sin guardar
   412→   412→                  </Badge>
   413→   413→                )}
   414→   414→              </div>
   415→   415→              <div className="flex items-center gap-1.5">
   416→   416→                <Button 
   417→   417→                  variant="outline" 
   418→   418→                  size="sm" 
   419→   419→                  className="h-8 text-xs"
   420→   420→                  onClick={handleSaveCollection}
   421→   421→                  disabled={!hasChanges || isSaving}
   422→   422→                >
   423→   423→                  {isSaving ? (
   424→   424→                    <Loader2 className="w-3.5 h-3.5 animate-spin" />
   425→   425→                  ) : (
   426→   426→                    <Save className="w-3.5 h-3.5" />
   427→   427→                  )}
   428→   428→                  <span className="ml-1 hidden sm:inline">Guardar</span>
   429→   429→                </Button>
   430→   430→                <Button 
   431→   431→                  variant="ghost" 
   432→   432→                  size="icon"
   433→   433→                  className="h-8 w-8"
   434→   434→                  onClick={() => fetchCollections()}
   435→   435→                  disabled={isLoading}
   436→   436→                >
   437→   437→                  {isLoading ? (
   438→   438→                    <Loader2 className="w-4 h-4 animate-spin" />
   439→   439→                  ) : (
   440→   440→                    <RefreshCw className="w-4 h-4" />
   441→   441→                  )}
   442→   442→                </Button>
   443→   443→                <Button 
   444→   444→                  variant="ghost" 
   445→   445→                  size="icon"
   446→   446→                  className="h-8 w-8 text-muted-foreground hover:text-foreground"
   447→   447→                  onClick={() => onOpenChange(false)}
   448→   448→                >
   449→   449→                  <X className="w-4 h-4" />
   450→   450→                </Button>
   451→   451→              </div>
   452→   452→            </motion.div>
   453→   453→
   454→   454→            {/* Body: Sidebar + Main Content */}
   455→   455→            <motion.div
   456→   456→              initial={{ opacity: 0, y: 4 }}
   457→   457→              animate={{ opacity: 1, y: 0 }}
   458→   458→              transition={{ duration: 0.2, delay: 0.1, ease: 'easeOut' }}
   459→   459→              className="flex-1 flex overflow-hidden"
   460→   460→            >
   461→   461→              {/* Collections Sidebar */}
   462→   462→              <div className="w-14 md:w-64 border-r bg-muted/30 flex flex-col flex-shrink-0">
   463→   463→                <div className="p-1.5 md:p-2 border-b">
   464→   464→                  <Tooltip>
   465→   465→                    <TooltipTrigger asChild>
   466→   466→                      <Button
   467→   467→                        variant="default"
   468→   468→                        size="sm"
   469→   469→                        className="w-full h-8"
   470→   470→                        onClick={() => setShowCreateDialog(true)}
   471→   471→                      >
   472→   472→                        <FolderPlus className="w-4 h-4 shrink-0" />
   473→   473→                        <span className="hidden md:inline ml-2">Nueva Colección</span>
   474→   474→                      </Button>
   475→   475→                    </TooltipTrigger>
   476→   476→                    <TooltipContent side="right" className="md:hidden">
   477→   477→                      Nueva Colección
   478→   478→                    </TooltipContent>
   479→   479→                  </Tooltip>
   480→   480→                </div>
   481→   481→                <ScrollArea className="flex-1">
   482→   482→                  <div className="p-1.5 md:p-2 space-y-0.5">
   483→   483→                    {collections.map(collection => (
   484→   484→                      <Tooltip key={collection.name}>
   485→   485→                        <TooltipTrigger asChild>
   486→   486→                          <div
   487→   487→                            className={cn(
   488→   488→                              "group flex items-center gap-2 p-2 rounded-md cursor-pointer transition-colors",
   489→   489→                              selectedCollectionName === collection.name
   490→   490→                                ? "bg-primary text-primary-foreground"
   491→   491→                                : "hover:bg-muted"
   492→   492→                            )}
   493→   493→                            onClick={() => handleSelectCollection(collection)}
   494→   494→                          >
   495→   495→                            <FolderOpen className="w-4 h-4 flex-shrink-0" />
   496→   496→                            <div className="flex-1 min-w-0 hidden md:block">
   497→   497→                              <p className="font-medium truncate text-sm">{collection.name}</p>
   498→   498→                              <p className="text-xs opacity-70">
   499→   499→                                {collection.files.length} archivo{collection.files.length !== 1 ? 's' : ''}
   500→   500→                              </p>
   501→   501→                            </div>
   502→   502→                            <Button
   503→   503→                              variant="ghost"
   504→   504→                              size="icon"
   505→   505→                              className={cn(
   506→   506→                                "h-6 w-6 shrink-0",
   507→   507→                                selectedCollectionName === collection.name 
   508→   508→                                  ? "text-primary-foreground hover:text-primary-foreground hover:bg-primary/80" 
   509→   509→                                  : "opacity-0 group-hover:opacity-100"
   510→   510→                              )}
   511→   511→                              onClick={(e) => {
