   463→function ExtraccionTab() {
   464→  const settings = useTavernStore((s) => s.settings);
   465→  const embeddingsChat = settings.embeddingsChat ?? DEFAULT_EMBEDDINGS_CHAT;
   466→  const updateSettings = useTavernStore((state) => state.updateSettings);
   467→
   468→  // Context settings with defaults
   469→  const contextSettings = settings.context ?? {
   470→    maxMessages: 50,
   471→    maxTokens: 4096,
   472→    keepFirstN: 1,
   473→    keepLastN: 20,
   474→  };
   475→
   476→  // Update context settings helper
   477→  const updateContextSettings = useCallback((updates: Partial<typeof contextSettings>) => {
   478→    updateSettings({
   479→      context: { ...contextSettings, ...updates }
   480→    });
   481→  }, [contextSettings, updateSettings]);
   482→
   483→  // Prompt editor state
   484→  const [activePromptTab, setActivePromptTab] = useState<'normal' | 'group'>('normal');
   485→  const [localPrompt, setLocalPrompt] = useState(() => embeddingsChat.memoryExtractionPrompt || DEFAULT_MEMORY_EXTRACTION_PROMPT);
   486→  const [localGroupPrompt, setLocalGroupPrompt] = useState(() => embeddingsChat.groupMemoryExtractionPrompt || DEFAULT_GROUP_MEMORY_EXTRACTION_PROMPT);
   487→  const [showPreview, setShowPreview] = useState(false);
   488→
   489→  const isNormal = activePromptTab === 'normal';
   490→  const currentPrompt = isNormal ? localPrompt : localGroupPrompt;
   491→  const currentStored = isNormal
   492→    ? (embeddingsChat.memoryExtractionPrompt || DEFAULT_MEMORY_EXTRACTION_PROMPT)
   493→    : (embeddingsChat.groupMemoryExtractionPrompt || DEFAULT_GROUP_MEMORY_EXTRACTION_PROMPT);
   494→  const previewData = isNormal ? NORMAL_PREVIEW : GROUP_PREVIEW;
   495→
   496→  const handleSavePrompt = () => {
   497→    if (isNormal) {
   498→      updateSettings({ embeddingsChat: { ...embeddingsChat, memoryExtractionPrompt: localPrompt } });
   499→    } else {
   500→      updateSettings({ embeddingsChat: { ...embeddingsChat, groupMemoryExtractionPrompt: localGroupPrompt } });
   501→    }
   502→  };
   503→
   504→  const handleRestoreDefault = () => {
   505→    if (isNormal) {
   506→      setLocalPrompt(DEFAULT_MEMORY_EXTRACTION_PROMPT);
   507→      updateSettings({ embeddingsChat: { ...embeddingsChat, memoryExtractionPrompt: DEFAULT_MEMORY_EXTRACTION_PROMPT } });
   508→    } else {
   509→      setLocalGroupPrompt(DEFAULT_GROUP_MEMORY_EXTRACTION_PROMPT);
   510→      updateSettings({ embeddingsChat: { ...embeddingsChat, groupMemoryExtractionPrompt: DEFAULT_GROUP_MEMORY_EXTRACTION_PROMPT } });
   511→    }
   512→  };
   513→
   514→  const handleChange = (value: string) => {
   515→    if (isNormal) setLocalPrompt(value);
   516→    else setLocalGroupPrompt(value);
   517→  };
   518→
   519→  const previewText = currentPrompt
   520→    .replace('{characterName}', previewData.characterName)
   521→    .replace('{chatContext}', previewData.chatContext)
   522→    .replace('{lastMessage}', previewData.lastMessage);
   523→
   524→  const hasChanges = currentPrompt !== currentStored;
   525→
   526→  return (
   527→    <div className="space-y-6">
   528→      {/* Note about needing Embeddings enabled */}
   529→      <div className="bg-amber-500/5 border border-amber-500/20 rounded-lg p-4">
   530→        <div className="flex items-start gap-2">
   531→          <Info className="w-4 h-4 text-amber-500 mt-0.5 shrink-0" />
   532→          <div className="space-y-1">
   533→            <p className="text-sm font-medium text-amber-600 dark:text-amber-400">Requiere Embeddings</p>
   534→            <p className="text-xs text-muted-foreground">
   535→              Estos ajustes requieren que Embeddings esté activado. Configúralo en la pestaña <strong>Base de Conocimiento</strong>.
   536→            </p>
   537→          </div>
   538→        </div>
   539→      </div>
   540→
   541→      {/* Memory Extraction Section */}
   542→      <Card>
   543→        <CardHeader className="pb-3">
   544→          <CardTitle className="flex items-center gap-2 text-base">
   545→            <Brain className="w-4 h-4 text-violet-500" />
   546→            Extracción Automática de Memoria
   547→          </CardTitle>
   548→          <CardDescription>
   549→            Extrae hechos memorables de las respuestas del personaje y los guarda como embeddings
   550→          </CardDescription>
   551→        </CardHeader>
   552→        <CardContent className="space-y-4">
   553→          <div className="flex items-center justify-between">
   554→            <div className="space-y-0.5">
   555→              <Label className="text-sm">Activar Extracción</Label>
   556→              <p className="text-[10px] text-muted-foreground">
