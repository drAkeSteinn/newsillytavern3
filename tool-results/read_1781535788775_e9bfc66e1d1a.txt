   128→// Format memory date to relative time
   129→function formatMemoryDate(date: Date): string {
   130→  const now = new Date();
   131→  const diff = now.getTime() - date.getTime();
   132→  const minutes = Math.floor(diff / 60000);
   133→  if (minutes < 1) return 'Ahora';
   134→  if (minutes < 60) return `Hace ${minutes}m`;
   135→  const hours = Math.floor(minutes / 60);
   136→  if (hours < 24) return `Hace ${hours}h`;
   137→  const days = Math.floor(hours / 24);
   138→  if (days < 7) return `Hace ${days}d`;
   139→  return date.toLocaleDateString('es-ES', { day: 'numeric', month: 'short' });
   140→}

   152→// Memory item component (outside main component for stable identity)

   162→
   163→  return (
   164→    <div className="group flex items-start gap-2 p-2 rounded-md bg-white/5 hover:bg-white/10 transition-colors">
   165→      {/* Type indicator bar */}
   166→      <div className={cn("w-1 h-full min-h-[2rem] rounded-full flex-shrink-0 mt-0.5", typeConfig.bgColor.replace('/20', '/60'))} />
   167→
   168→      <div className="flex-1 min-w-0">
   169→        {/* Top row: type badge + importance + date */}
   170→        <div className="flex items-center gap-1.5 mb-1">

   173→          </span>

   177→          </span>
   178→          {isConsolidated && (
   179→            <span className="text-[9px] text-cyan-400 bg-cyan-500/20 px-1 py-0.5 rounded">
   180→              Consolidada
   181→            </span>
   182→          )}
   183→          <span className="text-[9px] text-muted-foreground ml-auto">
   184→            {createdDate ? formatMemoryDate(createdDate) : ''}
   185→          </span>
   186→        </div>
   187→
   188→        {/* Memory content */}
   189→        <p className="text-xs leading-relaxed text-foreground/90 line-clamp-3">
   190→          {memory.content}
   191→        </p>
   192→      </div>
   193→
   194→      {/* Delete button (appears on hover) */}

   196→        onClick={() => onDelete(memory.id)}
   197→        className="opacity-0 group-hover:opacity-100 transition-opacity p-1 rounded hover:bg-red-500/20 text-muted-foreground hover:text-red-400 flex-shrink-0"

   201→      </button>
   202→    </div>
   203→  );
