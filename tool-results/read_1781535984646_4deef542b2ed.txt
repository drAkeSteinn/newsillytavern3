    97→    97→/**
    98→    98→ * Execute detected tool calls, send SSE events, and return results.
    99→    99→ * Returns { newContent: combined display messages, shouldContinue: true if tools were executed }
   100→   100→ */
   101→   101→async function executeToolCallsAndContinue(
   102→   102→  toolCalls: NativeToolCall[],
   103→   103→  availableTools: Array<{ id: string; name: string; label: string; icon: string }>,
   104→   104→  currentRound: number,
   105→   105→  maxRounds: number,
   106→   106→  character: CharacterCard,
   107→   107→  sessionId: string,
   108→   108→  userName: string,
   109→   109→  controller: { enqueue: (chunk: string) => void },
   110→   110→  sessionQuests?: SessionQuestInstance[],
   111→   111→  questTemplates?: QuestTemplate[],
   112→   112→  statsConfig?: CharacterStatsConfig,
   113→   113→  sessionStats?: SessionStats,
   114→   114→  allCharacters?: CharacterCard[],
   115→   115→  characterMemory?: CharacterMemory,
   116→   116→  lorebooks?: Lorebook[],
   117→   117→): Promise<{ newContent: string; shouldContinue: boolean; toolResults: Array<{ success: boolean; displayMessage: string }>; questActivations: QuestActivation[]; toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> }> {
   118→   118→  if (toolCalls.length === 0 || currentRound >= maxRounds) {
   119→   119→    return { newContent: '', shouldContinue: false, toolResults: [], questActivations: [], toolsUsed: [] };
   120→   120→  }
   121→   121→
   122→   122→  const toolResults: Array<{ success: boolean; displayMessage: string }> = [];
   123→   123→  const questActivations: QuestActivation[] = [];
   124→   124→  const toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> = [];
   125→   125→  let allDisplayMessages = '';
   126→   126→
   127→   127→  for (let i = 0; i < toolCalls.length; i++) {
   128→   128→    const tc = toolCalls[i];
   129→   129→    const toolDef = availableTools.find(t => t.name === tc.name);
   130→   130→    const callId = `${Date.now()}_${i}`;
   131→   131→
   132→   132→    // Send tool_call_start event
   133→   133→    controller.enqueue(createSSEJSON({
   134→   134→      type: 'tool_call_start',
   135→   135→      toolName: tc.name,
   136→   136→      toolLabel: toolDef?.label || tc.name,
   137→   137→      toolIcon: toolDef?.icon || 'Wrench',
   138→   138→      params: tc.arguments,
   139→   139→      callId,
   140→   140→    }));
   141→   141→
   142→   142→    console.log(`[Tools] Executing tool call: ${tc.name}`, tc.arguments);
   143→   143→
   144→   144→    // Execute the tool
   145→   145→    const toolResult = await executeTool(
   146→   146→      tc.name,
   147→   147→      tc.arguments,
   148→   148→      {
   149→   149→        characterId: character.id,
   150→   150→        characterName: character.name,
   151→   151→        sessionId,
   152→   152→        userName,
   153→   153→        sessionQuests,
   154→   154→        questTemplates,
   155→   155→        statsConfig: character.statsConfig,
   156→   156→        sessionStats,
   157→   157→        allCharacters,
   158→   158→        characterMemory,
   159→   159→        lorebooks,
   160→   160→      },
   161→   161→    );
   162→   162→
   163→   163→    // Send tool_call_result event
   164→   164→    controller.enqueue(createSSEJSON({
   165→   165→      type: 'tool_call_result',
   166→   166→      toolName: tc.name,
   167→   167→      success: toolResult.success,
   168→   168→      displayMessage: toolResult.displayMessage,
   169→   169→      duration: toolResult.duration || 0,
   170→   170→      callId,
   171→   171→    }));
   172→   172→
   173→   173→    // Check for quest activation and send SSE event
   174→   174→    if (toolResult.questActivation) {
   175→   175→      const activation = toolResult.questActivation;
   176→   176→      console.log(`[Tools] Quest activation from ${tc.name}:`, activation);
   177→   177→      
   178→   178→      controller.enqueue(createSSEJSON({
   179→   179→        type: 'quest_activation',
   180→   180→        toolName: tc.name,
   181→   181→        activationType: activation.type,
   182→   182→        key: activation.key,
   183→   183→        metadata: activation.metadata,
   184→   184→      }));
   185→   185→      
   186→   186→      questActivations.push(activation);
   187→   187→    }
   188→   188→
   189→   189→    // Check for action/skill activation and send SSE event
   190→   190→    if (toolResult.actionActivation) {
   191→   191→      const action = toolResult.actionActivation;
   192→   192→      console.log(`[Tools] Action activation from ${tc.name}:`, action.skillName);
   193→   193→      
   194→   194→      controller.enqueue(createSSEJSON({
   195→   195→        type: 'action_activation',
   196→   196→        toolName: tc.name,
   197→   197→        skillId: action.skillId,
   198→   198→        skillName: action.skillName,
   199→   199→        skillDescription: action.skillDescription,
   200→   200→        skillCompletedDescription: action.skillCompletedDescription,
   201→   201→        activationCosts: action.activationCosts,
   202→   202→        activationRewards: action.activationRewards,
   203→   203→        characterId: action.characterId,
   204→   204→      }));
   205→   205→    }
   206→   206→
   207→   207→    // Check for stat modification from modify_stat tool and send SSE event
   208→   208→    if (toolResult.statActivation) {
   209→   209→      const stat = toolResult.statActivation;
   210→   210→      console.log(`[Tools] Stat activation from ${tc.name}:`, stat.attributeKey, stat.oldValue, '→', stat.newValue);
   211→   211→      
   212→   212→      controller.enqueue(createSSEJSON({
   213→   213→        type: 'stat_activation',
   214→   214→        toolName: tc.name,
   215→   215→        characterId: stat.characterId,
   216→   216→        attributeKey: stat.attributeKey,
   217→   217→        attributeName: stat.attributeName,
   218→   218→        attributeType: stat.attributeType,
   219→   219→        oldValue: stat.oldValue,
   220→   220→        newValue: stat.newValue,
   221→   221→        reason: stat.reason,
   222→   222→      }));
   223→   223→    }
   224→   224→
   225→   225→    // Check for solicitud activation/completion and send SSE event
   226→   226→    if (toolResult.solicitudActivation) {
   227→   227→      const sol = toolResult.solicitudActivation;
   228→   228→      console.log(`[Tools] Solicitud activation from ${tc.name}:`, sol.type, sol.solicitudKey);
   229→   229→      
   230→   230→      controller.enqueue(createSSEJSON({
   231→   231→        type: 'solicitud_activation',
   232→   232→        toolName: tc.name,
   233→   233→        activationType: sol.type,
   234→   234→        solicitudKey: sol.solicitudKey,
   235→   235→        targetCharacterId: sol.targetCharacterId,
   236→   236→        targetCharacterName: sol.targetCharacterName,
   237→   237→        fromCharacterId: sol.fromCharacterId,
   238→   238→        fromCharacterName: sol.fromCharacterName,
   239→   239→        description: sol.description,
   240→   240→        completionDescription: sol.completionDescription,
   241→   241→        peticionKey: sol.peticionKey,
   242→   242→      }));
   243→   243→    }
   244→   244→
   245→   245→    // Check for memory activation (sync to client-side Character Memory)
   246→   246→    if (toolResult.memoryActivation) {
   247→   247→      const mem = toolResult.memoryActivation;
   248→   248→      console.log(`[Tools] Memory activation from ${tc.name}:`, mem.type);
   249→   249→      
   250→   250→      controller.enqueue(createSSEJSON({
   251→   251→        type: 'memory_activation',
   252→   252→        toolName: tc.name,
   253→   253→        activationType: mem.type,
   254→   254→        characterId: mem.characterId,
   255→   255→        eventData: mem.eventData,
   256→   256→        relationshipData: mem.relationshipData,
   257→   257→        noteContent: mem.noteContent,
   258→   258→        deleteEventId: mem.deleteEventId,
   259→   259→        deleteEmbeddingId: mem.deleteEmbeddingId,
   260→   260→      }));
   261→   261→    }
   262→   262→
   263→   263→    console.log(`[Tools] Tool ${tc.name}: success=${toolResult.success} duration=${toolResult.duration}ms`, toolResult.displayMessage);
   264→   264→
   265→   265→    toolResults.push({ success: toolResult.success, displayMessage: toolResult.displayMessage });
   266→   266→    toolsUsed.push({
   267→   267→      name: tc.name,
   268→   268→      label: toolDef?.label || tc.name,
   269→   269→      icon: toolDef?.icon || 'Wrench',
   270→   270→      success: toolResult.success,
   271→   271→    });
   272→   272→    if (toolResult.displayMessage) {
   273→   273→      allDisplayMessages += (allDisplayMessages ? '\n' : '') + toolResult.displayMessage;
   274→   274→    }
   275→   275→  }
   276→   276→
   277→   277→  // Return the display messages as content, tool results, and signal that a follow-up call is needed
   278→   278→  return {
   279→   279→    newContent: allDisplayMessages,
   280→   280→    shouldContinue: true, // Always continue to get the LLM's natural response
   281→   281→    toolResults,
   282→   282→    questActivations,
   283→   283→    toolsUsed,
   284→   284→  };
   285→   285→}

   287→   287→export async function POST(request: NextRequest) {
   288→   288→  try {
   289→   289→    // Capture auth headers from Z.ai gateway for token resolution
   290→   290→    // Only use actual JWT tokens - session IDs are NOT valid X-Tokens
   291→   291→    const incomingXToken = request.headers.get('X-Token');
   292→   292→    const fcSecurityToken = request.headers.get('x-fc-security-token');
   293→   293→
   294→   294→    const body = await request.json();
   295→   295→
   296→   296→    // Validate request (automatically detects request type)
   297→   297→    const validation = validateRequest(null, body);
   298→   298→    if (!validation.success) {
   299→   299→      return createErrorResponse(validation.error, 400);
   300→   300→    }
   301→   301→
   302→   302→    const {
   303→   303→      message,
   304→   304→      character,
   305→   305→      messages = [],
   306→   306→      llmConfig,
   307→   307→      userName = 'User',
   308→   308→      persona,
   309→   309→      sessionStats,
   310→   310→      allCharacters, // All characters for peticiones/solicitudes resolution
   311→   311→    } = validation.data;
   312→   312→
   313→   313→    // Extract lorebooks from body (not validated by validation.ts)
   314→   314→    const lorebooks: Lorebook[] = body.lorebooks || [];
   315→   315→
   316→   316→    // Extract HUD context from body
   317→   317→    const hudContext: HUDContextConfig | undefined = body.hudContext;
   318→   318→
   319→   319→    // Extract Quest data for pre-LLM integration (NEW FORMAT)
   320→   320→    const questTemplates: QuestTemplate[] = body.questTemplates || [];
   321→   321→    const sessionQuests: SessionQuestInstance[] = body.sessionQuests || [];
   322→   322→    const questSettings: QuestSettings = {
   323→   323→      ...DEFAULT_QUEST_SETTINGS,
   324→   324→      ...(body.questSettings || {})
   325→   325→    };
   326→   326→
   327→   327→    // Extract Sound data for {{sonidos}} key resolution
   328→   328→    const soundTriggers: SoundTrigger[] = body.soundTriggers || [];
   329→   329→    const soundSettings = body.settings?.sound;
   330→   330→
   331→   331→    // Extract summary for memory/context compression (single summary from session)
   332→   332→    const summary: SessionSummary | undefined = body.summary;
   333→   333→    
   334→   334→    // Extract embeddings chat settings for automatic context retrieval
   335→   335→    const embeddingsChat: Partial<EmbeddingsChatSettings> = body.embeddingsChat || {};
   336→   336→    const sessionId: string | undefined = body.sessionId;
   337→   337→    const characterId: string | undefined = body.characterId;
   338→   338→    
   339→   339→    // Extract character memory (events, relationships, notes from Zustand store)
   340→   340→    const characterMemory: CharacterMemory | undefined = body.characterMemory;
   341→   341→    
   342→   342→    // Extract tools settings for tool/action system (native tool calling only)
   343→   343→    const toolsSettings: ToolsSettings = {
   344→   344→      enabled: body.toolsSettings?.enabled ?? true,
   345→   345→      maxToolCallsPerTurn: body.toolsSettings?.maxToolCallsPerTurn ?? 4,
   346→   346→      characterConfigs: body.toolsSettings?.characterConfigs || [],
   347→   347→      usePromptBasedFallback: body.toolsSettings?.usePromptBasedFallback ?? false,
   348→   348→      disabledTools: body.toolsSettings?.disabledTools || [],
   349→   349→    };
   350→   350→
   351→   351→    // Extract inventory data for Inventory V2 system
   352→   352→    const inventoryData: InventoryPromptData | undefined = body.inventoryData;
   353→   353→
   354→   354→    if (!llmConfig) {
   355→   355→      return createErrorResponse('No LLM configuration provided', 400);
   356→   356→    }
   357→   357→
   358→   358→    // Validate API key for providers that require one
   359→   359→    const providersRequiringApiKey = ['openai', 'vllm', 'lm-studio', 'custom', 'anthropic', 'grok'];
   360→   360→    if (providersRequiringApiKey.includes(llmConfig.provider) && !llmConfig.apiKey) {
   361→   361→      return createErrorResponse(
   362→   362→        `API Key is required for ${llmConfig.provider} provider. Please configure it in Settings.`,
   363→   363→        400
   364→   364→      );
   365→   365→    }
   366→   366→
   367→   367→    // Validate endpoint for providers that require one
   368→   368→    const providersRequiringEndpoint = ['openai', 'vllm', 'lm-studio', 'custom', 'anthropic', 'ollama', 'text-generation-webui', 'koboldcpp'];
   369→   369→    if (providersRequiringEndpoint.includes(llmConfig.provider) && !llmConfig.endpoint) {
   370→   370→      return createErrorResponse(
   371→   371→        `Endpoint URL is required for ${llmConfig.provider} provider. Please configure it in Settings.`,
   372→   372→        400
   373→   373→      );
   374→   374→    }
   375→   375→
   376→   376→    // If using Z.ai provider, collect gateway-forwarded auth headers
   377→   377→    // The SDK v0.0.17+ reads config from /etc/.z-ai-config and uses the "token" field for X-Token.
   378→   378→    // Gateway headers are passed as runtime overrides to the provider.
   379→   379→    let zaiRuntimeToken: string | undefined;
   380→   380→    if (llmConfig.provider === 'z-ai') {
   381→   381→      // Only use actual auth tokens (not session IDs)
   382→   382→      const gatewayToken = incomingXToken || fcSecurityToken || undefined;
   383→   383→      if (gatewayToken) {
   384→   384→        zaiRuntimeToken = gatewayToken;
   385→   385→        console.log(`[Stream Route] Z.ai runtime token available (${gatewayToken.length} chars, source: ${incomingXToken ? 'X-Token' : 'fc-security-token'})`);
   386→   386→      } else {
   387→   387→        console.log(`[Stream Route] Z.ai: no gateway token available, using config file only`);
   388→   388→      }
   389→   389→    }
   390→   390→
   391→   391→    // Sanitize user message
   392→   392→    const sanitizedMessage = sanitizeInput(message);
   393→   393→
   394→   394→    // Create default character if none provided
   395→   395→    const effectiveCharacter: CharacterCard = character || DEFAULT_CHARACTER;
   396→   396→
   397→   397→    // Get effective user name from persona or use provided userName
   398→   398→    const effectiveUserName = getEffectiveUserName(persona, userName);
   399→   399→
   400→   400→    // Build context configuration from request or use defaults
   401→   401→    const contextConfig: Partial<ContextConfig> = body.contextConfig || {};
   402→   402→
   403→   403→    // Apply sliding window to messages
   404→   404→    const contextWindow = selectContextMessages(messages, llmConfig, contextConfig);
   405→   405→
   406→   406→    // Log context stats (for debugging)
   407→   407→    const stats = getContextStats(messages);
   408→   408→
   409→   409→    // Process lorebooks and get matched entries
   410→   410→    const { plan: lorebookPlan, lorebookAttributeKeys, lorebookEntryKeyMap, lorebookDebugEntries } = buildLorebookSectionForPrompt(
   411→   411→      messages,
   412→   412→      lorebooks,
   413→   413→      {
   414→   414→        scanDepth: contextConfig.scanDepth,
   415→   415→        userName: effectiveUserName,
   416→   416→        charName: effectiveCharacter?.name,
   417→   417→      },
   418→   418→      { sessionStats, characterId: effectiveCharacter?.id, characters: allCharacters }
   419→   419→    );
   420→   420→
   421→   421→    // ========================================
   422→   422→    // DEBUG: Build lorebook debug data for SSE event
   423→   423→    // ========================================
   424→   424→    const lorebookDebugData = {
   425→   425→      lorebookAttributeKeys,
   426→   426→      debugEntries: lorebookDebugEntries,
   427→   427→      availableStats: sessionStats
   428→   428→        ? Object.fromEntries(
   429→   429→            Object.entries(sessionStats.characterStats || {}).map(([charId, cs]) => [
   430→   430→              charId,
   431→   431→              cs?.attributeValues ? { ...cs.attributeValues } : '(no values)',
   432→   432→            ])
   433→   433→          )
   434→   434→        : '(no sessionStats)',
   435→   435→    };
   436→   436→
   437→   437→    // ========================================
   438→   438→    // Embeddings Context Retrieval
   439→   439→    // ========================================
   440→   440→    // Enrich search query with recent context for better semantic matching
   441→   441→    const searchContextDepth = embeddingsChat.searchContextDepth || 0;
   442→   442→    let enrichedSearchQuery = sanitizedMessage;
   443→   443→    if (searchContextDepth > 0) {
   444→   444→      // Collect recent messages from history for context enrichment
   445→   445→      const recentHistory = messages
   446→   446→        .filter(m => !m.isDeleted)
   447→   447→        .slice(-(searchContextDepth * 2 + 1)) // user+assistant pairs
   448→   448→        .map(m => m.content?.trim())
   449→   449→        .filter(Boolean)
   450→   450→        .slice(0, -1); // exclude current message (already in sanitizedMessage)
   451→   451→      
   452→   452→      if (recentHistory.length > 0) {
   453→   453→        enrichedSearchQuery = recentHistory.join(' ') + ' ' + sanitizedMessage;
   454→   454→      }
   455→   455→    }
   456→   456→
   457→   457→    // Smart truncation: limit query to the embedding model's context window.
   458→   458→    // Prioritize the user's CURRENT message over history when truncating.
   459→   459→    // The Ollama client also handles truncation, but this avoids sending huge payloads.
   460→   460→    try {
   461→   461→      const { MODEL_CONTEXT_LENGTHS, DEFAULT_CONTEXT_LENGTH, CHARS_PER_TOKEN } = await import('@/lib/embeddings/types');
   462→   462→      const { loadConfig } = await import('@/lib/embeddings/config-persistence');
   463→   463→      const embConfig = loadConfig();
   464→   464→      const modelKey = embConfig.model || 'bge-m3:567m';
   465→   465→      const modelCtx = MODEL_CONTEXT_LENGTHS[modelKey]
   466→   466→        || MODEL_CONTEXT_LENGTHS[modelKey.split(':')[0]]
   467→   467→        || DEFAULT_CONTEXT_LENGTH;
   468→   468→      const maxQueryChars = Math.floor(modelCtx * 0.75 * CHARS_PER_TOKEN);
   469→   469→
