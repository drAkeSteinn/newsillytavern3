    23→    23→const _spriteRotationIndex = new Map<string, Map<SpriteState, number>>();
    24→    24→
    25→    25→function getRotationIndex(characterId: string, state: SpriteState): number {
    26→    26→  let charMap = _spriteRotationIndex.get(characterId);
    27→    27→  if (!charMap) {
    28→    28→    charMap = new Map<SpriteState, number>();
    29→    29→    _spriteRotationIndex.set(characterId, charMap);
    30→    30→  }
    31→    31→  return charMap.get(state) ?? 0;
    32→    32→}

    34→    34→function advanceRotationIndex(characterId: string, state: SpriteState): number {
    35→    35→  let charMap = _spriteRotationIndex.get(characterId);
    36→    36→  if (!charMap) {
    37→    37→    charMap = new Map<SpriteState, number>();
    38→    38→    _spriteRotationIndex.set(characterId, charMap);
    39→    39→  }
    40→    40→  const current = charMap.get(state) ?? 0;
    41→    41→  const next = current + 1;
    42→    42→  charMap.set(state, next);
    43→    43→  return next;
    44→    44→}

    46→    46→// Get sprite from State Collection V2 (uses sprite packs)
    47→    47→function getSpriteFromStateCollectionV2(
    48→    48→  state: SpriteState,
    49→    49→  stateCollectionsV2: StateCollectionV2[],
    50→    50→  spritePacksV2: SpritePackV2[],
    51→    51→  characterId: string
    52→    52→): { url: string | null; label: string | null } {
    53→    53→  const stateCollection = stateCollectionsV2.find(c => c.state === state);
    54→    54→  if (!stateCollection) return { url: null, label: null };
    55→    55→  
    56→    56→  const pack = spritePacksV2.find(p => p.id === stateCollection.packId);
    57→    57→  if (!pack || pack.sprites.length === 0) return { url: null, label: null };
    58→    58→  
    59→    59→  switch (stateCollection.behavior) {
    60→    60→    case 'principal':
    61→    61→      if (stateCollection.principalSpriteId) {
    62→    62→        const principal = pack.sprites.find(s => s.id === stateCollection.principalSpriteId);
    63→    63→        if (principal) return { url: principal.url, label: principal.label };
    64→    64→      }
    65→    65→      return { url: pack.sprites[0].url, label: pack.sprites[0].label };
    66→    66→    
    67→    67→    case 'random': {
    68→    68→      const randomIndex = Math.floor(Math.random() * pack.sprites.length);
    69→    69→      const randomSprite = pack.sprites[randomIndex];
    70→    70→      return { url: randomSprite.url, label: randomSprite.label };
    71→    71→    }
    72→    72→    
    73→    73→    case 'list': {
    74→    74→      const index = getRotationIndex(characterId, state);
    75→    75→      const safeIndex = index % pack.sprites.length;
    76→    76→      const listSprite = pack.sprites[safeIndex];
    77→    77→      return { url: listSprite.url, label: listSprite.label };
    78→    78→    }
    79→    79→    
    80→    80→    default:
    81→    81→      return { url: pack.sprites[0].url, label: pack.sprites[0].label };
    82→    82→  }
    83→    83→}

    85→    85→// Format milliseconds to readable time
    86→    86→function formatTime(ms: number): string {
    87→    87→  if (ms <= 0) return '0s';
    88→    88→  const seconds = Math.floor(ms / 1000);
    89→    89→  if (seconds < 60) return `${seconds}s`;
    90→    90→  const minutes = Math.floor(seconds / 60);
    91→    91→  const remainingSeconds = seconds % 60;
    92→    92→  return `${minutes}m ${remainingSeconds}s`;
    93→    93→}

   149→   149→// Get the appropriate sprite URL based on state and config
   150→   150→// Priority: V2 State Collections > Avatar (legacy system removed)
   151→   151→function getSpriteUrl(
   152→   152→  state: SpriteState,
   153→   153→  character?: CharacterCard,
   154→   154→  characterId?: string
   155→   155→): { url: string; label: string | null } {
   156→   156→  const charId = characterId || character?.id || '';
   157→   157→  // Check for non-empty arrays (empty arrays should not be considered as "configured")
   158→   158→  const hasV2Collections = character?.stateCollectionsV2 && character.stateCollectionsV2.length > 0;
   159→   159→  const hasV2Packs = character?.spritePacksV2 && character.spritePacksV2.length > 0;
   160→   160→  
   161→   161→  // Try V2 state collections (new system)
   162→   162→  if (hasV2Collections && hasV2Packs) {
   163→   163→    const v2Result = getSpriteFromStateCollectionV2(
   164→   164→      state,
   165→   165→      character.stateCollectionsV2,
   166→   166→      character.spritePacksV2,
   167→   167→      charId
   168→   168→    );
   169→   169→    if (v2Result.url) {
   170→   170→      return v2Result;
   171→   171→    }
   172→   172→  }
   173→   173→
   174→   174→  // Additional fallbacks for talk and thinking states - try V2 idle collection
   175→   175→  if (state === 'talk' || state === 'thinking') {
   176→   176→    if (hasV2Collections && hasV2Packs) {
   177→   177→      const v2IdleResult = getSpriteFromStateCollectionV2(
   178→   178→        'idle',
   179→   179→        character!.stateCollectionsV2,
   180→   180→        character!.spritePacksV2,
   181→   181→        charId
   182→   182→      );
   183→   183→      if (v2IdleResult.url) {
   184→   184→        return v2IdleResult;
   185→   185→      }
   186→   186→    }
   187→   187→  }
   188→   188→
   189→   189→  // Final fallback to avatar
   190→   190→  if (character?.avatar) {
   191→   191→    return { url: character.avatar, label: 'avatar' };
   192→   192→  }
   193→   193→
   194→   194→  return { url: '', label: null };
   195→   195→}
