   489→function ExtraccionTab() {
   490→  const embeddingsChat = useTavernStore((state) => state.settings.embeddingsChat) ?? DEFAULT_EMBEDDINGS_CHAT;
   491→  const updateSettings = useTavernStore((state) => state.updateSettings);
   492→
   493→  // Prompt editor state
   494→  const [activePromptTab, setActivePromptTab] = useState<'normal' | 'group'>('normal');
   495→  const [localPrompt, setLocalPrompt] = useState(() => embeddingsChat.memoryExtractionPrompt || DEFAULT_MEMORY_EXTRACTION_PROMPT);
   496→  const [localGroupPrompt, setLocalGroupPrompt] = useState(() => embeddingsChat.groupMemoryExtractionPrompt || DEFAULT_GROUP_MEMORY_EXTRACTION_PROMPT);
   497→  const [showPreview, setShowPreview] = useState(false);
   498→
   499→  const isNormal = activePromptTab === 'normal';
   500→  const currentPrompt = isNormal ? localPrompt : localGroupPrompt;
   501→  const currentStored = isNormal
   502→    ? (embeddingsChat.memoryExtractionPrompt || DEFAULT_MEMORY_EXTRACTION_PROMPT)
   503→    : (embeddingsChat.groupMemoryExtractionPrompt || DEFAULT_GROUP_MEMORY_EXTRACTION_PROMPT);
   504→  const previewData = isNormal ? NORMAL_PREVIEW : GROUP_PREVIEW;
   505→
   506→  const handleSavePrompt = () => {
   507→    if (isNormal) {
   508→      updateSettings({ embeddingsChat: { ...embeddingsChat, memoryExtractionPrompt: localPrompt } });
   509→    } else {
   510→      updateSettings({ embeddingsChat: { ...embeddingsChat, groupMemoryExtractionPrompt: localGroupPrompt } });
   511→    }
   512→  };
   513→
   514→  const handleRestoreDefault = () => {
   515→    if (isNormal) {
   516→      setLocalPrompt(DEFAULT_MEMORY_EXTRACTION_PROMPT);
   517→      updateSettings({ embeddingsChat: { ...embeddingsChat, memoryExtractionPrompt: DEFAULT_MEMORY_EXTRACTION_PROMPT } });
   518→    } else {
   519→      setLocalGroupPrompt(DEFAULT_GROUP_MEMORY_EXTRACTION_PROMPT);
   520→      updateSettings({ embeddingsChat: { ...embeddingsChat, groupMemoryExtractionPrompt: DEFAULT_GROUP_MEMORY_EXTRACTION_PROMPT } });
   521→    }
   522→  };
   523→
   524→  const handleChange = (value: string) => {
   525→    if (isNormal) setLocalPrompt(value);
   526→    else setLocalGroupPrompt(value);
   527→  };
   528→
   529→  const previewText = currentPrompt
   530→    .replace('{characterName}', previewData.characterName)
   531→    .replace('{chatContext}', previewData.chatContext)
   532→    .replace('{lastMessage}', previewData.lastMessage);
   533→
   534→  const hasChanges = currentPrompt !== currentStored;
   535→
   536→  return (
   537→    <div className="space-y-6">
   538→      {/* Note about needing Embeddings enabled */}
   539→      <div className="bg-amber-500/5 border border-amber-500/20 rounded-lg p-4">
   540→        <div className="flex items-start gap-2">
   541→          <Info className="w-4 h-4 text-amber-500 mt-0.5 shrink-0" />
   542→          <div className="space-y-1">
   543→            <p className="text-sm font-medium text-amber-600 dark:text-amber-400">Requiere Embeddings</p>
   544→            <p className="text-xs text-muted-foreground">
   545→              Estos ajustes requieren que Embeddings esté activado. Configúralo en la pestaña <strong>Base de Conocimiento</strong>.
   546→            </p>
   547→          </div>
   548→        </div>
   549→      </div>
   550→
   551→      {/* Memory Extraction Section */}
   552→      <Card>
   553→        <CardHeader className="pb-3">
   554→          <CardTitle className="flex items-center gap-2 text-base">
   555→            <Brain className="w-4 h-4 text-violet-500" />
