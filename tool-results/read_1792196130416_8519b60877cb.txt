   278→        return;
   279→      }
   280→      thresholdSendTimerRef.current = null;
   281→      const text = pendingThresholdSendRef.current;
   282→      pendingThresholdSendRef.current = null;
   283→      if (text && text.trim()) {
   284→        handleSendRef.current(text);
   285→      }
   286→    };
   287→    if (!thresholdSendTimerRef.current) {
   288→      thresholdSendTimerRef.current = setTimeout(trySend, 400);
