   105→
   106→function RequirementOperatorToggle({ operator, onChange, requirementCount }: RequirementOperatorToggleProps) {
   107→  if (requirementCount < 2) return null;
   108→
   109→  const currentOperator = operator || 'AND';
   110→
   111→  return (
   112→    <div className="flex items-center gap-2 py-1">
   113→      <div className="flex items-center gap-1">
   114→        <button
   115→          type="button"
   116→          className={cn(
   117→            'px-2 py-0.5 text-xs rounded border transition-colors',
   118→            currentOperator === 'AND'
   119→              ? 'bg-blue-500/20 text-blue-400 border-blue-500/30'
   120→              : 'bg-muted/30 text-muted-foreground border-transparent'
   121→          )}
   122→          onClick={() => onChange('AND')}
   123→        >
   124→          Y (AND)
   125→        </button>
   126→        <button
   127→          type="button"
   128→          className={cn(
   129→            'px-2 py-0.5 text-xs rounded border transition-colors',
   130→            currentOperator === 'OR'
   131→              ? 'bg-purple-500/20 text-purple-400 border-purple-500/30'
   132→              : 'bg-muted/30 text-muted-foreground border-transparent'
   133→          )}
   134→          onClick={() => onChange('OR')}
   135→        >
   136→          O (OR)
   137→        </button>
   138→      </div>
   139→      <span className="text-[10px] text-muted-foreground">
   140→        {currentOperator === 'AND' ? 'Todas deben cumplirse' : 'Al menos una debe cumplirse'}
   141→      </span>
   142→    </div>
   143→  );

   150→interface RequirementEditorProps {
   151→  requirement: StatRequirement;
   152→  availableAttributes: AttributeDefinition[];
   153→  availableTargets?: { id: string; name: string; attributes: AttributeDefinition[] }[];

   157→
   158→function RequirementEditor({ requirement, availableAttributes, availableTargets = [], onChange, onDelete }: RequirementEditorProps) {
   159→  const isTargetMode = requirement.targetCharacterId !== undefined;
   160→  const selectedTarget = isTargetMode && requirement.targetCharacterId
   161→    ? availableTargets.find(t => t.id === requirement.targetCharacterId)
   162→    : undefined;
   163→  const targetAttrs = selectedTarget?.attributes || [];
   164→
   165→  // Determine the selected attribute and its type
   166→  const selectedSelfAttr = !isTargetMode ? availableAttributes.find(a => a.key === requirement.attributeKey) : undefined;
   167→  const selectedTargetAttr = isTargetMode ? targetAttrs.find(a => a.key === requirement.attributeKey) : undefined;
   168→  const selectedAttr = selectedSelfAttr || selectedTargetAttr;
   169→  const attrType = selectedAttr?.type || 'number';
   170→  const isTextType = attrType === 'text' || attrType === 'keyword';
   171→
   172→  const operatorOptions = isTextType ? TEXT_OPERATOR_OPTIONS : NUMERIC_OPERATOR_OPTIONS;
   173→  const selectedOperator = operatorOptions.find(op => op.value === requirement.operator);
   174→
   175→  const hasTargets = availableTargets.length > 0;
   176→
   177→  return (
   178→    <div className="flex items-center gap-2 bg-muted/50 rounded p-2 flex-wrap">
   179→      {/* Target/Mode indicator - only show Target option when availableTargets is provided */}
   180→      <Select
   181→        value={isTargetMode ? 'target' : 'self'}
   182→        onValueChange={(value) => {
   183→          if (value === 'target') {
   184→            onChange({ attributeKey: '', targetCharacterId: '', targetAttributeName: '', operator: '==', value: '' });
   185→          } else {
   186→            onChange({ attributeKey: '', targetCharacterId: undefined, targetAttributeName: undefined, operator: '>=', value: 0 });
   187→          }
   188→        }}
   189→      >
   190→        <SelectTrigger className="h-7 w-16 text-xs">
   191→          <SelectValue />
   192→        </SelectTrigger>
   193→        <SelectContent>
   194→          <SelectItem value="self">
   195→            <span className="flex items-center gap-1">🎭 Yo</span>
   196→          </SelectItem>
   197→          {hasTargets && (
   198→            <SelectItem value="target">
   199→              <span className="flex items-center gap-1">🎯 Target</span>
   200→            </SelectItem>
   201→          )}
   202→        </SelectContent>
   203→      </Select>
   204→
   205→      {isTargetMode ? (
   206→        <>
   207→          {/* Target selector */}
   208→          <Select
   209→            value={requirement.targetCharacterId || ''}
   210→            onValueChange={(value) => {
   211→              const target = availableTargets.find(t => t.id === value);
   212→              onChange({ targetCharacterId: value, attributeKey: '', targetAttributeName: target?.name || '' });
   213→            }}
   214→          >
   215→            <SelectTrigger className="h-7 w-28 text-xs">
   216→              <SelectValue placeholder="Target..." />
   217→            </SelectTrigger>
   218→            <SelectContent>
   219→              {availableTargets.map(t => (
   220→                <SelectItem key={t.id} value={t.id}>
   221→                  {t.id === '__user__' ? '👤 ' : '🎭 '}{t.name}
   222→                </SelectItem>
   223→              ))}
   224→            </SelectContent>
   225→          </Select>
   226→
   227→          {/* Target attribute selector */}
   228→          <Select
   229→            value={requirement.attributeKey}
   230→            onValueChange={(value) => {
   231→              const attr = targetAttrs.find(a => a.key === value);
   232→              const isText = attr?.type === 'text' || attr?.type === 'keyword';
   233→              onChange({
   234→                attributeKey: value,
   235→                targetAttributeName: attr?.name || '',
   236→                operator: isText ? '==' : '>=',
   237→                value: isText ? '' : 0,
   238→              });
   239→            }}
   240→            disabled={!requirement.targetCharacterId}
   241→          >
   242→            <SelectTrigger className="h-7 w-28 text-xs">
   243→              <SelectValue placeholder="Atributo..." />
   244→            </SelectTrigger>
   245→            <SelectContent>
   246→              {targetAttrs.map((attr, i) => (
   247→                <SelectItem key={attr.key || `attr-${i}`} value={attr.key}>
   248→                  <span className="flex items-center gap-1">
   249→                    <span className={attr.type === 'text' || attr.type === 'keyword' ? 'text-blue-400' : 'text-green-400'}>
   250→                      {attr.type === 'text' ? '📝' : attr.type === 'keyword' ? '🏷️' : '🔢'}
   251→                    </span>
   252→                    {attr.name}
   253→                  </span>
   254→                </SelectItem>
   255→              ))}
   256→            </SelectContent>
   257→          </Select>
   258→        </>
   259→      ) : (
   260→        /* Self attribute selector */
   261→        <Select
   262→          value={requirement.attributeKey}
   263→          onValueChange={(value) => {
   264→            const attr = availableAttributes.find(a => a.key === value);
   265→            const isText = attr?.type === 'text' || attr?.type === 'keyword';
   266→            onChange({
   267→              attributeKey: value,
   268→              operator: isText ? '==' : '>=',
   269→              value: isText ? '' : 0,
   270→            });
   271→          }}
   272→        >
   273→          <SelectTrigger className="h-7 w-24 text-xs">
   274→            <SelectValue placeholder="Atributo" />
   275→          </SelectTrigger>
   276→          <SelectContent>
   277→            {availableAttributes.map(attr => (
   278→              <SelectItem key={attr.id} value={attr.key}>
   279→                <span className="flex items-center gap-1">
   280→                  <span className={attr.type === 'text' || attr.type === 'keyword' ? 'text-blue-400' : 'text-green-400'}>
   281→                    {attr.type === 'text' ? '📝' : attr.type === 'keyword' ? '🏷️' : '🔢'}
   282→                  </span>
   283→                  {attr.name}
   284→                </span>
   285→              </SelectItem>
   286→            ))}
   287→          </SelectContent>
   288→        </Select>
   289→      )}
   290→
   291→      {/* Operator selector with descriptions */}
   292→      <Tooltip>
   293→        <TooltipTrigger asChild>
   294→          <Select
   295→            value={operatorOptions.some(op => op.value === requirement.operator) ? requirement.operator : operatorOptions[0].value}
   296→            onValueChange={(value: RequirementOperator) => onChange({ operator: value })}
   297→          >
   298→            <SelectTrigger className="h-7 w-16 text-xs">
   299→              <SelectValue />
   300→            </SelectTrigger>
   301→            <SelectContent>
   302→              {operatorOptions.map(op => (
   303→                <SelectItem key={op.value} value={op.value}>
   304→                  <div className="flex items-center gap-2">
   305→                    <span className="font-mono w-4">{op.label}</span>
   306→                    <span className="text-muted-foreground text-xs">{op.description}</span>
   307→                  </div>
   308→                </SelectItem>
   309→              ))}
   310→            </SelectContent>
   311→          </Select>
   312→        </TooltipTrigger>
   313→        <TooltipContent className="max-w-xs">
   314→          <p className="font-medium">{selectedOperator?.description}</p>
   315→          <p className="text-xs text-muted-foreground mt-1">
   316→            {requirement.operator === 'between'
   317→              ? `El valor debe estar entre ${requirement.value} y ${requirement.valueMax || '?'}`
   318→              : `El valor debe ser ${selectedOperator?.description} ${requirement.value}`
   319→            }
   320→          </p>
   321→        </TooltipContent>
   322→      </Tooltip>
   323→
   324→      {/* Value input - text or number based on attribute type */}
   325→      {isTextType ? (
   326→        <Input
   327→          type="text"
   328→          value={typeof requirement.value === 'string' ? requirement.value : String(requirement.value)}
   329→          onChange={(e) => onChange({ value: e.target.value })}
   330→          placeholder="Texto..."
   331→          className="h-7 w-24 text-xs"
   332→        />
   333→      ) : (
   334→        <Input
   335→          type="number"
   336→          value={requirement.value}
   337→          onChange={(e) => onChange({ value: parseFloat(e.target.value) || 0 })}
   338→          className="h-7 w-16 text-xs"
   339→        />
   340→      )}
   341→
   342→      {/* Max value for between operator (only for numeric) */}
   343→      {!isTextType && requirement.operator === 'between' && (
   344→        <>
   345→          <span className="text-xs text-muted-foreground">y</span>
   346→          <Input
   347→            type="number"
   348→            value={requirement.valueMax ?? ''}
   349→            onChange={(e) => {
   350→              const parsed = parseFloat(e.target.value);
   351→              onChange({ valueMax: isNaN(parsed) ? undefined : parsed });
   352→            }}
   353→            placeholder="max"
   354→            className="h-7 w-16 text-xs"
   355→          />
   356→        </>
   357→      )}
   358→
   359→      {/* Delete button */}
   360→      <Button variant="ghost" size="icon" className="h-7 w-7" onClick={onDelete}>
   361→        <Trash2 className="w-3 h-3 text-muted-foreground" />
   362→      </Button>
   363→    </div>
   364→  );
   365→}

   371→interface QuickRepliesPanelProps {

   380→  onChange: (quickReplies: CharacterQuickReply[]) => void;

   382→
   383→// Operation labels for display
   384→const OPERATION_LABELS: Record<QuickReplyModifierOperation, { label: string; symbol: string; description: string }> = {
   385→  set: { label: 'Establecer', symbol: '=', description: 'Reemplaza el valor actual' },
   386→  add: { label: 'Sumar', symbol: '+', description: 'Agrega al valor actual' },
   387→  subtract: { label: 'Restar', symbol: '-', description: 'Reduce del valor actual' },
   388→  multiply: { label: 'Multiplicar', symbol: '×', description: 'Multiplica el valor actual' },
   389→  divide: { label: 'Dividir', symbol: '÷', description: 'Divide el valor actual' },
   390→};
   391→
   392→function generateId(): string {
   393→  return `qr_${Date.now()}_${Math.random().toString(36).substring(2, 8)}`;
   394→}
   395→
   396→export function QuickRepliesPanel({
   397→  quickReplies,
   398→  statsConfig,
   399→  spritePacksV2,
   400→  triggerCollections,
   401→  availableTargets,
   402→  onChange,
   403→}: QuickRepliesPanelProps) {
   404→  const replies = quickReplies || [];
   405→  const attributes = statsConfig?.attributes || [];
   406→
   407→  // State for new reply form
   408→  const [newLabel, setNewLabel] = useState('');
   409→  const [newResponse, setNewResponse] = useState('');
   410→  const [newModifiers, setNewModifiers] = useState<QuickReplyAttributeModifier[]>([]);
   411→  const [newRequirements, setNewRequirements] = useState<StatRequirement[]>([]);
   412→  const [newRequirementOperator, setNewRequirementOperator] = useState<'AND' | 'OR'>('AND');
   413→
   414→  // State for editing existing reply
   415→  const [editingId, setEditingId] = useState<string | null>(null);
   416→  const [editLabel, setEditLabel] = useState('');
   417→  const [editResponse, setEditResponse] = useState('');
   418→  const [editModifiers, setEditModifiers] = useState<QuickReplyAttributeModifier[]>([]);
   419→  const [editRequirements, setEditRequirements] = useState<StatRequirement[]>([]);
   420→  const [editRequirementOperator, setEditRequirementOperator] = useState<'AND' | 'OR'>('AND');
   421→
   422→  // State for expanded modifiers section
   423→  const [expandedModifiers, setExpandedModifiers] = useState<string | null>(null);
   424→
   425→  // State for expanded conditions section
   426→  const [expandedConditions, setExpandedConditions] = useState<string | null>(null);
   427→
   428→  // State for sprite activation in new reply
   429→  const [newSpriteActivation, setNewSpriteActivation] = useState<QuickReplySpriteActivation | undefined>(undefined);
   430→
   431→  // State for sprite activation in editing reply
   432→  const [editSpriteActivation, setEditSpriteActivation] = useState<QuickReplySpriteActivation | undefined>(undefined);
   433→
   434→  // State for expanded sprite activation section
   435→  const [expandedSpriteActivation, setExpandedSpriteActivation] = useState<string | null>(null);
   436→
   437→  // Whether sprite activation is available (has packs or collections)
   438→  const hasSpriteOptions = (spritePacksV2 && spritePacksV2.length > 0) || (triggerCollections && triggerCollections.length > 0);
   439→
   440→  const isAdding = newLabel.trim() && newResponse.trim();
   441→
   442→  const handleAdd = () => {
   443→    if (!isAdding) return;
   444→    const newReply: CharacterQuickReply = {
   445→      id: generateId(),
   446→      label: newLabel.trim(),
   447→      response: newResponse.trim(),
   448→      modifiers: newModifiers.length > 0 ? newModifiers : undefined,
   449→      spriteActivation: newSpriteActivation,
   450→      requirements: newRequirements.length > 0 ? newRequirements : undefined,
   451→      requirementOperator: newRequirements.length > 1 ? newRequirementOperator : undefined,
   452→    };
   453→    onChange([...replies, newReply]);
   454→    setNewLabel('');
   455→    setNewResponse('');
   456→    setNewModifiers([]);
   457→    setNewSpriteActivation(undefined);
   458→    setNewRequirements([]);
   459→    setNewRequirementOperator('AND');
   460→  };
   461→
   462→  const handleDelete = (id: string) => {
   463→    onChange(replies.filter((r) => r.id !== id));
   464→  };
   465→
   466→  const handleStartEdit = (reply: CharacterQuickReply) => {
   467→    setEditingId(reply.id);
   468→    setEditLabel(reply.label);
   469→    setEditResponse(reply.response);
   470→    setEditModifiers(reply.modifiers ? [...reply.modifiers] : []);
   471→    setEditSpriteActivation(reply.spriteActivation ? { ...reply.spriteActivation } : undefined);
   472→    setEditRequirements(reply.requirements ? reply.requirements.map(r => ({ ...r })) : []);
   473→    setEditRequirementOperator(reply.requirementOperator || 'AND');
   474→  };
   475→
   476→  const handleSaveEdit = () => {
   477→    if (!editingId || !editLabel.trim() || !editResponse.trim()) return;
   478→    onChange(
   479→      replies.map((r) =>
   480→        r.id === editingId
   481→          ? {
   482→              ...r,
   483→              label: editLabel.trim(),
   484→              response: editResponse.trim(),
   485→              modifiers: editModifiers.length > 0 ? editModifiers : undefined,
   486→              spriteActivation: editSpriteActivation,
   487→              requirements: editRequirements.length > 0 ? editRequirements : undefined,
   488→              requirementOperator: editRequirements.length > 1 ? editRequirementOperator : undefined,
   489→            }
   490→          : r
   491→      )
   492→    );
   493→    setEditingId(null);
   494→    setEditLabel('');
   495→    setEditResponse('');
   496→    setEditModifiers([]);
   497→    setEditSpriteActivation(undefined);
   498→    setEditRequirements([]);
   499→    setEditRequirementOperator('AND');
   500→  };
   501→
   502→  const handleCancelEdit = () => {
   503→    setEditingId(null);
   504→    setEditLabel('');
   505→    setEditResponse('');
   506→    setEditModifiers([]);
   507→    setEditSpriteActivation(undefined);
   508→    setEditRequirements([]);
   509→    setEditRequirementOperator('AND');
   510→  };
   511→
   512→  // Add a modifier to the new reply form
   513→  const addModifierToNew = () => {
   514→    if (attributes.length === 0) return;
   515→    setNewModifiers([
   516→      ...newModifiers,
   517→      { attributeKey: attributes[0].key, operation: 'add', value: 1 },
   518→    ]);
   519→  };
   520→
   521→  const updateNewModifier = (index: number, field: keyof QuickReplyAttributeModifier, value: string | number) => {
   522→    const updated = [...newModifiers];
   523→    updated[index] = { ...updated[index], [field]: value };
   524→    setNewModifiers(updated);
   525→  };
   526→
   527→  const removeNewModifier = (index: number) => {
   528→    setNewModifiers(newModifiers.filter((_, i) => i !== index));
   529→  };
   530→
   531→  // Add a modifier to the editing reply form
   532→  const addModifierToEdit = () => {
   533→    if (attributes.length === 0) return;
   534→    setEditModifiers([
   535→      ...editModifiers,
   536→      { attributeKey: attributes[0].key, operation: 'add', value: 1 },
   537→    ]);
   538→  };
   539→
   540→  const updateEditModifier = (index: number, field: keyof QuickReplyAttributeModifier, value: string | number) => {
   541→    const updated = [...editModifiers];
   542→    updated[index] = { ...updated[index], [field]: value };
   543→    setEditModifiers(updated);
   544→  };
   545→
   546→  const removeEditModifier = (index: number) => {
   547→    setEditModifiers(editModifiers.filter((_, i) => i !== index));
   548→  };
   549→
   550→  // Render modifier row
   551→  const renderModifierRow = (
   552→    modifier: QuickReplyAttributeModifier,
   553→    index: number,
   554→    onUpdate: (index: number, field: keyof QuickReplyAttributeModifier, value: string | number) => void,
   555→    onRemove: (index: number) => void
   556→  ) => {
   557→    const attr = attributes.find((a) => a.key === modifier.attributeKey);
   558→    const isTextAttr = attr?.type === 'text' || attr?.type === 'keyword';
   559→    const availableOps: QuickReplyModifierOperation[] = isTextAttr
   560→      ? ['set']
   561→      : ['set', 'add', 'subtract', 'multiply', 'divide'];
   562→
