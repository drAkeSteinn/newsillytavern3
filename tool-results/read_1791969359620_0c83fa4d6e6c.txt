    92→/**
    93→ * Build a HUD context section for prompt injection
    94→ * 
    95→ * Now resolves ALL keys including:
    96→ * - Template variables: {{user}}, {{char}}, {{userpersona}}
    97→ * - Stats keys: {{resistencia}}, {{habilidades}}, etc.
    98→ */
    99→export function buildHUDContextSection(
   100→  contextConfig: HUDContextConfig,
   101→  keyContext?: KeyResolutionContext
   102→): PromptSection | null {
   103→  if (!contextConfig.enabled || !contextConfig.content.trim()) {
   104→    return null;
   105→  }
   106→
   107→  // Resolve all keys in the context content
   108→  let resolvedContent = contextConfig.content;
   109→  if (keyContext) {
   110→    resolvedContent = resolveAllKeys(contextConfig.content, keyContext);
   111→  }
   112→
   113→  return {
   114→    type: 'hud_context',
   115→    label: 'HUD Context',
   116→    content: resolvedContent,
   117→    color: SECTION_COLORS.hud_context
   118→  };
   119→}
   120→
   121→/**
   122→ * Inject HUD context at the specified position
   123→ *
   124→ * Positions:
   125→ * 0 = After system prompt
   126→ * 1 = After user message (after last user message)
   127→ * 2 = Before user message (before last user message)
   128→ * 3 = After assistant message (after last assistant message)
   129→ * 4 = Before assistant message (before last assistant message)
   130→ * 5 = At top of chat (before chat history)
   131→ * 6 = At bottom of chat (after all messages)
   132→ * 7 = After lorebook / Author's Note position
   133→ */
   134→export function injectHUDContextIntoMessages(
   135→  messages: ChatApiMessage[],
   136→  contextSection: PromptSection,
   137→  position: number
   138→): ChatApiMessage[] {
   139→  if (!contextSection.content.trim()) {
   140→    return messages;
   141→  }
   142→
   143→  const contextContent = `[${contextSection.label}]\n${contextSection.content}`;
   144→  const result: ChatApiMessage[] = [...messages];
   145→
   146→  switch (position) {
   147→    case 0: // After system prompt
   148→      // Find the first system/assistant message (the system prompt)
   149→      const sysIdx = result.findIndex(m => m.role === 'assistant' || m.role === 'system');
   150→      if (sysIdx >= 0) {
   151→        result[sysIdx] = {
   152→          ...result[sysIdx],
   153→          content: result[sysIdx].content + '\n\n' + contextContent
   154→        };
   155→      }
   156→      break;
   157→
   158→    case 1: // After user message
   159→      // Find the last user message
   160→      const lastUserIdx = result.map((m, i) => m.role === 'user' ? i : -1).filter(i => i >= 0).pop();
   161→      if (lastUserIdx !== undefined && lastUserIdx >= 0) {
   162→        result[lastUserIdx] = {
   163→          ...result[lastUserIdx],
   164→          content: result[lastUserIdx].content + '\n\n' + contextContent
   165→        };
   166→      }
   167→      break;
   168→
   169→    case 2: // Before user message
   170→      const lastUserIdx2 = result.map((m, i) => m.role === 'user' ? i : -1).filter(i => i >= 0).pop();
   171→      if (lastUserIdx2 !== undefined && lastUserIdx2 >= 0) {
   172→        result[lastUserIdx2] = {
   173→          ...result[lastUserIdx2],
   174→          content: contextContent + '\n\n' + result[lastUserIdx2].content
   175→        };
   176→      }
   177→      break;
   178→
   179→    case 3: // After assistant message
   180→      const lastAsstIdx = result.map((m, i) => m.role === 'assistant' ? i : -1).filter(i => i >= 0).pop();
   181→      if (lastAsstIdx !== undefined && lastAsstIdx >= 0) {
   182→        result[lastAsstIdx] = {
   183→          ...result[lastAsstIdx],
   184→          content: result[lastAsstIdx].content + '\n\n' + contextContent
   185→        };
   186→      }
   187→      break;
   188→
   189→    case 4: // Before assistant message
   190→      const lastAsstIdx4 = result.map((m, i) => m.role === 'assistant' ? i : -1).filter(i => i >= 0).pop();
   191→      if (lastAsstIdx4 !== undefined && lastAsstIdx4 >= 0) {
   192→        result[lastAsstIdx4] = {
   193→          ...result[lastAsstIdx4],
   194→          content: contextContent + '\n\n' + result[lastAsstIdx4].content
   195→        };
   196→      }
   197→      break;
   198→
   199→    case 5: // At top of chat (after system, before first message)
   200→      // Merge into the first system message to avoid breaking alternation
   201→      {
   202→        const sysMsgIdx5 = result.findIndex(m => m.role === 'system');
   203→        if (sysMsgIdx5 >= 0) {
   204→          result[sysMsgIdx5] = {
   205→            ...result[sysMsgIdx5],
   206→            content: result[sysMsgIdx5].content + '\n\n' + contextContent
   207→          };
   208→        }
   209→      }
   210→      break;
   211→
   212→    case 6: // At bottom of chat (after all messages)
   213→      // Merge into the first system message to avoid breaking alternation
   214→      {
   215→        const sysMsgIdx6 = result.findIndex(m => m.role === 'system');
   216→        if (sysMsgIdx6 >= 0) {
   217→          result[sysMsgIdx6] = {
   218→            ...result[sysMsgIdx6],
   219→            content: result[sysMsgIdx6].content + '\n\n' + contextContent
   220→          };
   221→        }
   222→      }
   223→      break;
   224→
   225→    case 7: // After lorebook / Author's Note position
   226→      // Merge into the first system message to avoid breaking alternation
   227→      {
   228→        const sysMsgIdx7 = result.findIndex(m => m.role === 'system' || m.role === 'assistant');
   229→        if (sysMsgIdx7 >= 0) {
   230→          result[sysMsgIdx7] = {
   231→            ...result[sysMsgIdx7],
   232→            content: result[sysMsgIdx7].content + '\n\n' + contextContent
   233→          };
   234→        }
   235→      }
   236→      break;
   237→
   238→    default:
   239→      // Merge into the first system message to avoid breaking alternation
   240→      {
   241→        const sysMsgIdxDef = result.findIndex(m => m.role === 'system');
   242→        if (sysMsgIdxDef >= 0) {
   243→          result[sysMsgIdxDef] = {
   244→            ...result[sysMsgIdxDef],
   245→            content: result[sysMsgIdxDef].content + '\n\n' + contextContent
   246→          };
   247→        } else {
   248→          result.push({
   249→            role: 'system',
   250→            content: contextContent
   251→          });
   252→        }
   253→      }
   254→      break;
   255→  }
   256→
   257→  return result;
   258→}

   260→/**
   261→ * Inject HUD context into prompt sections
   262→ * This is for the prompt viewer display
   263→ *
   264→ * Positions:
   265→ * 0 = After system prompt
   266→ * 1 = After user message (not applicable to sections, uses chat_history)
   267→ * 2 = Before user message (not applicable to sections, uses chat_history)
   268→ * 3 = After assistant message (not applicable to sections, uses chat_history)
   269→ * 4 = Before assistant message (not applicable to sections, uses chat_history)
   270→ * 5 = At top of chat (before chat history)
   271→ * 6 = At bottom of chat (after all sections)
   272→ * 7 = After lorebook
   273→ */
   274→export function injectHUDContextIntoSections(
   275→  sections: PromptSection[],
   276→  contextSection: PromptSection,
   277→  position: number
   278→): PromptSection[] {
   279→  if (!contextSection.content.trim()) {
   280→    return sections;
   281→  }
   282→
   283→  const result: PromptSection[] = [...sections];
   284→
   285→  switch (position) {
   286→    case 0: // After system prompt
   287→      const sysSectionIdx = result.findIndex(s => s.type === 'system');
   288→      if (sysSectionIdx >= 0) {
   289→        result.splice(sysSectionIdx + 1, 0, contextSection);
   290→      } else {
   291→        result.unshift(contextSection);
   292→      }
   293→      break;
   294→
   295→    case 1: // After user message - inject before chat history
   296→    case 2: // Before user message - inject before chat history
   297→    case 3: // After assistant message - inject before chat history
   298→    case 4: // Before assistant message - inject before chat history
   299→      // These positions apply to messages, for sections we inject before chat history
   300→      const chatHistoryIdx = result.findIndex(s => s.type === 'chat_history');
   301→      if (chatHistoryIdx >= 0) {
   302→        result.splice(chatHistoryIdx, 0, contextSection);
   303→      } else {
   304→        result.push(contextSection);
   305→      }
   306→      break;
   307→
   308→    case 5: // At top of chat (before chat history)
   309→      const chatIdx = result.findIndex(s => s.type === 'chat_history');
   310→      if (chatIdx >= 0) {
   311→        result.splice(chatIdx, 0, contextSection);
   312→      } else {
   313→        result.push(contextSection);
   314→      }
   315→      break;
   316→
   317→    case 7: // After lorebook
   318→      const lorebookIdx = result.findIndex(s => s.type === 'lorebook');
   319→      if (lorebookIdx >= 0) {
   320→        result.splice(lorebookIdx + 1, 0, contextSection);
   321→      } else {
   322→        // If no lorebook, insert after system prompt
   323→        const sysIdx = result.findIndex(s => s.type === 'system');
   324→        if (sysIdx >= 0) {
   325→          result.splice(sysIdx + 1, 0, contextSection);
   326→        } else {
   327→          result.unshift(contextSection);
   328→        }
   329→      }
   330→      break;
   331→
   332→    case 6: // At bottom of chat
   333→    default:
   334→      result.push(contextSection);
   335→      break;
   336→  }
   337→
   338→  return result;
   339→}

   394→/**
   395→ * Build the system prompt from character data (SillyTavern style)
   396→ *
   397→ * Uses unified key resolution for ALL sections:
   398→ * - Template variables: {{user}}, {{char}}, {{userpersona}}, etc.
   399→ * - Stats keys: {{resistencia}}, {{habilidades}}, etc.
   400→ * - Sound keys: {{sonidos}}
   401→ */
   402→export function buildSystemPrompt(
   403→  character: CharacterCard,
   404→  userName: string = 'User',
   405→  persona?: Persona,
   406→  lorebookPlan?: LorebookInjectionPlan | null,
   407→  sessionStats?: SessionStats,
   408→  allCharacters?: CharacterCard[],
   409→  soundTriggers?: SoundTrigger[],
   410→  soundSettings?: AppSettings['sound'],
   411→  questTemplates?: QuestTemplate[],
   412→  sessionQuests?: SessionQuestInstance[],
   413→  questSettings?: QuestSettings,
   414→  lorebookAttributeKeys?: Record<string, string>,
   415→  inventoryData?: InventoryPromptData,
   416→  lorebookEntryKeyMap?: Record<string, string>
   417→): { prompt: string; sections: PromptSection[]; lorebookChatInjections: LorebookChatInjection[]; exampleMessages: ChatApiMessage[] } {
   418→  const sections: PromptSection[] = [];
   419→
   420→  // Resolve stats for the persona FIRST (user attributes like {{resistencia}})
   421→  // This must be done before character stats so personaResolvedStats is available
   422→  let personaResolvedStats: ResolvedStats | null = null;
   423→  if (persona?.statsConfig?.enabled && sessionStats) {
   424→    personaResolvedStats = resolveStats({
   425→      characterId: '__user__',
   426→      statsConfig: persona.statsConfig,
   427→      sessionStats,
   428→      lorebookEntryKeys: lorebookEntryKeyMap,
   429→    });
   430→  }
   431→
   432→  // Resolve stats for this character (includes skills block with full key resolution)
   433→  const resolvedStats = resolveStats({
   434→    characterId: character.id,
   435→    statsConfig: character.statsConfig,
   436→    sessionStats,
   437→    allCharacters,
   438→    userName,
   439→    characterName: character.name,
   440→    questTemplates,
   441→    personaDescription: persona?.description,
   442→    personaResolvedStats,
   443→    lorebookEntryKeys: lorebookEntryKeyMap,
   444→  });
   445→
   446→  // Build outlet sections map from lorebook plan for {{outlet::name}} macro resolution
   447→  const outletSections: Record<string, string> = {};
   448→  if (lorebookPlan?.outletSections.length) {
   449→    for (const outletSection of lorebookPlan.outletSections) {
   450→      // Extract outlet name from label like "World Info (myOutlet)" → "myOutlet"
   451→      const match = outletSection.label.match(/^World Info \((.+)\)$/);
   452→      const outletName = match ? match[1] : outletSection.label;
   453→      outletSections[outletName] = outletSection.content;
   454→    }
   455→  }
   456→
   457→  // Build unified key resolution context (includes quest data for {{activeQuests}}, outlet sections, lorebook attribute keys, lorebook entry keys, and inventory data for {{slots}})
   458→  const keyContext = buildKeyResolutionContext(character, userName, persona, resolvedStats, sessionStats, soundTriggers, soundSettings, personaResolvedStats, questTemplates, sessionQuests, questSettings, outletSections, lorebookAttributeKeys, inventoryData ? {
   459→    personaItems: inventoryData.personaItems,
   460→    sessionEquipment: inventoryData.sessionEquipment || inventoryData.equippedItems?.flatMap(({ entry, item }) =>
   461→      entry.equippedSlotId ? [{ itemId: item.id, equippedSlotId: entry.equippedSlotId, slotEffectText: item.slotEffects?.find(se => se.slotId === entry.equippedSlotId)?.effectText }] : []
   462→    ) || [],
   463→    activeEffects: inventoryData.activeEffects,
   464→    currency: inventoryData.currency,
   465→    currencyName: inventoryData.currencyName,
   466→    currencyIcon: inventoryData.currencyIcon,
   467→    inventorySettings: inventoryData.inventorySettings,
   468→  } : undefined, lorebookEntryKeyMap);
   469→
   470→  // Main system instruction
   471→  // If character has a custom system prompt, use it instead of the default
   472→  const systemContent = character.systemPrompt?.trim()
   473→    ? character.systemPrompt
   474→    : `You are now in roleplay mode. You will act as ${character.name}.`;
   475→
   476→  sections.push({
   477→    type: 'system',
   478→    label: 'System Prompt',
   479→    content: systemContent,
   480→    color: SECTION_COLORS.system
   481→  });
   482→
   483→  // World Time (fictional clock) — compact context so the LLM knows when the scene happens
   484→  const worldClock = sessionStats?.worldClock;
   485→  if (!worldClock || worldClock.enabled !== false) {
   486→    sections.push({
   487→      type: 'scenario',
   488→      label: 'World Time',
   489→      content: `[TIEMPO DEL MUNDO] ${formatWorldClock(worldClock || createDefaultWorldClock())} (avanza ${worldClock?.minutesPerTurn || 20} min por turno)`,
   490→      color: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-300'
   491→    });
   492→  }
   493→
   494→  // Lorebook position 0: After system prompt
   495→  if (lorebookPlan?.position0Section) {
   496→    sections.push(lorebookPlan.position0Section);
   497→  }
   498→
   499→  // NOTE: User's Persona section removed - use {{persona}} key in character sections instead
   500→  // The persona content is now injected via the {{persona}} template key which can be
   501→  // placed anywhere in the character's description, scenario, etc.
   502→
   503→  // NOTE: Inventory section removed - use {{slots}} key in character sections instead.
   504→  // The {{slots}} key resolves slot equipment + active consumable effects and can be
   505→  // placed anywhere in the character's description, scenario, systemPrompt, etc.
   506→
   507→  // Add character description
   508→  if (character.description) {
   509→    sections.push({
   510→      type: 'character_description',
   511→      label: 'Character Description',
   512→      content: character.description,
   513→      color: SECTION_COLORS.character_description
   514→    });
   515→  }
   516→
   517→  // Add personality
   518→  if (character.personality) {
   519→    sections.push({
   520→      type: 'personality',
   521→      label: 'Personality',
   522→      content: character.personality,
   523→      color: SECTION_COLORS.personality
   524→    });
   525→  }
   526→
   527→  // FASE 5 (UPDATED): The automatic "Estado Emocional" prompt section was REMOVED.
   528→  // The emotional state is now injected manually by the user via the {{estado}}
   529→  // key, which the key resolver replaces (in card sections, lorebook entries or
   530→  // system prompt templates) with the character's current emotional state.
   531→  // This gives full control over WHERE and HOW the state appears in the prompt.
   532→
   533→  // Add scenario
   534→  if (character.scenario) {
   535→    sections.push({
   536→      type: 'scenario',
   537→      label: 'Scenario',
   538→      content: character.scenario,
   539→      color: SECTION_COLORS.scenario
   540→    });
   541→  }
   542→
   543→  // Add character's note (user-defined instructions for this character)
   544→  // NOTE: According to SillyTavern docs, Character Notes comes BEFORE Example Dialogue
   545→  if (character.characterNote) {
   546→    sections.push({
   547→      type: 'character_note',
   548→      label: "Character's Note",
   549→      content: character.characterNote,
   550→      color: SECTION_COLORS.character_note
   551→    });
   552→  }
   553→
   554→  // NOTE: [GESTIÓN DE ATRIBUTOS] is NOT hardcoded anymore. The block is only
   555→  // injected where the user places {{attributes}} / {{attributes_rulz}}
   556→  // (card sections or lorebook entries) — resolved by key-resolver Phase 5.97.
   557→
   558→  // GUARDARROPA V2: Wardrobe Section — tells the LLM about the wardrobe (outfits)

   838→/**
   839→ * Inject lorebook content at chat-level positions (1-4).
   840→ * Merges content into existing messages to avoid breaking message alternation.
   841→ *
   842→ * Position mapping:
   843→ * 1 = After last user message
   844→ * 2 = Before last user message
   845→ * 3 = After last assistant message
   846→ * 4 = Before last assistant message
   847→ */
   848→function applyChatInjections(
   849→  messages: ChatApiMessage[],
   850→  injections: LorebookChatInjection[]
   851→): void {
   852→  for (const injection of injections) {
   853→    const content = `\n\n[${injection.label}]\n${injection.content}`;
   854→
   855→    switch (injection.position) {
   856→      case 1: { // After last user message
   857→        const lastUserIdx = messages.map((m, i) => m.role === 'user' ? i : -1).filter(i => i >= 0).pop();
   858→        if (lastUserIdx !== undefined && lastUserIdx >= 0) {
   859→          messages[lastUserIdx] = {
   860→            ...messages[lastUserIdx],
   861→            content: messages[lastUserIdx].content + content
   862→          };
   863→        }
   864→        break;
   865→      }
   866→      case 2: { // Before last user message
   867→        const lastUserIdx2 = messages.map((m, i) => m.role === 'user' ? i : -1).filter(i => i >= 0).pop();
   868→        if (lastUserIdx2 !== undefined && lastUserIdx2 >= 0) {
   869→          messages[lastUserIdx2] = {
   870→            ...messages[lastUserIdx2],
   871→            content: content + messages[lastUserIdx2].content
   872→          };
   873→        }
   874→        break;
   875→      }
   876→      case 3: { // After last assistant message
   877→        const lastAsstIdx = messages.map((m, i) => m.role === 'assistant' ? i : -1).filter(i => i >= 0).pop();
   878→        if (lastAsstIdx !== undefined && lastAsstIdx >= 0) {
   879→          messages[lastAsstIdx] = {
   880→            ...messages[lastAsstIdx],
   881→            content: messages[lastAsstIdx].content + content
   882→          };
   883→        }
   884→        break;
   885→      }
   886→      case 4: { // Before last assistant message
   887→        const lastAsstIdx4 = messages.map((m, i) => m.role === 'assistant' ? i : -1).filter(i => i >= 0).pop();
   888→        if (lastAsstIdx4 !== undefined && lastAsstIdx4 >= 0) {
   889→          messages[lastAsstIdx4] = {
   890→            ...messages[lastAsstIdx4],
   891→            content: content + messages[lastAsstIdx4].content
   892→          };
   893→        }
   894→        break;
   895→      }
   896→    }
   897→  }
   898→}

   900→/**
   901→ * Build messages array for chat models
   902→ * 
   903→ * Order (SillyTavern style):
   904→ * 1. System message (system prompt)
   905→ * 2. Chat history
   906→ * 3. Author's Note (injected AFTER chat history, as system message)
   907→ * 4. Post-History Instructions (injected AFTER Author's Note, as system message)
   908→ */
   909→export function buildChatMessages(
   910→  systemPrompt: string,
   911→  messages: ChatMessage[],
   912→  character: CharacterCard,
   913→  userName: string = 'User',
   914→  postHistoryInstructions?: string,
   915→  authorNote?: string,
   916→  useSystemRole: boolean = false,
   917→  embeddingsContext?: string,  // embeddings injected before chat history
   918→  lorebookChatInjections?: LorebookChatInjection[],  // positions 1-4: inject into specific messages
   919→  exampleMessages?: ChatApiMessage[]  // SillyTavern-style example dialogue as chat messages
   920→): ChatApiMessage[] {
   921→  // =============================================
   922→  // Step 1: Build all system content as ONE message
   923→  // =============================================
   924→  // Consolidate system prompt + embeddings + author note + post-history
   925→  // into a single system message to avoid consecutive system messages
   926→  // which break Jinja templates in models like LM Studio.
   927→  const systemParts: string[] = [];
   928→
   929→  // Main system prompt
   930→  if (systemPrompt.trim()) {
   931→    systemParts.push(systemPrompt);
   932→  }
   933→
   934→  // Embeddings Context
   935→  if (embeddingsContext?.trim()) {
   936→    systemParts.push(embeddingsContext);
   937→  }
   938→
   939→  // Author's Note
   940→  if (authorNote?.trim()) {
   941→    systemParts.push(`[Author's Note]\n${authorNote}`);
   942→  }
   943→
   944→  // Post-History Instructions
   945→  if (postHistoryInstructions?.trim()) {
   946→    systemParts.push(postHistoryInstructions);
   947→  }
   948→
   949→  const chatMessages: ChatApiMessage[] = [];
   950→
   951→  // Single system message (or assistant if useSystemRole is false)
   952→  if (systemParts.length > 0) {
   953→    chatMessages.push({
   954→      role: useSystemRole ? 'system' : 'assistant',
   955→      content: systemParts.join('\n\n---\n\n')
   956→    });
   957→  }
   958→
   959→  // =============================================
   960→  // Step 2: Inject example dialogue as chat messages (SillyTavern style)
   961→  // =============================================
   962→  // These go between the system message and the actual chat history
   963→  // They serve as few-shot examples for the LLM
   964→  if (exampleMessages && exampleMessages.length > 0) {
   965→    chatMessages.push(...exampleMessages);
   966→  }
   967→
   968→  // =============================================
   969→  // Step 3: Build chat history with proper alternation
   970→  // =============================================
   971→  // Merge consecutive same-role messages and ensure
   972→  // strict user/assistant alternation for OpenAI-compatible APIs.
   973→  const visibleMessages = messages.filter(m => !m.isDeleted && !m.isNarratorMessage);
   974→
   975→  // First pass: merge consecutive same-role messages
   976→  const mergedMessages: ChatApiMessage[] = [];
   977→  for (const msg of visibleMessages) {
   978→    const role = msg.role === 'user' ? 'user' : 'assistant';
   979→    const last = mergedMessages[mergedMessages.length - 1];
   980→    if (last && last.role === role) {
   981→      // Merge into previous message
   982→      last.content += '\n' + msg.content;
   983→    } else {
   984→      mergedMessages.push({ role, content: msg.content });
   985→    }
   986→  }
   987→
   988→  // Second pass: enforce user/assistant alternation
   989→  // If the first chat message is 'assistant', prepend a synthetic user message
   990→  if (mergedMessages.length > 0 && mergedMessages[0].role === 'assistant') {
   991→    mergedMessages.unshift({ role: 'user', content: '*continúa*' });
   992→  }
   993→
   994→  // If two same-role messages are still adjacent (shouldn't happen after merge,
   995→  // but as safety), insert a bridging message
   996→  const finalHistory: ChatApiMessage[] = [];
   997→  for (const msg of mergedMessages) {
   998→    const last = finalHistory[finalHistory.length - 1];
   999→    if (last && last.role === msg.role) {
  1000→      // Insert bridging message
  1001→      const bridgeRole = msg.role === 'user' ? 'assistant' : 'user';
  1002→      finalHistory.push({ role: bridgeRole, content: '*continúa*' });
  1003→    }
  1004→    finalHistory.push(msg);
  1005→  }
  1006→
  1007→  // Bridge between example messages and chat history if needed
  1008→  // If the last example message has the same role as the first chat message,
  1009→  // insert a bridging message to maintain proper user/assistant alternation
  1010→  if (finalHistory.length > 0 && chatMessages.length > 0) {
  1011→    const lastExistingMsg = chatMessages[chatMessages.length - 1];
  1012→    const firstChatMsg = finalHistory[0];
  1013→    if (lastExistingMsg.role === firstChatMsg.role) {
  1014→      const bridgeRole = firstChatMsg.role === 'user' ? 'assistant' : 'user';
  1015→      chatMessages.push({ role: bridgeRole, content: '*continúa*' });
  1016→    }
  1017→  }
  1018→
  1019→  chatMessages.push(...finalHistory);
  1020→
  1021→  // =============================================
  1022→  // Step 4: Inject lorebook chat-level content (positions 1-4)
  1023→  // =============================================
  1024→  if (lorebookChatInjections?.length) {
  1025→    applyChatInjections(chatMessages, lorebookChatInjections);
  1026→  }
  1027→
  1028→  return chatMessages;
  1029→}

  1031→/**
  1032→ * Build prompt for completion-style APIs (Ollama, KoboldCPP, etc.)
  1033→ * 
  1034→ * Order (SillyTavern style):
  1035→ * 1. System prompt
  1036→ * 2. Chat history
  1037→ * 3. Author's Note
  1038→ * 4. Post-History Instructions
  1039→ * 5. Assistant prefix
  1040→ */
  1041→export function buildCompletionPrompt(config: CompletionPromptConfig): string {
  1042→  const { systemPrompt, messages, character, userName, postHistoryInstructions, authorNote, embeddingsContext, exampleMessages, allCharacters } = config;
  1043→  const parts: string[] = [];
  1044→
  1045→  parts.push(systemPrompt);
  1046→  parts.push('\n---\n');
  1047→
  1048→  // Embeddings Context - injected before chat history for recency primacy
  1049→  if (embeddingsContext?.trim()) {
  1050→    parts.push(embeddingsContext);
  1051→    parts.push('\n---\n');
  1052→  }
  1053→
  1054→  // Example dialogue is now included in the systemPrompt string (as [EJEMPLO] sections).
  1055→  // The exampleMessages parameter is kept for backward compatibility but will always be empty.
  1056→  // No separate injection needed here.
  1057→
  1058→  // Exclude narrator messages from prompt
  1059→  const visibleMessages = messages.filter(m => !m.isDeleted && !m.isNarratorMessage);
  1060→
  1061→  for (const msg of visibleMessages) {
  1062→    if (msg.role === 'user') {
  1063→      parts.push(`${userName}: ${msg.content}`);
  1064→    } else if (msg.role === 'assistant') {
  1065→      // Use actual speaker name from characterId (important for group chats where
  1066→      // multiple characters speak - each should be attributed to their own name)
  1067→      const speakerName = msg.characterId
  1068→        ? (allCharacters?.find(c => c.id === msg.characterId)?.name || character.name)
  1069→        : character.name;
  1070→      parts.push(`${speakerName}: ${msg.content}`);
  1071→    }
  1072→  }
  1073→
  1074→  // Author's Note - injected AFTER chat history, BEFORE post-history instructions
  1075→  if (authorNote?.trim()) {
  1076→    parts.push(`\n[Author's Note]\n${authorNote}`);
  1077→  }
  1078→
  1079→  // Post-History Instructions - injected AFTER Author's Note
  1080→  if (postHistoryInstructions) {
  1081→    parts.push(`\n${postHistoryInstructions}`);
  1082→  }
  1083→
  1084→  parts.push(`\n${character.name}:`);
  1085→
  1086→  return parts.join('\n');
  1087→}

  1437→/**
  1438→ * Build messages array for group chat
  1439→ *
  1440→ * Order (SillyTavern style):
  1441→ * 1. System message (system prompt)
  1442→ * 2. Chat history (all visible messages)
  1443→ * 3. Previous responses from this turn
  1444→ * 4. Author's Note (injected AFTER chat history, as system message)
  1445→ * 5. Post-History Instructions (injected AFTER Author's Note, as system message)
  1446→ *
  1447→ * @param isForNarrator - If true, narrator messages ARE included (narrator sees all)
  1448→ */
  1449→export function buildGroupChatMessages(
  1450→  systemPrompt: string,
  1451→  messages: ChatMessage[],
  1452→  character: CharacterCard,
  1453→  allCharacters: CharacterCard[],
  1454→  userName: string = 'User',
  1455→  previousResponses?: Array<{ characterName: string; content: string }>,
  1456→  postHistoryInstructions?: string,
  1457→  authorNote?: string,
  1458→  isForNarrator: boolean = false,
  1459→  embeddingsContext?: string,  // embeddings injected before chat history
  1460→  lorebookChatInjections?: LorebookChatInjection[],  // positions 1-4: inject into specific messages
  1461→  exampleMessages?: ChatApiMessage[],  // SillyTavern-style example dialogue as chat messages
  1462→  turnDirective?: string  // per-responder directive appended to the last user message (anti impersonation)
  1463→): GroupPromptBuildResult {
  1464→  // =============================================
  1465→  // Step 1: Build all system content as ONE message
  1466→  // =============================================
  1467→  const systemParts: string[] = [];
  1468→
  1469→  if (systemPrompt.trim()) {
  1470→    systemParts.push(systemPrompt);
  1471→  }
  1472→  if (embeddingsContext?.trim()) {
  1473→    systemParts.push(embeddingsContext);
  1474→  }
  1475→  if (authorNote?.trim()) {
  1476→    systemParts.push(`[Author's Note]\n${authorNote}`);
  1477→  }
  1478→  if (postHistoryInstructions?.trim()) {
  1479→    systemParts.push(postHistoryInstructions);
  1480→  }
  1481→
  1482→  const chatMessages: ChatApiMessage[] = [];
  1483→
  1484→  // Single system message
  1485→  // role 'system' (matches 1-to-1 buildChatMessages with useSystemRole=true):
  1486→  // sending the system prompt as 'assistant' weakens instruction-following and
  1487→  // encourages the model to keep narrating the whole scene by itself.
  1488→  if (systemParts.length > 0) {
  1489→    chatMessages.push({
  1490→      role: 'system',
  1491→      content: systemParts.join('\n\n---\n\n')
  1492→    });
  1493→  }
  1494→
  1495→  // =============================================
  1496→  // Step 2: Inject example dialogue as chat messages (SillyTavern style)
  1497→  // =============================================
  1498→  // These go between the system message and the actual chat history
  1499→  // They serve as few-shot examples for the LLM
  1500→  if (exampleMessages && exampleMessages.length > 0) {
  1501→    chatMessages.push(...exampleMessages);
  1502→  }
  1503→
  1504→  // =============================================
  1505→  // Step 3: Build chat history with proper alternation
  1506→  // =============================================
  1507→  const visibleMessages = isForNarrator
  1508→    ? messages.filter(m => !m.isDeleted)
  1509→    : messages.filter(m => !m.isDeleted && !m.isNarratorMessage);
  1510→
  1511→  const historyLines: string[] = [];
  1512→
  1513→  // Build history lines + API messages
  1514→  for (const msg of visibleMessages) {
  1515→    // FIX: summary blob gets no speaker attribution (see note below)
  1516→    if (msg.role !== 'user' && msg.content?.startsWith('[RECUERDOS ANTERIORES]')) {
  1517→      historyLines.push(msg.content);
  1518→      continue;
  1519→    }
  1520→    const speaker = msg.role === 'user' ? userName :
  1521→      (allCharacters.find(c => c.id === msg.characterId)?.name || 'Character');
  1522→    historyLines.push(`${speaker}: ${msg.content}`);
  1523→  }
  1524→
  1525→  // Build merged API messages with proper alternation
  1526→  // IMPORTANT: In group chats, multiple characters speak as 'assistant' role.
  1527→  // When merging consecutive assistant messages, we MUST include speaker names
  1528→  // so the LLM can distinguish which character said what.
  1529→  // FIX: the synthetic [RECUERDOS ANTERIORES] summary message must NOT get a
  1530→  // speaker prefix — it was rendered as "Aitana: [RECUERDOS ANTERIORES]…",
  1531→  // misattributing the memory blob to the character.
  1532→  const isSummaryMessage = (m: ChatMessage) => m.content?.startsWith('[RECUERDOS ANTERIORES]');
  1533→  const mergedMessages: ChatApiMessage[] = [];
  1534→  for (const msg of visibleMessages) {
  1535→    const role = msg.role === 'user' ? 'user' : 'assistant';
  1536→    if (msg.role !== 'user' && isSummaryMessage(msg)) {
  1537→      // Summary blob: no speaker attribution
  1538→      const last = mergedMessages[mergedMessages.length - 1];
  1539→      if (last && last.role === 'assistant') {
  1540→        last.content += '\n' + msg.content;
  1541→      } else {
  1542→        mergedMessages.push({ role: 'assistant', content: msg.content });
  1543→      }
  1544→      continue;
  1545→    }
  1546→    // Include speaker name for assistant messages so different characters' lines
  1547→    // are distinguishable even when merged into the same message
  1548→    const speakerName = msg.role === 'user' ? userName :
  1549→      (allCharacters.find(c => c.id === msg.characterId)?.name || character.name);
  1550→    const contentWithSpeaker = msg.role === 'user' ? msg.content : `${speakerName}: ${msg.content}`;
  1551→    const last = mergedMessages[mergedMessages.length - 1];
  1552→    if (last && last.role === role) {
  1553→      last.content += '\n' + contentWithSpeaker;
  1554→    } else {
  1555→      mergedMessages.push({ role, content: contentWithSpeaker });
  1556→    }
  1557→  }
  1558→
  1559→  // Add previous responses from this turn
  1560→  if (previousResponses && previousResponses.length > 0) {
  1561→    for (const resp of previousResponses) {
  1562→      historyLines.push(`${resp.characterName}: ${resp.content}`);
  1563→      const last = mergedMessages[mergedMessages.length - 1];
  1564→      const contentWithSpeaker = `${resp.characterName}: ${resp.content}`;
  1565→      if (last && last.role === 'assistant') {
  1566→        last.content += '\n' + contentWithSpeaker;
  1567→      } else {
  1568→        mergedMessages.push({ role: 'assistant', content: contentWithSpeaker });
  1569→      }
  1570→    }
  1571→  }
  1572→
  1573→  // Enforce alternation: if first is assistant, prepend synthetic user
  1574→  if (mergedMessages.length > 0 && mergedMessages[0].role === 'assistant') {
  1575→    mergedMessages.unshift({ role: 'user', content: '*continúa*' });
  1576→  }
  1577→
  1578→  // Safety: insert bridging messages if any same-role adjacency remains
  1579→  const finalHistory: ChatApiMessage[] = [];
  1580→  for (const msg of mergedMessages) {
  1581→    const last = finalHistory[finalHistory.length - 1];
  1582→    if (last && last.role === msg.role) {
  1583→      const bridgeRole = msg.role === 'user' ? 'assistant' : 'user';
  1584→      finalHistory.push({ role: bridgeRole, content: '*continúa*' });
  1585→    }
  1586→    finalHistory.push(msg);
  1587→  }
  1588→
  1589→  // Bridge between example messages and chat history if needed
  1590→  if (finalHistory.length > 0 && chatMessages.length > 0) {
  1591→    const lastExistingMsg = chatMessages[chatMessages.length - 1];
  1592→    const firstChatMsg = finalHistory[0];
  1593→    if (lastExistingMsg.role === firstChatMsg.role) {
  1594→      const bridgeRole = firstChatMsg.role === 'user' ? 'assistant' : 'user';
  1595→      chatMessages.push({ role: bridgeRole, content: '*continúa*' });
  1596→    }
  1597→  }
  1598→
  1599→  chatMessages.push(...finalHistory);
  1600→
  1601→  // =============================================
  1602→  // Step 4: Turn directive — appended to the LAST user message
  1603→  // =============================================
  1604→  // Recency bias makes the end of the prompt the strongest position.
  1605→  // Appending "respond ONLY as X" right after the user's message (instead of
  1606→  // burying it in the system block) is what actually stops cross-character
  1607→  // impersonation in group chats.
  1608→  if (turnDirective?.trim()) {
  1609→    const lastMsg = chatMessages[chatMessages.length - 1];
  1610→    if (lastMsg && lastMsg.role === 'user') {
  1611→      lastMsg.content = `${lastMsg.content}\n\n${turnDirective}`;
  1612→    } else {
  1613→      chatMessages.push({ role: 'user', content: turnDirective });
  1614→    }
  1615→  }
  1616→
  1617→  // Inject lorebook chat-level content (positions 1-4)
  1618→  if (lorebookChatInjections?.length) {
  1619→    applyChatInjections(chatMessages, lorebookChatInjections);
  1620→  }
  1621→
  1622→  // Build chat history section for prompt viewer
  1623→  let chatHistorySection: PromptSection | undefined;
  1624→  if (historyLines.length > 0) {
  1625→    chatHistorySection = {
  1626→      type: 'chat_history',
  1627→      label: 'Chat History',
  1628→      content: historyLines.join('\n\n'),
  1629→      color: SECTION_COLORS.chat_history
  1630→    };
  1631→  }
  1632→
  1633→  return {
  1634→    systemPrompt,
  1635→    sections: [],
  1636→    chatMessages,
  1637→    chatHistorySection
  1638→  };
  1639→}
  1640→
  1641→/**
  1642→ * Build the per-responder turn directive for group chats (anti impersonation).
  1643→ * Appended to the END of the last user message by buildGroupChatMessages.
  1644→ *
  1645→ * This mirrors SillyTavern's group "[Write the next reply only as ...]" pattern:
  1646→ * the model is told, at the position of maximum attention, exactly who it may
  1647→ * voice in this reply.
  1648→ */
  1649→export function buildGroupTurnDirective(
  1650→  character: CharacterCard,
  1651→  allCharacters?: CharacterCard[],
  1652→  userName: string = 'User',
  1653→  isNarrator: boolean = false
  1654→): string {
  1655→  // The narrator IS a scene voice: they narrate the world, but still must not
  1656→  // write dialogue "for" the other characters' minds.
  1657→  if (isNarrator) {
  1658→    return [
  1659→      `[INSTRUCCIÓN DE TURNO — GRUPO]`,
  1660→      `Es tu turno: responde como el NARRADOR (escenas, ambiente y eventos del mundo).`,
  1661→      `- NO escribas diálogo, acciones ni pensamientos "por" otros personajes (${(allCharacters || []).filter(c => c.id !== character.id && c.id !== '__user__').map(c => c.name).join(', ') || 'ninguno'}) ni por ${userName}.`,
  1662→      `- Responde directamente, sin prefijo de nombre ni encabezados.`,
  1663→    ].join('\n');
  1664→  }
  1665→
  1666→  const otherNames = (allCharacters || [])
  1667→    .filter(c => c.id !== character.id && c.id !== '__user__')
  1668→    .map(c => c.name);
  1669→
  1670→  return [
  1671→    `[INSTRUCCIÓN DE TURNO — GRUPO]`,
  1672→    `Es el turno de ${character.name}. Responde AHORA ÚNICAMENTE como ${character.name}.`,
  1673→    `- Escribe SOLO las líneas, acciones y pensamientos de ${character.name}.`,
  1674→    `- NO escribas nada "por" otros personajes (${otherNames.join(', ') || 'ninguno'}) ni por ${userName}: ellos responden en sus propios turnos.`,
  1675→    `- NO repitas ni parafrasees lo que otros personajes ya dijeron.`,
  1676→    `- Responde directamente, sin prefijo de nombre ("${character.name}:") ni encabezados.`,
  1677→  ].join('\n');
  1678→}

  1684→/**
  1685→ * Process character and return effective values
  1686→ * Uses the unified key resolver
  1687→ */
  1688→export function processCharacter(
  1689→  character: CharacterCard,
  1690→  userName: string,
  1691→  persona?: Persona,
  1692→  sessionStats?: SessionStats,
  1693→  allCharacters?: CharacterCard[],
  1694→  questTemplates?: QuestTemplate[]
  1695→): CharacterCard {
  1696→  // Resolve stats for the persona FIRST
  1697→  let personaResolvedStats: ResolvedStats | null = null;
  1698→  if (persona?.statsConfig?.enabled && sessionStats) {
  1699→    personaResolvedStats = resolveStats({
  1700→      characterId: '__user__',
  1701→      statsConfig: persona.statsConfig,
  1702→      sessionStats,
  1703→    });
  1704→  }
  1705→
  1706→  // Resolve stats for this character (includes skills block with full key resolution)
  1707→  const resolvedStats = resolveStats({
  1708→    characterId: character.id,
  1709→    statsConfig: character.statsConfig,
  1710→    sessionStats,
  1711→    allCharacters,
  1712→    userName,
  1713→    characterName: character.name,
  1714→    questTemplates,
  1715→    personaDescription: persona?.description,
  1716→    personaResolvedStats,
  1717→  });
  1718→
  1719→  // Build key resolution context
  1720→  const keyContext = buildKeyResolutionContext(character, userName, persona, resolvedStats, sessionStats, undefined, undefined, personaResolvedStats);
  1721→
  1722→  // Process all text fields
  1723→  return {
  1724→    ...character,
  1725→    description: resolveAllKeys(character.description, keyContext),
  1726→    personality: resolveAllKeys(character.personality, keyContext),
  1727→    scenario: resolveAllKeys(character.scenario, keyContext),
  1728→    firstMes: resolveAllKeys(character.firstMes, keyContext),
  1729→    mesExample: resolveAllKeys(character.mesExample, keyContext),
  1730→    systemPrompt: resolveAllKeys(character.systemPrompt, keyContext),
  1731→    postHistoryInstructions: resolveAllKeys(character.postHistoryInstructions, keyContext),
  1732→    characterNote: resolveAllKeys(character.characterNote, keyContext),
  1733→    authorNote: resolveAllKeys(character.authorNote, keyContext),
  1734→    alternateGreetings: (character.alternateGreetings || []).map(greeting =>
  1735→      resolveAllKeys(greeting, keyContext)
  1736→    )
  1737→  };
  1738→}

  1803→ * Get messages for summarization
  1804→ * Returns messages that should be included in summary generation
  1805→ * Note: Narrator messages are excluded from summaries
  1806→ */
  1807→export function getMessagesForSummary(
  1808→  messages: ChatMessage[],
  1809→  summarySettings: { triggerThreshold: number; keepRecentMessages: number }
  1810→): ChatMessage[] {
  1811→  // Exclude deleted and narrator messages from summaries
  1812→  const visibleMessages = messages.filter(m => !m.isDeleted && !m.isNarratorMessage);
  1813→
  1814→  if (visibleMessages.length <= summarySettings.triggerThreshold) {
  1815→    return [];
  1816→  }
  1817→
  1818→  // Exclude recent messages that should stay unsummarized
  1819→  const messagesToSummarize = visibleMessages.slice(
  1820→    0,
  1821→    visibleMessages.length - summarySettings.keepRecentMessages
  1822→  );
  1823→
  1824→  return messagesToSummarize;
  1825→}
