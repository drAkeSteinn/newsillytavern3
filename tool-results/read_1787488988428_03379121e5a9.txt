    93→/**
    94→ * Execute detected tool calls, send SSE events, and return results.
    95→ * Returns { newContent: combined display messages, shouldContinue: true if tools were executed }
    96→ */
    97→async function executeToolCallsAndContinue(
    98→  toolCalls: NativeToolCall[],
    99→  availableTools: Array<{ id: string; name: string; label: string; icon: string }>,
   100→  currentRound: number,
   101→  maxRounds: number,
   102→  character: CharacterCard,
   103→  sessionId: string,
   104→  userName: string,
   105→  controller: { enqueue: (chunk: string) => void },
   106→  sessionQuests?: SessionQuestInstance[],
   107→  questTemplates?: QuestTemplate[],
   108→  statsConfig?: CharacterStatsConfig,
   109→  sessionStats?: SessionStats,
   110→  allCharacters?: CharacterCard[],
   111→  characterMemory?: CharacterMemory,
   112→  lorebooks?: Lorebook[],
   113→): Promise<{ newContent: string; shouldContinue: boolean; toolResults: Array<{ success: boolean; displayMessage: string }>; questActivations: QuestActivation[]; toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> }> {
   114→  if (toolCalls.length === 0 || currentRound >= maxRounds) {
   115→    return { newContent: '', shouldContinue: false, toolResults: [], questActivations: [], toolsUsed: [] };
   116→  }
   117→
   118→  const toolResults: Array<{ success: boolean; displayMessage: string }> = [];
   119→  const questActivations: QuestActivation[] = [];
   120→  const toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> = [];
   121→  let allDisplayMessages = '';
   122→
   123→  for (let i = 0; i < toolCalls.length; i++) {
   124→    const tc = toolCalls[i];
   125→    const toolDef = availableTools.find(t => t.name === tc.name);
   126→    const callId = `${Date.now()}_${i}`;
   127→
   128→    // Send tool_call_start event
   129→    controller.enqueue(createSSEJSON({
   130→      type: 'tool_call_start',
   131→      toolName: tc.name,
   132→      toolLabel: toolDef?.label || tc.name,
   133→      toolIcon: toolDef?.icon || 'Wrench',
   134→      params: tc.arguments,
   135→      callId,
   136→    }));
   137→
   138→    console.log(`[Tools] Executing tool call: ${tc.name}`, tc.arguments);
   139→
   140→    // Execute the tool
   141→    const toolResult = await executeTool(
   142→      tc.name,
   143→      tc.arguments,
   144→      {
   145→        characterId: character.id,
   146→        characterName: character.name,
   147→        sessionId,
   148→        userName,
   149→        sessionQuests,
   150→        questTemplates,
   151→        statsConfig: character.statsConfig,
   152→        sessionStats,
   153→        allCharacters,
   154→        characterMemory,
   155→        lorebooks,
   156→      },
   157→    );
   158→
   159→    // Send tool_call_result event
   160→    controller.enqueue(createSSEJSON({
   161→      type: 'tool_call_result',
   162→      toolName: tc.name,
   163→      success: toolResult.success,
   164→      displayMessage: toolResult.displayMessage,
   165→      duration: toolResult.duration || 0,
   166→      callId,
   167→    }));
   168→
   169→    // Check for quest activation and send SSE event
   170→    if (toolResult.questActivation) {
   171→      const activation = toolResult.questActivation;
   172→      console.log(`[Tools] Quest activation from ${tc.name}:`, activation);
   173→      
   174→      controller.enqueue(createSSEJSON({
   175→        type: 'quest_activation',
   176→        toolName: tc.name,
   177→        activationType: activation.type,
   178→        key: activation.key,
   179→        metadata: activation.metadata,
   180→      }));
   181→      
   182→      questActivations.push(activation);
   183→    }
   184→
   185→    // Check for action/skill activation and send SSE event
   186→    if (toolResult.actionActivation) {
   187→      const action = toolResult.actionActivation;
   188→      console.log(`[Tools] Action activation from ${tc.name}:`, action.skillName);
   189→      
   190→      controller.enqueue(createSSEJSON({
   191→        type: 'action_activation',
   192→        toolName: tc.name,
   193→        skillId: action.skillId,
   194→        skillName: action.skillName,
   195→        skillDescription: action.skillDescription,
   196→        skillCompletedDescription: action.skillCompletedDescription,
   197→        activationCosts: action.activationCosts,
   198→        activationRewards: action.activationRewards,
   199→        characterId: action.characterId,
   200→      }));
   201→    }
   202→
   203→    // Check for stat modification from modify_stat tool and send SSE event
   204→    if (toolResult.statActivation) {
   205→      const stat = toolResult.statActivation;
   206→      console.log(`[Tools] Stat activation from ${tc.name}:`, stat.attributeKey, stat.oldValue, '→', stat.newValue);
   207→      
   208→      controller.enqueue(createSSEJSON({
   209→        type: 'stat_activation',
   210→        toolName: tc.name,
   211→        characterId: stat.characterId,
   212→        attributeKey: stat.attributeKey,
   213→        attributeName: stat.attributeName,
   214→        attributeType: stat.attributeType,
   215→        oldValue: stat.oldValue,
   216→        newValue: stat.newValue,
   217→        reason: stat.reason,
   218→      }));
   219→    }
   220→
   221→    // Check for solicitud activation/completion and send SSE event
   222→    if (toolResult.solicitudActivation) {
   223→      const sol = toolResult.solicitudActivation;
   224→      console.log(`[Tools] Solicitud activation from ${tc.name}:`, sol.type, sol.solicitudKey);
   225→      
   226→      controller.enqueue(createSSEJSON({
   227→        type: 'solicitud_activation',
   228→        toolName: tc.name,
   229→        activationType: sol.type,
   230→        solicitudKey: sol.solicitudKey,
   231→        targetCharacterId: sol.targetCharacterId,
   232→        targetCharacterName: sol.targetCharacterName,
   233→        fromCharacterId: sol.fromCharacterId,
   234→        fromCharacterName: sol.fromCharacterName,
   235→        description: sol.description,
   236→        completionDescription: sol.completionDescription,
   237→        peticionKey: sol.peticionKey,
   238→      }));
   239→    }
   240→
   241→    // Check for memory activation (sync to client-side Character Memory)
   242→    if (toolResult.memoryActivation) {
   243→      const mem = toolResult.memoryActivation;
   244→      console.log(`[Tools] Memory activation from ${tc.name}:`, mem.type);
   245→      
   246→      controller.enqueue(createSSEJSON({
   247→        type: 'memory_activation',
   248→        toolName: tc.name,
   249→        activationType: mem.type,
   250→        characterId: mem.characterId,
   251→        eventData: mem.eventData,
   252→        relationshipData: mem.relationshipData,
   253→        noteContent: mem.noteContent,
   254→        deleteEventId: mem.deleteEventId,
   255→        deleteEmbeddingId: mem.deleteEmbeddingId,
   256→      }));
   257→    }
   258→
   259→    console.log(`[Tools] Tool ${tc.name}: success=${toolResult.success} duration=${toolResult.duration}ms`, toolResult.displayMessage);
   260→
   261→    toolResults.push({ success: toolResult.success, displayMessage: toolResult.displayMessage });
   262→    toolsUsed.push({
   263→      name: tc.name,
   264→      label: toolDef?.label || tc.name,
   265→      icon: toolDef?.icon || 'Wrench',
   266→      success: toolResult.success,
   267→    });
   268→    if (toolResult.displayMessage) {
   269→      allDisplayMessages += (allDisplayMessages ? '\n' : '') + toolResult.displayMessage;
   270→    }
   271→  }
   272→
   273→  // Return the display messages as content, tool results, and signal that a follow-up call is needed
   274→  return {
   275→    newContent: allDisplayMessages,
   276→    shouldContinue: true, // Always continue to get the LLM's natural response
   277→    toolResults,
   278→    questActivations,
   279→    toolsUsed,
   280→  };
   281→}

   283→/**
   284→ * POST /api/chat/proactive
   285→ *
   286→ * Generates a proactive message for a character using SSE streaming.
   287→ * Full feature parity with the stream route including tools, embeddings,
   288→ * key resolution, memory reinforcement, and extraction.
   289→ */
   290→export async function POST(request: NextRequest) {
   291→  try {
   292→    // Capture auth headers from Z.ai gateway for token resolution
   293→    const incomingXToken = request.headers.get('X-Token');
   294→    const fcSecurityToken = request.headers.get('x-fc-security-token');
   295→
   296→    const body = await request.json();
   297→
   298→    // Extract required fields from body
   299→    const {
   300→      character,
   301→      messages = [],
   302→      llmConfig,
   303→      userName = 'User',
   304→      persona,
   305→      proactiveConfig,
   306→      reason = 'timer_idle',
   307→      lastActivityAt,
   308→      isGroupChat = false,
   309→      // FASE 11 v2: tracking de índices usados para la rotación linear/random de casos.
   310→      usedCaseIndices: clientUsedCaseIndices = {},
   311→    } = body;
   312→
   313→    const clientUsedCaseIndicesTyped: UsedCaseIndices =
   314→      (clientUsedCaseIndices && typeof clientUsedCaseIndices === 'object')
   315→        ? clientUsedCaseIndices as UsedCaseIndices
   316→        : {};
   317→
   318→    // Validate required fields
   319→    if (!character?.name || !llmConfig?.provider) {
   320→      return createErrorResponse('Missing required fields: character, llmConfig', 400);
   321→    }
   322→
   323→    if (!proactiveConfig?.enabled) {
   324→      return createErrorResponse('Proactive messages are disabled for this character', 400);
   325→    }
   326→
   327→    if (!llmConfig) {
   328→      return createErrorResponse('No LLM configuration provided', 400);
   329→    }
   330→
   331→    // Validate API key for providers that require one
   332→    const providersRequiringApiKey = ['openai', 'vllm', 'lm-studio', 'custom', 'anthropic', 'grok'];
   333→    if (providersRequiringApiKey.includes(llmConfig.provider) && !llmConfig.apiKey) {
   334→      return createErrorResponse(
   335→        `API Key is required for ${llmConfig.provider} provider. Please configure it in Settings.`,
   336→        400
   337→      );
   338→    }
   339→
   340→    // Validate endpoint for providers that require one
   341→    const providersRequiringEndpoint = ['openai', 'vllm', 'lm-studio', 'custom', 'anthropic', 'ollama', 'text-generation-webui', 'koboldcpp'];
   342→    if (providersRequiringEndpoint.includes(llmConfig.provider) && !llmConfig.endpoint) {
   343→      return createErrorResponse(
   344→        `Endpoint URL is required for ${llmConfig.provider} provider. Please configure it in Settings.`,
   345→        400
   346→      );
   347→    }
   348→
   349→    // If using Z.ai provider, collect gateway-forwarded auth headers
   350→    let zaiRuntimeToken: string | undefined;
   351→    if (llmConfig.provider === 'z-ai') {
   352→      const gatewayToken = incomingXToken || fcSecurityToken || undefined;
   353→      if (gatewayToken) {
   354→        zaiRuntimeToken = gatewayToken;
   355→        console.log(`[Proactive Route] Z.ai runtime token available (${gatewayToken.length} chars, source: ${incomingXToken ? 'X-Token' : 'fc-security-token'})`);
   356→      } else {
   357→        console.log(`[Proactive Route] Z.ai: no gateway token available, using config file only`);
   358→      }
   359→    }
   360→
   361→    console.log(`[Proactive] Generating proactive message for "${character.name}" (reason: ${reason})`);
   362→
   363→    // Extract all additional fields from body (matching stream route)
   364→    const lorebooks: Lorebook[] = body.lorebooks || [];
   365→    const sessionStats: SessionStats | undefined = body.sessionStats;
   366→    const allCharacters: CharacterCard[] = body.allCharacters || [character];
   367→
   368→    // Extract HUD context from body
   369→    const hudContext: HUDContextConfig | undefined = body.hudContext;
   370→
   371→    // Extract Quest data
   372→    const questTemplates: QuestTemplate[] = body.questTemplates || [];
   373→    const sessionQuests: SessionQuestInstance[] = body.sessionQuests || [];
   374→    const questSettings: QuestSettings = {
   375→      ...DEFAULT_QUEST_SETTINGS,
   376→      ...(body.questSettings || {})
   377→    };
   378→
   379→    // Extract Sound data for {{sonidos}} key resolution
   380→    const soundTriggers: SoundTrigger[] = body.soundTriggers || [];
   381→    const soundSettings = body.settings?.sound;
   382→
   383→    // Extract summary for memory/context compression
   384→    const summary: SessionSummary | undefined = body.summary;
   385→
   386→    // Extract embeddings chat settings
   387→    const embeddingsChat: Partial<EmbeddingsChatSettings> = body.embeddingsChat || {};
   388→    const characterMemory: CharacterMemory | undefined = body.characterMemory;
   389→    const sessionId: string | undefined = body.sessionId;
   390→    const characterId: string | undefined = body.characterId;
   391→    const inventoryData: InventoryPromptData | undefined = body.inventoryData;
   392→
   393→    // Extract tools settings
   394→    const toolsSettings: ToolsSettings = {
   395→      enabled: body.toolsSettings?.enabled ?? true,
   396→      maxToolCallsPerTurn: body.toolsSettings?.maxToolCallsPerTurn ?? 2,
   397→      characterConfigs: body.toolsSettings?.characterConfigs || [],
   398→      usePromptBasedFallback: body.toolsSettings?.usePromptBasedFallback ?? false,
   399→      disabledTools: body.toolsSettings?.disabledTools || [],
   400→    };
   401→
   402→    // Create default character if none provided
   403→    const effectiveCharacter: CharacterCard = character || DEFAULT_CHARACTER;
   404→
   405→    // Get effective user name from persona or use provided userName
   406→    const effectiveUserName = getEffectiveUserName(persona, userName);
   407→
   408→    // Build context configuration from request or use defaults
   409→    const contextConfig: Partial<ContextConfig> = body.contextConfig || {};
   410→
   411→    // Apply sliding window to messages
   412→    const contextWindow = selectContextMessages(messages, llmConfig, contextConfig);
   413→
   414→    // Log context stats (for debugging)
   415→    const stats = getContextStats(messages);
   416→
   417→    // Process lorebooks and get matched entries
   418→    const { plan: lorebookPlan, lorebookAttributeKeys, lorebookEntryKeyMap, lorebookDebugEntries } = buildLorebookSectionForPrompt(
   419→      messages,
   420→      lorebooks,
   421→      {
   422→        scanDepth: (contextConfig as any).scanDepth,
   423→        userName: effectiveUserName,
   424→        charName: effectiveCharacter?.name,
   425→      },
   426→      { sessionStats, characterId: effectiveCharacter?.id, characters: allCharacters }
   427→    );
   428→
   429→    // ========================================
   430→    // DEBUG: Build lorebook debug data for SSE event
   431→    // ========================================
   432→    const lorebookDebugData = {
   433→      lorebookAttributeKeys,
   434→      debugEntries: lorebookDebugEntries,
   435→      availableStats: sessionStats
   436→        ? Object.fromEntries(
   437→            Object.entries(sessionStats.characterStats || {}).map(([charId, cs]) => [
   438→              charId,
   439→              cs?.attributeValues ? { ...cs.attributeValues } : '(no values)',
   440→            ])
   441→          )
   442→        : '(no sessionStats)',
   443→    };
   444→
   445→    // ========================================
   446→    // Embeddings Context Retrieval
   447→    // ========================================
   448→    // For proactive messages, use recent chat context as the search query
   449→    // since there's no new user message
   450→    let enrichedSearchQuery = '';
   451→    const searchContextDepth = embeddingsChat.searchContextDepth || 0;
   452→    if (searchContextDepth > 0) {
   453→      const recentHistory = messages
   454→        .filter(m => !m.isDeleted)
   455→        .slice(-(searchContextDepth * 2))
   456→        .map(m => m.content?.trim())
   457→        .filter(Boolean);
   458→      
   459→      if (recentHistory.length > 0) {
   460→        enrichedSearchQuery = recentHistory.join(' ');
   461→      }
   462→    } else {
   463→      // Use last few messages as search context
   464→      const recentHistory = messages
   465→        .filter(m => !m.isDeleted)
   466→        .slice(-4)
   467→        .map(m => m.content?.trim())
   468→        .filter(Boolean);
   469→      
   470→      if (recentHistory.length > 0) {
   471→        enrichedSearchQuery = recentHistory.join(' ');
   472→      }
   473→    }
   474→
   475→    // Safety: smart truncation based on embedding model's context window.
   476→    try {
   477→      const { MODEL_CONTEXT_LENGTHS, DEFAULT_CONTEXT_LENGTH, CHARS_PER_TOKEN } = await import('@/lib/embeddings/types');
   478→      const { loadConfig } = await import('@/lib/embeddings/config-persistence');
   479→      const embConfig = loadConfig();
   480→      const modelKey = embConfig.model || 'bge-m3:567m';
   481→      const modelCtx = MODEL_CONTEXT_LENGTHS[modelKey]
   482→        || MODEL_CONTEXT_LENGTHS[modelKey.split(':')[0]]
   483→        || DEFAULT_CONTEXT_LENGTH;
   484→      const maxQueryChars = Math.floor(modelCtx * 0.75 * CHARS_PER_TOKEN);
   485→
   486→      if (enrichedSearchQuery.length > maxQueryChars) {
   487→        // For proactive: keep the most recent part (end of the string)
   488→        enrichedSearchQuery = enrichedSearchQuery.slice(-maxQueryChars);
   489→        console.warn(
   490→          `[Proactive Route] Search query trimmed to ${enrichedSearchQuery.length} chars ` +
   491→          `(model: ${modelKey}, context: ${modelCtx} tokens)`
   492→        );
   493→      }
   494→    } catch { /* fallback: Ollama client handles truncation */ }
   495→
   496→    // Retrieve relevant embeddings based on enriched query and settings
   497→    // Pass Character Memory events for deduplication (avoid duplicate memory in prompt)
   498→    const existingMemoryEvents = characterMemory?.events?.map(e => ({
   499→      content: e.content,
   500→      importance: e.importance,
   501→    }));
   502→
   503→    // Extract last assistant message for bidirectional search
   504→    const lastAssistantMsg = messages
   505→      .filter(m => !m.isDeleted && m.role === 'assistant')
   506→      .pop()?.content;
   507→
   508→    const embeddingsResult = await retrieveEmbeddingsContext(
   509→      enrichedSearchQuery,
   510→      characterId || effectiveCharacter.id,
   511→      sessionId,
   512→      embeddingsChat,
   513→      undefined, // groupId
   514→      existingMemoryEvents, // for deduplication
   515→      lastAssistantMsg, // bidirectional search with last assistant message
   516→    );
   517→    
   518→    if (embeddingsResult.found) {
   519→      console.log(`[Proactive Route] Retrieved ${embeddingsResult.count} embeddings from namespaces: ${embeddingsResult.searchedNamespaces.join(', ')}`);
   520→    }
   521→
   522→    // ========================================
   523→    // Build system prompt with unified key resolution
   524→    // ========================================
   525→    // FASE 11 v2: Si systemPromptOverride está configurado, REEMPLAZA character.systemPrompt.
   526→    // El resto de la card (description, personality, scenario, etc.) se mantiene igual.
   527→    // buildSystemPrompt resuelve las keys ({{user}}, {{char}}, {{vida}}, etc.) internamente.
   528→    const _systemPromptOverrideRaw = proactiveConfig.systemPromptOverride?.trim();
   529→    const characterForPrompt: CharacterCard = _systemPromptOverrideRaw
   530→      ? { ...effectiveCharacter, systemPrompt: _systemPromptOverrideRaw }
   531→      : effectiveCharacter;
   532→
   533→    const { prompt: systemPrompt, sections: systemSections, lorebookChatInjections, exampleMessages } = buildSystemPrompt(
   534→      characterForPrompt,
   535→      effectiveUserName,
   536→      persona,
   537→      lorebookPlan,
   538→      sessionStats,
   539→      allCharacters, // Pass all characters for peticiones/solicitudes resolution
   540→      soundTriggers,
   541→      soundSettings,
   542→      questTemplates,
   543→      sessionQuests,
   544→      questSettings,
   545→      lorebookAttributeKeys,
   546→      inventoryData,    // Pass inventory data for Inventory V2 section
   547→      lorebookEntryKeyMap // Pass lorebook entry key map for {{entryKey}} resolution
   548→    );
   549→
   550→    // Build key resolution context for HUD context and quest sections
   551→    let streamPersonaResolvedStats: import('@/types').ResolvedStats | null = null;
   552→    if (persona?.statsConfig?.enabled && sessionStats) {
   553→      streamPersonaResolvedStats = resolveStats({
   554→        characterId: '__user__',
   555→        statsConfig: persona.statsConfig,
   556→        sessionStats,
   557→      });
   558→    }
   559→    const resolvedStats = resolveStats({
   560→      characterId: effectiveCharacter.id,
   561→      statsConfig: effectiveCharacter.statsConfig,
   562→      sessionStats: sessionStats,
   563→      allCharacters,
   564→      userName: effectiveUserName,
   565→      characterName: effectiveCharacter.name,
   566→      questTemplates,
   567→      personaDescription: persona?.description,
   568→      personaResolvedStats: streamPersonaResolvedStats,
   569→    });
   570→
   571→    // Build outlet sections map from lorebook plan for {{outlet::name}} macro resolution
