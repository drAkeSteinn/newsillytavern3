    75→/**
    76→ * Get all keys from a trigger (main key + alternatives + legacy keywords)
    77→ */
    78→function getAllTriggerKeys(trigger: CharacterSpriteTrigger): string[] {
    79→  const allKeys: string[] = [];
    80→  
    81→  // Main key
    82→  if (trigger.key) {
    83→    allKeys.push(trigger.key);
    84→  }
    85→  
    86→  // Alternative keys
    87→  if (trigger.keys && trigger.keys.length > 0) {
    88→    allKeys.push(...trigger.keys);
    89→  }
    90→  
    91→  // Legacy keywords support
    92→  if (trigger.keywords && trigger.keywords.length > 0) {
    93→    allKeys.push(...trigger.keywords);
    94→  }
    95→  
    96→  return allKeys;
    97→}
    98→
    99→/**
   100→ * Parse keys from comma-separated string
   101→ */
   102→function parseKeysString(keysStr: string): string[] {
   103→  return keysStr
   104→    .split(',')
   105→    .map(k => k.trim())
   106→    .filter(Boolean);
   107→}
   108→
   109→/**
   110→ * Convert keys array to comma-separated string
   111→ */
   112→function keysToString(keys: string[] | undefined): string {
   113→  return (keys || []).join(', ');
   114→}

   116→export function CharacterTriggerEditor({ character, onChange }: CharacterTriggerEditorProps) {
   117→  // Get triggers from character data
   118→  const triggers: CharacterSpriteTrigger[] = character.spriteTriggers || [];
   119→

   121→  const [showEditor, setShowEditor] = useState(false);
   122→  const [availableSprites, setAvailableSprites] = useState<SpriteIndexEntry[]>([]);
   123→  const [loadingSprites, setLoadingSprites] = useState(false);
   124→  
   125→  // Key fields state
   126→  const [mainKey, setMainKey] = useState('');
   127→  const [alternativeKeys, setAlternativeKeys] = useState('');
   128→
   129→  // Load available sprites from character's collection
   130→  useEffect(() => {
   131→    const loadSprites = async () => {
   132→      setLoadingSprites(true);
   133→      try {
   134→        const response = await fetch('/api/sprites/index');
   135→        const data = await response.json();

   137→        // Filter by character's selected collection
   138→        const collectionName = character.spriteConfig?.collection;
   139→        if (collectionName && data.sprites) {
   140→          setAvailableSprites(data.sprites.filter((s: SpriteIndexEntry) => s.pack === collectionName));

   142→          setAvailableSprites(data.sprites || []);
   143→        }

   145→        console.error('Error loading sprites:', error);

   147→      } finally {

   149→      }

   158→    onChange({
   159→      spriteTriggers: newTriggers,

   163→  const handleAddTrigger = () => {

   165→      id: uuidv4(),
   166→      title: 'Nuevo Trigger',
   167→      active: true,
   168→      key: '',
   169→      keys: [],
   170→      requirePipes: true,
   171→      caseSensitive: false,
   172→      spriteUrl: '',
   173→      spriteState: '',

   175→      returnToMode: 'idle_collection',
   176→      returnToSpriteUrl: '',
   177→      cooldownMs: 1000,
   178→      priority: 1,

   183→    setShowEditor(true);
   184→  };

   186→  const handleEditTrigger = (trigger: CharacterSpriteTrigger) => {
   187→    // Initialize key fields from trigger
   188→    setMainKey(trigger.key || '');
   189→    
   190→    // Get alternative keys (excluding main key from legacy keywords)
   191→    const altKeys = trigger.keys || [];
   192→    // If legacy keywords exist but keys doesn't, use keywords minus main key
   193→    if (!trigger.keys && trigger.keywords && trigger.keywords.length > 0) {
   194→      const legacyAltKeys = trigger.keywords.filter(k => k !== trigger.key);
   195→      setAlternativeKeys(legacyAltKeys.join(', '));
   196→    } else {
   197→      setAlternativeKeys(keysToString(altKeys));
   198→    }
   199→    
   200→    setEditingTrigger({ ...trigger });
   201→    setShowEditor(true);
   202→  };
   203→
   204→  const handleDeleteTrigger = (id: string) => {
   205→    updateTriggers(triggers.filter(t => t.id !== id));
   206→  };
   207→
   208→  const handleSaveTrigger = () => {
   209→    if (!editingTrigger) return;
   210→    
   211→    // Build the trigger with keys
   212→    const updatedTrigger: CharacterSpriteTrigger = {
   213→      ...editingTrigger,
   214→      key: mainKey.trim(),
   215→      keys: parseKeysString(alternativeKeys),
   216→      // Clear legacy keywords if we have new key system
   217→      keywords: undefined,
   218→    };
   219→    
   220→    const existingIndex = triggers.findIndex(t => t.id === updatedTrigger.id);
   221→    if (existingIndex >= 0) {
   222→      // Update existing
   223→      const newTriggers = [...triggers];
   224→      newTriggers[existingIndex] = updatedTrigger;
   225→      updateTriggers(newTriggers);
   226→    } else {
   227→      // Add new
   228→      updateTriggers([...triggers, updatedTrigger]);
   229→    }
   230→    
   231→    setEditingTrigger(null);
   232→    setShowEditor(false);
   233→  };
   234→
   235→  // Get sprite preview URL from trigger
   236→  const getSpritePreviewUrl = (trigger: CharacterSpriteTrigger) => {
   237→    // If it's a state reference
   238→    if (trigger.spriteState && trigger.spriteState !== CUSTOM_SPRITE_VALUE) {
   239→      // Get from state collections first
   240→      const stateCollection = character.spriteConfig?.stateCollections?.[trigger.spriteState as SpriteState];
   241→      if (stateCollection && stateCollection.entries.length > 0) {
   242→        // Return principal sprite URL
   243→        const principal = stateCollection.entries.find(e => e.role === 'principal');
   244→        if (principal) return principal.spriteUrl;
   245→        return stateCollection.entries[0]?.spriteUrl;
   246→      }
   247→      // Fall back to legacy sprites
   248→      if (character.spriteConfig?.sprites?.[trigger.spriteState as SpriteState]) {
   249→        return character.spriteConfig.sprites[trigger.spriteState as SpriteState];
   250→      }
   251→    }
   252→    // Custom sprite URL
   253→    return trigger.spriteUrl;
   254→  };

   258→    if (trigger.spriteState && trigger.spriteState !== CUSTOM_SPRITE_VALUE) {
   259→      return SPRITE_STATES.find(s => s.value === trigger.spriteState)?.label || trigger.spriteState;
   260→    }
   261→    if (trigger.spriteUrl) {
   262→      // Find label from available sprites
   263→      const sprite = availableSprites.find(s => s.url === trigger.spriteUrl);
   264→      return sprite?.label || 'Sprite personalizado';
   265→    }
   266→    return 'Sin sprite';
   267→  };

   269→  // Format time for display
   270→  const formatTime = (ms: number) => {
   271→    if (ms <= 0) return 'Nunca';
   272→    if (ms < 1000) return `${ms}ms`;
   273→    const seconds = Math.floor(ms / 1000);
   274→    if (seconds < 60) return `${seconds}s`;
   275→    const minutes = Math.floor(seconds / 60);
   276→    const remainingSeconds = seconds % 60;
   277→    return remainingSeconds > 0 ? `${minutes}m ${remainingSeconds}s` : `${minutes}m`;
   278→  };
   279→
   280→  // Determine if we're in custom mode for trigger sprite
   281→  const isCustomMode = !editingTrigger?.spriteState || editingTrigger.spriteState === '';
   282→
   283→  return (
   284→    <div className="space-y-4">
   285→      {/* Header */}
   286→      <div className="flex items-center justify-between">
   287→        <div>
   288→          <h3 className="text-sm font-medium">Triggers Simples</h3>
   289→          <p className="text-xs text-muted-foreground">
   290→            Sistema básico: cuando se detecta <strong>cualquier key</strong>, muestra el sprite asignado.
   291→          </p>
   292→        </div>
   293→        <Button size="sm" onClick={handleAddTrigger}>
   294→          <Plus className="w-4 h-4 mr-1" />
   295→          Agregar
   296→        </Button>
   297→      </div>
   298→      
   299→      {/* How it works */}
   300→      <div className="text-xs bg-muted/50 border rounded-lg p-3 space-y-2">
   301→        <div className="flex items-center gap-2 font-medium text-foreground">
   302→          <Zap className="w-4 h-4 text-amber-500" />
   303→          ¿Cómo funciona?
   304→        </div>
   305→        <div className="grid grid-cols-3 gap-2 text-center">
   306→          <div className="p-2 bg-background rounded border">
   307→            <div className="font-mono text-xs">|spritewin|</div>
   308→            <div className="text-muted-foreground text-[10px] mt-1">Detectado en chat</div>
   309→          </div>
   310→          <div className="flex items-center justify-center text-muted-foreground">→</div>
   311→          <div className="p-2 bg-background rounded border">
   312→            <div className="font-mono text-xs">win.png</div>
   313→            <div className="text-muted-foreground text-[10px] mt-1">Sprite mostrado</div>
   314→          </div>
   315→        </div>
   316→        <p className="text-muted-foreground">
   317→          Los triggers simples son ideales para emociones básicas. Para lógica más compleja, usa <strong>Packs</strong>.
   318→        </p>
   319→      </div>
   320→
   321→      {/* Triggers List */}
   322→      {triggers.length === 0 ? (
   323→        <div className="text-center py-8 text-muted-foreground border rounded-lg">
   324→          <Zap className="w-10 h-10 mx-auto mb-2 opacity-30" />
   325→          <p className="text-sm">No hay triggers configurados</p>
   326→          <p className="text-xs mt-1">Los triggers permiten cambiar expresiones automáticamente</p>
   327→        </div>
   328→      ) : (
   329→        <ScrollArea className="h-[300px]">
   330→          <div className="space-y-2 pr-2">
   331→            {triggers.map((trigger) => {
   332→              const allKeys = getAllTriggerKeys(trigger);
   333→              return (
   334→                <div
   335→                  key={trigger.id}
   336→                  className={cn(
   337→                    "flex items-center gap-3 p-3 border rounded-lg",
   338→                    !trigger.active && "opacity-50"
   339→                  )}
   340→                >
   341→                  {/* Sprite Preview */}
   342→                  <div className="w-12 h-12 rounded border overflow-hidden bg-muted/50 flex items-center justify-center flex-shrink-0">
   343→                    {getSpritePreviewUrl(trigger) ? (
   344→                      <SpritePreview
   345→                        src={getSpritePreviewUrl(trigger)!}
   346→                        alt={trigger.title}
   347→                        className="w-full h-full"
   348→                        objectFit="contain"
   349→                      />
   350→                    ) : (
   351→                      <ImageIcon className="w-5 h-5 text-muted-foreground" />
   352→                    )}
   353→                  </div>
   354→
   355→                  {/* Info */}
   356→                  <div className="flex-1 min-w-0">
   357→                    <div className="flex items-center gap-2">
   358→                      <span className="font-medium text-sm truncate">{trigger.title}</span>
   359→                      <Badge variant="outline" className="text-xs">
   360→                        {getSpriteLabel(trigger)}
   361→                      </Badge>
   362→                    </div>
   363→                    <div className="flex flex-wrap gap-1 mt-1">
   364→                      {allKeys.slice(0, 4).map((k) => (
   365→                        <Badge key={k} variant="secondary" className="text-xs font-mono">
   366→                          {trigger.requirePipes ? `|${k}|` : k}
   367→                        </Badge>
   368→                      ))}
   369→                      {allKeys.length > 4 && (
   370→                        <Badge variant="secondary" className="text-xs">
   371→                          +{allKeys.length - 4}
   372→                        </Badge>
   373→                      )}
   374→                    </div>
   375→                    {/* Return info */}
   376→                    {trigger.returnToIdleMs > 0 && (
   377→                      <div className="flex items-center gap-1 mt-1 text-[10px] text-muted-foreground">
   378→                        <Timer className="w-3 h-3" />
   379→                        <span>
   380→                          Retorna en {formatTime(trigger.returnToIdleMs)} → 
   381→                          {trigger.returnToMode === 'custom_sprite' ? ' Sprite personalizado' : ' Colección Idle'}
   382→                        </span>
   383→                      </div>
   384→                    )}
   385→                  </div>
   386→
   387→                  {/* Actions */}
   388→                  <div className="flex items-center gap-1">
   389→                    <Switch
   390→                      checked={trigger.active}
   391→                      onCheckedChange={(active) => {
   392→                        const newTriggers = triggers.map(t =>
   393→                          t.id === trigger.id ? { ...t, active } : t
   394→                        );
   395→                        updateTriggers(newTriggers);
   396→                      }}
   397→                    />
   398→                    <Button
   399→                      variant="ghost"
   400→                      size="icon"
   401→                      className="h-8 w-8"
   402→                      onClick={() => handleEditTrigger(trigger)}
   403→                    >
   404→                      <Edit className="w-4 h-4" />
   405→                    </Button>
   406→                    <Button
   407→                      variant="ghost"
   408→                      size="icon"
   409→                      className="h-8 w-8 text-destructive"
   410→                      onClick={() => handleDeleteTrigger(trigger.id)}
   411→                    >
   412→                      <Trash2 className="w-4 h-4" />
   413→                    </Button>
   414→                  </div>
   415→                </div>
   416→              );
   417→            })}
   418→          </div>
   419→        </ScrollArea>
   420→      )}
   421→
   422→      {/* Info Box */}
   423→      <div className="text-xs text-muted-foreground bg-muted/50 p-3 rounded-lg space-y-1">
   424→        <p>💡 <strong>Keys:</strong> Key principal + alternativas separadas por coma.</p>
   425→        <p>💡 <strong>Pipes:</strong> Usa |key| para detectar solo cuando está entre pipes.</p>
   426→        <p>💡 <strong>Prioridad:</strong> Los triggers con mayor prioridad tienen preferencia.</p>
   427→      </div>
   428→
   429→      {/* Edit Dialog */}
   430→      <AlertDialog open={showEditor} onOpenChange={setShowEditor}>
   431→        <AlertDialogContent className="max-w-lg max-h-[90vh] overflow-y-auto">
   432→          <AlertDialogHeader>
   433→            <AlertDialogTitle>
   434→              {editingTrigger?.id && triggers.find(t => t.id === editingTrigger.id)
   435→                ? 'Editar Trigger'
   436→                : 'Nuevo Trigger'}
   437→            </AlertDialogTitle>
   438→          </AlertDialogHeader>
   439→
   440→          {editingTrigger && (
   441→            <div className="space-y-4 py-4">
   442→              {/* Title */}
   443→              <div>
   444→                <Label className="text-xs">Título</Label>
   445→                <Input
   446→                  value={editingTrigger.title}
   447→                  onChange={(e) => setEditingTrigger({ ...editingTrigger, title: e.target.value })}
   448→                  placeholder="Nombre del trigger"
   449→                  className="mt-1 h-8"
   450→                />
   451→              </div>
   452→
   453→              {/* Key Detection Section - HUD Style */}
   454→              <div className="space-y-3 p-4 rounded-xl border border-border/60 bg-gradient-to-br from-muted/30 to-muted/10">
   455→                <div className="flex items-center gap-2 text-sm font-medium text-foreground">
   456→                  <div className="p-1.5 rounded-md bg-cyan-500/10">
   457→                    <Key className="w-4 h-4 text-cyan-500" />
   458→                  </div>
   459→                  Detección de Keys
   460→                  <Badge variant="outline" className="text-[10px] px-1.5 py-0 h-4 bg-cyan-500/10 text-cyan-600 dark:text-cyan-400 border-cyan-500/20">
   461→                    Cómo se activa
   462→                  </Badge>
   463→                </div>
   464→                
   465→                {/* Primary Key */}
   466→                <div className="space-y-2">

   469→                    value={mainKey}
   470→                    onChange={(e) => setMainKey(e.target.value)}
   471→                    placeholder="spritewin, sprite:win, happy..."

   486→                    onChange={(e) => setAlternativeKeys(e.target.value)}

   494→                          {k}

   498→                  )}

   500→

   506→                      Si está desactivado, "Win" y "win" serán equivalentes

   510→                    checked={editingTrigger.caseSensitive}

   512→                  />
   513→                </div>
   514→                
   515→                {/* Require Pipes */}
   516→                <div className="flex items-center justify-between p-3 rounded-lg bg-background/50 border border-border/40">
   517→                  <div className="space-y-0.5">
   518→                    <Label className="text-xs font-medium">Requiere |pipes|</Label>
   519→                    <p className="text-[10px] text-muted-foreground">
   520→                      Si está activo, solo detecta keys entre | pipes |
   521→                    </p>
   522→                  </div>
   523→                  <Switch
   524→                    checked={editingTrigger.requirePipes}
   525→                    onCheckedChange={(requirePipes) => setEditingTrigger({ ...editingTrigger, requirePipes })}
   526→                  />
   527→                </div>
   528→              </div>
   529→
   530→              {/* Cooldown */}
   531→              <div className="flex items-center justify-between">
   532→                <Label className="text-xs">Cooldown</Label>
   533→                <div className="flex items-center gap-1">

   535→                    type="number"
   536→                    value={editingTrigger.cooldownMs}

   538→                      ...editingTrigger,

   759→                      </span>

   761→                  </>
   762→                )}

   765→              {/* Priority */}

   767→                <Label className="text-xs">Prioridad (mayor = más importante)</Label>

   769→                  type="number"

   771→                  onChange={(e) => setEditingTrigger({

   778→              </div>
   779→            </div>

   781→
   782→          <AlertDialogFooter>
