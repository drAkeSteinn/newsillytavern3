    81→const _spriteRotationIndex = new Map<string, Map<SpriteState, number>>();
    82→
    83→function getRotationIndex(characterId: string, state: SpriteState): number {
    84→  let charMap = _spriteRotationIndex.get(characterId);
    85→  if (!charMap) {
    86→    charMap = new Map<SpriteState, number>();
    87→    _spriteRotationIndex.set(characterId, charMap);
    88→  }
    89→  return charMap.get(state) ?? 0;
    90→}
    91→
    92→function advanceRotationIndex(characterId: string, state: SpriteState): number {
    93→  let charMap = _spriteRotationIndex.get(characterId);
    94→  if (!charMap) {
    95→    charMap = new Map<SpriteState, number>();
    96→    _spriteRotationIndex.set(characterId, charMap);
    97→  }
    98→  const current = charMap.get(state) ?? 0;
    99→  const next = current + 1;
   100→  charMap.set(state, next);
   101→  return next;
   102→}

   104→// Helper: resolve a sprite from a pack given behavior config
   105→function resolveSpriteFromPack(
   106→  pack: SpritePackV2,
   107→  behavior: 'principal' | 'random' | 'list',
   108→  principalSpriteId: string | undefined,
   109→  characterId: string,
   110→  state: SpriteState
   111→): { url: string | null; label: string | null } {
   112→  if (pack.sprites.length === 0) return { url: null, label: null };
   113→
   114→  switch (behavior) {
   115→    case 'principal':
   116→      // Use principal sprite if specified, otherwise first sprite
   117→      if (principalSpriteId) {
   118→        const principal = pack.sprites.find(s => s.id === principalSpriteId);
   119→        if (principal) return { url: principal.url, label: principal.label };
   120→      }
   121→      return { url: pack.sprites[0].url, label: pack.sprites[0].label };
   122→
   123→    case 'random':
   124→      // Random selection
   125→      const randomIndex = Math.floor(Math.random() * pack.sprites.length);
   126→      const randomSprite = pack.sprites[randomIndex];
   127→      return { url: randomSprite.url, label: randomSprite.label };
   128→
   129→    case 'list': {
   130→      // Rotate through sprites using the rotation tracker
   131→      const index = getRotationIndex(characterId, state);
   132→      const safeIndex = index % pack.sprites.length;
   133→      const listSprite = pack.sprites[safeIndex];
   134→      return { url: listSprite.url, label: listSprite.label };
   135→    }
   136→
   137→    default:
   138→      return { url: pack.sprites[0].url, label: pack.sprites[0].label };
   139→  }
   140→}

   142→// Get sprite from State Collection V2 (uses sprite packs)
   143→// Now supports conditional mode at the pack level
   144→function getSpriteFromStateCollectionV2(
   145→  state: SpriteState,
   146→  stateCollectionsV2: StateCollectionV2[],
   147→  spritePacksV2: SpritePackV2[],
   148→  characterId: string,
   149→  sessionStats?: SessionStats | null
   150→): { url: string | null; label: string | null } {
   151→  const stateCollection = stateCollectionsV2.find(c => c.state === state);
   152→  if (!stateCollection) return { url: null, label: null };
   153→
   154→  const pack = spritePacksV2.find(p => p.id === stateCollection.packId);
   155→  if (!pack || pack.sprites.length === 0) return { url: null, label: null };
   156→
   157→  // ============================================
   158→  // Evaluate pack-level conditional sprites (if pack has conditionalMode)
   159→  // ============================================
   160→  if (pack.conditionalMode) {
   161→    const winningSprite = evaluatePackConditionalSprites(
   162→      pack.sprites,
   163→      sessionStats,
   164→      characterId
   165→    );
   166→
   167→    if (winningSprite) {
   168→      return { url: winningSprite.url, label: winningSprite.label };
   169→    }
   170→
   171→    // No conditional sprite matched - try defaultSpriteId
   172→    if (pack.defaultSpriteId) {
   173→      const defaultSprite = pack.sprites.find(s => s.id === pack.defaultSpriteId);
   174→      if (defaultSprite) {
   175→        return { url: defaultSprite.url, label: defaultSprite.label };
   176→      }
   177→    }
   178→
   179→    // Also check for sprites marked as isDefault
   180→    const markedDefault = pack.sprites.find(s => s.isDefault);
   181→    if (markedDefault) {
   182→      return { url: markedDefault.url, label: markedDefault.label };
   183→    }
   184→
   185→    // Fall through to behavior-based resolution
   186→  }
   187→
   188→  // ============================================
   189→  // Default: use the state collection's behavior
   190→  // ============================================
   191→  return resolveSpriteFromPack(
   192→    pack,
   193→    stateCollection.behavior,
   194→    stateCollection.principalSpriteId,
   195→    characterId,
   196→    state
   197→  );
   198→}

   200→// Get the appropriate sprite URL based on state and config
   201→// Priority: V2 State Collections > Avatar (legacy system removed)
   202→function getSpriteUrl(
   203→  state: SpriteState,
   204→  avatarUrl?: string,
   205→  character?: CharacterCard,
   206→  characterId?: string,
   207→  sessionStats?: SessionStats | null
   208→): { url: string; label: string | null } {
   209→  const charId = characterId || '';
   210→  // Try V2 state collections (new system)
   211→  // Check for non-empty arrays (empty arrays should not be considered as "configured")
   212→  const hasV2Collections = character?.stateCollectionsV2 && character.stateCollectionsV2.length > 0;
   213→  const hasV2Packs = character?.spritePacksV2 && character.spritePacksV2.length > 0;
   214→  
   215→  if (hasV2Collections && hasV2Packs) {
   216→    const v2Result = getSpriteFromStateCollectionV2(
   217→      state,
   218→      character.stateCollectionsV2,
   219→      character.spritePacksV2,
   220→      charId,
   221→      sessionStats
   222→    );
   223→    if (v2Result.url) {
   224→      return v2Result;
   225→    }
   226→  }
   227→
   228→  // Additional fallback for talk and thinking states - try V2 idle collection
   229→  if (state === 'talk' || state === 'thinking') {
   230→    if (hasV2Collections && hasV2Packs) {
   231→      const v2IdleResult = getSpriteFromStateCollectionV2(
   232→        'idle',
   233→        character!.stateCollectionsV2,
   234→        character!.spritePacksV2,
   235→        charId,
   236→        sessionStats
   237→      );
   238→      if (v2IdleResult.url) {
   239→        return v2IdleResult;
   240→      }
   241→    }
   242→  }
   243→
   244→  // Final fallback to avatar
   245→  if (avatarUrl) {
   246→    return { url: avatarUrl, label: 'avatar' };
   247→  }
   248→
   249→  return { url: '', label: null };
   250→}

   252→// Format milliseconds to readable time
   253→function formatTime(ms: number): string {
   254→  if (ms <= 0) return '0s';
   255→  const seconds = Math.floor(ms / 1000);
   256→  if (seconds < 60) return `${seconds}s`;
   257→  const minutes = Math.floor(seconds / 60);
   258→  const remainingSeconds = seconds % 60;
   259→  return `${minutes}m ${remainingSeconds}s`;
   260→}

   262→export function CharacterSprite({ 
   263→  characterId, 
   264→  characterName, 
   265→  avatarUrl,
   266→  character,
   267→  isStreaming = false,
   268→  hasContent = false,
   269→  isTTSPlaying = false,
   270→  onSettingsChange 
   271→}: CharacterSpriteProps) {
   272→  const [settings, setSettings] = useState<SpriteSettings>(() => {
   273→    if (typeof window !== 'undefined') {
   274→      const saved = localStorage.getItem(`sprite-${characterId}`);
   275→      if (saved) {
   276→        try {
   277→          return { ...DEFAULT_SPRITE_SETTINGS, ...JSON.parse(saved) };
   278→        } catch {
   279→          return DEFAULT_SPRITE_SETTINGS;
   280→        }
   281→      }
   282→    }
   283→    return DEFAULT_SPRITE_SETTINGS;
   284→  });
   285→
   286→  const [isDragging, setIsDragging] = useState(false);
   287→  const [isResizing, setIsResizing] = useState(false);
   288→  const [showControls, setShowControls] = useState(false);
   289→  const [countdown, setCountdown] = useState<number>(0);
   290→  
   291→  const spriteRef = useRef<HTMLDivElement>(null);
   292→  const dragStartRef = useRef({ x: 0, y: 0, settingsX: 0, settingsY: 0 });
   293→  const resizeStartRef = useRef({ x: 0, y: 0, width: 0, height: 0 });
   294→
   295→  // ============================================
   296→  // UNIFIED SYSTEM: Get sprite state from store
   297→  // ============================================
   298→  // 
   299→  // ⚠️ SPRITE PRIORITY SYSTEM - DO NOT MODIFY ⚠️
   300→  // 
   301→  // The sprite system follows a STRICT priority order:
   302→  // 
   303→  // 1. TRIGGER SPRITE (HIGHEST PRIORITY)
   304→  //    - Activated by sprite triggers from AI response
   305→  //    - Once active, CANNOT be overridden by talk/thinking/idle
   306→  //    - Only changes when:
   307→  //      a) Timer expires (returnToIdleMs > 0)
   308→  //      b) Another trigger replaces it
   309→  //      c) User manually clears it
   310→  //
   311→  // 2. STATE COLLECTION V2 SPRITES
   312→  //    - Talk sprite (streaming with content)
   313→  //    - Thinking sprite (generating without content)
   314→  //    - Idle sprite (default state)
   315→  //    - Only shown when NO trigger sprite is active
   316→  //
   317→  // 3. AVATAR (lowest priority)
   318→  //    - Character avatar as final fallback
   319→  //
   320→  // See: /docs/SPRITE_PRIORITY_SYSTEM.md
   321→  //
   322→  // Use proper Zustand selectors to ensure reactivity
   323→  const characterSpriteStates = useTavernStore((state) => state.characterSpriteStates);
   324→  const isLocked = useTavernStore((state) => state.spriteLock.active);
   325→  const lockState = useTavernStore((state) => state.spriteLock);
   326→  const getReturnToIdleCountdownForCharacter = useTavernStore((state) => state.getReturnToIdleCountdownForCharacter);
   327→  const cancelReturnToIdleForCharacter = useTavernStore((state) => state.cancelReturnToIdleForCharacter);
   328→  const executeReturnToIdleForCharacter = useTavernStore((state) => state.executeReturnToIdleForCharacter);
   329→  const clearSpriteLock = useTavernStore((state) => state.clearSpriteLock);
   330→  const setSpriteStateForCharacter = useTavernStore((state) => state.setSpriteStateForCharacter);
   331→
   332→  // Get sessionStats from the active session for conditional sprite evaluation
   333→  const sessionStats = useTavernStore((state) => {
   334→    const activeSessionId = state.activeSessionId;
   335→    if (!activeSessionId) return null;
   336→    const session = state.sessions.find(s => s.id === activeSessionId);
   337→    return session?.sessionStats ?? null;
   338→  });
   339→  
   340→  // Get per-character sprite state from the unified store
   341→  // This is now reactive because characterSpriteStates is a proper selector
   342→  const charSpriteState = characterSpriteStates[characterId] || createDefaultCharacterState();
   343→  
   344→  // Debug logging for trigger sprite state
   345→  useEffect(() => {
   346→    if (charSpriteState.triggerSpriteUrl) {
   347→      console.log('[CharacterSprite] Trigger sprite active:', {
   348→        characterId,
   349→        triggerSpriteUrl: charSpriteState.triggerSpriteUrl,
   350→        triggerSpriteLabel: charSpriteState.triggerSpriteLabel,
   351→        isStreaming,
   352→        hasContent,
   353→      });
   354→    }
   355→  }, [charSpriteState.triggerSpriteUrl, charSpriteState.triggerSpriteLabel, characterId, isStreaming, hasContent]);
   356→  
   357→  // Check if sprite is locked (global state)
   358→  const spriteLockActive = isLocked && lockState.until === 0 ? true : (isLocked && Date.now() < lockState.until);
   359→  
   360→  // Get return to idle state for this character
   361→  const isReturnToIdleScheduled = charSpriteState.returnToIdle.active;
   362→  
   363→  // Determine the effective sprite state:
   364→  // Priority: Trigger sprite > TTS Talk > Streaming Thinking > Store State
   365→  // 
   366→  // When TTS is active:
   367→  //   - During streaming → 'thinking' (not talk, since the character is "thinking" while generating)
   368→  //   - During TTS playback → 'talk' (the character is "speaking")
   369→  //   - After TTS ends → 'idle'
   370→  // 
   371→  // If a trigger sprite is active, it always takes absolute priority.
   372→  const effectiveSpriteState = charSpriteState.triggerSpriteUrl
   373→    ? 'idle' // Trigger active, use trigger sprite (handled below)
   374→    : isTTSPlaying
   375→      ? 'talk'                    // TTS is playing → character is speaking
   376→      : isStreaming
   377→        ? 'thinking'               // Generating → thinking (not talk, even with content)
   378→        : charSpriteState.spriteState;
   379→
   380→  // ============================================
   381→  // LIST MODE ROTATION: Advance index on state transition
   382→  // When effectiveSpriteState transitions to 'talk' or 'thinking',
   383→  // advance the rotation index so the NEXT sprite in the list is shown.
   384→  // ============================================
   385→  const prevEffectiveStateRef = useRef<SpriteState>(effectiveSpriteState);
   386→  useEffect(() => {
   387→    const prev = prevEffectiveStateRef.current;
   388→    prevEffectiveStateRef.current = effectiveSpriteState;
   389→
   390→    // Only advance when transitioning INTO talk or thinking (not from one to the other)
   391→    if (prev !== 'talk' && prev !== 'thinking' && 
   392→        (effectiveSpriteState === 'talk' || effectiveSpriteState === 'thinking')) {
   393→      // Check if the target state collection uses 'list' behavior
   394→      const stateCollections = character?.stateCollectionsV2;
   395→      const collection = stateCollections?.find(c => c.state === effectiveSpriteState);
   396→      if (collection?.behavior === 'list') {
   397→        advanceRotationIndex(characterId, effectiveSpriteState);
   398→      }
   399→    }
   400→  }, [effectiveSpriteState, character?.stateCollectionsV2, characterId]);
   401→
   402→  // ============================================
   403→  // TTS ↔ SPRITE STATE SYNC
   404→  // When TTS playback ends, return the sprite to idle state
   405→  // (only if no trigger is active and not currently streaming)
   406→  // ============================================
   407→  const prevTTSToStateRef = useRef(isTTSPlaying);
   408→  useEffect(() => {
   409→    const wasPlaying = prevTTSToStateRef.current;
   410→    prevTTSToStateRef.current = isTTSPlaying;
   411→
   412→    // TTS just stopped playing → return to idle
   413→    if (wasPlaying && !isTTSPlaying && !isStreaming) {
   414→      const currentState = characterSpriteStates[characterId];
   415→      // Only update if no trigger is active (trigger has absolute priority)
   416→      if (currentState && !currentState.triggerSpriteUrl) {
   417→        setSpriteStateForCharacter(characterId, 'idle');
   418→      }
   419→    }
   420→  }, [isTTSPlaying, isStreaming, characterId, characterSpriteStates, setSpriteStateForCharacter]);
   421→
   422→  // Update countdown for return to idle
   423→  // PERF FIX: Changed from 100ms to 1000ms — the countdown is in seconds, so 1Hz
   424→  // is sufficient and avoids 10 React state updates per second per visible character.
   425→  useEffect(() => {
   426→    if (!isReturnToIdleScheduled) {
   427→      setCountdown(0);
   428→      return;
   429→    }
   430→
   431→    const updateCountdown = () => {
   432→      const remaining = getReturnToIdleCountdownForCharacter(characterId);
   433→      setCountdown(remaining);
   434→    };
   435→
   436→    updateCountdown();
   437→    const interval = setInterval(updateCountdown, 1000);
   438→
   439→    return () => clearInterval(interval);
   440→  }, [isReturnToIdleScheduled, getReturnToIdleCountdownForCharacter, characterId]);
   441→
   442→  // Determine the current sprite URL
   443→  // PRIORITY:
   444→  // 1. Trigger sprite from store (highest priority)
   445→  // 2. State collection V2 (if available)
   446→  // 3. Avatar (lowest priority)
   447→  const spriteResult = charSpriteState.triggerSpriteUrl 
   448→    ? { url: charSpriteState.triggerSpriteUrl, label: charSpriteState.triggerSpriteLabel }
   449→    : getSpriteUrl(
   450→        effectiveSpriteState,
   451→        avatarUrl,
   452→        character,
   453→        characterId,
   454→        sessionStats
   455→      );
   456→  const currentSpriteUrl = spriteResult.url;
   457→  const currentSpriteLabel = spriteResult.label;
   458→
   459→  // FASE 7: Determine the active transition config
   460→  // If a trigger sprite is active, use the trigger's transition config
   461→  // Otherwise, use the character's default transition for state-based changes
   462→  const activeTransition: SpriteTransitionConfig | undefined = charSpriteState.triggerSpriteUrl
   463→    ? (charSpriteState.triggerTransition ?? undefined)
   464→    : (charSpriteState.defaultTransition ?? character?.defaultTransition ?? undefined);
   465→
   466→  // Debug: Log when sprite data changes
   467→  useEffect(() => {
   468→    console.log('[CharacterSprite] Sprite data:', {
   469→      characterId,
   470→      hasCharacter: !!character,
   471→      spritePacksV2Count: character?.spritePacksV2?.length ?? 'undefined',
   472→      stateCollectionsV2Count: character?.stateCollectionsV2?.length ?? 'undefined',
   473→      currentSpriteUrl: currentSpriteUrl?.substring(0, 50),
   474→      avatarUrl: avatarUrl?.substring(0, 50),
   475→    });
   476→  }, [characterId, character?.spritePacksV2, character?.stateCollectionsV2, currentSpriteUrl, avatarUrl]);
   477→
   478→  // Save settings to localStorage
   479→  useEffect(() => {
   480→    localStorage.setItem(`sprite-${characterId}`, JSON.stringify(settings));
   481→    onSettingsChange?.(settings);
   482→  }, [settings, characterId, onSettingsChange]);
   483→
   484→  // ============================================
   485→  // ESCENARIO V3.2 — POSE LIGADA AL FONDO (por personaje): whenever the ACTIVE
   486→  // background changes, move the sprite to the pose saved for THAT fondo for
   487→  // THIS character. Poses live in the fondo's COLLECTION ENTRY
   488→  // (entry.spritePoses[characterId] → collection.json) with a settings
   489→  // fallback. This covers EVERY source of background change: manual
   490→  // (galleries), scenario switcher (escenario/lugar/punto), manage_escenario
   491→  // tool calling and activation keys — they all converge on activeBackground.
   492→  // If the fondo has no registry pose, fall back to the LEGACY node pose
   493→  // (Task 17). If nothing is saved, the sprite simply stays where it is.
   494→  // ============================================
   495→  const activeBackground = useTavernStore((s) => s.activeBackground);
   496→  const savedBackgroundPose = useTavernStore((s) =>
   497→    lookupSpritePoseForBackground(s, characterId, s.activeBackground)
   498→  );
   499→
   500→  useEffect(() => {
   501→    if (!activeBackground) return;
   502→    const pose = savedBackgroundPose
   503→      ?? resolveLegacyLocationPoseForBackground(characterId, activeBackground);
   504→    if (pose) {
   505→      setSettings(prev => ({ ...prev, x: pose.x, y: pose.y, width: pose.width, height: pose.height, opacity: pose.opacity }));
   506→      console.log('[CharacterSprite] Applied pose for background:', activeBackground);
   507→    }
   508→  }, [activeBackground, savedBackgroundPose, characterId]);
   509→
   510→  // ============================================
   511→  // ESCENARIO V3.2: save the current sprite position + scale LIGADOS AL FONDO
   512→  // ACTUAL (entry.spritePoses[characterId] in the collection JSON, settings
   513→  // fallback for orphan fondos). The write logic is SHARED with the
   514→  // ScenarioSwitcher button (saveSpritePoseToCurrentBackground).
   515→  // ============================================
   516→  const handleSavePoseToBackground = async () => {
   517→    const result = await saveSpritePoseToCurrentBackground(characterId, {
   518→      x: settings.x,
   519→      y: settings.y,
   520→      width: settings.width,
   521→      height: settings.height,
   522→      opacity: settings.opacity,
   523→    });
   524→    if (!result.ok) {
   525→      const messages: Record<string, string> = {
   526→        'no-background': 'No hay ningún fondo activo para guardar la pose',
   527→        'invalid-pose': 'No se pudo leer la posición actual del sprite',
   528→        'save-failed': 'No se pudo escribir la pose en el JSON de la colección — inténtalo de nuevo',
   529→      };
   530→      toast.error(messages[result.error ?? 'invalid-pose'] ?? 'No se pudo guardar la pose');
   531→      return;
   532→    }
   533→    toast.success(`🧷 Pose guardada para el fondo "${result.where}" (${result.storage === 'collection' ? 'collection.json' : 'ajustes locales'}) — el sprite volverá aquí al mostrar ese fondo`);
   534→  };
   535→
   536→  const handleClearPoseForBackground = async () => {
   537→    if (!activeBackground) return;
   538→    if (await clearSpritePoseForBackground(characterId, activeBackground)) {
