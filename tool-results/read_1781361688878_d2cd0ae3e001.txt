    92→/**
    93→ * Execute detected tool calls, send SSE events, and return results.
    94→ * Returns { newContent: combined display messages, shouldContinue: true if tools were executed }
    95→ */
    96→async function executeToolCallsAndContinue(
    97→  toolCalls: NativeToolCall[],
    98→  availableTools: Array<{ id: string; name: string; label: string; icon: string }>,
    99→  currentRound: number,
   100→  maxRounds: number,
   101→  character: CharacterCard,
   102→  sessionId: string,
   103→  userName: string,
   104→  controller: { enqueue: (chunk: string) => void },
   105→  sessionQuests?: SessionQuestInstance[],
   106→  questTemplates?: QuestTemplate[],
   107→  statsConfig?: CharacterStatsConfig,
   108→  sessionStats?: SessionStats,
   109→  allCharacters?: CharacterCard[],
   110→  characterMemory?: CharacterMemory,
   111→  lorebooks?: Lorebook[],
   112→): Promise<{ newContent: string; shouldContinue: boolean; toolResults: Array<{ success: boolean; displayMessage: string }>; questActivations: QuestActivation[]; toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> }> {
   113→  if (toolCalls.length === 0 || currentRound >= maxRounds) {
   114→    return { newContent: '', shouldContinue: false, toolResults: [], questActivations: [], toolsUsed: [] };
   115→  }
   116→
   117→  const toolResults: Array<{ success: boolean; displayMessage: string }> = [];
   118→  const questActivations: QuestActivation[] = [];
   119→  const toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> = [];
   120→  let allDisplayMessages = '';
   121→
   122→  for (let i = 0; i < toolCalls.length; i++) {
   123→    const tc = toolCalls[i];
   124→    const toolDef = availableTools.find(t => t.name === tc.name);
   125→    const callId = `${Date.now()}_${i}`;
   126→
   127→    // Send tool_call_start event
   128→    controller.enqueue(createSSEJSON({
   129→      type: 'tool_call_start',
   130→      toolName: tc.name,
   131→      toolLabel: toolDef?.label || tc.name,
   132→      toolIcon: toolDef?.icon || 'Wrench',
   133→      params: tc.arguments,
   134→      callId,
   135→    }));
   136→
   137→    console.log(`[Tools] Executing tool call: ${tc.name}`, tc.arguments);
   138→
   139→    // Execute the tool
   140→    const toolResult = await executeTool(
   141→      tc.name,
   142→      tc.arguments,
   143→      {
   144→        characterId: character.id,
   145→        characterName: character.name,
   146→        sessionId,
   147→        userName,
   148→        sessionQuests,
   149→        questTemplates,
   150→        statsConfig: character.statsConfig,
   151→        sessionStats,
   152→        allCharacters,
   153→        characterMemory,
   154→        lorebooks,
   155→      },
   156→    );
   157→
   158→    // Send tool_call_result event
   159→    controller.enqueue(createSSEJSON({
   160→      type: 'tool_call_result',
   161→      toolName: tc.name,
   162→      success: toolResult.success,
   163→      displayMessage: toolResult.displayMessage,
   164→      duration: toolResult.duration || 0,
   165→      callId,
   166→    }));
   167→
   168→    // Check for quest activation and send SSE event
   169→    if (toolResult.questActivation) {
   170→      const activation = toolResult.questActivation;
   171→      console.log(`[Tools] Quest activation from ${tc.name}:`, activation);
   172→      
   173→      controller.enqueue(createSSEJSON({
   174→        type: 'quest_activation',
   175→        toolName: tc.name,
   176→        activationType: activation.type,
   177→        key: activation.key,
   178→        metadata: activation.metadata,
   179→      }));
   180→      
   181→      questActivations.push(activation);
   182→    }
   183→
   184→    // Check for action/skill activation and send SSE event
   185→    if (toolResult.actionActivation) {
   186→      const action = toolResult.actionActivation;
   187→      console.log(`[Tools] Action activation from ${tc.name}:`, action.skillName);
   188→      
   189→      controller.enqueue(createSSEJSON({
   190→        type: 'action_activation',
   191→        toolName: tc.name,
   192→        skillId: action.skillId,
   193→        skillName: action.skillName,
   194→        skillDescription: action.skillDescription,
   195→        skillCompletedDescription: action.skillCompletedDescription,
   196→        activationCosts: action.activationCosts,
   197→        activationRewards: action.activationRewards,
   198→        characterId: action.characterId,
   199→      }));
   200→    }
   201→
   202→    // Check for stat modification from modify_stat tool and send SSE event
   203→    if (toolResult.statActivation) {
   204→      const stat = toolResult.statActivation;
   205→      console.log(`[Tools] Stat activation from ${tc.name}:`, stat.attributeKey, stat.oldValue, '→', stat.newValue);
   206→      
   207→      controller.enqueue(createSSEJSON({
   208→        type: 'stat_activation',
   209→        toolName: tc.name,
   210→        characterId: stat.characterId,
   211→        attributeKey: stat.attributeKey,
   212→        attributeName: stat.attributeName,
   213→        attributeType: stat.attributeType,
   214→        oldValue: stat.oldValue,
   215→        newValue: stat.newValue,
   216→        reason: stat.reason,
   217→      }));
   218→    }
   219→
   220→    // Check for solicitud activation/completion and send SSE event
   221→    if (toolResult.solicitudActivation) {
   222→      const sol = toolResult.solicitudActivation;
   223→      console.log(`[Tools] Solicitud activation from ${tc.name}:`, sol.type, sol.solicitudKey);
   224→      
   225→      controller.enqueue(createSSEJSON({
   226→        type: 'solicitud_activation',
   227→        toolName: tc.name,
   228→        activationType: sol.type,
   229→        solicitudKey: sol.solicitudKey,
   230→        targetCharacterId: sol.targetCharacterId,
   231→        targetCharacterName: sol.targetCharacterName,
   232→        fromCharacterId: sol.fromCharacterId,
   233→        fromCharacterName: sol.fromCharacterName,
   234→        description: sol.description,
   235→        completionDescription: sol.completionDescription,
   236→        peticionKey: sol.peticionKey,
   237→      }));
   238→    }
   239→
   240→    // Check for memory activation (sync to client-side Character Memory)
   241→    if (toolResult.memoryActivation) {
   242→      const mem = toolResult.memoryActivation;
   243→      console.log(`[Tools] Memory activation from ${tc.name}:`, mem.type);
   244→      
   245→      controller.enqueue(createSSEJSON({
   246→        type: 'memory_activation',
   247→        toolName: tc.name,
   248→        activationType: mem.type,
   249→        characterId: mem.characterId,
   250→        eventData: mem.eventData,
   251→        relationshipData: mem.relationshipData,
   252→        noteContent: mem.noteContent,
   253→        deleteEventId: mem.deleteEventId,
   254→        deleteEmbeddingId: mem.deleteEmbeddingId,
   255→      }));
   256→    }
   257→
   258→    console.log(`[Tools] Tool ${tc.name}: success=${toolResult.success} duration=${toolResult.duration}ms`, toolResult.displayMessage);
   259→
   260→    toolResults.push({ success: toolResult.success, displayMessage: toolResult.displayMessage });
   261→    toolsUsed.push({
   262→      name: tc.name,
   263→      label: toolDef?.label || tc.name,
   264→      icon: toolDef?.icon || 'Wrench',
   265→      success: toolResult.success,
   266→    });
   267→    if (toolResult.displayMessage) {
   268→      allDisplayMessages += (allDisplayMessages ? '\n' : '') + toolResult.displayMessage;
   269→    }
   270→  }
   271→
   272→  // Return the display messages as content, tool results, and signal that a follow-up call is needed
   273→  return {
   274→    newContent: allDisplayMessages,
   275→    shouldContinue: true, // Always continue to get the LLM's natural response
   276→    toolResults,
   277→    questActivations,
   278→    toolsUsed,
   279→  };
   280→}

   282→/**
   283→ * POST /api/chat/proactive
   284→ *
   285→ * Generates a proactive message for a character using SSE streaming.
   286→ * Full feature parity with the stream route including tools, embeddings,
   287→ * key resolution, memory reinforcement, and extraction.
   288→ */
   289→export async function POST(request: NextRequest) {
   290→  try {
   291→    // Capture auth headers from Z.ai gateway for token resolution
   292→    const incomingXToken = request.headers.get('X-Token');
   293→    const fcSecurityToken = request.headers.get('x-fc-security-token');
   294→
   295→    const body = await request.json();
   296→
   297→    // Extract required fields from body
   298→    const {
   299→      character,
   300→      messages = [],
   301→      llmConfig,
   302→      userName = 'User',
   303→      persona,
   304→      proactiveConfig,
   305→      reason = 'timer_idle',
   306→      lastActivityAt,
   307→      // FASE 3: Proactividad Inteligente
   308→      usedNudgeIndices: clientUsedNudgeIndices = [],
   309→      recentTopics: clientRecentTopics = [],
   310→      isGroupChat = false,
   311→      groupChatSpeakerId,
   312→    } = body;
   313→
   314→    // Validate required fields
   315→    if (!character?.name || !llmConfig?.provider) {
   316→      return createErrorResponse('Missing required fields: character, llmConfig', 400);
   317→    }
   318→
   319→    if (!proactiveConfig?.enabled) {
   320→      return createErrorResponse('Proactive messages are disabled for this character', 400);
   321→    }
   322→
   323→    if (!llmConfig) {
   324→      return createErrorResponse('No LLM configuration provided', 400);
   325→    }
   326→
   327→    // Validate API key for providers that require one
   328→    const providersRequiringApiKey = ['openai', 'vllm', 'lm-studio', 'custom', 'anthropic', 'grok'];
   329→    if (providersRequiringApiKey.includes(llmConfig.provider) && !llmConfig.apiKey) {
   330→      return createErrorResponse(
   331→        `API Key is required for ${llmConfig.provider} provider. Please configure it in Settings.`,
   332→        400
   333→      );
   334→    }
   335→
   336→    // Validate endpoint for providers that require one
   337→    const providersRequiringEndpoint = ['openai', 'vllm', 'lm-studio', 'custom', 'anthropic', 'ollama', 'text-generation-webui', 'koboldcpp'];
   338→    if (providersRequiringEndpoint.includes(llmConfig.provider) && !llmConfig.endpoint) {
   339→      return createErrorResponse(
   340→        `Endpoint URL is required for ${llmConfig.provider} provider. Please configure it in Settings.`,
   341→        400
   342→      );
   343→    }
   344→
   345→    // If using Z.ai provider, collect gateway-forwarded auth headers
   346→    let zaiRuntimeToken: string | undefined;
   347→    if (llmConfig.provider === 'z-ai') {
   348→      const gatewayToken = incomingXToken || fcSecurityToken || undefined;
   349→      if (gatewayToken) {
   350→        zaiRuntimeToken = gatewayToken;
   351→        console.log(`[Proactive Route] Z.ai runtime token available (${gatewayToken.length} chars, source: ${incomingXToken ? 'X-Token' : 'fc-security-token'})`);
   352→      } else {
   353→        console.log(`[Proactive Route] Z.ai: no gateway token available, using config file only`);
   354→      }
   355→    }
   356→
   357→    console.log(`[Proactive] Generating proactive message for "${character.name}" (reason: ${reason})`);
   358→
   359→    // Extract all additional fields from body (matching stream route)
   360→    const lorebooks: Lorebook[] = body.lorebooks || [];
   361→    const sessionStats: SessionStats | undefined = body.sessionStats;
   362→    const allCharacters: CharacterCard[] = body.allCharacters || [character];
   363→
   364→    // Extract HUD context from body
   365→    const hudContext: HUDContextConfig | undefined = body.hudContext;
   366→
   367→    // Extract Quest data
   368→    const questTemplates: QuestTemplate[] = body.questTemplates || [];
   369→    const sessionQuests: SessionQuestInstance[] = body.sessionQuests || [];
   370→    const questSettings: QuestSettings = {
   371→      ...DEFAULT_QUEST_SETTINGS,
   372→      ...(body.questSettings || {})
   373→    };
   374→
   375→    // Extract Sound data for {{sonidos}} key resolution
   376→    const soundTriggers: SoundTrigger[] = body.soundTriggers || [];
   377→    const soundSettings = body.settings?.sound;
   378→
   379→    // Extract summary for memory/context compression
   380→    const summary: SessionSummary | undefined = body.summary;
   381→
   382→    // Extract embeddings chat settings
   383→    const embeddingsChat: Partial<EmbeddingsChatSettings> = body.embeddingsChat || {};
   384→    const characterMemory: CharacterMemory | undefined = body.characterMemory;
   385→    const sessionId: string | undefined = body.sessionId;
   386→    const characterId: string | undefined = body.characterId;
   387→    const inventoryData: InventoryPromptData | undefined = body.inventoryData;
   388→
   389→    // Extract tools settings
   390→    const toolsSettings: ToolsSettings = {
   391→      enabled: body.toolsSettings?.enabled ?? true,
   392→      maxToolCallsPerTurn: body.toolsSettings?.maxToolCallsPerTurn ?? 2,
   393→      characterConfigs: body.toolsSettings?.characterConfigs || [],
   394→      usePromptBasedFallback: body.toolsSettings?.usePromptBasedFallback ?? false,
   395→      disabledTools: body.toolsSettings?.disabledTools || [],
   396→    };
   397→
   398→    // Create default character if none provided
   399→    const effectiveCharacter: CharacterCard = character || DEFAULT_CHARACTER;
   400→
   401→    // Get effective user name from persona or use provided userName
   402→    const effectiveUserName = getEffectiveUserName(persona, userName);
   403→
   404→    // Build context configuration from request or use defaults
   405→    const contextConfig: Partial<ContextConfig> = body.contextConfig || {};
   406→
   407→    // Apply sliding window to messages
   408→    const contextWindow = selectContextMessages(messages, llmConfig, contextConfig);
   409→
   410→    // Log context stats (for debugging)
   411→    const stats = getContextStats(messages);
   412→
   413→    // Process lorebooks and get matched entries
   414→    const { plan: lorebookPlan, lorebookAttributeKeys, lorebookEntryKeyMap, lorebookDebugEntries } = buildLorebookSectionForPrompt(
   415→      messages,
   416→      lorebooks,
   417→      {
   418→        scanDepth: (contextConfig as any).scanDepth,
   419→        userName: effectiveUserName,
   420→        charName: effectiveCharacter?.name,
   421→      },
   422→      { sessionStats, characterId: effectiveCharacter?.id, characters: allCharacters }
   423→    );
   424→
   425→    // ========================================
   426→    // DEBUG: Build lorebook debug data for SSE event
   427→    // ========================================
   428→    const lorebookDebugData = {
   429→      lorebookAttributeKeys,
   430→      debugEntries: lorebookDebugEntries,
   431→      availableStats: sessionStats
   432→        ? Object.fromEntries(
   433→            Object.entries(sessionStats.characterStats || {}).map(([charId, cs]) => [
   434→              charId,
   435→              cs?.attributeValues ? { ...cs.attributeValues } : '(no values)',
   436→            ])
   437→          )
   438→        : '(no sessionStats)',
   439→    };
   440→
   441→    // ========================================
   442→    // Embeddings Context Retrieval
   443→    // ========================================
   444→    // For proactive messages, use recent chat context as the search query
   445→    // since there's no new user message
   446→    let enrichedSearchQuery = '';
   447→    const searchContextDepth = embeddingsChat.searchContextDepth || 0;
   448→    if (searchContextDepth > 0) {
   449→      const recentHistory = messages
   450→        .filter(m => !m.isDeleted)
   451→        .slice(-(searchContextDepth * 2))
   452→        .map(m => m.content?.trim())
   453→        .filter(Boolean);
   454→      
   455→      if (recentHistory.length > 0) {
   456→        enrichedSearchQuery = recentHistory.join(' ');
   457→      }
   458→    } else {
   459→      // Use last few messages as search context
   460→      const recentHistory = messages
   461→        .filter(m => !m.isDeleted)
   462→        .slice(-4)
   463→        .map(m => m.content?.trim())
   464→        .filter(Boolean);
   465→      
   466→      if (recentHistory.length > 0) {
   467→        enrichedSearchQuery = recentHistory.join(' ');
   468→      }
   469→    }
   470→
   471→    // Safety: smart truncation based on embedding model's context window.
   472→    try {
   473→      const { MODEL_CONTEXT_LENGTHS, DEFAULT_CONTEXT_LENGTH, CHARS_PER_TOKEN } = await import('@/lib/embeddings/types');
   474→      const { loadConfig } = await import('@/lib/embeddings/config-persistence');
   475→      const embConfig = loadConfig();
   476→      const modelKey = embConfig.model || 'bge-m3:567m';
   477→      const modelCtx = MODEL_CONTEXT_LENGTHS[modelKey]
   478→        || MODEL_CONTEXT_LENGTHS[modelKey.split(':')[0]]
   479→        || DEFAULT_CONTEXT_LENGTH;
   480→      const maxQueryChars = Math.floor(modelCtx * 0.75 * CHARS_PER_TOKEN);
   481→
   482→      if (enrichedSearchQuery.length > maxQueryChars) {
   483→        // For proactive: keep the most recent part (end of the string)
   484→        enrichedSearchQuery = enrichedSearchQuery.slice(-maxQueryChars);
   485→        console.warn(
   486→          `[Proactive Route] Search query trimmed to ${enrichedSearchQuery.length} chars ` +
   487→          `(model: ${modelKey}, context: ${modelCtx} tokens)`
   488→        );
   489→      }
   490→    } catch { /* fallback: Ollama client handles truncation */ }
   491→
   492→    // Retrieve relevant embeddings based on enriched query and settings
   493→    // Pass Character Memory events for deduplication (avoid duplicate memory in prompt)
   494→    const existingMemoryEvents = characterMemory?.events?.map(e => ({
   495→      content: e.content,
   496→      importance: e.importance,
   497→    }));
   498→
   499→    // Extract last assistant message for bidirectional search
   500→    const lastAssistantMsg = messages
   501→      .filter(m => !m.isDeleted && m.role === 'assistant')
   502→      .pop()?.content;
   503→
   504→    const embeddingsResult = await retrieveEmbeddingsContext(
   505→      enrichedSearchQuery,
   506→      characterId || effectiveCharacter.id,
   507→      sessionId,
   508→      embeddingsChat,
   509→      undefined, // groupId
   510→      existingMemoryEvents, // for deduplication
   511→      lastAssistantMsg, // bidirectional search with last assistant message
   512→    );
   513→    
   514→    if (embeddingsResult.found) {
   515→      console.log(`[Proactive Route] Retrieved ${embeddingsResult.count} embeddings from namespaces: ${embeddingsResult.searchedNamespaces.join(', ')}`);
   516→    }
   517→
   518→    // ========================================
   519→    // Build system prompt with unified key resolution
   520→    // ========================================
   521→    const { prompt: systemPrompt, sections: systemSections, lorebookChatInjections, exampleMessages } = buildSystemPrompt(
   522→      effectiveCharacter,
   523→      effectiveUserName,
   524→      persona,
   525→      lorebookPlan,
   526→      sessionStats,
   527→      allCharacters, // Pass all characters for peticiones/solicitudes resolution
   528→      soundTriggers,
   529→      soundSettings,
   530→      questTemplates,
   531→      sessionQuests,
   532→      questSettings,
   533→      lorebookAttributeKeys,
   534→      inventoryData,    // Pass inventory data for Inventory V2 section
   535→      lorebookEntryKeyMap // Pass lorebook entry key map for {{entryKey}} resolution
   536→    );
   537→
   538→    // Build key resolution context for HUD context and quest sections
   539→    let streamPersonaResolvedStats: import('@/types').ResolvedStats | null = null;
   540→    if (persona?.statsConfig?.enabled && sessionStats) {
   541→      streamPersonaResolvedStats = resolveStats({
   542→        characterId: '__user__',
   543→        statsConfig: persona.statsConfig,
   544→        sessionStats,
   545→      });
   546→    }
   547→    const resolvedStats = resolveStats({
   548→      characterId: effectiveCharacter.id,
   549→      statsConfig: effectiveCharacter.statsConfig,
   550→      sessionStats: sessionStats,
   551→      allCharacters,
   552→      userName: effectiveUserName,
   553→      characterName: effectiveCharacter.name,
   554→      questTemplates,
   555→      personaDescription: persona?.description,
   556→      personaResolvedStats: streamPersonaResolvedStats,
   557→    });
   558→
   559→    // Build outlet sections map from lorebook plan for {{outlet::name}} macro resolution
   560→    const outletSections: Record<string, string> = {};
   561→    if (lorebookPlan?.outletSections.length) {
   562→      for (const outletSection of lorebookPlan.outletSections) {
   563→        const match = outletSection.label.match(/^World Info \((.+)\)$/);
   564→        const outletName = match ? match[1] : outletSection.label;
   565→        outletSections[outletName] = outletSection.content;
   566→      }
   567→    }
   568→
   569→    const keyContext = buildKeyResolutionContext(
   570→      effectiveCharacter,
   571→      effectiveUserName,
   572→      persona,
   573→      resolvedStats,
   574→      sessionStats,
   575→      soundTriggers,
   576→      soundSettings,
   577→      streamPersonaResolvedStats,
   578→      questTemplates,
   579→      sessionQuests,
   580→      questSettings,

  1551→                      if (toolResult.shouldContinue) {

  1554→                          ...baseChatMessages,

  1563→
  1564→                  // No tool calls detected - stream buffered content (clean artifacts)

  1573→                } else if (shouldUseTools && isToolRound) {

  1580→
  1581→              case 'text-generation-webui':
  1582→              case 'koboldcpp': {

  1600→                  generator = streamTextGenerationWebUIWithTools(chatMessages, proactiveLLMConfig, availableTools, accumulator);
  1601→

  1618→                      effectiveCharacter, sessionId || '', effectiveUserName, controller,

  1627→                        success: true, displayMessage: toolResult.newContent || `[${tc.name} ejecutada]`
  1628→                      }));

  1630→                        ...chatMessages,
  1631→                        ...buildToolMessagesForOpenAI(accumulator.toolCalls, toolResultPairs),
  1632→                      ];
  1633→                      accumulatedContent = '';
  1634→                      toolRound++;
  1635→                      toolContextMessages = tgwToolContextMessages as any;
  1636→                      continue;
  1637→                    }
  1638→                  }
  1639→
  1640→                  // No tool calls detected - stream buffered content (clean artifacts)

  1642→                  const cleanedTGWContent = cleanModelArtifacts(roundContent);
  1643→                  for (const chunk of splitIntoChunks(cleanedTGWContent)) {
  1644→                    controller.enqueue(createSSEJSON({ type: 'token', content: chunk }));
  1645→                  }
  1646→                  accumulatedContent = cleanModelArtifacts(accumulatedContent);
  1647→                  toolRound = maxToolRounds + 1;
  1648→                  continue;
  1649→                } else {

  1652→                  }
  1653→                  const prompt = buildCompletionPrompt({
  1654→                    systemPrompt: baseSystemPrompt || finalSystemPrompt,

  1662→                  generator = streamTextGenerationWebUI(prompt, proactiveLLMConfig);

  1665→              }
  1666→
  1667→              default: {
