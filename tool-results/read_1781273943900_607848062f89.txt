    96→// Inline editable label for sprite cards — click to edit, Enter/blur to save, Escape to cancel
    97→function SpriteLabelInput({
    98→  label,
    99→  onLabelChange,
   100→}: {
   101→  label: string;
   102→  onLabelChange: (newLabel: string) => void;
   103→}) {
   104→  const [isEditing, setIsEditing] = useState(false);
   105→  const [draft, setDraft] = useState(label);
   106→  const inputRef = useRef<HTMLInputElement>(null);
   107→
   108→  const startEditing = () => {
   109→    setDraft(label);
   110→    setIsEditing(true);
   111→    // Focus the input on next tick
   112→    setTimeout(() => inputRef.current?.focus(), 0);
   113→  };
   114→
   115→  const commit = () => {
   116→    const trimmed = draft.trim();
   117→    if (trimmed && trimmed !== label) {
   118→      onLabelChange(trimmed);
   119→    }
   120→    setIsEditing(false);
   121→  };
   122→
   123→  const cancel = () => {
   124→    setDraft(label);
   125→    setIsEditing(false);
   126→  };
   127→
   128→  if (isEditing) {
   129→    return (
   130→      <input
   131→        ref={inputRef}
   132→        type="text"
   133→        value={draft}
   134→        onChange={(e) => setDraft(e.target.value)}
   135→        onBlur={commit}
   136→        onKeyDown={(e) => {
   137→          if (e.key === 'Enter') commit();
   138→          if (e.key === 'Escape') cancel();
   139→        }}
   140→        className="w-full text-sm text-center font-medium bg-background border border-primary/50 rounded px-1.5 py-0.5 outline-none focus:ring-1 focus:ring-primary"
   141→      />
   142→    );
   143→  }
   144→
   145→  return (
   146→    <p
   147→      className="text-sm truncate text-center font-medium cursor-pointer hover:text-primary hover:underline underline-offset-2 transition-colors"
   148→      onClick={startEditing}
   149→      title="Click para editar la etiqueta"
   150→    >
   151→      {label}
   152→    </p>
   153→  );
   154→}

   156→// Sprite Condition Editor - Full-size version for dialog
   157→function SpriteConditionEditorFull({
   158→  conditions,
   159→  availableAttributes,
   160→  onConditionsChange,
   161→}: {
   162→  conditions: StatRequirement[];
   163→  availableAttributes: AttributeDefinition[];
   164→  onConditionsChange: (conditions: StatRequirement[]) => void;
   165→}) {
   166→  return (
   167→    <div className="space-y-3">
   168→      {conditions.length === 0 && (
   169→        <div className="text-center py-4 text-muted-foreground border border-dashed rounded-lg">
   170→          <GitBranch className="w-5 h-5 mx-auto mb-1.5 opacity-40" />
   171→          <p className="text-sm">Sin condiciones</p>
   172→          <p className="text-xs mt-0.5">Este sprite se mostrará si tiene la mayor prioridad</p>
   173→        </div>
   174→      )}
   175→      {conditions.map((cond, idx) => {
   176→        const attrDef = availableAttributes.find(a => a.key === cond.attributeKey);
   177→        return (
   178→          <div key={idx} className="flex items-start gap-3 p-3 rounded-lg bg-muted/40 border">
   179→            <div className="flex-1 space-y-2">
   180→              {/* Row 1: Attribute + Operator */}
   181→              <div className="flex items-center gap-2">
   182→                <div className="flex-1">
   183→                  <Label className="text-xs text-muted-foreground mb-1 block">Atributo</Label>
   184→                  <Select
   185→                    value={cond.attributeKey}
   186→                    onValueChange={(v) => {
   187→                      const newConditions = [...conditions];
   188→                      newConditions[idx] = { ...newConditions[idx], attributeKey: v };
   189→                      onConditionsChange(newConditions);
   190→                    }}
   191→                  >
   192→                    <SelectTrigger className="h-9 text-sm">
   193→                      <SelectValue placeholder="Seleccionar atributo" />
   194→                    </SelectTrigger>
   195→                    <SelectContent>
   196→                      {availableAttributes.map(attr => (
   197→                        <SelectItem key={attr.key} value={attr.key}>
   198→                          {attr.name || attr.key}
   199→                        </SelectItem>
   200→                      ))}
   201→                    </SelectContent>
   202→                  </Select>
   203→                </div>
   204→                <div className="w-40">
   205→                  <Label className="text-xs text-muted-foreground mb-1 block">Operador</Label>
   206→                  <Select
   207→                    value={cond.operator}
   208→                    onValueChange={(v) => {
   209→                      const newConditions = [...conditions];
   210→                      newConditions[idx] = { ...newConditions[idx], operator: v as RequirementOperator };
   211→                      onConditionsChange(newConditions);
   212→                    }}
   213→                  >
   214→                    <SelectTrigger className="h-9 text-sm">
   215→                      <SelectValue />
   216→                    </SelectTrigger>
   217→                    <SelectContent>
   218→                      {OPERATOR_OPTIONS.map(op => (
   219→                        <SelectItem key={op.value} value={op.value}>
   220→                          {op.symbol} {op.label}
   221→                        </SelectItem>
   222→                      ))}
   223→                    </SelectContent>
   224→                  </Select>
   225→                </div>
   226→              </div>
   227→              {/* Row 2: Value(s) */}
   228→              <div className="flex items-center gap-2">
   229→                <div className="flex-1">
   230→                  <Label className="text-xs text-muted-foreground mb-1 block">
   231→                    Valor {attrDef?.type === 'string' ? '(texto)' : '(número)'}
   232→                  </Label>
   233→                  <Input
   234→                    type={attrDef?.type === 'string' ? 'text' : 'number'}
   235→                    className="h-9 text-sm"
   236→                    value={typeof cond.value === 'number' ? cond.value : cond.value || ''}
   237→                    onChange={(e) => {
   238→                      const newConditions = [...conditions];
   239→                      const val = attrDef?.type === 'string' ? e.target.value : Number(e.target.value);
   240→                      newConditions[idx] = { ...newConditions[idx], value: val as number };
   241→                      onConditionsChange(newConditions);
   242→                    }}
   243→                  />
   244→                </div>
   245→                {cond.operator === 'between' && (
   246→                  <div className="flex-1">
   247→                    <Label className="text-xs text-muted-foreground mb-1 block">Valor máximo</Label>
   248→                    <Input
   249→                      type="number"
   250→                      className="h-9 text-sm"
   251→                      value={cond.valueMax || 0}
   252→                      onChange={(e) => {
   253→                        const newConditions = [...conditions];
   254→                        newConditions[idx] = { ...newConditions[idx], valueMax: Number(e.target.value) };
   255→                        onConditionsChange(newConditions);
   256→                      }}
   257→                    />
   258→                  </div>
   259→                )}
   260→              </div>
   261→              {/* Condition summary */}
   262→              <div className="text-xs text-muted-foreground">
   263→                {attrDef?.name || cond.attributeKey}{' '}
   264→                {OPERATOR_OPTIONS.find(o => o.value === cond.operator)?.symbol}{' '}
   265→                {cond.value}
   266→                {cond.operator === 'between' ? ` ~ ${cond.valueMax}` : ''}
   267→              </div>
   268→            </div>
   269→            {/* Delete button */}
   270→            <Button
   271→              variant="ghost"
   272→              size="sm"
   273→              className="h-9 w-9 p-0 shrink-0 text-red-500 hover:text-red-600 hover:bg-red-500/10 mt-5"
   274→              onClick={() => onConditionsChange(conditions.filter((_, i) => i !== idx))}
   275→              title="Eliminar condición"
   276→            >
   277→              <Trash2 className="w-4 h-4" />
   278→            </Button>
   279→          </div>
   280→        );
   281→      })}
   282→      <Button
   283→        variant="outline"
   284→        size="sm"
   285→        className="w-full h-9 text-sm text-purple-600 hover:text-purple-700 border-purple-500/30 hover:border-purple-500/50 hover:bg-purple-500/5"
   286→        onClick={() => {
   287→          const newCondition: StatRequirement = {
   288→            attributeKey: availableAttributes[0]?.key || '',
   289→            operator: '>=',
   290→            value: 0,
   291→          };
   292→          onConditionsChange([...conditions, newCondition]);
   293→        }}
   294→      >
   295→        <Plus className="w-4 h-4 mr-1.5" /> Agregar Condición
   296→      </Button>
   297→    </div>
   298→  );
   299→}

   301→interface SpritePackEditorV2Props {
   302→  character: CharacterCard;
   303→  customSprites: SpriteIndexEntry[];
   304→  collections: SpriteCollection[];
   305→  onChange: (updates: Partial<CharacterCard>) => void;

   307→
   308→export function SpritePackEditorV2({ 
   309→  character, 
   310→  customSprites,
   311→  collections,
   312→  onChange 
   313→}: SpritePackEditorV2Props) {
   314→  // State for dialogs
   315→  const [showCreatePackDialog, setShowCreatePackDialog] = useState(false);
   316→  const [showAddSpriteDialog, setShowAddSpriteDialog] = useState<string | null>(null); // packId
   317→  const [editingPackId, setEditingPackId] = useState<string | null>(null);
   318→  const [editingName, setEditingName] = useState('');
   319→  const [searchQuery, setSearchQuery] = useState('');
   320→  const [collectionFilter, setCollectionFilter] = useState<string>('__all__');
   321→  
   322→  // Sprite edit dialog state
   323→  const [editingSprite, setEditingSprite] = useState<{ packId: string; spriteId: string } | null>(null);
   324→  
   325→  // New pack form state
   326→  const [newPackName, setNewPackName] = useState('');
   327→  const [newPackDescription, setNewPackDescription] = useState('');
   328→
   329→  // Get current packs from character
   330→  const spritePacksV2: SpritePackV2[] = useMemo(() => {
   331→    return character.spritePacksV2 || [];
   332→  }, [character.spritePacksV2]);
   333→
   334→  // Available attributes from character statsConfig
   335→  const availableAttributes: AttributeDefinition[] = useMemo(() => {
   336→    return character.statsConfig?.attributes || [];
   337→  }, [character.statsConfig?.attributes]);
   338→
   339→  // Filter sprites by collection filter AND search
   340→  const filteredSprites = useMemo(() => {
   341→    let result = customSprites;
   342→    
   343→    // Filter by collection
   344→    if (collectionFilter && collectionFilter !== '__all__') {
   345→      result = result.filter(s => s.pack === collectionFilter);
   346→    }
   347→    
   348→    // Filter by search query
   349→    if (searchQuery.trim()) {
   350→      const query = searchQuery.toLowerCase();
   351→      result = result.filter(s => 
   352→        s.label.toLowerCase().includes(query) ||
   353→        s.filename.toLowerCase().includes(query) ||
   354→        (s.pack && s.pack.toLowerCase().includes(query))
   355→      );
   356→    }
   357→    
   358→    return result;
   359→  }, [customSprites, collectionFilter, searchQuery]);
   360→
   361→  // Get unique collection names from sprites for grouping
   362→  const collectionNames = useMemo(() => {
   363→    const names = new Set(customSprites.map(s => s.pack).filter(Boolean));
   364→    return Array.from(names).sort();
   365→  }, [customSprites]);
   366→
   367→  // Check if sprite is in any pack
   368→  const getSpritePackInfo = (spriteUrl: string) => {
   369→    for (const pack of spritePacksV2) {
   370→      const found = pack.sprites.find(s => s.url === spriteUrl);
   371→      if (found) return { packId: pack.id, packName: pack.name, sprite: found };
   372→    }
   373→    return null;
   374→  };
   375→
   376→  // Get current editing sprite data
   377→  const currentEditSprite = useMemo(() => {
   378→    if (!editingSprite) return null;
   379→    const pack = spritePacksV2.find(p => p.id === editingSprite.packId);
   380→    if (!pack) return null;
   381→    const sprite = pack.sprites.find(s => s.id === editingSprite.spriteId);
   382→    if (!sprite) return null;
   383→    return { pack, sprite };
   384→  }, [editingSprite, spritePacksV2]);
   385→
   386→  // Create new pack
   387→  const handleCreatePack = () => {
   388→    if (!newPackName.trim()) return;
   389→    
   390→    const now = new Date().toISOString();
   391→    const newPack: SpritePackV2 = {
   392→      id: uuidv4(),
   393→      name: newPackName.trim(),
   394→      description: newPackDescription.trim() || undefined,
   395→      sprites: [],
   396→      createdAt: now,
   397→      updatedAt: now,
   398→    };
   399→    
   400→    onChange({
   401→      spritePacksV2: [...spritePacksV2, newPack],
   402→    });
   403→    
   404→    // Reset form
   405→    setNewPackName('');
   406→    setNewPackDescription('');
   407→    setShowCreatePackDialog(false);
   408→    
   409→    logger.info('Created sprite pack', { packId: newPack.id, name: newPack.name });
   410→  };
   411→
   412→  // Delete pack
   413→  const handleDeletePack = (packId: string) => {
   414→    const pack = spritePacksV2.find(p => p.id === packId);
   415→    if (!pack) return;
   416→    
   417→    if (!confirm(`¿Eliminar el pack "${pack.name}"? Los sprites no se eliminarán, solo se quitarán del pack.\n\nSi este pack está siendo usado en Colecciones de Estado, se eliminará la referencia.`)) return;
   418→    
   419→    // Remove pack from spritePacksV2
   420→    const updatedPacks = spritePacksV2.filter(p => p.id !== packId);
   421→    
   422→    // Also clean up stateCollectionsV2 references to this pack
   423→    const stateCollectionsV2 = character.stateCollectionsV2 || [];
   424→    const updatedStateCollections = stateCollectionsV2
   425→      .filter(c => c.packId !== packId); // Remove collections that reference this pack
   426→    
   427→    // Also clean up trigger collections that reference this pack
   428→    const triggerCollections = character.triggerCollections || [];
   429→    const updatedTriggerCollections = triggerCollections.filter(c => c.packId !== packId);
   430→    
   431→    onChange({
   432→      spritePacksV2: updatedPacks,
   433→      stateCollectionsV2: updatedStateCollections,
   434→      triggerCollections: updatedTriggerCollections.length < triggerCollections.length ? updatedTriggerCollections : undefined,
   435→    });
   436→    
   437→    logger.info('Deleted sprite pack and cleaned up references', { 
   438→      packId,
   439→      removedStateCollections: stateCollectionsV2.length - updatedStateCollections.length,
   440→      removedTriggerCollections: triggerCollections.length - updatedTriggerCollections.length,
   441→    });
   442→  };
   443→
   444→  // Rename pack
   445→  const handleRenamePack = (packId: string) => {
   446→    if (!editingName.trim()) {
   447→      setEditingPackId(null);
   448→      return;
   449→    }
   450→    
   451→    onChange({
   452→      spritePacksV2: spritePacksV2.map(p => 
   453→        p.id === packId 
   454→          ? { ...p, name: editingName.trim(), updatedAt: new Date().toISOString() }
   455→          : p
   456→      ),
   457→    });
   458→    
   459→    setEditingPackId(null);
   460→    logger.info('Renamed sprite pack', { packId, newName: editingName });
   461→  };
   462→
   463→  // Add sprite to pack
   464→  const handleAddSpriteToPack = (packId: string, sprite: SpriteIndexEntry) => {
   465→    const pack = spritePacksV2.find(p => p.id === packId);
   466→    if (!pack) return;
   467→    
   468→    // Check if already in pack
   469→    if (pack.sprites.some(s => s.url === sprite.url)) {
   470→      return;
   471→    }
   472→    
   473→    const newEntry: SpritePackEntryV2 = {
   474→      id: uuidv4(),
   475→      label: sprite.label,
   476→      url: sprite.url,
   477→      thumbnail: sprite.thumb,
   478→      tags: sprite.expressions,
   479→      isAnimated: isAnimatedImage(sprite.url) || isVideoUrl(sprite.url),
   480→    };
   481→    
   482→    onChange({
   483→      spritePacksV2: spritePacksV2.map(p => 
   484→        p.id === packId 
   485→          ? { ...p, sprites: [...p.sprites, newEntry], updatedAt: new Date().toISOString() }
   486→          : p
   487→      ),
   488→    });
   489→    
   490→    logger.info('Added sprite to pack', { packId, spriteLabel: sprite.label });
   491→  };
   492→
   493→  // Remove sprite from pack
   494→  const handleRemoveSpriteFromPack = (packId: string, spriteId: string) => {
   495→    onChange({
   496→      spritePacksV2: spritePacksV2.map(p => 
   497→        p.id === packId 
   498→          ? { 
   499→              ...p, 
   500→              sprites: p.sprites.filter(s => s.id !== spriteId),
   501→              updatedAt: new Date().toISOString() 
   502→            }
   503→          : p
   504→      ),
   505→    });
   506→    
   507→    logger.info('Removed sprite from pack', { packId, spriteId });
   508→  };
   509→
   510→  // Update sprite within a pack
   511→  const handleUpdateSpriteInPack = (packId: string, spriteId: string, updates: Partial<SpritePackEntryV2>) => {
   512→    onChange({
   513→      spritePacksV2: spritePacksV2.map(p =>
   514→        p.id === packId
   515→          ? {
   516→              ...p,
   517→              sprites: p.sprites.map(s =>
   518→                s.id === spriteId ? { ...s, ...updates } : s
   519→              ),
   520→              updatedAt: new Date().toISOString()
   521→            }
   522→          : p
   523→      ),
   524→    });
   525→  };
   526→
   527→  // Set default sprite for a pack
   528→  const handleSetDefaultSprite = (packId: string, spriteId: string | undefined) => {
   529→    onChange({
   530→      spritePacksV2: spritePacksV2.map(p =>
   531→        p.id === packId
   532→          ? { ...p, defaultSpriteId: spriteId, updatedAt: new Date().toISOString() }
   533→          : p
   534→      ),
   535→    });
   536→  };
   537→
   538→  // Get sprites not in current pack for add dialog
   539→  const getSpritesNotInPack = (packId: string) => {
   540→    const pack = spritePacksV2.find(p => p.id === packId);
   541→    if (!pack) return filteredSprites;
   542→    
   543→    const packUrls = new Set(pack.sprites.map(s => s.url));
   544→    return filteredSprites.filter(s => !packUrls.has(s.url));
   545→  };
   546→
   547→  // Get condition summary text for compact display
   548→  const getConditionSummary = (sprite: SpritePackEntryV2): string => {
   549→    if (!sprite.conditions || sprite.conditions.length === 0) return 'Sin condiciones';
   550→    return sprite.conditions.map(c => {
   551→      const attr = availableAttributes.find(a => a.key === c.attributeKey);
   552→      const op = OPERATOR_OPTIONS.find(o => o.value === c.operator);
   553→      return `${attr?.name || c.attributeKey} ${op?.symbol || c.operator} ${c.value}${c.operator === 'between' ? `~${c.valueMax}` : ''}`;
   554→    }).join(', ');
   555→  };
   556→
   557→  return (
   558→    <TooltipProvider>
   559→      <div className="space-y-4">
   560→        {/* Header */}
   561→        <div className="flex items-center justify-between">
   562→          <div>
   563→            <h4 className="text-sm font-medium flex items-center gap-2">
   564→              <Layers className="w-4 h-4 text-purple-500" />
   565→              Sprite Packs
   566→            </h4>
   567→            <p className="text-xs text-muted-foreground">
   568→              Crea packs de sprites para usar en colecciones de estado y triggers.
   569→            </p>
   570→          </div>
   571→          <Button
   572→            size="sm"
   573→            className="h-8"
   574→            onClick={() => setShowCreatePackDialog(true)}
   575→          >
   576→            <Plus className="w-4 h-4 mr-1" />
   577→            Nuevo Pack
   578→          </Button>
   579→        </div>
