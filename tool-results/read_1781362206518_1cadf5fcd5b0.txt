   167→/**
   168→ * Clamp a numeric value to the attribute's min/max bounds
   169→ * Returns the clamped value, or the original value if not numeric or no bounds defined
   170→ */
   171→function clampAttributeValue(
   172→  value: number | string,
   173→  attributeDef: AttributeDefinition | undefined
   174→): number | string {
   175→  // Only clamp numeric values
   176→  if (typeof value !== 'number') {
   177→    return value;
   178→  }
   179→
   180→  // If no attribute definition, return as-is
   181→  if (!attributeDef) {
   182→    return value;
   183→  }
   184→
   185→  let clampedValue = value;
   186→
   187→  // Apply min bound - handle both number and string min values
   188→  if (attributeDef.min !== undefined) {
   189→    const minVal = typeof attributeDef.min === 'number'
   190→      ? attributeDef.min
   191→      : parseFloat(String(attributeDef.min));
   192→
   193→    if (!isNaN(minVal)) {
   194→      clampedValue = Math.max(clampedValue, minVal);
   195→    }
   196→  }
   197→
   198→  // Apply max bound - handle both number and string max values
   199→  if (attributeDef.max !== undefined) {
   200→    const maxVal = typeof attributeDef.max === 'number'
   201→      ? attributeDef.max
   202→      : parseFloat(String(attributeDef.max));
   203→
   204→    if (!isNaN(maxVal)) {
   205→      clampedValue = Math.min(clampedValue, maxVal);
   206→    }
   207→  }
   208→
   209→  return clampedValue;
   210→}

   212→/**
   213→ * Create default character stats from config
   214→ */
   215→function createDefaultCharacterStats(
   216→  statsConfig?: CharacterStatsConfig
   217→): CharacterSessionStats {
   218→  const attributeValues: Record<string, number | string> = {};
   219→  const lastUpdated: Record<string, number> = {};
   220→  const now = Date.now();
   221→  
   222→  if (statsConfig?.attributes) {
   223→    for (const attr of statsConfig.attributes) {
   224→      attributeValues[attr.key] = attr.defaultValue;
   225→      lastUpdated[attr.key] = now;
   226→    }
   227→  }
   228→  
   229→  return {
   230→    attributeValues,
   231→    lastUpdated,
   232→    changeLog: [],
   233→  };
   234→}

   236→/**
   237→ * Add entry to change log
   238→ */
   239→function addChangeLogEntry(
   240→  stats: CharacterSessionStats,
   241→  attribute: AttributeDefinition | undefined,
   242→  attributeKey: string,
   243→  oldValue: number | string | undefined,
   244→  newValue: number | string,
   245→  reason: StatChangeLogEntry['reason']
   246→): void {
   247→  if (!stats.changeLog) {
   248→    stats.changeLog = [];
   249→  }
   250→  
   251→  stats.changeLog.push({
   252→    attributeId: attribute?.id || attributeKey,
   253→    attributeKey,
   254→    attributeName: attribute?.name || attributeKey,
   255→    oldValue: oldValue ?? '',
   256→    newValue,
   257→    reason,
   258→    timestamp: Date.now(),
   259→  });
   260→  
   261→  // Keep only last 100 entries
   262→  if (stats.changeLog.length > 100) {
   263→    stats.changeLog = stats.changeLog.slice(-100);
   264→  }
   265→}

   279→  initializeSessionStats: (sessionId, characters) => {

   291→    
   292→    // Initialize stats for each character
   293→    for (const char of characters) {
   294→      const stats = createDefaultCharacterStats(char.statsConfig);
   295→      // FASE 5: Initialize emotional state if emotional config is present
   296→      if (char.emotionalConfig?.enabled && char.emotionalConfig.initialState) {
   297→        stats.emotionalState = char.emotionalConfig.initialState;
   298→        stats.emotionalStateLastEval = now;
   299→        stats.emotionalStateTurnCount = 0;
   300→        // Also set as attribute value for {{emocion}} key resolution
   301→        stats.attributeValues['emocion'] = char.emotionalConfig.initialState;
   302→      }
   303→      characterStats[char.id] = stats;
   304→    }
   305→    
   306→    const newSessionStats: SessionStats = {
   307→      characterStats,
   308→      solicitudes: {
   309→        characterSolicitudes: {},
   310→        lastModified: now,
   311→      },
   312→      initialized: true,
   313→      lastModified: now,
   314→      // Initialize timer state
   315→      lastTimerUpdate: now,
   316→      keywordCycleIndex: {},
   317→    };
   318→    
   319→    // Update session with new stats
   320→    set((state: any) => ({
   321→      sessions: state.sessions.map((s: any) =>
   322→        s.id === sessionId 
   323→          ? { ...s, sessionStats: newSessionStats, updatedAt: new Date().toISOString() }
   324→          : s
   325→      ),
   326→    }));
   327→  },
   328→
   329→  updateCharacterStat: (sessionId, characterId, attributeKey, value, reason = 'manual') => {
   330→    // Default result
   331→    const defaultResult: UpdateCharacterStatResult = {
   332→      oldValue: undefined,
   333→      newValue: typeof value === 'number' ? value : value,
   334→      clamped: false,
   335→      thresholdsReached: [],
   336→    };
   337→
   338→    // Read current state
   339→    const state = get();
   340→    const sessions = state.sessions as Array<{ 
   341→      id: string; 
   342→      sessionStats?: SessionStats;
   343→      characterId?: string;
   344→      groupId?: string;
   345→    }>;
   346→    const sessionIndex = sessions.findIndex(s => s.id === sessionId);
   347→    
   348→    if (sessionIndex === -1) return defaultResult;
   349→    
   350→    const session = sessions[sessionIndex];
   351→    let sessionStats = session.sessionStats;
   352→    
   353→    // Auto-initialize sessionStats if missing
   354→    if (!sessionStats) {
   355→      sessionStats = {
   356→        characterStats: {},
   357→        solicitudes: {
   358→          characterSolicitudes: {},
   359→          lastModified: Date.now(),
   360→        },
   361→        ultimo_objetivo_completado: undefined,
   362→        ultima_solicitud_completada: undefined,
   363→        ultima_solicitud_realizada: undefined,
   364→        ultima_accion_realizada: undefined,
   365→        initialized: true,
   366→        lastModified: Date.now(),
   367→      };
   368→    }
   369→    
   370→    // Auto-initialize character stats if missing
   371→    if (!sessionStats.characterStats[characterId]) {
   372→      let statsConfig: CharacterStatsConfig | undefined;
   373→      if (characterId === '__user__') {
   374→        // Look up active persona's statsConfig for __user__
   375→        const activePersonaId = (state as any).activePersonaId;
   376→        const personas: any[] = (state as any).personas || [];
   377→        const activePersona = personas.find((p: any) => p.id === activePersonaId);
   378→        statsConfig = activePersona?.statsConfig;
   379→      } else {
   380→        const character = state.characters.find((c: any) => c.id === characterId);
   381→        statsConfig = character?.statsConfig;
   382→      }
   383→      let newStats = createDefaultCharacterStats(statsConfig);
   384→      // For persona (__user__), also initialize equipment slot values
   385→      if (characterId === '__user__') {
   386→        const equipmentSlots = (state as any).inventorySettings?.equipmentSlots;
   387→        if (equipmentSlots && equipmentSlots.length > 0) {
   388→          for (const slot of equipmentSlots) {
   389→            if (!(slot.key in newStats.attributeValues)) {
   390→              newStats.attributeValues[slot.key] = '';
   391→              newStats.lastUpdated[slot.key] = Date.now();
   392→            }
   393→          }
   394→        }
   395→      }
   396→      sessionStats = {
   397→        ...sessionStats,
   398→        characterStats: {
   399→          ...sessionStats.characterStats,
   400→          [characterId]: newStats,
   401→        },
   402→      };
   403→    }
   404→    
   405→    const stats = sessionStats.characterStats[characterId];
   406→    if (!stats) return defaultResult;
   407→
   408→    const oldValue = stats.attributeValues[attributeKey];
   409→
   410→    // Find attribute definition for logging and clamping
   411→    let attributeDef: AttributeDefinition | undefined;
   412→    if (characterId === '__user__') {
   413→      // Look up active persona's statsConfig for __user__
   414→      const activePersonaId = (state as any).activePersonaId;
   415→      const personas: any[] = (state as any).personas || [];
   416→      const activePersona = personas.find((p: any) => p.id === activePersonaId);
   417→      attributeDef = activePersona?.statsConfig?.attributes?.find(
   418→        (a: AttributeDefinition) => a.key === attributeKey
   419→      );
   420→    } else {
   421→      const character = state.characters.find((c: any) => c.id === characterId);
   422→      attributeDef = character?.statsConfig?.attributes?.find(
   423→        (a: AttributeDefinition) => a.key === attributeKey
   424→      );
   425→    }
   426→
   427→    // Clamp value to min/max bounds
   428→    const clampedValue = clampAttributeValue(value, attributeDef);
   429→    const clamped = clampedValue !== value;
   430→
   431→    // Log if clamping occurred
   432→    if (clamped) {
   433→      console.log(`[StatsSlice] Clamped ${attributeKey}: ${value} → ${clampedValue} (min: ${attributeDef?.min}, max: ${attributeDef?.max})`);
   434→    }
   435→
   436→    // Detect threshold reached
   437→    const thresholdsReached: ThresholdReachedInfo[] = [];
   438→    
   439→    if (attributeDef) {
   440→      // V2: Evaluate new thresholdEffects (flexible conditions with priority)
   441→      if (attributeDef.thresholdEffects && attributeDef.thresholdEffects.length > 0) {
   442→        // Build a temporary session stats with the new value for evaluation
   443→        const tempSessionStats: SessionStats = {
   444→          ...sessionStats,
   445→          characterStats: {
   446→            ...sessionStats.characterStats,
   447→            [characterId]: {
   448→              ...stats,
   449→              attributeValues: {
   450→                ...stats.attributeValues,
   451→                [attributeKey]: clampedValue,
   452→              },
   453→            },
   454→          },
   455→        };
   456→        
   457→        const matchingEffects = evaluateThresholdEffects(
   458→          attributeDef.thresholdEffects,
   459→          tempSessionStats,
   460→          characterId
   461→        );
   462→        
   463→        for (const effect of matchingEffects) {
   464→          thresholdsReached.push({
   465→            attributeKey: attributeDef.key,
   466→            attributeName: attributeDef.name,
   467→            thresholdType: 'custom',
   468→            effectName: effect.name,
   469→            effectId: effect.id,
   470→            priority: effect.priority,
   471→            rewards: effect.rewards,
   472→          });
   473→          console.log(`[StatsSlice] Threshold effect "${effect.name}" triggered for ${attributeDef.name} (priority: ${effect.priority})`);
   474→        }
   475→      }
   476→      
   477→      // Legacy: Check old onMinReached/onMaxReached (only if no thresholdEffects defined)
   478→      if (!attributeDef.thresholdEffects || attributeDef.thresholdEffects.length === 0) {
   479→        if (typeof clampedValue === 'number') {
   480→          // Check if reached minimum
   481→          if (attributeDef.min !== undefined && clampedValue === attributeDef.min) {
   482→            if (attributeDef.onMinReached?.enabled && attributeDef.onMinReached.rewards.length > 0) {
   483→              thresholdsReached.push({
   484→                attributeKey: attributeDef.key,
   485→                attributeName: attributeDef.name,
   486→                thresholdType: 'min',
   487→                thresholdValue: attributeDef.min,
   488→                rewards: attributeDef.onMinReached.rewards,
   489→              });
   490→              console.log(`[StatsSlice] Threshold reached: ${attributeDef.name} hit minimum (${attributeDef.min})`);
   491→            }
   492→          }
   493→          
   494→          // Check if reached maximum
   495→          if (attributeDef.max !== undefined && clampedValue === attributeDef.max) {
   496→            if (attributeDef.onMaxReached?.enabled && attributeDef.onMaxReached.rewards.length > 0) {
   497→              thresholdsReached.push({
   498→                attributeKey: attributeDef.key,
   499→                attributeName: attributeDef.name,
   500→                thresholdType: 'max',
   501→                thresholdValue: attributeDef.max,
   502→                rewards: attributeDef.onMaxReached.rewards,
   503→              });
   504→              console.log(`[StatsSlice] Threshold reached: ${attributeDef.name} hit maximum (${attributeDef.max})`);
   505→            }
   506→          }
   507→        }
   508→      }
   509→    }
   510→
   511→    // Build the updated state
   512→    const updatedCharacterStats = {
   513→      ...sessionStats.characterStats,
   514→      [characterId]: {
   515→        ...stats,
   516→        attributeValues: {
   517→          ...stats.attributeValues,
   518→          [attributeKey]: clampedValue,
   519→        },
   520→        lastUpdated: {
   521→          ...stats.lastUpdated,
   522→          [attributeKey]: Date.now(),
   523→        },
   524→      },
   525→    };
   526→
   527→    // Add to change log
   528→    addChangeLogEntry(
   529→      updatedCharacterStats[characterId],
   530→      attributeDef,
   531→      attributeKey,
   532→      oldValue,
   533→      clampedValue,
   534→      reason
   535→    );
   536→    
   537→    const newSessionStats: SessionStats = {
   538→      ...sessionStats,
   539→      characterStats: updatedCharacterStats,
   540→      lastModified: Date.now(),
   541→    };
   542→    
   543→    // Update the store
   544→    set({
   545→      sessions: state.sessions.map((s: any) =>
   546→        s.id === sessionId
   547→          ? { 
   548→              ...s, 
   549→              sessionStats: newSessionStats,
   550→              updatedAt: new Date().toISOString() 
   551→            }
   552→          : s
   553→      ),
   554→    });
   555→
   556→    // Sync currency changes to persona (Inventory V2)
   557→    // When currency is updated via quest rewards/triggers on __user__,
   558→    // also update persona.currency so the InventoryPanel/HUD stay in sync
   559→    if (characterId === '__user__' && attributeKey === 'currency' && typeof clampedValue === 'number') {
   560→      try {
   561→        const activePersonaId = (get() as any).activePersonaId;
   562→        if (activePersonaId) {
   563→          (get() as any).updatePersona?.(activePersonaId, { currency: clampedValue });
   564→        }
   565→      } catch { /* non-critical sync */ }
   566→    }
   567→
   568→    // Refresh objective visibility since attribute conditions may have changed
   569→    try {
   570→      (get() as any).refreshAllObjectiveVisibility?.(sessionId);
   571→    } catch { /* non-critical */ }
   572→
   573→    // Refresh activation conditions since attribute conditions may have changed
   574→    try {
   575→      (get() as any).refreshAllActivationConditions?.(sessionId);
   576→    } catch { /* non-critical */ }
   577→
   578→    // Return result with threshold info
   579→    return {
   580→      oldValue,
   581→      newValue: clampedValue,
   582→      clamped,
   583→      thresholdsReached,
   584→    };
   585→  },
   586→
   587→  batchUpdateCharacterStats: (sessionId, characterId, updates, reason = 'llm_detection') => {
   588→    set((state: any) => {
   589→      const sessions = state.sessions as Array<{
   590→        id: string; 
   591→        sessionStats?: SessionStats;
   592→      }>;
   593→      const sessionIndex = sessions.findIndex(s => s.id === sessionId);
   594→      
   595→      if (sessionIndex === -1) return state;
   596→      
   597→      const session = sessions[sessionIndex];
   598→      let sessionStats = session.sessionStats;
   599→      
   600→      // Auto-initialize sessionStats if missing (includes event fields reset)
   601→      if (!sessionStats) {
   602→        sessionStats = {
   603→          characterStats: {},
   604→          solicitudes: {
   605→            characterSolicitudes: {},
   606→            lastModified: Date.now(),
   607→          },
   608→          ultimo_objetivo_completado: undefined,
   609→          ultima_solicitud_completada: undefined,
   610→          ultima_solicitud_realizada: undefined,
   611→          ultima_accion_realizada: undefined,
   612→          initialized: true,
   613→          lastModified: Date.now(),
   614→        };
   615→      }
   616→      
   617→      // Auto-initialize character stats if missing
   618→      if (!sessionStats.characterStats[characterId]) {
   619→        let statsConfig: CharacterStatsConfig | undefined;
   620→        if (characterId === '__user__') {
   621→          // Look up active persona's statsConfig for __user__
   622→          const activePersonaId = (state as any).activePersonaId;
   623→          const personas: any[] = (state as any).personas || [];
   624→          const activePersona = personas.find((p: any) => p.id === activePersonaId);
   625→          statsConfig = activePersona?.statsConfig;
   626→        } else {
   627→          const character = state.characters.find((c: any) => c.id === characterId);
   628→          statsConfig = character?.statsConfig;
   629→        }
   630→        let autoStats = createDefaultCharacterStats(statsConfig);
   631→        // For persona (__user__), also initialize equipment slot values
   632→        if (characterId === '__user__') {

   637→                autoStats.attributeValues[slot.key] = '';
   638→                autoStats.lastUpdated[slot.key] = Date.now();

   640→            }
   641→          }
   642→        }
   643→        sessionStats = {
   644→          ...sessionStats,
   645→          characterStats: {
   646→            ...sessionStats.characterStats,
   647→            [characterId]: autoStats,
   648→          },
   649→        };
   650→      }
   651→      
   652→      const stats = sessionStats.characterStats[characterId];
   653→      if (!stats) return state;
   654→      
   655→      // Look up statsConfig for the character (or persona for __user__)
   656→      let statsConfig: CharacterStatsConfig | undefined;
   657→      if (characterId === '__user__') {
   658→        const activePersonaId = (state as any).activePersonaId;
   659→        const personas: any[] = (state as any).personas || [];
   660→        const activePersona = personas.find((p: any) => p.id === activePersonaId);
   661→        statsConfig = activePersona?.statsConfig;
   662→      } else {
   663→        const character = state.characters.find((c: any) => c.id === characterId);
   664→        statsConfig = character?.statsConfig;
   665→      }
   666→      const now = Date.now();
   667→      
   668→      // Apply all updates
   669→      const newAttributeValues = { ...stats.attributeValues };
   670→      const newLastUpdated = { ...stats.lastUpdated };
   671→      const newChangeLog = [...(stats.changeLog || [])];
   672→
   673→      for (const update of updates) {
   674→        const oldValue = newAttributeValues[update.attributeKey];
   675→        const attributeDef = statsConfig?.attributes?.find(
   676→          (a: AttributeDefinition) => a.key === update.attributeKey
   677→        );
   678→
   679→        // Clamp value to min/max bounds
   680→        const clampedValue = clampAttributeValue(update.value, attributeDef);
   681→
   682→        // Log if clamping occurred
   683→        if (clampedValue !== update.value) {
   684→          console.log(`[StatsSlice] Clamped ${update.attributeKey}: ${update.value} → ${clampedValue} (min: ${attributeDef?.min}, max: ${attributeDef?.max})`);
   685→        }
   686→
   687→        newAttributeValues[update.attributeKey] = clampedValue;
   688→        newLastUpdated[update.attributeKey] = now;
   689→
   690→        newChangeLog.push({
   691→          attributeId: attributeDef?.id || update.attributeKey,
   692→          attributeKey: update.attributeKey,
   693→          attributeName: attributeDef?.name || update.attributeKey,
   694→          oldValue: oldValue ?? '',
   695→          newValue: clampedValue,
   696→          reason,
   697→          timestamp: now,
   698→        });
   699→      }
   700→      
   701→      // Keep only last 100 entries
   702→      const trimmedChangeLog = newChangeLog.slice(-100);
   703→      
   704→      const updatedCharacterStats = {
   705→        ...sessionStats.characterStats,
   706→        [characterId]: {
   707→          ...stats,
   708→          attributeValues: newAttributeValues,
   709→          lastUpdated: newLastUpdated,
   710→          changeLog: trimmedChangeLog,
   711→        },
   712→      };
   713→      
   714→      const newSessionStats: SessionStats = {
   715→        ...sessionStats,
   716→        characterStats: updatedCharacterStats,
   717→        lastModified: now,
   718→      };
   719→      
   720→      return {
   721→        sessions: state.sessions.map((s: any) =>
   722→          s.id === sessionId
   723→            ? { 
   724→                ...s, 
   725→                sessionStats: newSessionStats,
   726→                updatedAt: new Date().toISOString() 
   727→              }
   728→            : s
   729→        ),
   730→      };
   731→    });
   732→
   733→    // Refresh objective visibility since attribute conditions may have changed
   734→    try {
   735→      (get() as any).refreshAllObjectiveVisibility?.(sessionId);
   736→    } catch { /* non-critical */ }
   737→
   738→    // Refresh activation conditions since attribute conditions may have changed
   739→    try {
   740→      (get() as any).refreshAllActivationConditions?.(sessionId);
   741→    } catch { /* non-critical */ }
   742→  },

  1712→/**
  1713→ * Evaluate a single requirement against current stats
  1714→ * Supports target requirements (checking attributes of other characters or persona)
  1715→ */
  1716→export function evaluateRequirement(
  1717→  requirement: StatRequirement,
  1718→  attributeValues: Record<string, number | string>,
  1719→  sessionStats?: SessionStats | null
  1720→): boolean {
  1721→  let currentValue: number | string | undefined;
  1722→
  1723→  if (requirement.targetCharacterId) {
  1724→    // Target mode - look up attribute from another character or persona
  1725→    currentValue = sessionStats?.characterStats?.[requirement.targetCharacterId]?.attributeValues?.[requirement.attributeKey];
  1726→  } else {
  1727→    // Self mode - use own attribute values
  1728→    currentValue = attributeValues[requirement.attributeKey];
  1729→  }
  1730→
  1731→  if (currentValue === undefined) return false;
  1732→
  1733→  const currentNum = typeof currentValue === 'number' ? currentValue : parseFloat(currentValue);
  1734→  const valueNum = typeof requirement.value === 'number' ? requirement.value : parseFloat(requirement.value);
  1735→
  1736→  // String comparison for text operators
  1737→  const isTextOperator = requirement.operator === 'contains' || requirement.operator === 'not_contains';
  1738→  if (isTextOperator || isNaN(currentNum) || isNaN(valueNum)) {
  1739→    const currentStr = String(currentValue).toLowerCase();
  1740→    const valueStr = String(requirement.value).toLowerCase();
  1741→
  1742→    switch (requirement.operator) {
  1743→      case '==': return currentStr === valueStr;
  1744→      case '!=': return currentStr !== valueStr;
  1745→      case 'contains': return currentStr.includes(valueStr);
  1746→      case 'not_contains': return !currentStr.includes(valueStr);
  1747→      default: return false;
  1748→    }
  1749→  }
  1750→
  1751→  switch (requirement.operator) {
  1752→    case '<': return currentNum < valueNum;
  1753→    case '<=': return currentNum <= valueNum;
  1754→    case '>': return currentNum > valueNum;
  1755→    case '>=': return currentNum >= valueNum;
  1756→    case '==': return currentNum === valueNum;
  1757→    case '!=': return currentNum !== valueNum;
  1758→    case 'between': {
  1759→      const maxNum = typeof requirement.valueMax === 'number'
  1760→        ? requirement.valueMax
  1761→        : parseFloat(requirement.valueMax?.toString() || '0');
  1762→      return currentNum >= valueNum && currentNum <= maxNum;
  1763→    }
  1764→    default: return false;
  1765→  }
  1766→}
  1767→
  1768→/**
  1769→ * Evaluate all requirements (supports AND and OR logic)
  1770→ * Default is AND logic for backward compatibility
  1771→ */
  1772→export function evaluateRequirements(
  1773→  requirements: StatRequirement[],
  1774→  attributeValues: Record<string, number | string>,
  1775→  sessionStats?: SessionStats | null,
  1776→  operator?: 'AND' | 'OR'
  1777→): boolean {
  1778→  if (!requirements || requirements.length === 0) return true;
  1779→  const logicFn = operator === 'OR' ? requirements.some.bind(requirements) : requirements.every.bind(requirements);
  1780→  return logicFn(req => evaluateRequirement(req, attributeValues, sessionStats));
  1781→}
