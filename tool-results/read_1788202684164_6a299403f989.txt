   958→  tickEffects: (personaId) => {
   959→    const expiredMessages: string[] = [];
   960→    const stateAny = get() as any;
   961→
   962→    // 1. Apply dynamic consumable effects before decrementing
   963→    const consumableEffects = get().activeConsumableEffects.filter(e => e.personaId === personaId);
   964→    for (const activeEffect of consumableEffects) {
   965→      const dynamicEffects = activeEffect.effects.filter(e => e.mode === 'dynamic');
   966→      const activeTurns = activeEffect.totalTurns - activeEffect.remainingTurns;
   967→      for (const dynEffect of dynamicEffects) {
   968→        applyDynamicEffectToSessionStats(stateAny, dynEffect, activeTurns);
   969→      }
   970→    }
   971→
   972→    // 2. Apply dynamic equipment effects and increment turn counters
   973→    const dynamicEqState = { ...get().dynamicEquipmentState };
   974→    const equippedItems = get().getEquippedItems(personaId);
   975→    for (const { entry, item } of equippedItems) {
   976→      const dynamicEffects = (item.attributeEffects || []).filter((e: ItemAttributeEffect) => e.mode === 'dynamic');
   977→      if (dynamicEffects.length > 0) {
   978→        const stateKey = `${personaId}:${item.id}`;
   979→        const currentState = dynamicEqState[stateKey] || { activeTurns: 0, appliedAt: new Date().toISOString() };
   980→        for (const dynEffect of dynamicEffects) {
   981→          const resolvedEffect = entry.targetOverrideId ? { ...dynEffect, targetId: entry.targetOverrideId } : dynEffect;
   982→          applyDynamicEffectToSessionStats(stateAny, resolvedEffect, currentState.activeTurns);
   983→        }
   984→        dynamicEqState[stateKey] = { ...currentState, activeTurns: currentState.activeTurns + 1 };
   985→      }
   986→    }
   987→
   988→    // 3. Decrement consumable turns
   989→    set((state) => ({
   990→      activeConsumableEffects: state.activeConsumableEffects.map(effect => {
   991→        if (effect.personaId !== personaId) return effect;
   992→        const newRemaining = effect.remainingTurns - 1;
   993→        if (newRemaining <= 0) {
   994→          const msg = effect.expireMessage || `El efecto de ${effect.itemName} ha expirado`;
   995→          expiredMessages.push(msg);
   996→        }
   997→        return { ...effect, remainingTurns: Math.max(0, newRemaining) };
   998→      }),
   999→      dynamicEquipmentState: dynamicEqState,
  1000→    }));
  1001→
  1002→    // Sync to session for per-session storage
  1003→    syncEffectsToSession(stateAny, get().activeConsumableEffects);
  1004→
  1005→    return expiredMessages;
  1006→  },
