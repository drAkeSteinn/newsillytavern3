   223→// ============================================
   224→// CONDITIONAL STATE VARIANT
   225→// ============================================
   226→
   227→/**
   228→ * @deprecated Conditions are now defined at the SpritePack level instead of State Collections.
   229→ * This interface is kept for backward compatibility but should not be used in new code.
   230→ * Use SpritePackV2.conditionalMode + SpritePackEntryV2.conditions instead.
   231→ *
   232→ * Conditional State Variant - Attribute-driven sprite variant for a state collection.
   233→ *
   234→ * When a character has attributes (e.g., "exhaustion", "vida"), these variants
   235→ * allow the idle/talk/thinking sprite to change dynamically based on attribute values.
   236→ *
   237→ * Priority system (like lorebook dynamic conditions):
   238→ * - Higher priority wins when multiple variants match
   239→ * - If no variant matches, the default state collection pack is used
   240→ *
   241→ * Example:
   242→ *   Variant "Exhausted" → Priority 2, Condition: exhaustion >= 80 → pack_exhausted
   243→ *   Variant "Tired"     → Priority 1, Condition: exhaustion >= 50 → pack_tired
   244→ *   Variant "Normal"    → (no conditions, default pack)
   245→ *
   246→ * When exhaustion = 90: Both match, but "Exhausted" (P2) wins
   247→ * When exhaustion = 60: "Tired" (P1) wins
   248→ * When exhaustion = 30: Default pack is used
   249→ */
   250→export interface ConditionalStateVariant {
   251→  id: string;
   252→  name: string;                        // Display name: "Exhausted Idle", "Injured Idle"
   253→  enabled: boolean;                    // Enable/disable this variant
   254→
   255→  // Priority - Higher number = higher priority (wins when multiple match)
   256→  priority: number;                    // Default: 0
   257→
   258→  // Conditions - evaluated with AND or OR logic (controlled by conditionOperator)
   259→  // Reuses StatRequirement which supports:
   260→  //   - Operators: <, <=, >, >=, ==, !=, between, contains, not_contains
   261→  //   - Cross-character: targetCharacterId for other characters' or persona's attributes
   262→  conditions: StatRequirement[];
   263→  conditionOperator?: 'AND' | 'OR'; // Logic operator for conditions (default: 'AND')
   264→
   265→  // Which pack to use when this variant is active
   266→  packId: string;                      // References SpritePackV2.id
   267→
   268→  // Behavior when selecting sprite from the pack
   269→  behavior: 'principal' | 'random' | 'list';
   270→  principalSpriteId?: string;          // For 'principal' mode
   271→  spriteOrder?: string[];              // For 'list' mode custom ordering
   272→  excludedSpriteIds?: string[];        // Sprites to exclude
   273→}

   389→/**
   390→ * Conditional Sprite Entry - Attribute-driven sprite within a trigger collection.
   391→ *
   392→ * Used when a Trigger Collection is in conditional mode. Instead of matching keys
   393→ * to select a sprite, it evaluates attribute conditions with priority.
   394→ *
   395→ * When a Skill/Action reward activates a conditional sprite collection,
   396→ * the entries are evaluated by priority (highest first). The first entry
   397→ * whose conditions all match determines the sprite to display.
   398→ * If no entry matches, the default sprite (principalSpriteId) is used.
   399→ *
   400→ * Example (Trigger Collection "Attack Faces"):
   401→ *   Entry "Desperate"  → Priority 2, Condition: vida <= 20  → attack_desperate.webp
   402→ *   Entry "Empowered"  → Priority 1, Condition: mana >= 50  → attack_empowered.webp
   403→ *   Default             → attack_normal.webp
   404→ */
   405→export interface ConditionalSpriteEntry {
   406→  id: string;
   407→  name: string;                        // Display name: "Desperate Attack", "Low HP Face"
   408→  enabled: boolean;                    // Enable/disable this entry
   409→
   410→  // Priority - Higher number = higher priority (wins when multiple match)
   411→  priority: number;                    // Default: 0
   412→
   413→  // Conditions - evaluated with AND or OR logic (controlled by conditionOperator)
   414→  // Reuses StatRequirement for cross-character and all operator support
   415→  conditions: StatRequirement[];
   416→  conditionOperator?: 'AND' | 'OR'; // Logic operator for conditions (default: 'AND')
   417→
   418→  // Which sprite from the pack to show when this entry matches
   419→  spriteId: string;                    // References SpritePackEntryV2.id
   420→}

  1345→// ============================================
  1346→// Proactive Attribute-Driven Messages (FASE 11)
  1347→// Permite condicionar los mensajes proactivos a un atributo del personaje
  1348→// (ej. "codicia") y definir múltiples casos (lineal/random) por condición.
  1349→// ============================================
  1350→
  1351→/**
  1352→ * Operadores disponibles para comparar atributos en condiciones proactivas.
  1353→ * (Espejo de AttributeComparator usado por los lorebooks de tipo atributo.)
  1354→ */
  1355→export type ProactiveComparator = AttributeComparator;
  1356→
  1357→/**
  1358→ * Modo de selección de casos cuando una condición aplica y hay múltiples casos.
  1359→ * - 'linear': recorre la lista de casos de forma cíclica (1, 2, 3, 1, 2, 3, ...)
  1360→ * - 'random': elige un caso al azar (evitando repetir el último usado).
  1361→ */
  1362→export type ProactiveCaseMode = 'linear' | 'random';
  1363→
  1364→/**
  1365→ * Un caso (mensaje) que se enviará cuando su condición padre aplique.
  1366→ * El contenido soporta keys de lorebook ({{key}}), atributos ({{vida}}),
  1367→ * y variables de plantilla ({{user}}, {{char}}, etc.).
  1368→ */
  1369→export interface ProactiveCase {
  1370→  id: string;
  1371→  /** Etiqueta opcional para identificar el caso en la UI. */
  1372→  label?: string;
  1373→  /** Contenido del mensaje. Soporta todas las keys de resolución. */
  1374→  content: string;
  1375→  /** Si el caso está activo (false = se omite al seleccionar). */
  1376→  enabled?: boolean;
  1377→}
  1378→
  1379→/**
  1380→ * Una condición sobre el atributo configurado. Cuando aplica (según operador + valor),
  1381→ * se selecciona uno de sus `cases` según el `caseMode`.
  1382→ * Las condiciones se evalúan en orden de `priority` descendente (mayor prioridad primero),
  1383→ * igual que los lorebooks de tipo atributo dinámico.
  1384→ */
  1385→export interface ProactiveAttributeCondition {
  1386→  id: string;
  1387→  /** Etiqueta opcional para identificar la condición en la UI. */
  1388→  label?: string;
  1389→  /** Operador de comparación contra el valor actual del atributo. */
  1390→  operator: ProactiveComparator;
  1391→  /** Valor de comparación (numérico o texto según el tipo del atributo). */
  1392→  value: number | string;
  1393→  /**
  1394→   * Prioridad (mayor = más importante). En caso de que varias condiciones apliquen
  1395→   * simultáneamente, gana la de mayor prioridad. Por defecto 0.
  1396→   */
  1397→  priority?: number;
  1398→  /** Cómo seleccionar entre los `cases` cuando hay más de uno. */
  1399→  caseMode: ProactiveCaseMode;
  1400→  /** Lista de casos (mensajes) para esta condición. */
  1401→  cases: ProactiveCase[];
  1402→  /** Si la condición está activa (false = se omite al evaluar). */
  1403→  enabled?: boolean;
  1404→}
  1405→
  1406→/**
  1407→ * Configuración de mensajes proactivos condicionados por atributo del personaje.
  1408→ * Si está habilitado, el sistema:
  1409→ *   1. Lee el valor actual del atributo configurado.
  1410→ *   2. Evalúa las condiciones en orden de prioridad.
  1411→ *   3. La primera condición que aplique selecciona uno de sus casos (lineal/random).
  1412→ *   4. Si ninguna condición aplica, usa los `defaultCases` (con `defaultCaseMode`).
  1413→ * El mensaje seleccionado se interpola entre `proactivePrefix` y `proactiveSuffix`.
  1414→ */
  1415→export interface ProactiveAttributeConfig {
  1416→  /** Activa/desactiva la lógica condicional por atributo. */
  1417→  enabled: boolean;
  1418→  /** Personaje cuyo atributo se evalúa: '__user__' (persona), '__char__' (actual) o ID específico. */
  1419→  characterId: string;
  1420→  /** Key del atributo a evaluar (ej. 'codicia', 'vida', 'confianza'). */
  1421→  attributeKey: string;
  1422→  /** Condiciones ordenables por prioridad. */
  1423→  conditions: ProactiveAttributeCondition[];
  1424→  /** Modo de selección de los casos por defecto (ninguna condición aplica). */
  1425→  defaultCaseMode: ProactiveCaseMode;
  1426→  /** Casos por defecto cuando ninguna condición aplica. */
  1427→  defaultCases: ProactiveCase[];
  1428→}

  1459→  postHistoryOverride?: string;
