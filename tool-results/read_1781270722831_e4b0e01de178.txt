   221→   221→// ============================================
   222→   222→// CONDITIONAL STATE VARIANT
   223→   223→// ============================================
   224→   224→
   225→   225→/**
   226→   226→ * @deprecated Conditions are now defined at the SpritePack level instead of State Collections.
   227→   227→ * This interface is kept for backward compatibility but should not be used in new code.
   228→   228→ * Use SpritePackV2.conditionalMode + SpritePackEntryV2.conditions instead.
   229→   229→ *
   230→   230→ * Conditional State Variant - Attribute-driven sprite variant for a state collection.
   231→   231→ *
   232→   232→ * When a character has attributes (e.g., "exhaustion", "vida"), these variants
   233→   233→ * allow the idle/talk/thinking sprite to change dynamically based on attribute values.
   234→   234→ *
   235→   235→ * Priority system (like lorebook dynamic conditions):
   236→   236→ * - Higher priority wins when multiple variants match
   237→   237→ * - If no variant matches, the default state collection pack is used
   238→   238→ *
   239→   239→ * Example:
   240→   240→ *   Variant "Exhausted" → Priority 2, Condition: exhaustion >= 80 → pack_exhausted
   241→   241→ *   Variant "Tired"     → Priority 1, Condition: exhaustion >= 50 → pack_tired
   242→   242→ *   Variant "Normal"    → (no conditions, default pack)
   243→   243→ *
   244→   244→ * When exhaustion = 90: Both match, but "Exhausted" (P2) wins
   245→   245→ * When exhaustion = 60: "Tired" (P1) wins
   246→   246→ * When exhaustion = 30: Default pack is used
   247→   247→ */
   248→   248→export interface ConditionalStateVariant {
   249→   249→  id: string;
   250→   250→  name: string;                        // Display name: "Exhausted Idle", "Injured Idle"
   251→   251→  enabled: boolean;                    // Enable/disable this variant
   252→   252→
   253→   253→  // Priority - Higher number = higher priority (wins when multiple match)
   254→   254→  priority: number;                    // Default: 0
   255→   255→
   256→   256→  // Conditions - ALL must match (AND logic)
   257→   257→  // Reuses StatRequirement which supports:
   258→   258→  //   - Operators: <, <=, >, >=, ==, !=, between, contains, not_contains
   259→   259→  //   - Cross-character: targetCharacterId for other characters' or persona's attributes
   260→   260→  conditions: StatRequirement[];
   261→   261→
   262→   262→  // Which pack to use when this variant is active
   263→   263→  packId: string;                      // References SpritePackV2.id
   264→   264→
   265→   265→  // Behavior when selecting sprite from the pack
   266→   266→  behavior: 'principal' | 'random' | 'list';
   267→   267→  principalSpriteId?: string;          // For 'principal' mode
   268→   268→  spriteOrder?: string[];              // For 'list' mode custom ordering
   269→   269→  excludedSpriteIds?: string[];        // Sprites to exclude
   270→   270→}

   353→   353→/**
   354→   354→ * Conditional Sprite Entry - Attribute-driven sprite within a trigger collection.
   355→   355→ *
   356→   356→ * Used when a Trigger Collection is in conditional mode. Instead of matching keys
   357→   357→ * to select a sprite, it evaluates attribute conditions with priority.
   358→   358→ *
   359→   359→ * When a Skill/Action reward activates a conditional sprite collection,
   360→   360→ * the entries are evaluated by priority (highest first). The first entry
   361→   361→ * whose conditions all match determines the sprite to display.
   362→   362→ * If no entry matches, the default sprite (principalSpriteId) is used.
   363→   363→ *
   364→   364→ * Example (Trigger Collection "Attack Faces"):
   365→   365→ *   Entry "Desperate"  → Priority 2, Condition: vida <= 20  → attack_desperate.webp
   366→   366→ *   Entry "Empowered"  → Priority 1, Condition: mana >= 50  → attack_empowered.webp
   367→   367→ *   Default             → attack_normal.webp
   368→   368→ */
   369→   369→export interface ConditionalSpriteEntry {
   370→   370→  id: string;
   371→   371→  name: string;                        // Display name: "Desperate Attack", "Low HP Face"
   372→   372→  enabled: boolean;                    // Enable/disable this entry
   373→   373→
   374→   374→  // Priority - Higher number = higher priority (wins when multiple match)
   375→   375→  priority: number;                    // Default: 0
   376→   376→
   377→   377→  // Conditions - ALL must match (AND logic)
   378→   378→  // Reuses StatRequirement for cross-character and all operator support
   379→   379→  conditions: StatRequirement[];
   380→   380→
   381→   381→  // Which sprite from the pack to show when this entry matches
   382→   382→  spriteId: string;                    // References SpritePackEntryV2.id
   383→   383→}
