     1→'use client';
     2→
     3→/**
     4→ * HUDDisplay Component
     5→ * 
     6→ * Displays a HUD (Heads-Up Display) overlay in the chat interface.
     7→ * Shows field values from the active HUD template AND character attributes with showInHUD=true.
     8→ * 
     9→ * Features:
    10→ * - Multiple display styles (default, progress, badge, icon, chip, status, gauge, etc.)
    11→ * - Positionable (top-left, top-right, bottom-left, bottom-right)
    12→ * - Real-time updates from trigger detection
    13→ * - Progress bars for gauges for numeric fields
    14→ * - Color-coded badges for enum fields
    15→ * - Automatic integration with character stats system
    16→ * - Group chat support: shows attributes for all group members
    17→ * - Animated visual effects (value change, shine, glow, particles)
    18→ */
    19→
    20→import { useTavernStore } from '@/store';
    21→import type { HUDTemplate, HUDField, HUDPosition, HUDStyle, HUDFieldStyle, AttributeDefinition, CharacterCard } from '@/types';
    22→import { cn } from '@/lib/utils';
    23→import {
    24→  AnimatedProgress,
    25→  AnimatedGauge,
    26→  AnimatedMeter,
    27→  AnimatedDots,
    28→  AnimatedBadge,
    29→} from './hud-effects';
    30→
    31→interface HUDDisplayProps {
    32→  position?: HUDPosition;
    33→  className?: string;
    34→}
    35→
    36→// Position classes
    37→const positionClasses: Record<HUDPosition, string> = {
    38→  'top-left': 'top-4 left-4',
    39→  'top-right': 'top-4 right-4',
    40→  'bottom-left': 'bottom-4 left-4',
    41→  'bottom-right': 'bottom-4 right-4',
    42→};
    43→
    44→// Color classes
    45→const colorClasses: Record<string, string> = {
    46→  red: 'text-red-400',
    47→  green: 'text-green-400',
    48→  blue: 'text-blue-400',
    49→  yellow: 'text-yellow-400',
    50→  purple: 'text-purple-400',
    51→  orange: 'text-orange-400',
    52→  pink: 'text-pink-400',
    53→  cyan: 'text-cyan-400',
    54→  default: 'text-white/80',
    55→};
    56→

    58→  red: 'bg-red-500/20 border-red-500/30',

    60→  blue: 'bg-blue-500/20 border-blue-500/30',
    61→  yellow: 'bg-yellow-500/20 border-yellow-500/30',
    62→  purple: 'bg-purple-500/20 border-purple-500/30',
    63→  orange: 'bg-orange-500/20 border-orange-500/30',
    64→  pink: 'bg-pink-500/20 border-pink-500/30',
    65→  cyan: 'bg-cyan-500/20 border-cyan-500/30',
    66→  default: 'bg-white/10 border-white/20',

    68→
    69→const progressColorClasses: Record<string, string> = {
    70→  red: 'bg-red-500',
    71→  green: 'bg-green-500',
    72→  blue: 'bg-blue-500',
    73→  yellow: 'bg-yellow-500',
    74→  purple: 'bg-purple-500',
    75→  orange: 'bg-orange-500',
    76→  pink: 'bg-pink-500',
    77→  cyan: 'bg-cyan-500',
    78→  default: 'bg-white/50',
    79→};
    80→
    81→export function HUDDisplay({ className }: HUDDisplayProps) {
    82→  const hudTemplates = useTavernStore((state) => state.hudTemplates);
    83→  const hudSessionState = useTavernStore((state) => state.hudSessionState);
    84→  const activeCharacterId = useTavernStore((state) => state.activeCharacterId);
    85→  const activeGroupId = useTavernStore((state) => state.activeGroupId);
    86→  const groups = useTavernStore((state) => state.groups);
    87→  const characters = useTavernStore((state) => state.characters);
    88→  const sessions = useTavernStore((state) => state.sessions);
    89→  const activeSessionId = useTavernStore((state) => state.activeSessionId);
    90→  const activePersonaId = useTavernStore((state) => state.activePersonaId);
    91→  const personas = useTavernStore((state) => state.personas);
    92→
    93→  // Get derived values from subscribed state
    94→  const activeCharacter = characters.find((c) => c.id === activeCharacterId);
    95→  const activeGroup = activeGroupId ? groups.find((g) => g.id === activeGroupId) : null;
    96→
    97→  // Get active template
    98→  const activeTemplate = hudTemplates.find(
    99→    (t) => t.id === hudSessionState.activeTemplateId
   100→  );
   101→
   102→  // Use template's position setting (defaults to top-right if no template)
   103→  const position = activeTemplate?.position || 'top-right';
   104→  
   105→  // Get current session for stats values
   106→  const activeSession = sessions.find((s) => s.id === activeSessionId);
   107→  const sessionStats = activeSession?.sessionStats;
   108→  
   109→  // Determine if this is a group chat
   110→  const isGroupChat = !!activeGroup;
   111→  
   112→  // Get all characters with their attributes
   113→  let charactersWithAttributes: Array<{
   114→    character: CharacterCard;
   115→    attributes: AttributeDefinition[];
   116→    values: Record<string, number | string>;
   117→  }> = [];
   118→  
   119→  if (isGroupChat && activeGroup?.members) {
   120→    // Group chat: get all group members with their attributes
   121→    charactersWithAttributes = activeGroup.members
   122→      .map((member) => {
   123→        const char = characters.find((c) => c.id === member.characterId);
   124→        if (!char) return null;
   125→
   126→        const attributes = char.statsConfig?.enabled
   127→          ? char.statsConfig.attributes.filter((attr) => attr.showInHUD !== false)
   128→          : [];
   129→
   130→        // Get values from sessionStats, or use defaults if not initialized
   131→        // IMPORTANT: Always use defaults as base, then override with stored values
   132→        // This ensures new attributes added after session creation are shown
   133→        const storedValues = sessionStats?.characterStats?.[char.id]?.attributeValues;
   134→        const values: Record<string, number | string> = {};
   135→
   136→        // Start with default values for ALL attributes
   137→        for (const attr of attributes) {
   138→          values[attr.key] = attr.defaultValue;
   139→        }
   140→
   141→        // Override with stored values if available
   142→        if (storedValues) {
   143→          Object.assign(values, storedValues);
   144→        }
   145→
   146→        return { character: char, attributes, values };
   147→      })
   148→      .filter((item): item is NonNullable<typeof item> => item !== null)
   149→      .filter((item) => item.attributes.length > 0);
   150→  } else if (activeCharacter) {
   151→    // Single character chat
   152→    const attributes = activeCharacter.statsConfig?.enabled
   153→      ? activeCharacter.statsConfig.attributes.filter((attr) => attr.showInHUD !== false)
   154→      : [];
   155→
   156→    // Get values from sessionStats, or use defaults if not initialized
   157→    // IMPORTANT: Always use defaults as base, then override with stored values
   158→    // This ensures new attributes added after session creation are shown
   159→    const storedValues = sessionStats?.characterStats?.[activeCharacter.id]?.attributeValues;
   160→    const values: Record<string, number | string> = {};
   161→
   162→    // Start with default values for ALL attributes
   163→    for (const attr of attributes) {
   164→      values[attr.key] = attr.defaultValue;
   165→    }
   166→
   167→    // Override with stored values if available
   168→    if (storedValues) {
   169→      Object.assign(values, storedValues);
   170→    }
   171→
   172→    if (attributes.length > 0) {
   173→      charactersWithAttributes = [{ character: activeCharacter, attributes, values }];
   174→    }
   175→  }
   176→
   177→  // Also include persona (user) attributes if available
   178→  const activePersona = activePersonaId ? personas.find((p) => p.id === activePersonaId) : null;
   179→  let personaAttributes: { attributes: AttributeDefinition[]; values: Record<string, number | string>; personaName: string } | null = null;
   180→
   181→  if (activePersona?.statsConfig?.enabled && activePersona.statsConfig.attributes?.length > 0) {
   182→    const attributes = activePersona.statsConfig.attributes.filter((attr) => attr.showInHUD !== false);
   183→    const storedValues = sessionStats?.characterStats?.['__user__']?.attributeValues;
   184→    const values: Record<string, number | string> = {};
   185→
   186→    for (const attr of attributes) {
   187→      values[attr.key] = attr.defaultValue;
   188→    }
   189→    if (storedValues) {
   190→      // Only copy values for DEFINED persona attributes — skips internal mirrors
   191→      // (relacion/relacion_etapa/hora/momento_del_dia/dia/estacion) that live in the
   192→      // same record but shouldn't flood the persona HUD panel.
   193→      for (const attr of attributes) {
   194→        if (storedValues[attr.key] !== undefined) {
   195→          values[attr.key] = storedValues[attr.key];
   196→        }
   197→      }
   198→    }
   199→
   200→    if (attributes.length > 0) {
   201→      personaAttributes = { attributes, values, personaName: activePersona.name };
   202→    }
   203→  }
   204→  
   205→  // No active HUD and no character attributes
   206→  const hasTemplate = activeTemplate && activeTemplate.fields.length > 0;
   207→  const hasAttributes = charactersWithAttributes.length > 0;
   208→  const hasPersonaAttributes = personaAttributes !== null;
   209→
   210→  if (!hasTemplate && !hasAttributes && !hasPersonaAttributes) {
   211→    return null;
   212→  }
   213→
   214→  // Get style from template
   215→  const hudStyle = activeTemplate?.style || 'card';
   216→
   217→  return (
   218→    <div
   219→      className={cn(
   220→        'absolute z-30 pointer-events-none',
   221→        positionClasses[position],
   222→        className
   223→      )}
   224→      style={{ opacity: activeTemplate?.opacity ?? 0.9 }}
   225→    >
   226→      {/* HUD Template Fields */}
   227→      {hasTemplate && (
   228→        <HUDContainer template={activeTemplate} values={hudSessionState.fieldValues} />
   229→      )}
   230→
   231→      {/* Character Attributes HUD */}
   232→      {(hasAttributes || hasPersonaAttributes) && (
   233→        <MultiCharacterAttributesHUD
   234→          charactersWithAttributes={charactersWithAttributes}
   235→          compact={activeTemplate?.compact}
   236→          isGroupChat={isGroupChat}
   237→          style={hudStyle}
   238→          personaAttributes={personaAttributes}
   239→        />
   240→      )}
   241→    </div>
   242→  );
   243→}

   245→// ============================================
   246→// Multi-Character Attributes HUD Component
   247→// ============================================
   248→
   249→interface CharacterWithAttributes {
   250→  character: CharacterCard;
   251→  attributes: AttributeDefinition[];
   252→  values: Record<string, number | string>;
   253→}
   254→
   255→interface PersonaAttributesData {
   256→  attributes: AttributeDefinition[];
   257→  values: Record<string, number | string>;
   258→  personaName: string;
   259→}
   260→
   261→interface MultiCharacterAttributesHUDProps {
   262→  charactersWithAttributes: CharacterWithAttributes[];
   263→  compact?: boolean;
   264→  isGroupChat: boolean;
   265→  style?: HUDStyle;
   266→  personaAttributes?: PersonaAttributesData | null;
   267→}
   268→
   269→function MultiCharacterAttributesHUD({ charactersWithAttributes, compact, isGroupChat, style = 'card', personaAttributes }: MultiCharacterAttributesHUDProps) {
   270→  // Get container classes based on style
   271→  const getStyleClasses = () => {
   272→    switch (style) {
   273→      case 'glass':
   274→        return 'backdrop-blur-xl bg-white/5 border-white/20 shadow-[0_8px_32px_rgba(0,0,0,0.3)]';
   275→      case 'neon':
   276→        return 'backdrop-blur-md bg-slate-900/80 border-2 border-cyan-500/50 shadow-[0_0_20px_rgba(6,182,212,0.3)]';
   277→      case 'holographic':
   278→        return 'backdrop-blur-lg bg-gradient-to-r from-cyan-500/10 via-purple-500/10 to-pink-500/10 border-cyan-400/30';
   279→      case 'fantasy':
   280→        return 'bg-gradient-to-br from-amber-950/80 via-stone-900/80 to-amber-950/80 border-2 border-amber-600/40';
   281→      case 'retro':
   282→        return 'bg-black/90 border-4 border-green-500/70 font-mono shadow-[0_0_10px_rgba(34,197,94,0.5)]';
   283→      case 'panel':
   284→        return 'backdrop-blur-lg bg-gradient-to-br from-slate-900/90 via-slate-800/80 to-slate-900/90 border-2 border-white/20';
   285→      case 'minimal':
   286→        return 'bg-transparent border-transparent';
   287→      default:
   288→        return 'backdrop-blur-md bg-gradient-to-br from-slate-900/80 to-slate-800/60 border border-white/10';
   289→    }
   290→  };
   291→
   292→  return (
   293→    <div className="flex flex-col gap-3">
   294→      {charactersWithAttributes.map(({ character, attributes, values }) => (
   295→        <div
   296→          key={character.id}
   297→          className={cn(
   298→            'pointer-events-auto rounded-xl border transition-all duration-300',
   299→            getStyleClasses(),
   300→            compact ? 'p-2' : 'p-4',
   301→            'shadow-xl'
   302→          )}
   303→        >
   304→          {/* Character Name Header (only in group chats) */}
   305→          {isGroupChat && (
   306→            <div className={cn(
   307→              'font-medium mb-2 pb-1 border-b border-white/10',
   308→              compact ? 'text-[10px]' : 'text-xs',
   309→              style === 'neon' && 'text-cyan-400 uppercase tracking-wider',
   310→              style === 'fantasy' && 'text-amber-400',
   311→              style === 'retro' && 'text-green-400',
   312→              style === 'holographic' && 'text-cyan-300',
   313→              (!['neon', 'fantasy', 'retro', 'holographic'].includes(style)) && 'text-white/80'
   314→            )}>
   315→              <span aria-hidden>🎭</span>
   316→              <span className="truncate max-w-[120px]">{character.name}</span>
   317→            </div>
   318→          )}
   319→
   320→          {/* Attributes - Ensure all are visible (scrollable after ~5 rows) */}
   321→          <div className={cn('flex flex-col max-h-56 overflow-y-auto pr-0.5', compact ? 'gap-1.5' : 'gap-2.5')}>
   322→            {attributes.map((attr) => (
   323→              <AttributeHUDField
   324→                key={attr.id}
   325→                attribute={attr}
   326→                value={values[attr.key] ?? attr.defaultValue}
   327→                compact={compact}
   328→              />
   329→            ))}
   330→          </div>
   331→        </div>
   332→      ))}
   333→
   334→      {/* Persona (User) Attributes */}
   335→      {personaAttributes && (
   336→        <div
   337→          className={cn(
   338→            'pointer-events-auto rounded-xl border transition-all duration-300',
   339→            getStyleClasses(),
   340→            compact ? 'p-2' : 'p-4',
   341→            'shadow-xl'
   342→          )}
   343→        >
   344→          {/* User Persona Header */}
   345→          <div className={cn(
   346→            'font-medium mb-2 pb-1 border-b border-white/10',
   347→            compact ? 'text-[10px]' : 'text-xs',
   348→            style === 'neon' && 'text-cyan-400 uppercase tracking-wider',
   349→            style === 'fantasy' && 'text-amber-400',
   350→            style === 'retro' && 'text-green-400',
   351→            style === 'holographic' && 'text-cyan-300',
   352→            (!['neon', 'fantasy', 'retro', 'holographic'].includes(style)) && 'text-white/80'
   353→          )}>
   354→            <span aria-hidden>👤</span>
   355→            <span className="truncate max-w-[120px]">{personaAttributes.personaName}</span>
   356→            <span className="text-[9px] opacity-60 font-normal">(tú)</span>
   357→          </div>
   358→
   359→          <div className={cn('flex flex-col', compact ? 'gap-1.5' : 'gap-2.5')}>
   360→            {personaAttributes.attributes.map((attr) => (
   361→              <AttributeHUDField
   362→                key={attr.id}
   363→                attribute={attr}
   364→                value={personaAttributes.values[attr.key] ?? attr.defaultValue}
   365→                compact={compact}
   366→              />
   367→            ))}
   368→          </div>
   369→        </div>
   370→      )}
   371→    </div>
   372→  );
   373→}

   584→// ============================================
   585→// HUD Container
   586→// ============================================
   587→
   588→interface HUDContainerProps {
   589→  template: HUDTemplate;
   590→  values: Record<string, string | number | boolean>;
   591→}
   592→
   593→function HUDContainer({ template, values }: HUDContainerProps) {
   594→  const style = template.style;
   595→  const compact = template.compact;
   596→
   597→  // Style configurations with animations and effects
   598→  const getContainerClasses = () => {
   599→    const baseClasses = 'pointer-events-auto transition-all duration-300';
   600→
   601→    switch (style) {
   602→      case 'minimal':
   603→        return cn(
   604→          baseClasses,
   605→          'bg-transparent p-2',
   606→          'text-white/90'
   607→        );
   608→
   609→      case 'card':
   610→        return cn(
   611→          baseClasses,
   612→          'backdrop-blur-md rounded-xl border',
   613→          'bg-gradient-to-br from-slate-900/80 to-slate-800/60',
   614→          'border-white/10 shadow-xl shadow-black/20',
   615→          compact ? 'p-2' : 'p-4'
   616→        );
   617→

   764→interface HUDFieldDisplayProps {

   767→  compact?: boolean;

   771→function HUDFieldDisplay({ field, value, compact, style }: HUDFieldDisplayProps) {
   772→  const fieldStyle = field.style || 'default';
   773→  const color = field.color || 'default';
   774→  
   775→  const textColor = colorClasses[color] || colorClasses.default;
   776→  const bgColor = bgColorClasses[color] || bgColorClasses.default;
   777→  const progressColor = progressColorClasses[color] || progressColorClasses.default;
   778→  
   779→  // Format value
   780→  const formatValue = (): string => {
   781→    if (typeof value === 'boolean') return value ? '✓' : '✗';
   782→    if (typeof value === 'number') return String(value);
   783→    return String(value);
   784→  };
   785→  
   786→  // Render based on style
   787→  switch (fieldStyle) {
   788→    case 'progress':
   789→      return (
   790→        <ProgressField field={field} value={Number(value)} color={color} compact={compact} />
   791→      );
   792→    
   793→    case 'badge':
   794→      return (
   795→        <BadgeField field={field} value={String(value)} color={color} compact={compact} />
   796→      );
   797→    
   798→    case 'icon':
   799→      return (
   800→        <IconField field={field} value={String(value)} color={color} compact={compact} />
   801→      );
   802→    
   803→    case 'chip':
   804→      return (
   805→        <ChipField field={field} value={String(value)} color={color} compact={compact} />
   806→      );
   807→    
   808→    case 'status':
   809→      return (
   810→        <StatusField field={field} value={value} color={color} compact={compact} />
   811→      );
   812→    
   813→    case 'gauge':
   814→      return (
   815→        <GaugeField field={field} value={Number(value)} color={color} compact={compact} />
   816→      );
   817→    
   818→    case 'separator':
   819→      return (
   820→        <SeparatorField field={field} compact={compact} />
   821→      );
   822→    
   823→    case 'label-only':
   824→      return (
   825→        <LabelOnlyField field={field} color={color} compact={compact} />
   826→      );
   827→    
   828→    case 'pill':
   829→      return (
   830→        <PillField field={field} value={String(value)} color={color} compact={compact} />
   831→      );
   832→    
   833→    case 'meter':
   834→      return (
   835→        <MeterField field={field} value={Number(value)} color={color} compact={compact} />
   836→      );
   837→    
   838→    case 'dots':
   839→      return (
   840→        <DotsField field={field} value={value} color={color} compact={compact} />
   841→      );
   842→    
   843→    default:
   844→      return (
   845→        <DefaultField field={field} value={value} color={color} compact={compact} style={style} />
   846→      );
   847→  }
   848→}
