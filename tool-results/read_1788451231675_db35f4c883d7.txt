   107→
   108→function RequirementOperatorToggle({ operator, onChange, requirementCount }: RequirementOperatorToggleProps) {
   109→  if (requirementCount < 2) return null;
   110→
   111→  const currentOperator = operator || 'AND';
   112→
   113→  return (
   114→    <div className="flex items-center gap-2 py-1">
   115→      <div className="flex items-center gap-1">
   116→        <button
   117→          type="button"
   118→          className={cn(
   119→            'px-2 py-0.5 text-xs rounded border transition-colors',
   120→            currentOperator === 'AND'
   121→              ? 'bg-blue-500/20 text-blue-400 border-blue-500/30'
   122→              : 'bg-muted/30 text-muted-foreground border-transparent'
   123→          )}
   124→          onClick={() => onChange('AND')}
   125→        >
   126→          Y (AND)
   127→        </button>
   128→        <button
   129→          type="button"
   130→          className={cn(
   131→            'px-2 py-0.5 text-xs rounded border transition-colors',
   132→            currentOperator === 'OR'
   133→              ? 'bg-purple-500/20 text-purple-400 border-purple-500/30'
   134→              : 'bg-muted/30 text-muted-foreground border-transparent'
   135→          )}
   136→          onClick={() => onChange('OR')}
   137→        >
   138→          O (OR)
   139→        </button>
   140→      </div>
   141→      <span className="text-[10px] text-muted-foreground">

   143→      </span>

   152→interface RequirementEditorProps {
   153→  requirement: StatRequirement;

   160→function RequirementEditor({ requirement, availableAttributes, availableTargets = [], onChange, onDelete }: RequirementEditorProps) {
   161→  const isTargetMode = requirement.targetCharacterId !== undefined;
   162→  const selectedTarget = isTargetMode && requirement.targetCharacterId
   163→    ? availableTargets.find(t => t.id === requirement.targetCharacterId)
   164→    : undefined;
   165→  const targetAttrs = selectedTarget?.attributes || [];
   166→
   167→  // Determine the selected attribute and its type
   168→  const selectedSelfAttr = !isTargetMode ? availableAttributes.find(a => a.key === requirement.attributeKey) : undefined;
   169→  const selectedTargetAttr = isTargetMode ? targetAttrs.find(a => a.key === requirement.attributeKey) : undefined;
   170→  const selectedAttr = selectedSelfAttr || selectedTargetAttr;
   171→  const attrType = selectedAttr?.type || 'number';
   172→  const isTextType = attrType === 'text' || attrType === 'keyword';
   173→
   174→  const operatorOptions = isTextType ? TEXT_OPERATOR_OPTIONS : NUMERIC_OPERATOR_OPTIONS;
   175→  const selectedOperator = operatorOptions.find(op => op.value === requirement.operator);
   176→
   177→  const hasTargets = availableTargets.length > 0;
   178→
   179→  return (
   180→    <div className="flex items-center gap-2 bg-muted/50 rounded p-2 flex-wrap">
   181→      {/* Target/Mode indicator - only show Target option when availableTargets is provided */}
   182→      <Select
   183→        value={isTargetMode ? 'target' : 'self'}
   184→        onValueChange={(value) => {
   185→          if (value === 'target') {
   186→            onChange({ attributeKey: '', targetCharacterId: '', targetAttributeName: '', operator: '==', value: '' });
   187→          } else {
   188→            onChange({ attributeKey: '', targetCharacterId: undefined, targetAttributeName: undefined, operator: '>=', value: 0 });
   189→          }
   190→        }}
   191→      >
   192→        <SelectTrigger className="h-7 w-16 text-xs">
   193→          <SelectValue />
   194→        </SelectTrigger>
   195→        <SelectContent>
   196→          <SelectItem value="self">
   197→            <span className="flex items-center gap-1">🎭 Yo</span>
   198→          </SelectItem>
   199→          {hasTargets && (
   200→            <SelectItem value="target">
   201→              <span className="flex items-center gap-1">🎯 Target</span>
   202→            </SelectItem>
   203→          )}
   204→        </SelectContent>
   205→      </Select>
   206→
   207→      {isTargetMode ? (
   208→        <>
   209→          {/* Target selector */}
   210→          <Select
   211→            value={requirement.targetCharacterId || ''}
   212→            onValueChange={(value) => {
   213→              const target = availableTargets.find(t => t.id === value);
   214→              onChange({ targetCharacterId: value, attributeKey: '', targetAttributeName: target?.name || '' });
   215→            }}
   216→          >
   217→            <SelectTrigger className="h-7 w-28 text-xs">
   218→              <SelectValue placeholder="Target..." />
   219→            </SelectTrigger>
   220→            <SelectContent>
   221→              {availableTargets.map(t => (
   222→                <SelectItem key={t.id} value={t.id}>
   223→                  {t.id === '__user__' ? '👤 ' : '🎭 '}{t.name}
   224→                </SelectItem>
   225→              ))}
   226→            </SelectContent>
   227→          </Select>
   228→
   229→          {/* Target attribute selector */}
   230→          <Select
   231→            value={requirement.attributeKey}
   232→            onValueChange={(value) => {
   233→              const attr = targetAttrs.find(a => a.key === value);
   234→              const isText = attr?.type === 'text' || attr?.type === 'keyword';
   235→              onChange({
   236→                attributeKey: value,
   237→                targetAttributeName: attr?.name || '',
   238→                operator: isText ? '==' : '>=',
   239→                value: isText ? '' : 0,
   240→              });
   241→            }}
   242→            disabled={!requirement.targetCharacterId}
   243→          >
   244→            <SelectTrigger className="h-7 w-28 text-xs">
   245→              <SelectValue placeholder="Atributo..." />
   246→            </SelectTrigger>
   247→            <SelectContent>
   248→              {targetAttrs.map((attr, i) => (
   249→                <SelectItem key={attr.key || `attr-${i}`} value={attr.key}>
   250→                  <span className="flex items-center gap-1">
   251→                    <span className={attr.type === 'text' || attr.type === 'keyword' ? 'text-blue-400' : 'text-green-400'}>
   252→                      {attr.type === 'text' ? '📝' : attr.type === 'keyword' ? '🏷️' : '🔢'}
   253→                    </span>
   254→                    {attr.name}
   255→                  </span>
   256→                </SelectItem>
   257→              ))}
   258→            </SelectContent>
   259→          </Select>
   260→        </>
   261→      ) : (
   262→        /* Self attribute selector */
   263→        <Select
   264→          value={requirement.attributeKey}
   265→          onValueChange={(value) => {
   266→            const attr = availableAttributes.find(a => a.key === value);
   267→            const isText = attr?.type === 'text' || attr?.type === 'keyword';
   268→            onChange({
   269→              attributeKey: value,
   270→              operator: isText ? '==' : '>=',
   271→              value: isText ? '' : 0,
   272→            });
   273→          }}
   274→        >
   275→          <SelectTrigger className="h-7 w-24 text-xs">
   276→            <SelectValue placeholder="Atributo" />
   277→          </SelectTrigger>
   278→          <SelectContent>
   279→            {availableAttributes.map(attr => (
   280→              <SelectItem key={attr.id} value={attr.key}>
   281→                <span className="flex items-center gap-1">
   282→                  <span className={attr.type === 'text' || attr.type === 'keyword' ? 'text-blue-400' : 'text-green-400'}>
   283→                    {attr.type === 'text' ? '📝' : attr.type === 'keyword' ? '🏷️' : '🔢'}
   284→                  </span>
   285→                  {attr.name}
   286→                </span>
   287→              </SelectItem>
   288→            ))}
   289→          </SelectContent>
   290→        </Select>
   291→      )}
   292→
   293→      {/* Operator selector with descriptions */}
   294→      <Tooltip>
   295→        <TooltipTrigger asChild>
   296→          <Select
   297→            value={operatorOptions.some(op => op.value === requirement.operator) ? requirement.operator : operatorOptions[0].value}
   298→            onValueChange={(value: RequirementOperator) => onChange({ operator: value })}
   299→          >
   300→            <SelectTrigger className="h-7 w-16 text-xs">
   301→              <SelectValue />
   302→            </SelectTrigger>
   303→            <SelectContent>
   304→              {operatorOptions.map(op => (
   305→                <SelectItem key={op.value} value={op.value}>
   306→                  <div className="flex items-center gap-2">
   307→                    <span className="font-mono w-4">{op.label}</span>
   308→                    <span className="text-muted-foreground text-xs">{op.description}</span>
   309→                  </div>
   310→                </SelectItem>
   311→              ))}
   312→            </SelectContent>
   313→          </Select>
   314→        </TooltipTrigger>
   315→        <TooltipContent className="max-w-xs">
   316→          <p className="font-medium">{selectedOperator?.description}</p>
   317→          <p className="text-xs text-muted-foreground mt-1">
   318→            {requirement.operator === 'between'
   319→              ? `El valor debe estar entre ${requirement.value} y ${requirement.valueMax || '?'}`
   320→              : `El valor debe ser ${selectedOperator?.description} ${requirement.value}`
   321→            }
   322→          </p>
   323→        </TooltipContent>
   324→      </Tooltip>
   325→
   326→      {/* Value input - text or number based on attribute type */}
   327→      {isTextType ? (
   328→        <Input
   329→          type="text"
   330→          value={typeof requirement.value === 'string' ? requirement.value : String(requirement.value)}
   331→          onChange={(e) => onChange({ value: e.target.value })}
   332→          placeholder="Texto..."
   333→          className="h-7 w-24 text-xs"
   334→        />
   335→      ) : (
   336→        <Input
   337→          type="number"
   338→          value={requirement.value}
   339→          onChange={(e) => onChange({ value: parseFloat(e.target.value) || 0 })}
   340→          className="h-7 w-16 text-xs"
   341→        />
   342→      )}
   343→
   344→      {/* Max value for between operator (only for numeric) */}
   345→      {!isTextType && requirement.operator === 'between' && (
   346→        <>
   347→          <span className="text-xs text-muted-foreground">y</span>
   348→          <Input
   349→            type="number"
   350→            value={requirement.valueMax ?? ''}
   351→            onChange={(e) => {
   352→              const parsed = parseFloat(e.target.value);
   353→              onChange({ valueMax: isNaN(parsed) ? undefined : parsed });
   354→            }}
   355→            placeholder="max"
   356→            className="h-7 w-16 text-xs"
   357→          />
   358→        </>
   359→      )}
   360→
   361→      {/* Delete button */}
   362→      <Button variant="ghost" size="icon" className="h-7 w-7" onClick={onDelete}>
   363→        <Trash2 className="w-3 h-3 text-muted-foreground" />
   364→      </Button>
   365→    </div>
   366→  );
   367→}

   507→    setEditLabel('');
   508→    setEditResponse('');
   509→    setEditModifiers([]);
   510→    setEditSpriteActivation(undefined);

   517→    setEditLabel('');

   521→    setEditRequirements([]);
   522→    setEditRequirementOperator('AND');

   524→

   529→      ...newModifiers,
   530→      { attributeKey: attributes[0].key, operation: 'add', value: 1 },
   531→    ]);
   532→  };
   533→
   534→  const updateNewModifier = (index: number, field: keyof QuickReplyAttributeModifier, value: string | number) => {
   535→    const updated = [...newModifiers];
   536→    updated[index] = { ...updated[index], [field]: value };
   537→    setNewModifiers(updated);
   538→  };
   539→
   540→  const removeNewModifier = (index: number) => {
   541→    setNewModifiers(newModifiers.filter((_, i) => i !== index));
   542→  };
   543→
   544→  // Add a modifier to the editing reply form
   545→  const addModifierToEdit = () => {
   546→    if (attributes.length === 0) return;
   547→    setEditModifiers([
   548→      ...editModifiers,
   549→      { attributeKey: attributes[0].key, operation: 'add', value: 1 },
   550→    ]);
   551→  };
   552→
   553→  const updateEditModifier = (index: number, field: keyof QuickReplyAttributeModifier, value: string | number) => {
   554→    const updated = [...editModifiers];
   555→    updated[index] = { ...updated[index], [field]: value };
   556→    setEditModifiers(updated);
   557→  };
   558→
   559→  const removeEditModifier = (index: number) => {
   560→    setEditModifiers(editModifiers.filter((_, i) => i !== index));
