    98→    98→/**
    99→    99→ * Get characters that have pending solicitudes (requests they need to respond to)
   100→   100→ */
   101→   101→function getCharactersWithPendingSolicitudes(
   102→   102→  sessionStats: SessionStats | undefined,
   103→   103→  eligibleCharacterIds: string[]
   104→   104→): { characterId: string; fromCharacterName: string }[] {
   105→   105→  if (!sessionStats?.solicitudes?.characterSolicitudes) {
   106→   106→    return [];
   107→   107→  }
   108→   108→
   109→   109→  const result: { characterId: string; fromCharacterName: string }[] = [];
   110→   110→
   111→   111→  for (const characterId of eligibleCharacterIds) {
   112→   112→    const solicitudes = sessionStats.solicitudes.characterSolicitudes[characterId];
   113→   113→    if (solicitudes) {
   114→   114→      const pendingSolicitudes = solicitudes.filter(s => s.status === 'pending');
   115→   115→      if (pendingSolicitudes.length > 0) {
   116→   116→        result.push({
   117→   117→          characterId,
   118→   118→          fromCharacterName: pendingSolicitudes[0].fromCharacterName
   119→   119→        });
   120→   120→      }
   121→   121→    }
   122→   122→  }
   123→   123→
   124→   124→  return result;
   125→   125→}

   127→   127→/**
   128→   128→ * Check if there are pending solicitudes targeting the user

   130→   130→function hasPendingSolicitudesForUser(
   131→   131→  sessionStats: SessionStats | undefined
   132→   132→): boolean {
   133→   133→  if (!sessionStats?.solicitudes?.characterSolicitudes) {
   134→   134→    return false;
   135→   135→  }
   136→   136→
   137→   137→  const userSolicitudes = sessionStats.solicitudes.characterSolicitudes['__user__'];
   138→   138→  return userSolicitudes?.some(s => s.status === 'pending') ?? false;
   139→   139→}

   141→   141→/**
   142→   142→ * Determine responders based on strategy
   143→   143→ * Note: Narrators are excluded from normal response flow and handled separately
   144→   144→ */
   145→   145→function getResponders(
   146→   146→  message: string,
   147→   147→  characters: CharacterCard[],
   148→   148→  group: CharacterGroup,
   149→   149→  lastResponderId?: string,
   150→   150→  sessionStats?: SessionStats
   151→   151→): ResponderSelectionResult {
   152→   152→  const strategy = group.activationStrategy;
   153→   153→  const minResponses = group.minResponsesPerTurn ?? 1;
   154→   154→  const maxResponses = group.maxResponsesPerTurn ?? 3;
   155→   155→
   156→   156→  // Get active members, EXCLUDING narrators (they have their own response logic)
   157→   157→  const activeMemberIds = (group.members || [])
   158→   158→    .filter(m => m.isActive && m.isPresent !== false && !m.isNarrator)
   159→   159→    .map(m => m.characterId);
   160→   160→
   161→   161→  // If no members defined, use characterIds (excluding narrators)
   162→   162→  const eligibleIds = activeMemberIds.length > 0
   163→   163→    ? activeMemberIds
   164→   164→    : (group.characterIds || []).filter(id => {
   165→   165→        const member = (group.members || []).find(m => m.characterId === id);
   166→   166→        return !member?.isNarrator;
   167→   167→      });
   168→   168→
   169→   169→  // Filter to only characters that exist and are eligible
   170→   170→  const eligibleCharacters = characters.filter(c => eligibleIds.includes(c.id));
   171→   171→
   172→   172→  if (eligibleCharacters.length === 0) {
   173→   173→    return { responders: [], stopForUser: false, reasons: new Map() };
   174→   174→  }
   175→   175→
   176→   176→  // Get ordered member IDs (excluding narrators)
   177→   177→  const orderedIds = (group.members || [])
   178→   178→    .filter(m => !m.isNarrator)
   179→   179→    .sort((a, b) => a.joinOrder - b.joinOrder)
   180→   180→    .map(m => m.characterId);
   181→   181→
   182→   182→  const reasons = new Map<string, string>();
   183→   183→
   184→   184→  switch (strategy) {
   185→   185→    case 'all': {
   186→   186→      // All active members respond (no limit for 'all' strategy)
   187→   187→      eligibleCharacters.forEach(c => reasons.set(c.id, 'Todos responden'));
   188→   188→      return { responders: eligibleCharacters, stopForUser: false, reasons };
   189→   189→    }
   190→   190→
   191→   191→    case 'reactive': {
   192→   192→      // ========================================
   193→   193→      // REACTIVE STRATEGY with Solicitud Support
   194→   194→      // ========================================
   195→   195→      // Priority:
   196→   196→      // 1. If peticion targets user -> STOP, let user respond
   197→   197→      // 2. Characters with pending solicitudes respond
   198→   198→      // 3. Mentioned characters respond
   199→   199→      // 4. Fill to minResponses if needed
   200→   200→
   201→   201→      // Check if user has pending solicitudes
   202→   202→      const stopForUser = hasPendingSolicitudesForUser(sessionStats);
   203→   203→      if (stopForUser) {
   204→   204→        console.log('[getResponders] Peticion targets user, stopping turn for user response');
   205→   205→        return { responders: [], stopForUser: true, reasons: new Map() };
   206→   206→      }
   207→   207→
   208→   208→      // Detect mentions
   209→   209→      const mentions = detectMentions(message, characters, group);
   210→   210→      const mentionedIds = mentions.map(m => m.characterId);
   211→   211→      const mentionedCharacters = eligibleCharacters.filter(c => mentionedIds.includes(c.id));
   212→   212→      mentionedCharacters.forEach(c => reasons.set(c.id, 'Mencionado en el mensaje'));
   213→   213→
   214→   214→      // Get characters with pending solicitudes
   215→   215→      const charactersWithSolicitudes = getCharactersWithPendingSolicitudes(sessionStats, eligibleIds);
   216→   216→      const solicitudCharacterIds = charactersWithSolicitudes.map(s => s.characterId);
   217→   217→      const solicitudCharacters = eligibleCharacters.filter(c => solicitudCharacterIds.includes(c.id));
   218→   218→
   219→   219→      // Add solicitud reasons (don't overwrite mention reasons)
   220→   220→      solicitudCharacters.forEach(c => {
   221→   221→        if (!reasons.has(c.id)) {
   222→   222→          const solicitudInfo = charactersWithSolicitudes.find(s => s.characterId === c.id);
   223→   223→          reasons.set(c.id, `Tiene solicitud pendiente de ${solicitudInfo?.fromCharacterName || 'otro personaje'}`);
   224→   224→        }
   225→   225→      });
   226→   226→
   227→   227→      // Combine: unique characters from mentions and solicitudes
   228→   228→      const combinedIds = new Set([...mentionedIds, ...solicitudCharacterIds]);
   229→   229→      let selectedCharacters = eligibleCharacters.filter(c => combinedIds.has(c.id));
   230→   230→
   231→   231→      // If no mentions or solicitudes, fill to minResponses
   232→   232→      if (selectedCharacters.length === 0) {
   233→   233→        // Select first eligible character as default
   234→   234→        selectedCharacters = [eligibleCharacters[0]];
   235→   235→        reasons.set(eligibleCharacters[0].id, 'Personaje por defecto (sin menciones ni solicitudes)');
   236→   236→      }
   237→   237→
   238→   238→      // Ensure we have at least minResponses (but respect maxResponses)
   239→   239→      if (selectedCharacters.length < minResponses) {
   240→   240→        const remaining = eligibleCharacters
   241→   241→          .filter(c => !combinedIds.has(c.id))
   242→   242→          .slice(0, minResponses - selectedCharacters.length);
   243→   243→        remaining.forEach(c => reasons.set(c.id, 'Para alcanzar mínimo de respuestas'));
   244→   244→        selectedCharacters = [...selectedCharacters, ...remaining];
   245→   245→      }
   246→   246→
   247→   247→      // Limit to maxResponses
   248→   248→      const limitedResponders = selectedCharacters.slice(0, maxResponses);
   249→   249→
   250→   250→      console.log('[getResponders] Reactive selection:', {
   251→   251→        mentionedIds,
   252→   252→        solicitudCharacterIds,
   253→   253→        selectedCount: limitedResponders.length,
   254→   254→        minResponses,
   255→   255→        maxResponses,
   256→   256→        stopForUser
   257→   257→      });
   258→   258→
   259→   259→      return { responders: limitedResponders, stopForUser, reasons };
   260→   260→    }
   261→   261→
   262→   262→    case 'round_robin': {
   263→   263→      // Take turns in order
   264→   264→      const sortedIds = orderedIds.length > 0 ? orderedIds : eligibleIds;
   265→   265→
   266→   266→      let nextIndex = 0;
   267→   267→      if (lastResponderId) {
   268→   268→        const lastIndex = sortedIds.indexOf(lastResponderId);
   269→   269→        if (lastIndex !== -1) {
   270→   270→          nextIndex = (lastIndex + 1) % sortedIds.length;
   271→   271→        }
   272→   272→      }
   273→   273→
   274→   274→      const roundRobinChar = characters.find(c => c.id === sortedIds[nextIndex]);
   275→   275→      if (roundRobinChar) {
   276→   276→        reasons.set(roundRobinChar.id, 'Turno rotativo');
   277→   277→      }
   278→   278→      return { responders: roundRobinChar ? [roundRobinChar] : [], stopForUser: false, reasons };
   279→   279→    }
   280→   280→
   281→   281→    case 'random': {
   282→   282→      // Random selection
   283→   283→      const shuffled = [...eligibleCharacters].sort(() => Math.random() - 0.5);
   284→   284→      const selected = shuffled.slice(0, Math.min(maxResponses, shuffled.length));
   285→   285→      selected.forEach(c => reasons.set(c.id, 'Selección aleatoria'));
   286→   286→      return { responders: selected, stopForUser: false, reasons };
   287→   287→    }
   288→   288→
   289→   289→    case 'smart': {
   290→   290→      // AI-like decision: mentioned characters + contextually relevant
   291→   291→      const mentions = detectMentions(message, characters, group);
   292→   292→      const mentionedIds = mentions.map(m => m.characterId);
   293→   293→      const mentionedChars = eligibleCharacters.filter(c => mentionedIds.includes(c.id));
   294→   294→      mentionedChars.forEach(c => reasons.set(c.id, 'Mencionado en el mensaje'));
   295→   295→
   296→   296→      // Add contextually relevant characters
   297→   297→      const remainingChars = eligibleCharacters.filter(c => !mentionedIds.includes(c.id));
   298→   298→      const additionalCount = Math.max(0, Math.min(maxResponses - mentionedChars.length, 1));
   299→   299→
   300→   300→      // Check if character name or tags appear in message
   301→   301→      const relevantChars = remainingChars.filter(c => {
   302→   302→        const keywords = [...c.tags, c.name.toLowerCase()];
   303→   303→        return keywords.some(kw => message.toLowerCase().includes(kw.toLowerCase()));
   304→   304→      }).slice(0, additionalCount);
   305→   305→      relevantChars.forEach(c => reasons.set(c.id, 'Contextualmente relevante'));
   306→   306→
   307→   307→      const result = [...mentionedChars, ...relevantChars].slice(0, maxResponses);
   308→   308→
   309→   309→      // If no one selected, default to first
   310→   310→      if (result.length === 0 && eligibleCharacters.length > 0) {
   311→   311→        result.push(eligibleCharacters[0]);
   312→   312→        reasons.set(eligibleCharacters[0].id, 'Personaje por defecto');
   313→   313→      }
   314→   314→
   315→   315→      return { responders: result, stopForUser: false, reasons };
   316→   316→    }
   317→   317→
   318→   318→    default: {
   319→   319→      // Default to first active character
   320→   320→      if (eligibleCharacters.length > 0) {
   321→   321→        reasons.set(eligibleCharacters[0].id, 'Personaje por defecto');
   322→   322→      }
   323→   323→      return { responders: eligibleCharacters.slice(0, 1), stopForUser: false, reasons };
   324→   324→    }
   325→   325→  }
   326→   326→}

   338→   338→/**
   339→   339→ * Execute detected tool calls for group chat and send SSE events.
   340→   340→ * Returns { results: display messages, shouldContinue: true, questActivations: quest activations }
   341→   341→ */
   342→   342→async function executeGroupToolCalls(
   343→   343→  toolCalls: NativeToolCall[],
   344→   344→  availableTools: Array<{ id: string; name: string; label: string; icon: string }>,
   345→   345→  character: CharacterCard,
   346→   346→  sessionId: string,
   347→   347→  userName: string,
   348→   348→  controller: { enqueue: (chunk: string) => void },
   349→   349→  sessionQuests?: SessionQuestInstance[],
   350→   350→  questTemplates?: QuestTemplate[],
   351→   351→  statsConfig?: CharacterStatsConfig,
   352→   352→  sessionStats?: SessionStats,
   353→   353→  allCharacters?: CharacterCard[],
   354→   354→  characterMemory?: CharacterMemory,
   355→   355→  lorebooks?: Lorebook[],
   356→   356→): Promise<{ results: string; shouldContinue: boolean; questActivations: QuestActivation[]; toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> }> {
   357→   357→  if (toolCalls.length === 0) {
   358→   358→    return { results: '', shouldContinue: false, questActivations: [], toolsUsed: [] };
   359→   359→  }
   360→   360→
   361→   361→  let allDisplayMessages = '';
   362→   362→  const questActivations: QuestActivation[] = [];
   363→   363→  const toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> = [];
   364→   364→
   365→   365→  for (let i = 0; i < toolCalls.length; i++) {
   366→   366→    const tc = toolCalls[i];
   367→   367→    const toolDef = availableTools.find(t => t.name === tc.name);
   368→   368→    const callId = `${Date.now()}_${i}`;
   369→   369→
   370→   370→    // Send tool_call_start event
   371→   371→    controller.enqueue(createSSEJSON({
   372→   372→      type: 'tool_call_start',
   373→   373→      toolName: tc.name,
   374→   374→      toolLabel: toolDef?.label || tc.name,
   375→   375→      toolIcon: toolDef?.icon || 'Wrench',
   376→   376→      params: tc.arguments,
   377→   377→      callId,
   378→   378→    }));
   379→   379→
   380→   380→    console.log(`[GroupStream-Tools] Executing: ${tc.name}`, tc.arguments);
   381→   381→
   382→   382→    // Execute the tool
   383→   383→    const toolResult = await executeTool(
   384→   384→      tc.name,
   385→   385→      tc.arguments,
   386→   386→      {
   387→   387→        characterId: character.id,
   388→   388→        characterName: character.name,
   389→   389→        sessionId,
   390→   390→        userName,
   391→   391→        sessionQuests,
   392→   392→        questTemplates,
   393→   393→        statsConfig: character.statsConfig,
   394→   394→        sessionStats,
   395→   395→        allCharacters,
   396→   396→        characterMemory,
   397→   397→        lorebooks,
   398→   398→      },
   399→   399→    );
   400→   400→
   401→   401→    // Check for quest activation and send SSE event
   402→   402→    if (toolResult.questActivation) {
   403→   403→      const activation = toolResult.questActivation;
   404→   404→      console.log(`[GroupStream-Tools] Quest activation from ${tc.name}:`, activation);
   405→   405→      
   406→   406→      controller.enqueue(createSSEJSON({
   407→   407→        type: 'quest_activation',
   408→   408→        toolName: tc.name,
   409→   409→        activationType: activation.type,
   410→   410→        key: activation.key,
   411→   411→        metadata: activation.metadata,
   412→   412→      }));
   413→   413→      
   414→   414→      questActivations.push(activation);
   415→   415→    }
   416→   416→
   417→   417→    // Check for action/skill activation and send SSE event
   418→   418→    if (toolResult.actionActivation) {
   419→   419→      const action = toolResult.actionActivation;
   420→   420→      console.log(`[GroupStream-Tools] Action activation from ${tc.name}:`, action.skillName);
   421→   421→      
   422→   422→      controller.enqueue(createSSEJSON({
   423→   423→        type: 'action_activation',
   424→   424→        toolName: tc.name,
   425→   425→        skillId: action.skillId,
   426→   426→        skillName: action.skillName,
   427→   427→        skillDescription: action.skillDescription,
   428→   428→        skillCompletedDescription: action.skillCompletedDescription,
   429→   429→        activationCosts: action.activationCosts,
   430→   430→        activationRewards: action.activationRewards,
   431→   431→        characterId: action.characterId,
   432→   432→      }));
   433→   433→    }
   434→   434→
   435→   435→    // Check for stat modification from modify_stat tool and send SSE event
   436→   436→    if (toolResult.statActivation) {
   437→   437→      const stat = toolResult.statActivation;
   438→   438→      console.log(`[GroupStream-Tools] Stat activation from ${tc.name}:`, stat.attributeKey, stat.oldValue, '→', stat.newValue);
   439→   439→      
   440→   440→      controller.enqueue(createSSEJSON({
   441→   441→        type: 'stat_activation',
   442→   442→        toolName: tc.name,
   443→   443→        characterId: stat.characterId,
   444→   444→        attributeKey: stat.attributeKey,
   445→   445→        attributeName: stat.attributeName,
   446→   446→        attributeType: stat.attributeType,
   447→   447→        oldValue: stat.oldValue,
   448→   448→        newValue: stat.newValue,
   449→   449→        reason: stat.reason,
   450→   450→      }));
   451→   451→    }
   452→   452→
   453→   453→    // Check for solicitud activation/completion and send SSE event
   454→   454→    if (toolResult.solicitudActivation) {
   455→   455→      const sol = toolResult.solicitudActivation;
   456→   456→      console.log(`[GroupStream-Tools] Solicitud activation from ${tc.name}:`, sol.type, sol.solicitudKey);
   457→   457→      
   458→   458→      controller.enqueue(createSSEJSON({
   459→   459→        type: 'solicitud_activation',
   460→   460→        toolName: tc.name,
   461→   461→        activationType: sol.type,
   462→   462→        solicitudKey: sol.solicitudKey,
   463→   463→        targetCharacterId: sol.targetCharacterId,
   464→   464→        targetCharacterName: sol.targetCharacterName,
   465→   465→        fromCharacterId: sol.fromCharacterId,
   466→   466→        fromCharacterName: sol.fromCharacterName,
   467→   467→        description: sol.description,
   468→   468→        completionDescription: sol.completionDescription,
   469→   469→        peticionKey: sol.peticionKey,
   470→   470→      }));
   471→   471→    }
   472→   472→
   473→   473→    // Check for memory activation (sync to client-side Character Memory)
   474→   474→    if (toolResult.memoryActivation) {
   475→   475→      const mem = toolResult.memoryActivation;
