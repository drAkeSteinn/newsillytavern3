    33→// e.g., Objective A reward completes Objective B, whose reward completes Objective A
    34→const _processingCompletions = new Set<string>();

    36→// ============================================
    37→// Shared Objective Key Matching
    38→// ============================================
    39→// Unified matching logic for all paths (tool, POST-LLM, cascading rewards).
    40→// Matches: exact, case-insensitive, prefix ("obj-" + key), partial substring.
    41→
    42→function objectiveKeyMatches(completionKey: string, searchKey: string): boolean {
    43→  if (!completionKey || !searchKey) return false;
    44→  // Exact match
    45→  if (completionKey === searchKey) return true;
    46→  // Case-insensitive
    47→  const ck = completionKey.toLowerCase();
    48→  const sk = searchKey.toLowerCase();
    49→  if (ck === sk) return true;
    50→  // Prefix match (e.g. completion key = "obj-troncos_abedul", search = "troncos_abedul")
    51→  if (ck === `obj-${sk}`) return true;
    52→  // Partial substring match (fallback)
    53→  if (ck.includes(sk) || sk.includes(ck)) return true;
    54→  return false;
    55→}

    57→/**
    58→ * Shared logic to find and complete an objective by completion key.
    59→ * Used by activateSkillByTool and executeCompletionRewards closures.
    60→ * Returns true if the objective was found and completed, false otherwise.
    61→ */
    62→function findAndCompleteObjectiveByKey(
    63→  get: () => any,
    64→  sessionId: string,
    65→  questId: string,
    66→  objectiveKey: string,
    67→  characterId: string,
    68→): boolean {
    69→  const session = get().sessions.find((s: ChatSession) => s.id === sessionId);
    70→  if (!session) {
    71→    console.warn(`[findAndCompleteObjectiveByKey] Session not found: ${sessionId}`);
    72→    return false;
    73→  }
    74→
    75→  const templates = get().questTemplates || [];
    76→  const sessionQuests = session.sessionQuests || [];
    77→
    78→  console.log(`[findAndCompleteObjectiveByKey] Searching for objective key "${objectiveKey}" in quest ${questId || '(any)'}, session=${sessionId}`);
    79→  console.log(`[findAndCompleteObjectiveByKey] Active quests in session:`, sessionQuests.map(q => `${q.templateId} (${q.status})`).join(', '));
    80→
    81→  // Phase 1: Search with questId filter (if provided)
    82→  for (const quest of sessionQuests) {
    83→    if (quest.status !== 'active' && quest.status !== 'available') continue;
    84→    if (questId && quest.templateId !== questId) continue;
    85→
    86→    const tmpl = templates.find((t: QuestTemplate) => t.id === quest.templateId);
    87→    if (!tmpl) {
    88→      console.warn(`[findAndCompleteObjectiveByKey] Template "${quest.templateId}" not found in store. Available templates:`, templates.map(t => t.id).join(', '));
    89→      continue;
    90→    }
    91→
    92→    if (tryCompleteObjectiveInQuest(get, sessionId, quest, tmpl, objectiveKey, characterId)) {
    93→      return true;
    94→    }
    95→  }
    96→
    97→  // Phase 2: Fallback — search ALL active quests (in case questId is stale/wrong)
    98→  if (questId) {
    99→    console.log(`[findAndCompleteObjectiveByKey] QuestId "${questId}" filter didn't match. Retrying without filter...`);
   100→    for (const quest of sessionQuests) {
   101→      if (quest.status !== 'active' && quest.status !== 'available') continue;
   102→
   103→      const tmpl = templates.find((t: QuestTemplate) => t.id === quest.templateId);
   104→      if (!tmpl) continue;
   105→
   106→      if (tryCompleteObjectiveInQuest(get, sessionId, quest, tmpl, objectiveKey, characterId)) {
   107→        console.log(`[findAndCompleteObjectiveByKey] Found objective in fallback search (quest ${quest.templateId} instead of ${questId})`);
   108→        return true;
   109→      }
   110→    }
   111→  }
   112→
   113→  console.warn(`[findAndCompleteObjectiveByKey] No active objective found with key "${objectiveKey}" in any active quest`);
   114→  return false;
   115→}

   117→function tryCompleteObjectiveInQuest(
   118→  get: () => any,
   119→  sessionId: string,
   120→  quest: SessionQuestInstance,
   121→  template: QuestTemplate,
   122→  objectiveKey: string,
   123→  characterId: string,
   124→): boolean {
   125→  for (const obj of template.objectives || []) {
   126→    const completionKeys = [
   127→      obj.completion?.key,
   128→      ...(obj.completion?.keys || []),
   129→    ].filter(Boolean);
   130→
   131→    const matched = completionKeys.some((k: string) => objectiveKeyMatches(k, objectiveKey));
   132→    if (!matched) continue;
   133→
   134→    // Check if already completed in session
   135→    const sessionObj = quest.objectives?.find((o: any) => o.templateId === obj.id);
   136→    if (sessionObj?.isCompleted) {
   137→      console.log(`[findAndCompleteObjectiveByKey] Objective "${obj.description || objectiveKey}" already completed, skipping.`);
   138→      return true; // Return true so caller doesn't keep searching
   139→    }
   140→
   141→    console.log(`[findAndCompleteObjectiveByKey] ✓ Found matching objective "${obj.description || objectiveKey}" in quest "${template.name}" (${quest.templateId})`);
   142→    // Use progressQuestObjective (same path quest-detector uses)
   143→    get().progressQuestObjective?.(sessionId, quest.templateId, obj.id, 999, characterId);
   144→    return true;
   145→  }
   146→  return false;
   147→}

   149→/**
   150→ * Execute rewards after objective or quest completion.
   151→ * Called from store functions to ensure rewards run for ALL callers (UI, trigger system, tools).
   152→ * 
   153→ * Flow:
   154→ * 1. Execute objective rewards (if objectiveId provided)
   155→ * 2. Check if quest auto-completed → execute quest rewards
   156→ */
   157→function executeCompletionRewards(
   158→  get: () => any,
   159→  sessionId: string,
   160→  questTemplateId: string,
   161→  objectiveId?: string,
   162→  characterId?: string,
   163→): void {
   164→  const guardKey = objectiveId 
   165→    ? `${sessionId}:${questTemplateId}:${objectiveId}`
   166→    : `${sessionId}:${questTemplateId}:QUEST`;
   167→  
   168→  if (_processingCompletions.has(guardKey)) {
   169→    console.log(`[Quest Rewards] Skipping (already processing): ${guardKey}`);
   170→    return;
   171→  }
   172→  _processingCompletions.add(guardKey);
   173→  
   174→  try {
   175→    const session = get().sessions.find((s: ChatSession) => s.id === sessionId);
   176→    if (!session) return;
   177→    
   178→    const template = get().getTemplateById?.(questTemplateId);
   179→    if (!template) return;
   180→    
   181→    const resolvedCharacterId = characterId || session.characterId || '';
   182→    const character = get().getCharacterById?.(resolvedCharacterId);
   183→    
   184→    // Build allCharacters for group chat
   185→    let allCharacters: CharacterCard[] = [];
   186→    if (session.groupId) {
   187→      const group = get().getGroupById?.(session.groupId);
   188→      if (group?.members) {
   189→        allCharacters = group.members
   190→          .map((m: any) => get().getCharacterById(m.characterId))
   191→          .filter((c: any): c is CharacterCard => c !== undefined);
   192→      }
   193→    } else if (character) {
   194→      allCharacters = [character];
   195→    }
   196→    
   197→    const context: RewardExecutionContext = {
   198→      sessionId,
   199→      characterId: resolvedCharacterId,
   200→      character,
   201→      allCharacters,
   202→      sessionStats: session.sessionStats,
   203→      timestamp: Date.now(),
   204→      soundCollections: get().soundCollections,
   205→      soundTriggers: get().soundTriggers,
   206→      soundSequenceTriggers: get().soundSequenceTriggers,
   207→      backgroundPacks: get().backgroundTriggerPacks,
   208→      soundSettings: {
   209→        enabled: get().settings?.sound?.enabled ?? false,
   210→        globalVolume: get().settings?.sound?.globalVolume ?? 0.85,
   211→      },
   212→      backgroundSettings: {
   213→        transitionDuration: get().settings?.backgroundTriggers?.transitionDuration ?? 500,
   214→        defaultTransitionType: get().settings?.backgroundTriggers?.defaultTransitionType ?? 'fade',
   215→      },
   216→    };
   217→    
   218→    const actions: RewardStoreActions = {
   219→      updateCharacterStat: (sid: string, cid: string, key: string, value: number | string, reason?: string) => {
   220→        get().updateCharacterStat?.(sid, cid, key, value, reason as any);
   221→      },
   222→      // Allow chained objective rewards (objective reward that completes another objective)
   223→      completeQuestObjective: (sid: string, qid: string, objKey: string, cid?: string) => {
   224→        return findAndCompleteObjectiveByKey(get, sid, qid || '', objKey, cid || resolvedCharacterId);
   225→      },
   226→      // Allow solicitud rewards from quest objectives
   227→      completeSolicitud: (sid: string, cid: string, solicitudKey: string) => {
   228→        return get().completeSolicitud?.(sid, cid, solicitudKey) || null;
   229→      },
   230→      applyTriggerForCharacter: (cid: string, hit: any) => {
   231→        get().applyTriggerForCharacter?.(cid, hit);
   232→      },
   233→      scheduleReturnToIdleForCharacter: (cid: string, triggerSpriteUrl: string, returnToMode: any, returnSpriteUrl: string, returnSpriteLabel: string | null, returnToIdleMs: number) => {
   234→        get().scheduleReturnToIdleForCharacter?.(cid, triggerSpriteUrl, returnToMode, returnSpriteUrl, returnSpriteLabel, returnToIdleMs);
   235→      },
   236→      isSpriteLocked: () => get().isSpriteLocked?.() ?? false,
   237→      playSound: (collection: string, filename: string, volume?: number) => {
   238→        get().playSound?.(collection, filename, volume);
   239→      },
   240→      setBackground: (url: string) => {
   241→        get().setBackground?.(url);
   242→      },
   243→      setActiveOverlays: (overlays: any) => {
   244→        get().setActiveOverlays?.(overlays);
   245→      },
   246→    };
   247→    
   248→    // Step 1: Execute objective rewards (if an objective was completed)
   249→    if (objectiveId) {
   250→      const targetObjective = template.objectives.find((o: any) => o.id === objectiveId);
   251→      if (targetObjective?.rewards && targetObjective.rewards.length > 0) {
   252→        console.log(`[Quest Rewards] Executing ${targetObjective.rewards.length} objective rewards for "${targetObjective.description}"`);
   253→        const objResult = executeObjectiveRewards(targetObjective.rewards, context, actions);
   254→        console.log(`[Quest Rewards] Objective rewards: ${objResult.successCount} succeeded, ${objResult.failureCount} failed`);
   255→        
   256→        // Add notification with objective reward details
   257→        if (objResult.successCount > 0 && get().questSettings?.showNotifications) {
   258→          const rewardMessages = objResult.results
   259→            .filter((r: any) => r.success)
   260→            .map((r: any) => r.message)
   261→            .filter(Boolean);
   262→          if (rewardMessages.length > 0) {
   263→            get().addQuestNotification?.({
   264→              questId: questTemplateId,
   265→              questName: template.name,
   266→              type: 'objective_complete',
   267→              message: `Objetivo completado: ${targetObjective.description}. Recompensas: ${rewardMessages.join(', ')}`,
   268→              rewards: targetObjective.rewards,
   269→            });
   270→          }
   271→        }
   272→      }
   273→    }
   274→    
   275→    // Step 2: Check if quest was auto-completed → execute quest rewards
   276→    const updatedSession = get().sessions.find((s: ChatSession) => s.id === sessionId);
   277→    const updatedQuest = updatedSession?.sessionQuests?.find(
   278→      (q: SessionQuestInstance) => q.templateId === questTemplateId
   279→    );
   280→    
   281→    if (updatedQuest?.status === 'completed' && template.rewards && template.rewards.length > 0) {
   282→      console.log(`[Quest Rewards] Quest "${template.name}" completed! Executing ${template.rewards.length} quest rewards`);
   283→      const questResult = executeQuestCompletionRewards(template, context, actions);
   284→      console.log(`[Quest Rewards] Quest rewards: ${questResult.successCount} succeeded, ${questResult.failureCount} failed`);
   285→      
   286→      // Add detailed notification with quest reward info
   287→      if (questResult.successCount > 0 && get().questSettings?.showNotifications) {
   288→        const rewardMessages = questResult.results
   289→          .filter((r: any) => r.success)
   290→          .map((r: any) => r.message)
   291→          .filter(Boolean);
   292→        if (rewardMessages.length > 0) {
   293→          get().addQuestNotification?.({
   294→            questId: questTemplateId,
   295→            questName: template.name,
   296→            type: 'quest_complete',
   297→            message: `¡Misión completada: ${template.name}! Recompensas: ${rewardMessages.join(', ')}`,
   298→            rewards: template.rewards,
   299→          });
   300→        }
   301→      }
   302→    }
   303→  } catch (err) {
   304→    console.error('[Quest Rewards] Error executing completion rewards:', err);
   305→  } finally {
   306→    _processingCompletions.delete(guardKey);
   307→  }
   308→}

   314→/**
   315→ * Create default character stats from statsConfig
   316→ */
   317→function createDefaultCharacterStats(
   318→  statsConfig?: { enabled?: boolean; attributes?: Array<{ key: string; defaultValue: number | string }> }
   319→): CharacterSessionStats {
   320→  const attributeValues: Record<string, number | string> = {};
   321→  const lastUpdated: Record<string, number> = {};
   322→  const now = Date.now();
   323→  
   324→  if (statsConfig?.enabled && statsConfig.attributes) {
   325→    for (const attr of statsConfig.attributes) {
   326→      attributeValues[attr.key] = attr.defaultValue;
   327→      lastUpdated[attr.key] = now;
   328→    }
   329→  }
   330→  
   331→  return {
   332→    attributeValues,
   333→    lastUpdated,
   334→    changeLog: [],
   335→  };
   336→}

   338→/**
   339→ * Add equipment slot defaults to persona session stats.
   340→ * Slots are initialized as empty text attributes so {{slotkey}} resolves in prompts.
   341→ */
   342→function addEquipmentSlotsToPersonaStats(
   343→  stats: CharacterSessionStats,
   344→  equipmentSlots?: EquipmentSlotDefinition[]
   345→): CharacterSessionStats {
   346→  if (!equipmentSlots || equipmentSlots.length === 0) return stats;
   347→  const now = Date.now();
   348→  const updated = { ...stats };
   349→  updated.attributeValues = { ...stats.attributeValues };
   350→  updated.lastUpdated = { ...stats.lastUpdated };
   351→  for (const slot of equipmentSlots) {
   352→    // Don't overwrite existing values (could be set by equip/unequip)
   353→    if (!(slot.key in updated.attributeValues)) {
   354→      updated.attributeValues[slot.key] = '';
   355→      updated.lastUpdated[slot.key] = now;
   356→    }
   357→  }
   358→  return updated;
   359→}

   361→/**
   362→ * Initialize session stats for a character or group of characters
   363→ * Resets all stats, solicitudes, and session events to default values
   364→ */
   365→function initializeSessionStatsForCharacters(
   366→  characters: Array<{ id: string; statsConfig?: { enabled?: boolean; attributes?: Array<{ key: string; defaultValue: number | string }> }; emotionalConfig?: { enabled?: boolean; initialState?: string } }>
   367→): SessionStats {
   368→  const now = Date.now();
   369→  const characterStats: Record<string, CharacterSessionStats> = {};
   370→  
   371→  for (const char of characters) {
   372→    const stats = createDefaultCharacterStats(char.statsConfig);
   373→    // FASE 5: Initialize emotional state if emotional config is present
   374→    if (char.emotionalConfig?.enabled && char.emotionalConfig.initialState) {
   375→      stats.emotionalState = char.emotionalConfig.initialState;
   376→      stats.emotionalStateLastEval = now;
   377→      stats.emotionalStateTurnCount = 0;
   378→      // Also set as attribute value for {{emocion}} key resolution
   379→      stats.attributeValues['emocion'] = char.emotionalConfig.initialState;
   380→    }
   381→    characterStats[char.id] = stats;
   382→  }
   383→  
   384→  return {
   385→    characterStats,
   386→    solicitudes: {
   387→      characterSolicitudes: {},
   388→      lastModified: now,
   389→    },
   390→    // Reset session events to undefined (clean state)
   391→    ultimo_objetivo_completado: undefined,
   392→    ultima_solicitud_completada: undefined,
   393→    ultima_solicitud_realizada: undefined,
   394→    ultima_accion_realizada: undefined,
   395→    initialized: true,
   396→    lastModified: now,
   397→    // Reset timer state - start fresh from now
   398→    lastTimerUpdate: now,
   399→    keywordCycleIndex: {},
   400→  };
   401→}

   407→/**
   408→ * Create quest instances from templates
   409→ * Creates 'available' status quests (not yet activated)
   410→ * 'automatic' quests without prerequisites AND with 'normal' activation type start as 'active' immediately
   411→ * Quests with by_attribute/by_objective activation types always start as 'available'
   412→ */
   413→function createQuestInstancesFromTemplates(
   414→  templates: QuestTemplate[]
   415→): SessionQuestInstance[] {
   416→  return templates.map(template => {
   417→    const isAutomatic = template.activation.method === 'automatic';
   418→    const hasPrerequisites = template.prerequisites && template.prerequisites.length > 0;
   419→    const activationType = template.activation?.activationType || 'normal';
   420→
   421→    // Automatic quests without prerequisites AND with normal activation type become active immediately.
   422→    // Quests with by_attribute or by_objective activation types must start as 'available'
   423→    // and wait for their conditions to be evaluated before being activated.
   424→    const isActive = isAutomatic && !hasPrerequisites && activationType === 'normal';
   425→
   426→    return {
   427→      templateId: template.id,
   428→      status: isActive ? ('active' as const) : ('available' as const),
   429→      objectives: template.objectives.map(obj => ({
   430→        templateId: obj.id,
   431→        currentCount: 0,
   432→        isCompleted: false,
   433→        isVisible: (obj.visibilityType || 'normal') === 'normal', // normal = always visible, conditional = starts hidden until evaluated
   434→      })),
   435→      progress: 0,
   436→      activatedAt: isActive ? new Date().toISOString() : undefined,
   437→      completedAt: undefined,
   438→      activatedAtTurn: isActive ? 0 : undefined,
   439→    };
   440→  });
   441→}

   443→/**
   444→ * Check and auto-activate quests whose prerequisites are now met.
   445→ * Called after any quest or objective completion.
   446→ * 
   447→ * Activates quests that:
   448→ * - Are in 'available' status
   449→ * - Have prerequisites, and ALL prerequisites are completed
   450→ * - Are NOT already the quest that just completed
   451→ * 
   452→ * This handles both:
   453→ * - 'chain' method quests (traditional chain via template.chain)
   454→ * - 'automatic' method quests (auto-activate on prerequisites)
   455→ * - Any quest with prerequisites that should chain-activate
   456→ */
   457→function activateQuestsWhosePrerequisitesAreMet(
   458→  get: any,
   459→  sessionId: string,
   460→  completedQuestId?: string
   461→): void {
   462→  try {
   463→    const session = get().sessions.find((s: ChatSession) => s.id === sessionId);
   464→    if (!session?.sessionQuests) return;
   465→
   466→    const completedQuestIds = session.sessionQuests
   467→      .filter(q => q.status === 'completed')
   468→      .map(q => q.templateId);
   469→
   470→    const questTemplates = get().questTemplates || [];
   471→
   472→    // Find candidates: available quests that aren't the one just completed
   473→    const candidates = session.sessionQuests.filter(
   474→      (q: SessionQuestInstance) =>
   475→        q.status === 'available' &&
   476→        q.templateId !== completedQuestId &&
   477→        q.templateId !== undefined
   478→    );
   479→
   480→    if (candidates.length === 0) return;
   481→
   482→    for (const candidate of candidates) {
   483→      const candidateTemplate = get().getTemplateById?.(candidate.templateId);
   484→      if (!candidateTemplate?.prerequisites?.length) continue;
   485→
   486→      // Check if ALL prerequisites are completed
   487→      const allMet = candidateTemplate.prerequisites.every(id =>
   488→        completedQuestIds.includes(id)
   489→      );
   490→
   491→      if (allMet) {
   492→        // Also check activation conditions (by_attribute / by_objective)
   493→        const canActivate = evaluateActivationConditions(
   494→          candidateTemplate,
   495→          session.sessionStats,
   496→          session.sessionQuests,
   497→          questTemplates
   498→        );
   499→        if (!canActivate) {
   500→          console.log(`[Quest Auto] Prerequisites met for "${candidateTemplate.name}" but activation conditions not met, skipping`);
   501→          continue;
   502→        }
   503→
   504→        console.log(`[Quest Auto] Activating "${candidateTemplate.name}" (${candidateTemplate.activation.method}) — all prerequisites met`);
   505→        get().activateQuest(sessionId, candidate.templateId);
   506→      }
   507→    }
   508→  } catch (e) {
   509→    console.warn('[Quest Auto] Error checking prerequisite-based activation:', e);
   510→  }
   511→}

   513→/**
   514→ * Calculate quest progress from objectives
   515→ */
   516→function calculateQuestProgress(objectives: Array<{ templateId: string; currentCount: number; isCompleted: boolean } & { isOptional?: boolean }>): number {
   517→  if (!objectives || objectives.length === 0) return 0;
   518→  
   519→  const requiredObjectives = objectives.filter(o => !o.isOptional);
   520→  
   521→  if (requiredObjectives.length === 0) return 100;
   522→  
   523→  const completedCount = requiredObjectives.filter(o => o.isCompleted).length;
   524→  return Math.round((completedCount / requiredObjectives.length) * 100);
   525→}
