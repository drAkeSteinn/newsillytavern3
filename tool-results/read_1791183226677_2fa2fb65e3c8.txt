    13→// - {{solicitudes}} → Block of received requests

    90→export function getCharacterSessionStats(
    91→  sessionStats: SessionStats | undefined,
    92→  characterId: string
    93→): CharacterSessionStats | null {
    94→  if (!sessionStats?.characterStats?.[characterId]) {
    95→    return null;
    96→  }
    97→  return sessionStats.characterStats[characterId];
    98→}

   100→/**
   101→ * Get attribute value (from session or default)
   102→ */
   103→export function getAttributeValue(
   104→  attribute: AttributeDefinition,
   105→  sessionStats: CharacterSessionStats | null
   106→): number | string {
   107→  if (sessionStats?.attributeValues?.[attribute.key] !== undefined) {
   108→    return sessionStats.attributeValues[attribute.key];
   109→  }
   110→  return attribute.defaultValue;
   111→}
   112→
   113→/**
   114→ * Format attribute value for prompt
   115→ * 
   116→ * Format:
   117→ * - Number type: "Nombre: (valor/max)" e.g., "Resistencia física: (40/100)"
   118→ * - Keyword/Text type: "Nombre: valor" e.g., "Detección: mágica"
   119→ * - Custom outputFormat takes precedence if defined
   120→ */
   121→export function formatAttributeValue(
   122→  attribute: AttributeDefinition,
   123→  value: number | string
   124→): string {
   125→  // Use new outputFormat field first (custom format takes precedence)
   126→  if (attribute.outputFormat) {
   127→    return attribute.outputFormat.replace('{value}', String(value));
   128→  }
   129→  // Fallback to legacy keywordFormat for backward compatibility
   130→  if (attribute.keywordFormat) {
   131→    return attribute.keywordFormat.replace('{value}', String(value));
   132→  }
   133→  
   134→  // Default formatting based on attribute type
   135→  const attributeType = attribute.type || 'number';
   136→  
   137→  if (attributeType === 'number') {
   138→    // For numeric attributes, show (current/max) format
   139→    const max = attribute.max ?? 100;
   140→    return `${attribute.name}: (${value}/${max})`;
   141→  } else {
   142→    // For keyword and text types, show just the value
   143→    return `${attribute.name}: ${value}`;
   144→  }
   145→}

   147→/**
   148→ * Resolve a single attribute key
   149→ */
   150→export function resolveAttributeKey(
   151→  key: string,
   152→  statsConfig: CharacterStatsConfig | undefined,
   153→  sessionStats: CharacterSessionStats | null
   154→): string | null {
   155→  if (!statsConfig?.attributes) return null;
   156→  
   157→  const attribute = statsConfig.attributes.find(a => a.key === key);
   158→  if (!attribute) return null;
   159→  
   160→  const value = getAttributeValue(attribute, sessionStats);
   161→  return formatAttributeValue(attribute, value);
   162→}

   164→/**
   165→ * Resolve all attributes for a character
   166→ */
   167→export function resolveAllAttributes(
   168→  statsConfig: CharacterStatsConfig | undefined,
   169→  sessionStats: CharacterSessionStats | null
   170→): ResolvedAttribute[] {
   171→  if (!statsConfig?.attributes) return [];
   172→  
   173→  return statsConfig.attributes.map(attribute => {
   174→    const value = getAttributeValue(attribute, sessionStats);
   175→    return {
   176→      key: attribute.key,
   177→      name: attribute.name,
   178→      value,
   179→      formatted: formatAttributeValue(attribute, value),
   180→    };
   181→  });
   182→}

   184→/**
   185→ * Resolve template keys in text using the full key resolution pipeline
   186→ * Now handles ALL key types: {{user}}, {{char}}, {{userpersona}}, {{last_events}},
   187→ * {{solicitante}}, {{solicitado}}, stat attribute keys, etc.
   188→ *
   189→ * Falls back to basic regex replacement when full context is not available.
   190→ */
   191→function resolveTemplateKeys(
   192→  text: string,
   193→  userName?: string,
   194→  characterName?: string,
   195→  solicitanteName?: string,
   196→  solicitadoName?: string,
   197→  fullContext?: {
   198→    characterName: string;
   199→    userName: string;
   200→    personaDescription?: string;
   201→    sessionStats?: SessionStats;
   202→    characterId?: string;
   203→    resolvedStats?: import('@/types').ResolvedStats | null;
   204→    personaResolvedStats?: import('@/types').ResolvedStats | null;
   205→    lorebookEntryKeys?: Record<string, string>;
   206→  }
   207→): string {
   208→  if (!text) return text;
   209→
   210→  // If full context is available, use the comprehensive resolver
   211→  if (fullContext) {
   212→    const keyContext = buildKeyResolutionContext(
   213→      { id: '', name: fullContext.characterName } as import('@/types').CharacterCard,
   214→      fullContext.userName,
   215→      fullContext.personaDescription
   216→        ? { name: fullContext.userName, description: fullContext.personaDescription } as import('@/types').Persona
   217→        : undefined,
   218→      fullContext.resolvedStats,
   219→      fullContext.sessionStats,
   220→      undefined, // soundTriggers
   221→      undefined, // soundSettings
   222→      fullContext.personaResolvedStats,
   223→      undefined, // questTemplates
   224→      undefined, // sessionQuests
   225→      undefined, // questSettings
   226→      undefined, // outletSections
   227→      undefined, // lorebookAttributeKeys
   228→      undefined, // inventoryData
   229→      fullContext.lorebookEntryKeys
   230→    );
   231→    // Override characterId for event keys
   232→    (keyContext as Record<string, unknown>).characterId = fullContext.characterId;
   233→    return resolveAllKeys(text, keyContext);
   234→  }
   235→
   236→  // Fallback: basic regex replacement for common keys
   237→  let result = text;
   238→  if (userName) {
   239→    result = result.replace(/\{\{user\}\}/gi, userName);
   240→  }
   241→  if (characterName) {
   242→    result = result.replace(/\{\{char\}\}/gi, characterName);
   243→  }
   244→  if (solicitanteName) {
   245→    result = result.replace(/\{\{solicitante\}\}/gi, solicitanteName);
   246→  }
   247→  if (solicitadoName) {
   248→    result = result.replace(/\{\{solicitado\}\}/gi, solicitadoName);
   249→  }
   250→  return result;
   251→}

   253→/**
   254→ * Find objective name by completion key
   255→ */
   256→function findObjectiveNameByKey(
   257→  objectiveKey: string,
   258→  questTemplates: { objectives?: { completion?: { key?: string; keys?: string[] }; description?: string }[] }[]
   259→): string | null {
   260→  const normalizedKey = objectiveKey.toLowerCase().trim();
   261→  
   262→  for (const template of questTemplates) {
   263→    for (const objective of template.objectives || []) {
   264→      const keys = [
   265→        objective.completion?.key,
   266→        ...(objective.completion?.keys || [])
   267→      ].filter(Boolean);
   268→      
   269→      for (const key of keys) {
   270→        if (key?.toLowerCase().trim() === normalizedKey) {
   271→          return objective.description || null;
   272→        }
   273→      }
   274→    }
   275→  }
   276→  
   277→  return null;
   278→}

   280→/**
   281→ * Build skills/actions block for injection
   282→ *
   283→ * NEW READABLE FORMAT:
   284→ * [ACCIONES DEL PERSONAJE]
   285→ * El personaje puede realizar las siguientes acciones:
   286→ *
   287→ * • Examen psicológico
   288→ *   Tipo: ejecución
   289→  *   Descripción: Sabes aplicar examen psicológico...
   290→  *   Resultado esperado: Completará "Analizar al usuario"
   291→  */
   292→export function buildSkillsBlock(
   293→  skills: SkillDefinition[],
   294→  attributeValues: Record<string, number | string>,
   295→  header: string,
   296→  questTemplates: { objectives?: { completion?: { key?: string; keys?: string[] }; description?: string }[] }[] = [],
   297→  characterName?: string,
   298→  sessionStats?: SessionStats,
   299→  userName?: string,
   300→  fullContext?: {
   301→    characterId?: string;
   302→    personaDescription?: string;
   303→    resolvedStats?: import('@/types').ResolvedStats | null;
   304→    personaResolvedStats?: import('@/types').ResolvedStats | null;
   305→    userName?: string;
   306→    characterName?: string;
   307→    sessionStats?: SessionStats;
   308→    lorebookEntryKeys?: Record<string, string>;
   309→  }
   310→): string {
   311→  const availableSkills = filterSkillsByRequirements(skills, attributeValues, sessionStats);
   312→
   313→  if (availableSkills.length === 0) {
   314→    return '';
   315→  }
   316→
   317→  const charName = characterName || '{{char}}';
   318→  
   319→  // Build instruction text that goes BEFORE the header
   320→  // This ensures the LLM sees the rules before the action list
   321→  const introLines: string[] = [];
   322→  if (header.includes('ACCIONES')) {
   323→    introLines.push(`${charName} DEBE usar acciones SIEMPRE que realice algo significativo en el roleplay. Las acciones son la forma principal de interactuar con el mundo. Cada vez que ${charName} haga algo más allá de hablar (moverse, atacar, usar habilidades, reaccionar físicamente, etc.), DEBE usar la TOOL "manage_action" con la key correspondiente.`);
   324→    introLines.push('');
   325→    introLines.push('REGLAS DE USO DE ACCIONES:');
   326→    introLines.push('- USA ACCIONES ACTIVAMENTE. No esperes a que haya objetivos o misiones para usarlas.');
   327→    introLines.push('- Cada acción narrativa importante DEBE ir acompañada de un manage_action call.');
   328→    introLines.push('- Si una acción describe algo que el personaje hace físicamente, úsala.');
   329→    introLines.push('- Puedes usar múltiples acciones en una sola respuesta si la situación lo requiere.');
   330→    introLines.push('- NUNCA dejes de usar una acción disponible cuando el personaje la podría realizar.');
   331→    introLines.push('');
   332→    introLines.push('Si una acción indica "Puede completar", usa la TOOL "manage_quest" o "manage_solicitud" con la key correspondiente para marcar como completado inmediatamente después de realizar la acción.');
   333→  }
   334→  const introText = introLines.length > 0 ? introLines.join('\n') : '';
   335→
   336→  // Build the block: instructions → header → action list
   337→  const lines: string[] = [];
   338→  if (introText) {
   339→    lines.push(introText);
   340→  }
   341→  lines.push(header);
   342→
   343→  // Build full context for comprehensive key resolution if data is available
   344→  const keyFullContext = (userName && characterName) ? {
   345→    characterName,
   346→    userName,
   347→    personaDescription: fullContext?.personaDescription,
   348→    sessionStats,
   349→    characterId: fullContext?.characterId,
   350→    resolvedStats: fullContext?.resolvedStats,
   351→    personaResolvedStats: fullContext?.personaResolvedStats,
   352→    lorebookEntryKeys: fullContext?.lorebookEntryKeys,
   353→  } : undefined;
   354→
   355→  availableSkills.forEach((skill) => {
   356→    // Resolve template keys in skill text fields using full key resolution
   357→    const resolvedName = resolveTemplateKeys(skill.name, userName, characterName, undefined, undefined, keyFullContext);
   358→    const resolvedDescription = resolveTemplateKeys(skill.description, userName, characterName, undefined, undefined, keyFullContext);
   359→
   360→    // Check for custom inject format first
   361→    if (skill.injectFormat) {
   362→      const resolvedInjectFormat = resolveTemplateKeys(skill.injectFormat, userName, characterName, undefined, undefined, keyFullContext);
   363→      const formatted = resolvedInjectFormat
   364→        .replace('{name}', resolvedName)
   365→        .replace('{description}', resolvedDescription)
   366→        .replace('{key}', skill.activationKey || skill.key || '');
   367→      lines.push(`- ${formatted}`);
   368→    } else {
   369→      // New readable format
   370→      lines.push(`- Nombre: ${resolvedName}`);
   371→
   372→      // Type (preparacion/ejecucion)
   373→      if (skill.type) {
   374→        const tipoLabel = skill.type === 'preparacion' ? 'preparación' : 'ejecución';
   375→        lines.push(`  Tipo: ${tipoLabel}`);
   376→      }
   377→
   378→      lines.push(`  Descripción: ${resolvedDescription}`);
   379→
   380→      // Collect objectives and solicitudes that this skill can complete
   381→      const objectives: string[] = [];
   382→      const solicitudes: string[] = [];
   383→
   384→      for (const reward of skill.activationRewards || []) {
   385→        if (reward.type === 'objective' && reward.objective?.objectiveKey) {
   386→          const objectiveName = findObjectiveNameByKey(reward.objective.objectiveKey, questTemplates);
   387→          objectives.push(objectiveName || reward.objective.objectiveKey);
   388→        } else if (reward.type === 'solicitud' && reward.solicitud?.solicitudKey) {
   389→          solicitudes.push(reward.solicitud.solicitudName || reward.solicitud.solicitudKey);
   390→        }
   391→      }
   392→
   393→      // Build "Puede completar" section
   394→      if (objectives.length > 0 || solicitudes.length > 0) {
   395→        lines.push(`  Puede completar:`);
   396→        for (const obj of objectives) {
   397→          lines.push(`    - Objetivo: ${obj}`);
   398→        }
   399→        for (const sol of solicitudes) {
   400→          lines.push(`    - Solicitud: ${sol}`);
   401→        }
   402→      }
   403→    }
   404→  });
   405→
   406→  return lines.join('\n');
   407→}

   409→/**
   410→ * Build intentions block for injection
   411→ *
   412→ * Format (same as skills):
   413→ * 1) Intention Name
   414→ *    - Descripción: description text
   415→ *    - key de activación: activation_key (only if key exists)
   416→ *
   417→ * Numbers are dynamic based on available intentions count.
   418→ */
   419→export function buildIntentionsBlock(
   420→  intentions: IntentionDefinition[],
   421→  attributeValues: Record<string, number | string>,
   422→  header: string,
   423→  sessionStats?: SessionStats,
   424→  userName?: string,
   425→  characterName?: string,
   426→  fullContext?: {
   427→    characterId?: string;
   428→    personaDescription?: string;
   429→    resolvedStats?: import('@/types').ResolvedStats | null;
   430→    personaResolvedStats?: import('@/types').ResolvedStats | null;
   431→    userName?: string;
   432→    characterName?: string;
   433→    sessionStats?: SessionStats;
   434→    lorebookEntryKeys?: Record<string, string>;
   435→  }
   436→): string {
   437→  const availableIntentions = filterIntentionsByRequirements(intentions, attributeValues, sessionStats);
   438→
   439→  if (availableIntentions.length === 0) {
   440→    return '';
   441→  }
   442→
   443→  const lines: string[] = [header];
   444→
   445→  // Build full context for comprehensive key resolution if data is available
   446→  const keyFullContext = (userName && characterName) ? {
   447→    characterName,
   448→    userName,
   449→    personaDescription: fullContext?.personaDescription,
   450→    sessionStats,
   451→    characterId: fullContext?.characterId,
   452→    resolvedStats: fullContext?.resolvedStats,
   453→    personaResolvedStats: fullContext?.personaResolvedStats,
   454→    lorebookEntryKeys: fullContext?.lorebookEntryKeys,
   455→  } : undefined;
   456→
   457→  availableIntentions.forEach((intention, index) => {
   458→    const intentionNumber = index + 1;
   459→
   460→    // Resolve template keys in intention text fields using full key resolution
   461→    const resolvedName = resolveTemplateKeys(intention.name, userName, characterName, undefined, undefined, keyFullContext);
   462→    const resolvedDescription = resolveTemplateKeys(intention.description, userName, characterName, undefined, undefined, keyFullContext);
   463→
   464→    // Check for custom inject format first
   465→    if (intention.injectFormat) {
   466→      const resolvedInjectFormat = resolveTemplateKeys(intention.injectFormat, userName, characterName, undefined, undefined, keyFullContext);
   467→      const formatted = resolvedInjectFormat
   468→        .replace('{name}', resolvedName)
   469→        .replace('{description}', resolvedDescription)
   470→        .replace('{key}', intention.key || '');
   471→      lines.push(`${intentionNumber}) ${formatted}`);
   472→    } else {
   473→      // Default format with description and activation key
   474→      lines.push(`${intentionNumber}) ${resolvedName}`);
   475→      lines.push(`   - Descripción: ${resolvedDescription}`);
   476→
   477→      // Show activation key only if it exists
   478→      if (intention.key) {
   479→        lines.push(`   - key de activación: ${intention.key}`);
   480→      }
   481→    }
   482→  });
   483→
   484→  return lines.join('\n');
   485→}

   487→/**
   488→ * Build invitations/peticiones block for injection
   489→ *
   490→ * NEW FORMAT - Gets key and description from target's solicitudDefinition:
   491→ * [PETICIONES POSIBLES]
   492→ * - key: pedir_madera
   493→ *   dirigido_a: Carpintero
   494→ *   descripcion: Solicitar madera para construcción
   495→ *
   496→ * The invitation must reference a solicitud from another character.
   497→ * Only shows if the target character meets the requirements of their own solicitud.
   498→ */
   499→export function buildInvitationsBlock(
   500→  invitations: InvitationDefinition[],
   501→  attributeValues: Record<string, number | string>,
   502→  header: string,
   503→  allCharacters?: CharacterCard[],
   504→  sessionStats?: SessionStats,
   505→  userName?: string,
   506→  characterName?: string,
   507→  fullContext?: {
   508→    characterId?: string;
   509→    personaDescription?: string;
   510→    resolvedStats?: import('@/types').ResolvedStats | null;
   511→    personaResolvedStats?: import('@/types').ResolvedStats | null;
   512→    userName?: string;
   513→    characterName?: string;
   514→    sessionStats?: SessionStats;
   515→    lorebookEntryKeys?: Record<string, string>;
   516→  }
   517→): string {
   518→  const availableInvitations = filterInvitationsByRequirements(invitations, attributeValues, sessionStats);
   519→
   520→  if (availableInvitations.length === 0) {
   521→    return '';
   522→  }
   523→
   524→  const lines: string[] = [header];
   525→
   526→  availableInvitations.forEach((invitation) => {
   527→    // Skip if no target configured
   528→    if (!invitation.objetivo?.characterId || !invitation.objetivo?.solicitudId) {
   529→      return;
   530→    }
   531→
   532→    // Find target character
   533→    const targetCharacter = allCharacters?.find(c => c.id === invitation.objetivo!.characterId);
   534→    if (!targetCharacter) {
   535→      return;
   536→    }
   537→
   538→    // Find the specific solicitud on the target
   539→    const solicitud = targetCharacter.statsConfig?.solicitudDefinitions?.find(
   540→      s => s.id === invitation.objetivo!.solicitudId
   541→    );
   542→    if (!solicitud) {
   543→      return;
   544→    }
   545→
   546→    // Check if target character meets the solicitud's requirements
   547→    // (the target needs to have the required attributes to fulfill the request)
   548→    const targetAttributeValues = sessionStats?.characterStats?.[targetCharacter.id]?.attributeValues || {};
   549→    const targetMeetsRequirements = evaluateRequirements(solicitud.requirements, targetAttributeValues, undefined, solicitud.requirementOperator);
   550→
   551→    if (!targetMeetsRequirements) {
   552→      // Target doesn't meet requirements - don't show this invitation
   553→      return;
   554→    }
   555→
   556→    // Build full context for comprehensive key resolution
   557→    const invKeyFullContext = (userName && characterName) ? {
   558→      characterName,
   559→      userName,
   560→      personaDescription: fullContext?.personaDescription,
   561→      sessionStats,
   562→      characterId: fullContext?.characterId,
   563→      resolvedStats: fullContext?.resolvedStats,
   564→      personaResolvedStats: fullContext?.personaResolvedStats,
   565→      lorebookEntryKeys: fullContext?.lorebookEntryKeys,
   566→    } : undefined;
   567→
   568→    // Resolve keys in description:
   569→    // - {{solicitante}} = characterName (who makes the request - current character)
   570→    // - {{solicitado}} = targetCharacter.name (who receives the request)
   571→    const resolvedDescription = resolveTemplateKeys(
   572→      solicitud.peticionDescription,
   573→      userName,
   574→      characterName,
   575→      characterName,       // solicitante = current character (who asks)
   576→      targetCharacter.name, // solicitado = target (who is asked)
   577→      invKeyFullContext
   578→    );
   579→
   580→    // Use custom inject format if available
   581→    if (invitation.injectFormat) {
   582→      const formatted = invitation.injectFormat
   583→        .replace('{name}', invitation.name)
   584→        .replace('{key}', solicitud.peticionKey)
   585→        .replace('{descripcion}', resolvedDescription)
   586→        .replace('{objetivo}', targetCharacter.name);
   587→      lines.push(formatted);
   588→    } else {
   589→      // New YAML-like format
   590→      lines.push(`- key: ${solicitud.peticionKey}`);
   591→      lines.push(`  dirigido_a: ${targetCharacter.name}`);
   592→      lines.push(`  descripcion: ${resolvedDescription}`);
   593→    }
   594→  });
   595→
   596→  // Return empty if no valid invitations after filtering
   597→  if (lines.length === 1) {
   598→    return '';
   599→  }
   600→
   601→  return lines.join('\n');
   602→}

   604→/**
   605→ * Resolve invitations to get their actual keys and descriptions
   606→ * (for use in detection system and UI components)
   607→ */
   608→export function resolveInvitations(
   609→  invitations: InvitationDefinition[],
   610→  attributeValues: Record<string, number | string>,
   611→  allCharacters?: CharacterCard[],
   612→  sessionStats?: SessionStats,
   613→  userName?: string,
   614→  characterName?: string
   615→): ResolvedInvitation[] {
   616→  const availableInvitations = filterInvitationsByRequirements(invitations, attributeValues, sessionStats);
   617→  const resolved: ResolvedInvitation[] = [];
   618→
   619→  availableInvitations.forEach((invitation) => {
   620→    if (!invitation.objetivo?.characterId || !invitation.objetivo?.solicitudId) {
   621→      return;
   622→    }
   623→
   624→    const targetCharacter = allCharacters?.find(c => c.id === invitation.objetivo!.characterId);
   625→    if (!targetCharacter) {
   626→      return;
   627→    }
   628→
   629→    const solicitud = targetCharacter.statsConfig?.solicitudDefinitions?.find(
   630→      s => s.id === invitation.objetivo!.solicitudId
   631→    );
   632→    if (!solicitud) {
   633→      return;
   634→    }
   635→
   636→    // Check if target meets solicitud requirements
   637→    const targetAttributeValues = sessionStats?.characterStats?.[targetCharacter.id]?.attributeValues || {};
   638→    const targetMeetsRequirements = evaluateRequirements(solicitud.requirements, targetAttributeValues, undefined, solicitud.requirementOperator);
   639→
   640→    if (!targetMeetsRequirements) {
   641→      return;
   642→    }
   643→
   644→    // Resolve keys in peticionDescription (shown to SOLICITANTE - who asks):
   645→    // - {{solicitante}} = characterName (who makes the request - current character)
   646→    // - {{solicitado}} = targetCharacter.name (who receives the request)
   647→    const resolvedPeticionDescription = resolveTemplateKeys(
   648→      solicitud.peticionDescription,
   649→      userName,
   650→      characterName,
   651→      characterName,       // solicitante = current character (who asks)
   652→      targetCharacter.name // solicitado = target (who is asked)
   653→    );
   654→
   655→    // Resolve keys in solicitudDescription (shown to SOLICITADO - who receives):
   656→    // - {{solicitante}} = characterName (who makes the request)
   657→    // - {{solicitado}} = targetCharacter.name (current character who receives)
   658→    const resolvedSolicitudDescription = resolveTemplateKeys(
   659→      solicitud.solicitudDescription,
   660→      userName,
   661→      targetCharacter.name,
   662→      characterName,        // solicitante = who makes the request
   663→      targetCharacter.name  // solicitado = who receives
   664→    );
   665→
   666→    resolved.push({
   667→      id: invitation.id,
   668→      name: invitation.name,
   669→      peticionKey: solicitud.peticionKey,
   670→      solicitudKey: solicitud.solicitudKey,  // Key for completing the solicitud
   671→      peticionDescription: resolvedPeticionDescription,
   672→      solicitudDescription: resolvedSolicitudDescription,
   673→      targetCharacterId: targetCharacter.id,
   674→      targetCharacterName: targetCharacter.name,
   675→      solicitudId: solicitud.id,
   676→    });
   677→  });
   678→
   679→  return resolved;
   680→}
   681→
   682→/**
   683→ * Build solicitudes block for injection
   684→ *
   685→ * Shows requests received from other characters
   686→ *
   687→ * FORMAT:
   688→ * [SOLICITUDES RECIBIDAS]
   689→ * - key: preparar_troncos
   690→ *   de: Aitana
   691→ *   descripcion: Aitana necesita que dejes listos los troncos de abedul.
   692→ */
   693→export function buildSolicitudesBlock(
   694→  solicitudes: SolicitudInstance[],
   695→  header: string,
   696→  userName?: string,
   697→  characterName?: string
   698→): string {
   699→  // Filter only pending solicitudes
   700→  const pendingSolicitudes = solicitudes.filter(s => s.status === 'pending');
   701→
   702→  if (pendingSolicitudes.length === 0) {
   703→    return '';
   704→  }
   705→
   706→  const lines: string[] = [header];
   707→
   708→  pendingSolicitudes.forEach((solicitud) => {
   709→    // Resolve keys in description:
   710→    // - {{solicitante}} = solicitud.fromCharacterName (who sent the request)
   711→    // - {{solicitado}} = characterName (current character who receives)
   712→    const resolvedDescription = resolveTemplateKeys(
   713→      solicitud.description,
   714→      userName,
   715→      characterName,
   716→      solicitud.fromCharacterName, // solicitante = who sent the request
   717→      characterName                 // solicitado = current character (who receives)
   718→    );
   719→    lines.push(`- key: ${solicitud.key}`);
   720→    lines.push(`  de: ${solicitud.fromCharacterName}`);
   721→    lines.push(`  descripcion: ${resolvedDescription}`);
   722→  });
   723→
   724→  return lines.join('\n');
   725→}

   727→/**
   728→ * Full stats resolution for a character
   729→ */
   730→export function resolveStats(
   731→  context: StatsResolutionContext
   732→): ResolvedStats | null {
   733→  const { characterId, statsConfig, sessionStats } = context;
   734→  
   735→  if (!statsConfig || !statsConfig.enabled) {
   736→    return null;
   737→  }
   738→  
   739→  const charStats = getCharacterSessionStats(sessionStats, characterId);
   740→  
   741→  // Resolve all attributes
   742→  const attributes = resolveAllAttributes(statsConfig, charStats);
   743→  const attributeValues = charStats?.attributeValues || 
   744→    Object.fromEntries(
   745→      (statsConfig.attributes || []).map(a => [a.key, a.defaultValue])
   746→    );
   747→  
   748→  // Build attribute map for template resolution
   749→  const attributesMap: Record<string, string> = {};
   750→  for (const attr of attributes) {
   751→    attributesMap[attr.key] = attr.formatted;
   752→  }
   753→
   754→  // Also include slot values and other dynamically-set attributes from session stats
   755→  // that aren't already in statsConfig.attributes (e.g., equipment slot keys like {{cabeza}})
   756→  if (charStats?.attributeValues) {
   757→    for (const [key, value] of Object.entries(charStats.attributeValues)) {
   758→      if (!(key in attributesMap) && value !== undefined && value !== null && value !== '') {
   759→        attributesMap[key] = String(value);
   760→      } else if (!(key in attributesMap) && value !== undefined && value !== null) {
   761→        // Include even empty values so the key resolves (to empty string) rather than staying as {{key}}
   762→        attributesMap[key] = String(value);
   763→      }
   764→    }
   765→  }
   766→
   767→  // FASE 5: Include emotional state as {{emocion}} key
   768→  // This allows the emotion to be resolved in prompts and sprite conditions
   769→  if (charStats?.emotionalState && !('emocion' in attributesMap)) {
   770→    attributesMap['emocion'] = charStats.emotionalState;
   771→  }
   772→  
   773→  // Build blocks with full key resolution context
   774→  // Include a partial resolvedStats with just the attributes map so that
   775→  // stat attribute keys ({{vida}}, {{mana}}, etc.) in skill/intention descriptions get resolved
   776→  const partialResolvedStats: ResolvedStats = {
   777→    attributes: attributesMap,
   778→    availableSkills: [],
   779→    availableIntentions: [],
   780→    availableInvitations: [],
   781→    availableSolicitudes: [],
   782→    skillsBlock: '',
   783→    intentionsBlock: '',
   784→    invitationsBlock: '',
   785→    solicitudesBlock: '',
   786→    mainAttribute: null,
   787→  };
   788→
   789→  const keyFullContext = {
   790→    characterId,
   791→    personaDescription: context.personaDescription,
   792→    personaResolvedStats: context.personaResolvedStats,
   793→    resolvedStats: partialResolvedStats,
   794→    userName: context.userName,
   795→    characterName: context.characterName,
   796→    sessionStats,
   797→    lorebookEntryKeys: context.lorebookEntryKeys,
   798→  };
   799→
   800→  // Defensive access: statsConfig may have incomplete properties when sent from frontend
   801→  // (e.g., only {enabled: true, attributes: [...]} without skills/intentions/blockHeaders)
   802→  const skills = statsConfig.skills || [];
   803→  const intentions = statsConfig.intentions || [];
   804→  const invitations = statsConfig.invitations || [];
   805→  const blockHeaders = statsConfig.blockHeaders || {};
   806→
   807→  const skillsBlock = buildSkillsBlock(
   808→    skills,
   809→    attributeValues,
   810→    blockHeaders.skills || '[ACCIONES DISPONIBLES]',
   811→    context.questTemplates || [],
   812→    context.characterName,
   813→    sessionStats,
   814→    context.userName,
   815→    keyFullContext
   816→  );
   817→  
   818→  const intentionsBlock = buildIntentionsBlock(
   819→    intentions,
   820→    attributeValues,
   821→    blockHeaders.intentions || 'Intenciones disponibles:',
   822→    sessionStats,
   823→    context.userName,
   824→    context.characterName,
   825→    keyFullContext
   826→  );
   827→  
   828→  const invitationsBlock = buildInvitationsBlock(
   829→    invitations,
   830→    attributeValues,
   831→    blockHeaders.invitations || '[PETICIONES DISPONIBLES]',
   832→    context.allCharacters,
   833→    sessionStats,
   834→    context.userName,
   835→    context.characterName,
   836→    keyFullContext
   837→  );
   838→
   839→  // Build solicitudes block (requests received from other characters)
   840→  const solicitudes = sessionStats?.solicitudes?.characterSolicitudes?.[characterId] || [];
   841→  const solicitudesBlock = buildSolicitudesBlock(
   842→    solicitudes,
   843→    blockHeaders.solicitudesRecibidas || '[SOLICITUDES RECIBIDAS]',
   844→    context.userName,
   845→    context.characterName
   846→  );
   847→
   848→  // Filter available items
   849→  const availableSkills = filterSkillsByRequirements(skills, attributeValues, sessionStats);
   850→  const availableIntentions = filterIntentionsByRequirements(intentions, attributeValues, sessionStats);
   851→  const availableInvitations = filterInvitationsByRequirements(invitations, attributeValues, sessionStats);
   852→
   853→  // Find the main attribute (if any isMain=true is set)
   854→  // Used by proactive messages default, Director tension, modify_stat tool hint, etc.
   855→  const mainAttrDef = (statsConfig.attributes || []).find(a => a.isMain === true);
   856→  let mainAttribute: ResolvedStats['mainAttribute'] = null;
   857→  if (mainAttrDef) {
   858→    const mainResolved = attributes.find(a => a.key === mainAttrDef.key);
   859→    if (mainResolved) {
   860→      mainAttribute = {
   861→        key: mainResolved.key,
   862→        name: mainResolved.name,
   863→        value: mainResolved.value,
   864→        formatted: mainResolved.formatted,
   865→        definition: mainAttrDef,
   866→      };
   867→    }
   868→  }
   869→
   870→  return {
   871→    attributes: attributesMap,
   872→    availableSkills,
   873→    availableIntentions,
   874→    availableInvitations,
   875→    availableSolicitudes: solicitudes.filter(s => s.status === 'pending'),
   876→    skillsBlock,
   877→    intentionsBlock,
   878→    invitationsBlock,
   879→    solicitudesBlock,
   880→    mainAttribute,
   881→  };
   882→}

   888→/**
   889→ * Regex pattern for stats keys
   890→ * Matches {{key}} where key is alphanumeric with underscores
   891→ */
   892→const STATS_KEY_PATTERN_SOURCE = '\\{\\{([a-zA-Z_][a-zA-Z0-9_]*)\\}\\}';
   893→
   894→/**
   895→ * Check if a key is a block key (acciones, habilidades, intenciones, invitaciones, peticiones, solicitudes)
   896→ * Also accepts alternate spellings for backward compatibility
   897→ */
   898→export function isBlockKey(key: string): boolean {
   899→  return key === 'acciones' || key === 'habilidades' || 
   900→         key === 'intenciones' || key === 'intensiones' || 
   901→         key === 'invitaciones' || key === 'peticiones' || 
   902→         key === 'solicitudes';
   903→}

   905→/**
   906→ * Resolve all stats keys in a text
   907→ */
   908→export function resolveStatsInText(
   909→  text: string,
   910→  resolvedStats: ResolvedStats | null
   911→): string {
   912→  const pattern = new RegExp(STATS_KEY_PATTERN_SOURCE, 'g');
   913→  return text.replace(pattern, (match, key) => {
   914→    // Block keys (acciones, habilidades, intenciones, invitaciones, peticiones, solicitudes)
   915→    // Return empty string if stats disabled, empty, or no items available
   916→    // Support both {{acciones}} (new) and {{habilidades}} (legacy)
   917→    if (key === 'acciones' || key === 'habilidades') {
   918→      return resolvedStats?.skillsBlock ?? '';
   919→    }
   920→    // Accept both "intenciones" (correct Spanish) and "intensiones" (typo, for backward compatibility)
   921→    if (key === 'intenciones' || key === 'intensiones') {
   922→      return resolvedStats?.intentionsBlock ?? '';
   923→    }
   924→    // Support both {{peticiones}} (new) and {{invitaciones}} (legacy)
   925→    if (key === 'peticiones' || key === 'invitaciones') {
   926→      return resolvedStats?.invitationsBlock ?? '';
   927→    }
   928→    // New {{solicitudes}} key - requests received from other characters
   929→    if (key === 'solicitudes') {
   930→      return resolvedStats?.solicitudesBlock ?? '';
   931→    }
   932→
   933→    // Attribute keys - only replace if defined in stats config
   934→    // If stats are disabled or attribute not defined, leave the key alone
   935→    if (resolvedStats?.attributes && key in resolvedStats.attributes) {
   936→      return resolvedStats.attributes[key];
   937→    }
   938→
   939→    // Unknown key - leave it alone (might be handled by other template systems)
   940→    return match;
   941→  });
   942→}

   944→/**
   945→ * Get all stats keys from a text
   946→ */
   947→export function extractStatsKeys(text: string): string[] {
   948→  const keys: string[] = [];
   949→  let match;
   950→  
   951→  const pattern = new RegExp(STATS_KEY_PATTERN_SOURCE, 'g');
   952→  
   953→  while ((match = pattern.exec(text)) !== null) {
   954→    if (!keys.includes(match[1])) {
   955→      keys.push(match[1]);
   956→    }
   957→  }
   958→  
   959→  return keys;
   960→}

   962→/**
   963→ * Check if text contains stats keys
   964→ */
   965→export function hasStatsKeys(text: string): boolean {
   966→  const pattern = new RegExp(STATS_KEY_PATTERN_SOURCE);
   967→  return pattern.test(text);
   968→}

   990→      label: `${characterName} Stats`,
   991→      content: attrValues.join('\n'),

   998→    sections.push({
   999→      type: 'skills',
  1000→      label: 'Habilidades',
  1001→      content: resolvedStats.skillsBlock,
  1002→      color: 'bg-fuchsia-100 text-fuchsia-700 dark:bg-fuchsia-900/30 dark:text-fuchsia-300',
  1003→    });
  1004→  }
  1005→  
  1006→  // Add intentions block if available
  1007→  if (resolvedStats.intentionsBlock) {
  1008→    sections.push({
  1009→      type: 'intentions',
  1010→      label: 'Intenciones',
  1011→      content: resolvedStats.intentionsBlock,
  1012→      color: 'bg-violet-100 text-violet-700 dark:bg-violet-900/30 dark:text-violet-300',
  1013→    });
  1014→  }

  1021→      content: resolvedStats.invitationsBlock,
  1022→      color: 'bg-rose-100 text-rose-700 dark:bg-rose-900/30 dark:text-rose-300',
  1023→    });
  1024→  }
  1025→  
  1026→  return sections;
  1027→}
