     8→// This module implements the sprite priority system. The rules are:
     9→//
    10→// 1. TRIGGER SPRITE HAS ABSOLUTE PRIORITY
    11→//    - Once set, triggerSpriteUrl MUST NOT be cleared by:
    12→//      * startGenerationForCharacter()
    13→//      * endGenerationForCharacter()
    14→//      * State changes (talk/thinking/idle)
    15→//    - Only cleared by:
    16→//      * Timer expiration (returnToIdleMs > 0)
    17→//      * New trigger replacing it
    18→//      * User manual action
    19→//
    20→// 2. TIMER BEHAVIOR
    21→//    - returnToIdleMs = 0: Trigger persists indefinitely
    22→//    - returnToIdleMs > 0: Trigger clears after X ms
    23→//
    24→// 3. RETURN MODE
    25→//    - 'clear': Clear trigger, show state-based sprite (talk/thinking/idle)
    26→//    - 'idle': Clear trigger, show idle sprite
    27→//    - 'talk': Clear trigger, show talk sprite
    28→//    - 'thinking': Clear trigger, show thinking sprite
    29→//
    30→// See: /docs/SPRITE_PRIORITY_SYSTEM.md for full documentation

    62→// Active sound chain tracking
    63→const activeSoundChains = new Map<string, { timeouts: NodeJS.Timeout[]; audios: HTMLAudioElement[] }>();
    64→
    65→/**
    66→ * Play a sound chain for a character
    67→ * This is called directly when startSoundChain is invoked
    68→ */
    69→function playSoundChainForCharacter(
    70→  characterId: string,
    71→  chain: SoundChain,
    72→  soundTriggers: SoundTrigger[],
    73→  soundCollections: SoundCollection[],
    74→  globalVolume: number
    75→): void {
    76→  // Stop any existing chain for this character
    77→  stopSoundChainForCharacter(characterId);
    78→  
    79→  const chainState = { timeouts: [] as NodeJS.Timeout[], audios: [] as HTMLAudioElement[] };
    80→  activeSoundChains.set(characterId, chainState);
    81→  
    82→  const chainVolume = (chain.volume ?? 1) * globalVolume;
    83→  
    84→  console.log('[SpriteSlice] Playing sound chain:', {
    85→    characterId,
    86→    steps: chain.steps.length,
    87→    volume: chainVolume,
    88→  });
    89→  
    90→  chain.steps.forEach((step: SoundChainStep, index: number) => {
    91→    const timeout = setTimeout(() => {
    92→      playSoundChainStep(
    93→        characterId,
    94→        step,
    95→        soundTriggers,
    96→        soundCollections,
    97→        chainVolume
    98→      );
    99→    }, step.delayMs);
   100→    
   101→    chainState.timeouts.push(timeout);
   102→  });
   103→}

   105→/**
   106→ * Play a single sound chain step
   107→ */
   108→function playSoundChainStep(
   109→  characterId: string,
   110→  step: SoundChainStep,
   111→  soundTriggers: SoundTrigger[],
   112→  soundCollections: SoundCollection[],
   113→  chainVolume: number
   114→): void {
   115→  let soundUrl: string | null = null;
   116→  let volume = (step.volume ?? 1) * chainVolume;
   117→  
   118→  // Try sound trigger key first
   119→  if (step.soundTriggerKey) {
   120→    const trigger = soundTriggers.find(t => t.name === step.soundTriggerKey || t.id === step.soundTriggerKey);
   121→    if (trigger) {
   122→      const collection = soundCollections.find(c => c.name === trigger.collection);
   123→      if (collection && collection.files.length > 0) {
   124→        // Pick sound based on play mode
   125→        let soundFile: string;
   126→        if (trigger.playMode === 'random') {
   127→          soundFile = collection.files[Math.floor(Math.random() * collection.files.length)];
   128→        } else {
   129→          const index = (trigger.currentIndex || 0) % collection.files.length;
   130→          soundFile = collection.files[index];
   131→        }
   132→        soundUrl = soundFile;
   133→        volume *= trigger.volume;
   134→      }
   135→    }
   136→  }
   137→  // Fall back to direct URL
   138→  else if (step.soundUrl) {
   139→    soundUrl = step.soundUrl;
   140→  }
   141→  
   142→  if (!soundUrl) {
   143→    console.warn('[SpriteSlice] No sound URL for step:', step);
   144→    return;
   145→  }
   146→  
   147→  try {
   148→    const audio = new Audio(soundUrl);
   149→    audio.volume = Math.min(1, Math.max(0, volume));
   150→    
   151→    // Emit comic sound visual event
   152→    const stepName = step.soundTriggerKey || 'chain_sound';
   153→    emitComicSoundEvent(stepName, stepName, characterId);
   154→    
   155→    audio.play().catch(e => {
   156→      console.warn('[SpriteSlice] Audio play failed:', e);
   157→    });
   158→    
   159→    // Track the audio
   160→    const chainState = activeSoundChains.get(characterId);
   161→    if (chainState) {
   162→      chainState.audios.push(audio);
   163→    }
   164→    
   165→    audio.onended = () => {
   166→      const state = activeSoundChains.get(characterId);
   167→      if (state) {
   168→        const idx = state.audios.indexOf(audio);
   169→        if (idx > -1) state.audios.splice(idx, 1);
   170→      }
   171→    };
   172→  } catch (error) {
   173→    console.error('[SpriteSlice] Failed to play sound:', error);
   174→  }
   175→}

   177→/**
   178→ * Stop sound chain for a character
   179→ */
   180→function stopSoundChainForCharacter(characterId: string): void {
   181→  const chainState = activeSoundChains.get(characterId);
   182→  if (!chainState) return;
   183→  
   184→  // Clear all timeouts
   185→  chainState.timeouts.forEach(t => clearTimeout(t));
   186→  
   187→  // Stop all audio
   188→  chainState.audios.forEach(a => {
   189→    a.pause();
   190→    a.remove();
   191→  });
   192→  
   193→  activeSoundChains.delete(characterId);
   194→}
