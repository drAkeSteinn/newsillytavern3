    70→// Sprite List Rotation Tracker
    71→// Tracks rotation indices per character per state
    72→// Avoids modifying character data (no persistence noise)
    73→// ============================================
    74→const _spriteRotationIndex = new Map<string, Map<SpriteState, number>>();
    75→
    76→function getRotationIndex(characterId: string, state: SpriteState): number {
    77→  let charMap = _spriteRotationIndex.get(characterId);
    78→  if (!charMap) {
    79→    charMap = new Map<SpriteState, number>();
    80→    _spriteRotationIndex.set(characterId, charMap);
    81→  }
    82→  return charMap.get(state) ?? 0;
    83→}
    84→
    85→function advanceRotationIndex(characterId: string, state: SpriteState): number {
    86→  let charMap = _spriteRotationIndex.get(characterId);
    87→  if (!charMap) {
    88→    charMap = new Map<SpriteState, number>();
    89→    _spriteRotationIndex.set(characterId, charMap);
    90→  }
    91→  const current = charMap.get(state) ?? 0;
    92→  const next = current + 1;
    93→  charMap.set(state, next);
    94→  return next;
    95→}

    97→// Helper: resolve a sprite from a pack given behavior config
    98→function resolveSpriteFromPack(
    99→  pack: SpritePackV2,
   100→  behavior: 'principal' | 'random' | 'list',
   101→  principalSpriteId: string | undefined,
   102→  characterId: string,
   103→  state: SpriteState
   104→): { url: string | null; label: string | null } {
   105→  if (pack.sprites.length === 0) return { url: null, label: null };
   106→
   107→  switch (behavior) {
   108→    case 'principal':
   109→      // Use principal sprite if specified, otherwise first sprite
   110→      if (principalSpriteId) {
   111→        const principal = pack.sprites.find(s => s.id === principalSpriteId);
   112→        if (principal) return { url: principal.url, label: principal.label };
   113→      }
   114→      return { url: pack.sprites[0].url, label: pack.sprites[0].label };
   115→
   116→    case 'random':
   117→      // Random selection
   118→      const randomIndex = Math.floor(Math.random() * pack.sprites.length);
   119→      const randomSprite = pack.sprites[randomIndex];
   120→      return { url: randomSprite.url, label: randomSprite.label };
   121→
   122→    case 'list': {
   123→      // Rotate through sprites using the rotation tracker
   124→      const index = getRotationIndex(characterId, state);
   125→      const safeIndex = index % pack.sprites.length;
   126→      const listSprite = pack.sprites[safeIndex];
   127→      return { url: listSprite.url, label: listSprite.label };
   128→    }
   129→
   130→    default:
   131→      return { url: pack.sprites[0].url, label: pack.sprites[0].label };
   132→  }
   133→}

   135→// Get sprite from State Collection V2 (uses sprite packs)
   136→// Now supports conditional mode at the pack level
   137→function getSpriteFromStateCollectionV2(
   138→  state: SpriteState,
   139→  stateCollectionsV2: StateCollectionV2[],
   140→  spritePacksV2: SpritePackV2[],
   141→  characterId: string,
   142→  sessionStats?: SessionStats | null
   143→): { url: string | null; label: string | null } {
   144→  const stateCollection = stateCollectionsV2.find(c => c.state === state);
   145→  if (!stateCollection) return { url: null, label: null };
   146→
   147→  const pack = spritePacksV2.find(p => p.id === stateCollection.packId);
   148→  if (!pack || pack.sprites.length === 0) return { url: null, label: null };
   149→
   150→  // ============================================
   151→  // Evaluate pack-level conditional sprites (if pack has conditionalMode)
   152→  // ============================================
   153→  if (pack.conditionalMode) {
   154→    const winningSprite = evaluatePackConditionalSprites(
   155→      pack.sprites,
   156→      sessionStats,
   157→      characterId
   158→    );
   159→
   160→    if (winningSprite) {
   161→      return { url: winningSprite.url, label: winningSprite.label };
   162→    }
   163→
   164→    // No conditional sprite matched - try defaultSpriteId
   165→    if (pack.defaultSpriteId) {
   166→      const defaultSprite = pack.sprites.find(s => s.id === pack.defaultSpriteId);
   167→      if (defaultSprite) {
   168→        return { url: defaultSprite.url, label: defaultSprite.label };
   169→      }
   170→    }
   171→
   172→    // Also check for sprites marked as isDefault
   173→    const markedDefault = pack.sprites.find(s => s.isDefault);
   174→    if (markedDefault) {
   175→      return { url: markedDefault.url, label: markedDefault.label };
   176→    }
   177→
   178→    // Fall through to behavior-based resolution
   179→  }
   180→
   181→  // ============================================
   182→  // Default: use the state collection's behavior
   183→  // ============================================
   184→  return resolveSpriteFromPack(
   185→    pack,
   186→    stateCollection.behavior,
   187→    stateCollection.principalSpriteId,
   188→    characterId,
   189→    state
   190→  );
   191→}

   193→// Get the appropriate sprite URL based on state and config
   194→// Priority: V2 State Collections > Avatar (legacy system removed)
   195→function getSpriteUrl(
   196→  state: SpriteState,
   197→  avatarUrl?: string,
   198→  character?: CharacterCard,
   199→  characterId?: string,
   200→  sessionStats?: SessionStats | null
   201→): { url: string; label: string | null } {
   202→  const charId = characterId || '';
   203→  // Try V2 state collections (new system)
   204→  // Check for non-empty arrays (empty arrays should not be considered as "configured")
   205→  const hasV2Collections = character?.stateCollectionsV2 && character.stateCollectionsV2.length > 0;
   206→  const hasV2Packs = character?.spritePacksV2 && character.spritePacksV2.length > 0;
   207→  
   208→  if (hasV2Collections && hasV2Packs) {
   209→    const v2Result = getSpriteFromStateCollectionV2(
   210→      state,
   211→      character.stateCollectionsV2,
   212→      character.spritePacksV2,
   213→      charId,
   214→      sessionStats
   215→    );
   216→    if (v2Result.url) {
   217→      return v2Result;
   218→    }
   219→  }
   220→
   221→  // Additional fallback for talk and thinking states - try V2 idle collection
   222→  if (state === 'talk' || state === 'thinking') {
   223→    if (hasV2Collections && hasV2Packs) {
   224→      const v2IdleResult = getSpriteFromStateCollectionV2(
   225→        'idle',
   226→        character!.stateCollectionsV2,
   227→        character!.spritePacksV2,
   228→        charId,
   229→        sessionStats
   230→      );
   231→      if (v2IdleResult.url) {
   232→        return v2IdleResult;
   233→      }
   234→    }
   235→  }
   236→
   237→  // Final fallback to avatar
   238→  if (avatarUrl) {
   239→    return { url: avatarUrl, label: 'avatar' };
   240→  }
   241→
   242→  return { url: '', label: null };
   243→}

   245→// Format milliseconds to readable time
   246→function formatTime(ms: number): string {
   247→  if (ms <= 0) return '0s';
   248→  const seconds = Math.floor(ms / 1000);
   249→  if (seconds < 60) return `${seconds}s`;
   250→  const minutes = Math.floor(seconds / 60);
   251→  const remainingSeconds = seconds % 60;
   252→  return `${minutes}m ${remainingSeconds}s`;
   253→}

   255→export function CharacterSprite({ 
   256→  characterId, 
   257→  characterName, 
   258→  avatarUrl,
   259→  character,
   260→  isStreaming = false,
   261→  hasContent = false,
   262→  isTTSPlaying = false,
   263→  onSettingsChange 
   264→}: CharacterSpriteProps) {
   265→  const [settings, setSettings] = useState<SpriteSettings>(() => {
   266→    if (typeof window !== 'undefined') {
   267→      const saved = localStorage.getItem(`sprite-${characterId}`);
   268→      if (saved) {
   269→        try {
   270→          return { ...DEFAULT_SPRITE_SETTINGS, ...JSON.parse(saved) };
   271→        } catch {
   272→          return DEFAULT_SPRITE_SETTINGS;
   273→        }
   274→      }
   275→    }
   276→    return DEFAULT_SPRITE_SETTINGS;
   277→  });
   278→
   279→  const [isDragging, setIsDragging] = useState(false);
   280→  const [isResizing, setIsResizing] = useState(false);
   281→  const [showControls, setShowControls] = useState(false);
   282→  const [countdown, setCountdown] = useState<number>(0);
   283→  
   284→  const spriteRef = useRef<HTMLDivElement>(null);
   285→  const dragStartRef = useRef({ x: 0, y: 0, settingsX: 0, settingsY: 0 });
   286→  const resizeStartRef = useRef({ x: 0, y: 0, width: 0, height: 0 });
   287→
   288→  // ============================================
   289→  // UNIFIED SYSTEM: Get sprite state from store
   290→  // ============================================
   291→  // 
   292→  // ⚠️ SPRITE PRIORITY SYSTEM - DO NOT MODIFY ⚠️
   293→  // 
   294→  // The sprite system follows a STRICT priority order:
   295→  // 
   296→  // 1. TRIGGER SPRITE (HIGHEST PRIORITY)
   297→  //    - Activated by sprite triggers from AI response
   298→  //    - Once active, CANNOT be overridden by talk/thinking/idle
   299→  //    - Only changes when:
   300→  //      a) Timer expires (returnToIdleMs > 0)
   301→  //      b) Another trigger replaces it
   302→  //      c) User manually clears it
   303→  //
   304→  // 2. STATE COLLECTION V2 SPRITES
   305→  //    - Talk sprite (streaming with content)
   306→  //    - Thinking sprite (generating without content)
   307→  //    - Idle sprite (default state)
   308→  //    - Only shown when NO trigger sprite is active
   309→  //
   310→  // 3. AVATAR (lowest priority)
   311→  //    - Character avatar as final fallback
   312→  //
   313→  // See: /docs/SPRITE_PRIORITY_SYSTEM.md
   314→  //
   315→  // Use proper Zustand selectors to ensure reactivity
   316→  const characterSpriteStates = useTavernStore((state) => state.characterSpriteStates);
   317→  const isLocked = useTavernStore((state) => state.spriteLock.active);
   318→  const lockState = useTavernStore((state) => state.spriteLock);
   319→  const getReturnToIdleCountdownForCharacter = useTavernStore((state) => state.getReturnToIdleCountdownForCharacter);
   320→  const cancelReturnToIdleForCharacter = useTavernStore((state) => state.cancelReturnToIdleForCharacter);
   321→  const executeReturnToIdleForCharacter = useTavernStore((state) => state.executeReturnToIdleForCharacter);
   322→  const clearSpriteLock = useTavernStore((state) => state.clearSpriteLock);
   323→  const setSpriteStateForCharacter = useTavernStore((state) => state.setSpriteStateForCharacter);
   324→
   325→  // Get sessionStats from the active session for conditional sprite evaluation
   326→  const sessionStats = useTavernStore((state) => {
   327→    const activeSessionId = state.activeSessionId;
   328→    if (!activeSessionId) return null;
   329→    const session = state.sessions.find(s => s.id === activeSessionId);
   330→    return session?.sessionStats ?? null;
   331→  });
   332→  
   333→  // Get per-character sprite state from the unified store
   334→  // This is now reactive because characterSpriteStates is a proper selector
   335→  const charSpriteState = characterSpriteStates[characterId] || createDefaultCharacterState();
   336→  
   337→  // Debug logging for trigger sprite state
   338→  useEffect(() => {
   339→    if (charSpriteState.triggerSpriteUrl) {
   340→      console.log('[CharacterSprite] Trigger sprite active:', {
   341→        characterId,
   342→        triggerSpriteUrl: charSpriteState.triggerSpriteUrl,
   343→        triggerSpriteLabel: charSpriteState.triggerSpriteLabel,
   344→        isStreaming,
   345→        hasContent,
   346→      });
   347→    }
   348→  }, [charSpriteState.triggerSpriteUrl, charSpriteState.triggerSpriteLabel, characterId, isStreaming, hasContent]);
   349→  
   350→  // Check if sprite is locked (global state)
   351→  const spriteLockActive = isLocked && lockState.until === 0 ? true : (isLocked && Date.now() < lockState.until);
   352→  
   353→  // Get return to idle state for this character
   354→  const isReturnToIdleScheduled = charSpriteState.returnToIdle.active;
   355→  
   356→  // Determine the effective sprite state:
   357→  // Priority: Trigger sprite > TTS Talk > Streaming Thinking > Store State
   358→  // 
   359→  // When TTS is active:
   360→  //   - During streaming → 'thinking' (not talk, since the character is "thinking" while generating)
   361→  //   - During TTS playback → 'talk' (the character is "speaking")
   362→  //   - After TTS ends → 'idle'
   363→  // 
   364→  // If a trigger sprite is active, it always takes absolute priority.
   365→  const effectiveSpriteState = charSpriteState.triggerSpriteUrl
   366→    ? 'idle' // Trigger active, use trigger sprite (handled below)
   367→    : isTTSPlaying
   368→      ? 'talk'                    // TTS is playing → character is speaking
   369→      : isStreaming
   370→        ? 'thinking'               // Generating → thinking (not talk, even with content)
   371→        : charSpriteState.spriteState;
   372→
   373→  // ============================================
   374→  // LIST MODE ROTATION: Advance index on state transition
   375→  // When effectiveSpriteState transitions to 'talk' or 'thinking',
   376→  // advance the rotation index so the NEXT sprite in the list is shown.
   377→  // ============================================
   378→  const prevEffectiveStateRef = useRef<SpriteState>(effectiveSpriteState);
   379→  useEffect(() => {
   380→    const prev = prevEffectiveStateRef.current;
   381→    prevEffectiveStateRef.current = effectiveSpriteState;
   382→
   383→    // Only advance when transitioning INTO talk or thinking (not from one to the other)
   384→    if (prev !== 'talk' && prev !== 'thinking' && 
   385→        (effectiveSpriteState === 'talk' || effectiveSpriteState === 'thinking')) {
   386→      // Check if the target state collection uses 'list' behavior
   387→      const stateCollections = character?.stateCollectionsV2;
   388→      const collection = stateCollections?.find(c => c.state === effectiveSpriteState);
   389→      if (collection?.behavior === 'list') {
   390→        advanceRotationIndex(characterId, effectiveSpriteState);
   391→      }
   392→    }
   393→  }, [effectiveSpriteState, character?.stateCollectionsV2, characterId]);
   394→
   395→  // ============================================
   396→  // TTS ↔ SPRITE STATE SYNC
   397→  // When TTS playback ends, return the sprite to idle state
   398→  // (only if no trigger is active and not currently streaming)
   399→  // ============================================
   400→  const prevTTSToStateRef = useRef(isTTSPlaying);
   401→  useEffect(() => {
   402→    const wasPlaying = prevTTSToStateRef.current;
   403→    prevTTSToStateRef.current = isTTSPlaying;
   404→
   405→    // TTS just stopped playing → return to idle
   406→    if (wasPlaying && !isTTSPlaying && !isStreaming) {
   407→      const currentState = characterSpriteStates[characterId];
   408→      // Only update if no trigger is active (trigger has absolute priority)
   409→      if (currentState && !currentState.triggerSpriteUrl) {
   410→        setSpriteStateForCharacter(characterId, 'idle');
   411→      }
   412→    }
   413→  }, [isTTSPlaying, isStreaming, characterId, characterSpriteStates, setSpriteStateForCharacter]);
   414→
   415→  // Update countdown for return to idle
   416→  // PERF FIX: Changed from 100ms to 1000ms — the countdown is in seconds, so 1Hz
   417→  // is sufficient and avoids 10 React state updates per second per visible character.
   418→  useEffect(() => {
   419→    if (!isReturnToIdleScheduled) {
   420→      setCountdown(0);
   421→      return;
   422→    }
   423→
   424→    const updateCountdown = () => {
   425→      const remaining = getReturnToIdleCountdownForCharacter(characterId);
   426→      setCountdown(remaining);
   427→    };
   428→
   429→    updateCountdown();
   430→    const interval = setInterval(updateCountdown, 1000);
   431→
   432→    return () => clearInterval(interval);
   433→  }, [isReturnToIdleScheduled, getReturnToIdleCountdownForCharacter, characterId]);
   434→
   435→  // Determine the current sprite URL
   436→  // PRIORITY:
   437→  // 1. Trigger sprite from store (highest priority)
   438→  // 2. State collection V2 (if available)
   439→  // 3. Avatar (lowest priority)
   440→  const spriteResult = charSpriteState.triggerSpriteUrl 
   441→    ? { url: charSpriteState.triggerSpriteUrl, label: charSpriteState.triggerSpriteLabel }
   442→    : getSpriteUrl(
   443→        effectiveSpriteState,
   444→        avatarUrl,
   445→        character,
   446→        characterId,
   447→        sessionStats
   448→      );
   449→  const currentSpriteUrl = spriteResult.url;
   450→  const currentSpriteLabel = spriteResult.label;
   451→
   452→  // FASE 7: Determine the active transition config
   453→  // If a trigger sprite is active, use the trigger's transition config
   454→  // Otherwise, use the character's default transition for state-based changes
   455→  const activeTransition: SpriteTransitionConfig | undefined = charSpriteState.triggerSpriteUrl
   456→    ? (charSpriteState.triggerTransition ?? undefined)
   457→    : (charSpriteState.defaultTransition ?? character?.defaultTransition ?? undefined);
   458→
   459→  // Debug: Log when sprite data changes
   460→  useEffect(() => {
   461→    console.log('[CharacterSprite] Sprite data:', {
   462→      characterId,
   463→      hasCharacter: !!character,
   464→      spritePacksV2Count: character?.spritePacksV2?.length ?? 'undefined',
   465→      stateCollectionsV2Count: character?.stateCollectionsV2?.length ?? 'undefined',
   466→      currentSpriteUrl: currentSpriteUrl?.substring(0, 50),
   467→      avatarUrl: avatarUrl?.substring(0, 50),
   468→    });
   469→  }, [characterId, character?.spritePacksV2, character?.stateCollectionsV2, currentSpriteUrl, avatarUrl]);
   470→
   471→  // Save settings to localStorage
   472→  useEffect(() => {
   473→    localStorage.setItem(`sprite-${characterId}`, JSON.stringify(settings));
   474→    onSettingsChange?.(settings);
   475→  }, [settings, characterId, onSettingsChange]);
   476→
   477→  // ============================================
   478→  // ESCENARIO V3: apply a queued sprite pose (position + scale) when the
   479→  // scene moves to a scenario location that has a saved pose. The queue is
   480→  // filled by the scenario_activation SSE handler (tool) or the manual
   481→  // scenario switcher. `at` timestamps dedupe repeats.
   482→  // ============================================
   483→  const appliedPoseAtRef = useRef(0);
   484→  const pendingPose = charSpriteState.pendingPose;
   485→  useEffect(() => {
   486→    if (!pendingPose || pendingPose.at === appliedPoseAtRef.current) return;
   487→    appliedPoseAtRef.current = pendingPose.at;
   488→    setSettings(prev => ({ ...prev, ...pendingPose.pose }));
   489→    console.log('[CharacterSprite] Applied scenario pose:', pendingPose.pose);
   490→  }, [pendingPose]);
   491→
   492→  // ============================================
   493→  // ESCENARIO V3: save the current sprite position + scale into the ACTIVE
   494→  // scenario location (spot → place, or the legacy location). When the scene
   495→  // later moves to that location the pose is applied automatically.
   496→  // The write logic is SHARED with the ScenarioSwitcher button
   497→  // (saveSpritePoseToActiveLocation in @/lib/scenario/save-pose).
   498→  // ============================================
   499→  const handleSavePoseToLocation = () => {
   500→    const result = saveSpritePoseToActiveLocation(characterId, {
   501→      x: settings.x,
   502→      y: settings.y,
   503→      width: settings.width,
   504→      height: settings.height,
   505→      opacity: settings.opacity,
   506→    });
   507→    if (!result.ok) {
   508→      const messages: Record<string, string> = {
   509→        'no-location': 'No hay ninguna ubicación de escenario activa para guardar la pose',
   510→        'no-collection': 'No se encontró la colección de escenario',
   511→        'no-character': 'No se encontró el personaje activo',
   512→        'no-config': 'No se encontró el escenario del personaje',
   513→      };
   514→      toast.error(messages[result.error ?? 'no-location'] ?? 'No se pudo guardar la pose');
   515→      return;
   516→    }
   517→    toast.success(`📍 Pose guardada en "${result.where}" — el sprite se moverá aquí al cambiar de escenario`);
   518→  };
   519→
   520→  // Drag handlers - click anywhere on sprite to drag
   521→  const handleDragStart = (e: React.MouseEvent) => {
   522→    // Don't start drag if clicking on controls
   523→    if ((e.target as HTMLElement).closest('.sprite-controls')) return;
   524→    if (isResizing) return;
   525→    
   526→    e.preventDefault();
   527→    e.stopPropagation();
   528→    setIsDragging(true);
   529→    dragStartRef.current = {
   530→      x: e.clientX,
   531→      y: e.clientY,
   532→      settingsX: settings.x,
   533→      settingsY: settings.y
   534→    };
   535→  };
   536→
   537→  useEffect(() => {
   538→    if (!isDragging) return;
   539→
   540→    const handleDragMove = (e: MouseEvent) => {
   541→      const container = spriteRef.current?.parentElement;
   542→      if (!container) return;
   543→
   544→      const rect = container.getBoundingClientRect();
   545→      const deltaX = ((e.clientX - dragStartRef.current.x) / rect.width) * 100;
