   222→/**
   223→ * Clamp a numeric value to the attribute's min/max bounds
   224→ * Returns the clamped value, or the original value if not numeric or no bounds defined
   225→ */
   226→function clampAttributeValue(
   227→  value: number | string,
   228→  attributeDef: AttributeDefinition | undefined
   229→): number | string {
   230→  // Only clamp numeric values
   231→  if (typeof value !== 'number') {
   232→    return value;
   233→  }
   234→
   235→  // If no attribute definition, return as-is
   236→  if (!attributeDef) {
   237→    return value;
   238→  }
   239→
   240→  let clampedValue = value;
   241→
   242→  // Apply min bound - handle both number and string min values
   243→  if (attributeDef.min !== undefined) {
   244→    const minVal = typeof attributeDef.min === 'number'
   245→      ? attributeDef.min
   246→      : parseFloat(String(attributeDef.min));
   247→
   248→    if (!isNaN(minVal)) {
   249→      clampedValue = Math.max(clampedValue, minVal);
   250→    }
   251→  }
   252→
   253→  // Apply max bound - handle both number and string max values
   254→  if (attributeDef.max !== undefined) {
   255→    const maxVal = typeof attributeDef.max === 'number'
   256→      ? attributeDef.max
   257→      : parseFloat(String(attributeDef.max));
   258→
   259→    if (!isNaN(maxVal)) {
   260→      clampedValue = Math.min(clampedValue, maxVal);
   261→    }
   262→  }
   263→
   264→  return clampedValue;
   265→}

   267→/**
   268→ * Create default character stats from config
   269→ */
   270→function createDefaultCharacterStats(
   271→  statsConfig?: CharacterStatsConfig
   272→): CharacterSessionStats {
   273→  const attributeValues: Record<string, number | string> = {};
   274→  const lastUpdated: Record<string, number> = {};
   275→  const now = Date.now();
   276→  
   277→  if (statsConfig?.attributes) {
   278→    for (const attr of statsConfig.attributes) {
   279→      attributeValues[attr.key] = attr.defaultValue;
   280→      lastUpdated[attr.key] = now;
   281→    }
   282→  }
   283→  
   284→  return {
   285→    attributeValues,
   286→    lastUpdated,
   287→    changeLog: [],
   288→  };
   289→}

   291→/**
   292→ * Add entry to change log
   293→ */
   294→function addChangeLogEntry(
   295→  stats: CharacterSessionStats,
   296→  attribute: AttributeDefinition | undefined,
   297→  attributeKey: string,
   298→  oldValue: number | string | undefined,
   299→  newValue: number | string,
   300→  reason: StatChangeLogEntry['reason']
   301→): void {
   302→  if (!stats.changeLog) {
   303→    stats.changeLog = [];
   304→  }
   305→  
   306→  stats.changeLog.push({
   307→    attributeId: attribute?.id || attributeKey,
   308→    attributeKey,
   309→    attributeName: attribute?.name || attributeKey,
   310→    oldValue: oldValue ?? '',
   311→    newValue,
   312→    reason,
   313→    timestamp: Date.now(),
   314→  });
   315→  
   316→  // Keep only last 100 entries
   317→  if (stats.changeLog.length > 100) {
   318→    stats.changeLog = stats.changeLog.slice(-100);
   319→  }
   320→}

   325→
   326→export const createStatsSlice = (set: any, get: any): StatsSlice => ({
   327→  // Initial State
   328→  sessionStats: null,
   329→
   330→  // ============================================
   331→  // Session Stats Actions
   332→  // ============================================
   333→
   334→  initializeSessionStats: (sessionId, characters) => {
   335→    const state = get();
   336→    const sessions = state.sessions as Array<{ id: string; sessionStats?: SessionStats }>;
   337→    const session = sessions.find(s => s.id === sessionId);
   338→    
   339→    if (!session) return;
   340→    
   341→    // Check if already initialized
   342→    if (session.sessionStats?.initialized) return;
   343→    
   344→    const now = Date.now();
   345→    const characterStats: Record<string, CharacterSessionStats> = {};
   346→    
   347→    // Initialize stats for each character
   348→    for (const char of characters) {
   349→      const stats = createDefaultCharacterStats(char.statsConfig);
   350→      // FASE 5: Initialize emotional state if emotional config is present
   351→      if (char.emotionalConfig?.enabled && char.emotionalConfig.initialState) {
   352→        stats.emotionalState = char.emotionalConfig.initialState;
   353→        stats.emotionalStateLastEval = now;
   354→        stats.emotionalStateTurnCount = 0;
   355→        // Also set as attribute value for {{emocion}} key resolution
   356→        stats.attributeValues['emocion'] = char.emotionalConfig.initialState;
   357→      }
   358→      characterStats[char.id] = stats;
   359→    }
   360→    
   361→    const newSessionStats: SessionStats = {
   362→      characterStats,
   363→      solicitudes: {
   364→        characterSolicitudes: {},
   365→        lastModified: now,
   366→      },
   367→      initialized: true,
   368→      lastModified: now,
   369→      // Initialize timer state
   370→      lastTimerUpdate: now,
   371→      keywordCycleIndex: {},
   372→    };
   373→    
   374→    // Update session with new stats
   375→    set((state: any) => ({
   376→      sessions: state.sessions.map((s: any) =>
   377→        s.id === sessionId 
   378→          ? { ...s, sessionStats: newSessionStats, updatedAt: new Date().toISOString() }
   379→          : s
   380→      ),
   381→    }));
   382→  },
   383→
   384→  updateCharacterStat: (sessionId, characterId, attributeKey, value, reason = 'manual') => {
   385→    // Default result
   386→    const defaultResult: UpdateCharacterStatResult = {
   387→      oldValue: undefined,
   388→      newValue: typeof value === 'number' ? value : value,
   389→      clamped: false,
   390→      thresholdsReached: [],
   391→    };
   392→
   393→    // Read current state
   394→    const state = get();
   395→    const sessions = state.sessions as Array<{ 
   396→      id: string; 
   397→      sessionStats?: SessionStats;
   398→      characterId?: string;
   399→      groupId?: string;
   400→    }>;
   401→    const sessionIndex = sessions.findIndex(s => s.id === sessionId);
   402→    
   403→    if (sessionIndex === -1) return defaultResult;
   404→    
   405→    const session = sessions[sessionIndex];
   406→    let sessionStats = session.sessionStats;
   407→    
   408→    // Auto-initialize sessionStats if missing
   409→    if (!sessionStats) {
   410→      sessionStats = {
   411→        characterStats: {},
   412→        solicitudes: {
   413→          characterSolicitudes: {},
   414→          lastModified: Date.now(),
   415→        },
   416→        ultimo_objetivo_completado: undefined,
   417→        ultima_solicitud_completada: undefined,
   418→        ultima_solicitud_realizada: undefined,
   419→        ultima_accion_realizada: undefined,
   420→        initialized: true,
   421→        lastModified: Date.now(),
   422→      };
   423→    }
   424→    
   425→    // Auto-initialize character stats if missing
   426→    if (!sessionStats.characterStats[characterId]) {
   427→      let statsConfig: CharacterStatsConfig | undefined;
   428→      if (characterId === '__user__') {
   429→        // Look up active persona's statsConfig for __user__
   430→        const activePersonaId = (state as any).activePersonaId;
   431→        const personas: any[] = (state as any).personas || [];
   432→        const activePersona = personas.find((p: any) => p.id === activePersonaId);
   433→        statsConfig = activePersona?.statsConfig;
   434→      } else {
   435→        const character = state.characters.find((c: any) => c.id === characterId);
   436→        statsConfig = character?.statsConfig;
   437→      }
   438→      let newStats = createDefaultCharacterStats(statsConfig);
   439→      // For persona (__user__), also initialize equipment slot values
   440→      if (characterId === '__user__') {
   441→        const equipmentSlots = (state as any).inventorySettings?.equipmentSlots;
   442→        if (equipmentSlots && equipmentSlots.length > 0) {
   443→          for (const slot of equipmentSlots) {
   444→            if (!(slot.key in newStats.attributeValues)) {
   445→              newStats.attributeValues[slot.key] = '';
   446→              newStats.lastUpdated[slot.key] = Date.now();
   447→            }
   448→          }
   449→        }
   450→      }
   451→      sessionStats = {
   452→        ...sessionStats,
   453→        characterStats: {
   454→          ...sessionStats.characterStats,
   455→          [characterId]: newStats,
   456→        },
   457→      };
   458→    }
   459→    
   460→    const stats = sessionStats.characterStats[characterId];
   461→    if (!stats) return defaultResult;
   462→
   463→    const oldValue = stats.attributeValues[attributeKey];
   464→
   465→    // Find attribute definition for logging and clamping
   466→    let attributeDef: AttributeDefinition | undefined;
   467→    if (characterId === '__user__') {
   468→      // Look up active persona's statsConfig for __user__
   469→      const activePersonaId = (state as any).activePersonaId;
   470→      const personas: any[] = (state as any).personas || [];
   471→      const activePersona = personas.find((p: any) => p.id === activePersonaId);
   472→      attributeDef = activePersona?.statsConfig?.attributes?.find(
   473→        (a: AttributeDefinition) => a.key === attributeKey
   474→      );
   475→    } else {
   476→      const character = state.characters.find((c: any) => c.id === characterId);
   477→      attributeDef = character?.statsConfig?.attributes?.find(
   478→        (a: AttributeDefinition) => a.key === attributeKey
   479→      );
   480→    }
   481→
   482→    // Clamp value to min/max bounds
   483→    const clampedValue = clampAttributeValue(value, attributeDef);
   484→    const clamped = clampedValue !== value;
   485→
   486→    // Log if clamping occurred
   487→    if (clamped) {
   488→      console.log(`[StatsSlice] Clamped ${attributeKey}: ${value} → ${clampedValue} (min: ${attributeDef?.min}, max: ${attributeDef?.max})`);
   489→    }
   490→
   491→    // Detect threshold reached
   492→    const thresholdsReached: ThresholdReachedInfo[] = [];
   493→    
   494→    if (attributeDef) {
   495→      // V2: Evaluate new thresholdEffects (flexible conditions with priority)
   496→      if (attributeDef.thresholdEffects && attributeDef.thresholdEffects.length > 0) {
   497→        // Build a temporary session stats with the new value for evaluation
   498→        const tempSessionStats: SessionStats = {
   499→          ...sessionStats,
   500→          characterStats: {
   501→            ...sessionStats.characterStats,
   502→            [characterId]: {
   503→              ...stats,
   504→              attributeValues: {
   505→                ...stats.attributeValues,
   506→                [attributeKey]: clampedValue,
   507→              },
   508→            },
   509→          },
   510→        };
   511→        
   512→        const matchingEffects = evaluateThresholdEffects(
   513→          attributeDef.thresholdEffects,
   514→          tempSessionStats,
   515→          characterId
   516→        );
   517→        
   518→        for (const effect of matchingEffects) {
   519→          thresholdsReached.push({
   520→            attributeKey: attributeDef.key,
   521→            attributeName: attributeDef.name,
   522→            thresholdType: 'custom',
   523→            effectName: effect.name,
   524→            effectId: effect.id,
   525→            priority: effect.priority,
   526→            rewards: effect.rewards,
   527→          });
   528→          console.log(`[StatsSlice] Threshold effect "${effect.name}" triggered for ${attributeDef.name} (priority: ${effect.priority})`);
   529→        }
   530→      }
   531→      
   532→      // Legacy: Check old onMinReached/onMaxReached (only if no thresholdEffects defined)
   533→      if (!attributeDef.thresholdEffects || attributeDef.thresholdEffects.length === 0) {
   534→        if (typeof clampedValue === 'number') {
   535→          // Check if reached minimum
   536→          if (attributeDef.min !== undefined && clampedValue === attributeDef.min) {
   537→            if (attributeDef.onMinReached?.enabled && attributeDef.onMinReached.rewards.length > 0) {
   538→              thresholdsReached.push({
   539→                attributeKey: attributeDef.key,
   540→                attributeName: attributeDef.name,
   541→                thresholdType: 'min',
   542→                thresholdValue: attributeDef.min,
   543→                rewards: attributeDef.onMinReached.rewards,
   544→              });
   545→              console.log(`[StatsSlice] Threshold reached: ${attributeDef.name} hit minimum (${attributeDef.min})`);
   546→            }
   547→          }
   548→          
   549→          // Check if reached maximum
   550→          if (attributeDef.max !== undefined && clampedValue === attributeDef.max) {
   551→            if (attributeDef.onMaxReached?.enabled && attributeDef.onMaxReached.rewards.length > 0) {
   552→              thresholdsReached.push({
   553→                attributeKey: attributeDef.key,
   554→                attributeName: attributeDef.name,
   555→                thresholdType: 'max',
   556→                thresholdValue: attributeDef.max,
   557→                rewards: attributeDef.onMaxReached.rewards,
   558→              });
   559→              console.log(`[StatsSlice] Threshold reached: ${attributeDef.name} hit maximum (${attributeDef.max})`);
   560→            }
   561→          }
   562→        }
   563→      }
   564→    }
   565→
   566→    // Build the updated state
   567→    const updatedCharacterStats = {
   568→      ...sessionStats.characterStats,
   569→      [characterId]: {
   570→        ...stats,
   571→        attributeValues: {
   572→          ...stats.attributeValues,
   573→          [attributeKey]: clampedValue,
   574→        },
   575→        lastUpdated: {
   576→          ...stats.lastUpdated,
   577→          [attributeKey]: Date.now(),
   578→        },
   579→      },
   580→    };
   581→
   582→    // Add to change log
   583→    addChangeLogEntry(
   584→      updatedCharacterStats[characterId],
   585→      attributeDef,
   586→      attributeKey,
   587→      oldValue,
   588→      clampedValue,
   589→      reason
   590→    );
   591→    
   592→    const newSessionStats: SessionStats = {
   593→      ...sessionStats,
   594→      characterStats: updatedCharacterStats,
   595→      lastModified: Date.now(),
   596→    };
   597→    
   598→    // Update the store
   599→    set({
   600→      sessions: state.sessions.map((s: any) =>
   601→        s.id === sessionId
   602→          ? { 
   603→              ...s, 
   604→              sessionStats: newSessionStats,
   605→              updatedAt: new Date().toISOString() 
   606→            }
   607→          : s
   608→      ),
   609→    });
   610→
   611→    // Sync currency changes to persona (Inventory V2)
   612→    // When currency is updated via quest rewards/triggers on __user__,
   613→    // also update persona.currency so the InventoryPanel/HUD stay in sync
   614→    if (characterId === '__user__' && attributeKey === 'currency' && typeof clampedValue === 'number') {
   615→      try {
   616→        const activePersonaId = (get() as any).activePersonaId;
   617→        if (activePersonaId) {
   618→          (get() as any).updatePersona?.(activePersonaId, { currency: clampedValue });
   619→        }
   620→      } catch { /* non-critical sync */ }
   621→    }
   622→
   623→    // Refresh objective visibility since attribute conditions may have changed
   624→    try {
   625→      (get() as any).refreshAllObjectiveVisibility?.(sessionId);
   626→    } catch { /* non-critical */ }
   627→
   628→    // Refresh activation conditions since attribute conditions may have changed
   629→    try {
   630→      (get() as any).refreshAllActivationConditions?.(sessionId);
   631→    } catch { /* non-critical */ }
   632→
   633→    // Return result with threshold info
   634→    return {
   635→      oldValue,
   636→      newValue: clampedValue,
   637→      clamped,
   638→      thresholdsReached,
   639→    };
   640→  },

   719→        statsConfig = character?.statsConfig;

   746→          attributeId: attributeDef?.id || update.attributeKey,
   747→          attributeKey: update.attributeKey,
   748→          attributeName: attributeDef?.name || update.attributeKey,
   749→          oldValue: oldValue ?? '',
   750→          newValue: clampedValue,
   751→          reason,
   752→          timestamp: now,
   753→        });

   755→
