    89→/**
    90→ * Result of responder selection including metadata about why certain characters were selected
    91→ */
    92→interface ResponderSelectionResult {

    95→  reasons: Map<string, string>; // characterId -> reason for selection

    98→/**
    99→ * Get characters that have pending solicitudes (requests they need to respond to)
   100→ */
   101→function getCharactersWithPendingSolicitudes(
   102→  sessionStats: SessionStats | undefined,
   103→  eligibleCharacterIds: string[]
   104→): { characterId: string; fromCharacterName: string }[] {
   105→  if (!sessionStats?.solicitudes?.characterSolicitudes) {
   106→    return [];
   107→  }
   108→
   109→  const result: { characterId: string; fromCharacterName: string }[] = [];
   110→
   111→  for (const characterId of eligibleCharacterIds) {
   112→    const solicitudes = sessionStats.solicitudes.characterSolicitudes[characterId];
   113→    if (solicitudes) {
   114→      const pendingSolicitudes = solicitudes.filter(s => s.status === 'pending');
   115→      if (pendingSolicitudes.length > 0) {
   116→        result.push({
   117→          characterId,
   118→          fromCharacterName: pendingSolicitudes[0].fromCharacterName
   119→        });
   120→      }
   121→    }
   122→  }
   123→
   124→  return result;
   125→}

   127→/**
   128→ * Check if there are pending solicitudes targeting the user
   129→ */
   130→function hasPendingSolicitudesForUser(
   131→  sessionStats: SessionStats | undefined
   132→): boolean {
   133→  if (!sessionStats?.solicitudes?.characterSolicitudes) {
   134→    return false;
   135→  }
   136→
   137→  const userSolicitudes = sessionStats.solicitudes.characterSolicitudes['__user__'];
   138→  return userSolicitudes?.some(s => s.status === 'pending') ?? false;
   139→}

   141→/**
   142→ * Determine responders based on strategy
   143→ * Note: Narrators are excluded from normal response flow and handled separately
   144→ */
   145→function getResponders(
   146→  message: string,
   147→  characters: CharacterCard[],
   148→  group: CharacterGroup,
   149→  lastResponderId?: string,
   150→  sessionStats?: SessionStats
   151→): ResponderSelectionResult {
   152→  const strategy = group.activationStrategy;
   153→  const minResponses = group.minResponsesPerTurn ?? 1;
   154→  const maxResponses = group.maxResponsesPerTurn ?? 3;
   155→
   156→  // Get active members, EXCLUDING narrators (they have their own response logic)
   157→  const activeMemberIds = (group.members || [])
   158→    .filter(m => m.isActive && m.isPresent !== false && !m.isNarrator)
   159→    .map(m => m.characterId);
   160→
   161→  // If no members defined, use characterIds (excluding narrators)
   162→  const eligibleIds = activeMemberIds.length > 0
   163→    ? activeMemberIds
   164→    : (group.characterIds || []).filter(id => {
   165→        const member = (group.members || []).find(m => m.characterId === id);
   166→        return !member?.isNarrator;
   167→      });
   168→
   169→  // Filter to only characters that exist and are eligible
   170→  const eligibleCharacters = characters.filter(c => eligibleIds.includes(c.id));
   171→
   172→  if (eligibleCharacters.length === 0) {
   173→    return { responders: [], stopForUser: false, reasons: new Map() };
   174→  }
   175→
   176→  // Get ordered member IDs (excluding narrators)
   177→  const orderedIds = (group.members || [])
   178→    .filter(m => !m.isNarrator)
   179→    .sort((a, b) => a.joinOrder - b.joinOrder)
   180→    .map(m => m.characterId);
   181→
   182→  const reasons = new Map<string, string>();
   183→
   184→  switch (strategy) {
   185→    case 'all': {
   186→      // All active members respond (no limit for 'all' strategy)
   187→      eligibleCharacters.forEach(c => reasons.set(c.id, 'Todos responden'));
   188→      return { responders: eligibleCharacters, stopForUser: false, reasons };
   189→    }
   190→
   191→    case 'reactive': {
   192→      // ========================================
   193→      // REACTIVE STRATEGY with Solicitud Support
   194→      // ========================================
   195→      // Priority:
   196→      // 1. If peticion targets user -> STOP, let user respond
   197→      // 2. Characters with pending solicitudes respond
   198→      // 3. Mentioned characters respond
   199→      // 4. Fill to minResponses if needed
   200→
   201→      // Check if user has pending solicitudes
   202→      const stopForUser = hasPendingSolicitudesForUser(sessionStats);
   203→      if (stopForUser) {
   204→        console.log('[getResponders] Peticion targets user, stopping turn for user response');
   205→        return { responders: [], stopForUser: true, reasons: new Map() };
   206→      }
   207→
   208→      // Detect mentions
   209→      const mentions = detectMentions(message, characters, group);
   210→      const mentionedIds = mentions.map(m => m.characterId);
   211→      const mentionedCharacters = eligibleCharacters.filter(c => mentionedIds.includes(c.id));
   212→      mentionedCharacters.forEach(c => reasons.set(c.id, 'Mencionado en el mensaje'));
   213→
   214→      // Get characters with pending solicitudes
   215→      const charactersWithSolicitudes = getCharactersWithPendingSolicitudes(sessionStats, eligibleIds);
   216→      const solicitudCharacterIds = charactersWithSolicitudes.map(s => s.characterId);
   217→      const solicitudCharacters = eligibleCharacters.filter(c => solicitudCharacterIds.includes(c.id));
   218→
   219→      // Add solicitud reasons (don't overwrite mention reasons)
   220→      solicitudCharacters.forEach(c => {
   221→        if (!reasons.has(c.id)) {
   222→          const solicitudInfo = charactersWithSolicitudes.find(s => s.characterId === c.id);
   223→          reasons.set(c.id, `Tiene solicitud pendiente de ${solicitudInfo?.fromCharacterName || 'otro personaje'}`);
   224→        }
   225→      });
   226→
   227→      // Combine: unique characters from mentions and solicitudes
   228→      const combinedIds = new Set([...mentionedIds, ...solicitudCharacterIds]);
   229→      let selectedCharacters = eligibleCharacters.filter(c => combinedIds.has(c.id));
   230→
   231→      // If no mentions or solicitudes, fill to minResponses
   232→      if (selectedCharacters.length === 0) {
   233→        // Select first eligible character as default
   234→        selectedCharacters = [eligibleCharacters[0]];
   235→        reasons.set(eligibleCharacters[0].id, 'Personaje por defecto (sin menciones ni solicitudes)');
   236→      }
   237→
   238→      // Ensure we have at least minResponses (but respect maxResponses)
   239→      if (selectedCharacters.length < minResponses) {
   240→        const remaining = eligibleCharacters
   241→          .filter(c => !combinedIds.has(c.id))
   242→          .slice(0, minResponses - selectedCharacters.length);
   243→        remaining.forEach(c => reasons.set(c.id, 'Para alcanzar mínimo de respuestas'));
   244→        selectedCharacters = [...selectedCharacters, ...remaining];
   245→      }
   246→
   247→      // Limit to maxResponses
   248→      const limitedResponders = selectedCharacters.slice(0, maxResponses);
   249→
   250→      console.log('[getResponders] Reactive selection:', {
   251→        mentionedIds,
   252→        solicitudCharacterIds,
   253→        selectedCount: limitedResponders.length,
   254→        minResponses,
   255→        maxResponses,
   256→        stopForUser
   257→      });
   258→
   259→      return { responders: limitedResponders, stopForUser, reasons };
   260→    }
   261→
   262→    case 'round_robin': {
   263→      // Take turns in order
   264→      const sortedIds = orderedIds.length > 0 ? orderedIds : eligibleIds;
   265→
   266→      let nextIndex = 0;
   267→      if (lastResponderId) {
   268→        const lastIndex = sortedIds.indexOf(lastResponderId);
   269→        if (lastIndex !== -1) {
   270→          nextIndex = (lastIndex + 1) % sortedIds.length;
   271→        }
   272→      }
   273→
   274→      const roundRobinChar = characters.find(c => c.id === sortedIds[nextIndex]);
   275→      if (roundRobinChar) {
   276→        reasons.set(roundRobinChar.id, 'Turno rotativo');
   277→      }
   278→      return { responders: roundRobinChar ? [roundRobinChar] : [], stopForUser: false, reasons };
   279→    }
   280→
   281→    case 'random': {
   282→      // Random selection
   283→      const shuffled = [...eligibleCharacters].sort(() => Math.random() - 0.5);
   284→      const selected = shuffled.slice(0, Math.min(maxResponses, shuffled.length));
   285→      selected.forEach(c => reasons.set(c.id, 'Selección aleatoria'));
   286→      return { responders: selected, stopForUser: false, reasons };
   287→    }
   288→
   289→    case 'smart': {
   290→      // AI-like decision: mentioned characters + contextually relevant
   291→      const mentions = detectMentions(message, characters, group);
   292→      const mentionedIds = mentions.map(m => m.characterId);
   293→      const mentionedChars = eligibleCharacters.filter(c => mentionedIds.includes(c.id));
   294→      mentionedChars.forEach(c => reasons.set(c.id, 'Mencionado en el mensaje'));
   295→
   296→      // Add contextually relevant characters
   297→      const remainingChars = eligibleCharacters.filter(c => !mentionedIds.includes(c.id));
   298→      const additionalCount = Math.max(0, Math.min(maxResponses - mentionedChars.length, 1));
   299→
   300→      // Check if character name or tags appear in message
   301→      const relevantChars = remainingChars.filter(c => {
   302→        const keywords = [...c.tags, c.name.toLowerCase()];
   303→        return keywords.some(kw => message.toLowerCase().includes(kw.toLowerCase()));
   304→      }).slice(0, additionalCount);
   305→      relevantChars.forEach(c => reasons.set(c.id, 'Contextualmente relevante'));
   306→
   307→      const result = [...mentionedChars, ...relevantChars].slice(0, maxResponses);
   308→
   309→      // If no one selected, default to first
   310→      if (result.length === 0 && eligibleCharacters.length > 0) {
   311→        result.push(eligibleCharacters[0]);
   312→        reasons.set(eligibleCharacters[0].id, 'Personaje por defecto');
   313→      }
   314→
   315→      return { responders: result, stopForUser: false, reasons };
   316→    }
   317→
   318→    default: {
   319→      // Default to first active character
   320→      if (eligibleCharacters.length > 0) {
   321→        reasons.set(eligibleCharacters[0].id, 'Personaje por defecto');
   322→      }
   323→      return { responders: eligibleCharacters.slice(0, 1), stopForUser: false, reasons };
   324→    }
   325→  }
   326→}

   338→/**
   339→ * Execute detected tool calls for group chat and send SSE events.
   340→ * Returns { results: display messages, shouldContinue: true, questActivations: quest activations }
   341→ */
   342→async function executeGroupToolCalls(
   343→  toolCalls: NativeToolCall[],
   344→  availableTools: Array<{ id: string; name: string; label: string; icon: string }>,
   345→  character: CharacterCard,
   346→  sessionId: string,
   347→  userName: string,
   348→  controller: { enqueue: (chunk: string) => void },
   349→  sessionQuests?: SessionQuestInstance[],
   350→  questTemplates?: QuestTemplate[],
   351→  statsConfig?: CharacterStatsConfig,
   352→  sessionStats?: SessionStats,
   353→  allCharacters?: CharacterCard[],
   354→  characterMemory?: CharacterMemory,
   355→  lorebooks?: Lorebook[],
   356→): Promise<{ results: string; shouldContinue: boolean; questActivations: QuestActivation[]; toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> }> {
   357→  if (toolCalls.length === 0) {
   358→    return { results: '', shouldContinue: false, questActivations: [], toolsUsed: [] };
   359→  }
   360→
   361→  let allDisplayMessages = '';
   362→  const questActivations: QuestActivation[] = [];
   363→  const toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> = [];
   364→
   365→  for (let i = 0; i < toolCalls.length; i++) {
   366→    const tc = toolCalls[i];
   367→    const toolDef = availableTools.find(t => t.name === tc.name);
   368→    const callId = `${Date.now()}_${i}`;
   369→
   370→    // Send tool_call_start event
   371→    controller.enqueue(createSSEJSON({
   372→      type: 'tool_call_start',
   373→      toolName: tc.name,
   374→      toolLabel: toolDef?.label || tc.name,
   375→      toolIcon: toolDef?.icon || 'Wrench',
   376→      params: tc.arguments,
   377→      callId,
   378→    }));
   379→
   380→    console.log(`[GroupStream-Tools] Executing: ${tc.name}`, tc.arguments);
   381→
   382→    // Execute the tool
   383→    const toolResult = await executeTool(
   384→      tc.name,
   385→      tc.arguments,
   386→      {
   387→        characterId: character.id,
   388→        characterName: character.name,
   389→        sessionId,
   390→        userName,
   391→        sessionQuests,
   392→        questTemplates,
   393→        statsConfig: character.statsConfig,
   394→        sessionStats,
   395→        allCharacters,
   396→        characterMemory,
   397→        lorebooks,
   398→      },
   399→    );
   400→
   401→    // Check for quest activation and send SSE event
   402→    if (toolResult.questActivation) {
   403→      const activation = toolResult.questActivation;
   404→      console.log(`[GroupStream-Tools] Quest activation from ${tc.name}:`, activation);
   405→      
   406→      controller.enqueue(createSSEJSON({
   407→        type: 'quest_activation',
   408→        toolName: tc.name,
   409→        activationType: activation.type,
   410→        key: activation.key,
   411→        metadata: activation.metadata,
   412→      }));
   413→      
   414→      questActivations.push(activation);
   415→    }
   416→
   417→    // Check for action/skill activation and send SSE event
   418→    if (toolResult.actionActivation) {
   419→      const action = toolResult.actionActivation;
   420→      console.log(`[GroupStream-Tools] Action activation from ${tc.name}:`, action.skillName);
   421→      
   422→      controller.enqueue(createSSEJSON({
   423→        type: 'action_activation',
   424→        toolName: tc.name,
   425→        skillId: action.skillId,
   426→        skillName: action.skillName,
   427→        skillDescription: action.skillDescription,
   428→        skillCompletedDescription: action.skillCompletedDescription,
   429→        activationCosts: action.activationCosts,
   430→        activationRewards: action.activationRewards,
   431→        characterId: action.characterId,
   432→      }));
   433→    }
   434→
   435→    // Check for stat modification from modify_stat tool and send SSE event
   436→    if (toolResult.statActivation) {
   437→      const stat = toolResult.statActivation;
   438→      console.log(`[GroupStream-Tools] Stat activation from ${tc.name}:`, stat.attributeKey, stat.oldValue, '→', stat.newValue);
   439→      
   440→      controller.enqueue(createSSEJSON({
   441→        type: 'stat_activation',
   442→        toolName: tc.name,
   443→        characterId: stat.characterId,
   444→        attributeKey: stat.attributeKey,
   445→        attributeName: stat.attributeName,
   446→        attributeType: stat.attributeType,
   447→        oldValue: stat.oldValue,
   448→        newValue: stat.newValue,
   449→        reason: stat.reason,
   450→      }));
   451→    }
   452→
   453→    // Check for solicitud activation/completion and send SSE event
   454→    if (toolResult.solicitudActivation) {
   455→      const sol = toolResult.solicitudActivation;
   456→      console.log(`[GroupStream-Tools] Solicitud activation from ${tc.name}:`, sol.type, sol.solicitudKey);
   457→      
   458→      controller.enqueue(createSSEJSON({
   459→        type: 'solicitud_activation',
   460→        toolName: tc.name,
   461→        activationType: sol.type,
   462→        solicitudKey: sol.solicitudKey,
   463→        targetCharacterId: sol.targetCharacterId,
   464→        targetCharacterName: sol.targetCharacterName,
   465→        fromCharacterId: sol.fromCharacterId,
   466→        fromCharacterName: sol.fromCharacterName,
   467→        description: sol.description,
   468→        completionDescription: sol.completionDescription,
   469→        peticionKey: sol.peticionKey,
   470→      }));
   471→    }
   472→
   473→    // Check for memory activation (sync to client-side Character Memory)
   474→    if (toolResult.memoryActivation) {
   475→      const mem = toolResult.memoryActivation;
   476→      console.log(`[GroupStream-Tools] Memory activation from ${tc.name}:`, mem.type);
   477→      
   478→      controller.enqueue(createSSEJSON({
   479→        type: 'memory_activation',
   480→        toolName: tc.name,
   481→        activationType: mem.type,
   482→        characterId: mem.characterId,
   483→        eventData: mem.eventData,
   484→        relationshipData: mem.relationshipData,
   485→        noteContent: mem.noteContent,
   486→        deleteEventId: mem.deleteEventId,
   487→        deleteEmbeddingId: mem.deleteEmbeddingId,
   488→      }));
   489→    }
   490→
   491→    // Send tool_call_result event
   492→    controller.enqueue(createSSEJSON({
   493→      type: 'tool_call_result',
   494→      toolName: tc.name,
   495→      success: toolResult.success,
   496→      displayMessage: toolResult.displayMessage,
   497→      duration: toolResult.duration || 0,
   498→      callId,
   499→    }));
   500→
   501→    console.log(`[GroupStream-Tools] ${tc.name}: success=${toolResult.success}`);
   502→
   503→    toolsUsed.push({
   504→      name: tc.name,
   505→      label: toolDef?.label || tc.name,
   506→      icon: toolDef?.icon || 'Wrench',
   507→      success: toolResult.success,
   508→    });
   509→
   510→    if (toolResult.displayMessage) {
   511→      allDisplayMessages += (allDisplayMessages ? '\n' : '') + toolResult.displayMessage;
   512→    }
   513→  }
   514→
   515→  return { results: allDisplayMessages, shouldContinue: true, questActivations, toolsUsed };
   516→}

   522→export async function POST(request: NextRequest) {
   523→  try {
   524→    // Capture auth headers from Z.ai gateway for token resolution
   525→    // Only use actual JWT tokens - session IDs are NOT valid X-Tokens
   526→    const incomingXToken = request.headers.get('X-Token');
   527→    const fcSecurityToken = request.headers.get('x-fc-security-token');
   528→
   529→    const body = await request.json();
   530→
   531→    // Validate request (automatically detects group request)
   532→    const validation = validateRequest(null, body);
   533→    if (!validation.success) {
   534→      return createErrorResponse(validation.error, 400);
   535→    }
   536→
   537→    const {
   538→      message,
   539→      group,
   540→      characters,
   541→      messages = [],
   542→      llmConfig,
   543→      userName = 'User',
   544→      persona,
   545→      lastResponderId,
   546→      sessionStats,
   547→      hudContext
   548→    } = validation.data;
   549→
   550→    // Extract lorebooks from body (not validated by validation.ts)
   551→    const lorebooks: Lorebook[] = body.lorebooks || [];
   552→
   553→    // Build allCharacters including persona for cross-character lookups (target requirements, peticiones)
   554→    const allCharacters: CharacterCard[] = [...(characters || [])];
   555→    if (persona?.statsConfig?.enabled) {
   556→      allCharacters.push({
   557→        id: '__user__',
   558→        name: persona.name || 'User',
   559→        statsConfig: persona.statsConfig,
   560→      } as CharacterCard);
   561→    }
   562→
   563→    // Quest data still needed for buildGroupSystemPrompt {{activeQuests}} key resolution
   564→    const questTemplates: QuestTemplate[] = body.questTemplates || [];
   565→    const sessionQuests: SessionQuestInstance[] = body.sessionQuests || [];
   566→    const questSettings = { ...DEFAULT_QUEST_SETTINGS, ...(body.questSettings || {}) };
   567→
   568→    // Extract summaries for memory/context compression
   569→    const summary: SessionSummary | undefined = body.summary;
   570→
   571→    // Extract embeddings chat settings
   572→    const embeddingsChat: Partial<EmbeddingsChatSettings> = body.embeddingsChat || {};
   573→    const sessionId: string | undefined = body.sessionId;
   574→    const characterMemory: CharacterMemory | undefined = body.characterMemory;
   575→
   576→    // Extract tools settings for tool/action system (native tool calling only)
   577→    const toolsSettings: ToolsSettings = {
   578→      enabled: body.toolsSettings?.enabled ?? true,
