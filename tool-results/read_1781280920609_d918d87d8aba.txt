   962→
   963→interface AttributeHUDPreviewProps {
   964→  attribute: AttributeDefinition;
   965→}
   966→
   967→function AttributeHUDPreview({ attribute }: AttributeHUDPreviewProps) {
   968→  const value = attribute.defaultValue ?? (attribute.type === 'number' ? 0 : '');
   969→  const style = attribute.hudStyle || 'default';
   970→  const color = attribute.color || 'default';
   971→  const icon = attribute.icon;
   972→  const unit = attribute.hudUnit;
   973→  const min = attribute.min ?? 0;
   974→  const max = attribute.max ?? 100;
   975→  
   976→  // Color classes
   977→  const colorClasses: Record<string, string> = {
   978→    red: 'text-red-400',
   979→    green: 'text-green-400',
   980→    blue: 'text-blue-400',
   981→    yellow: 'text-yellow-400',
   982→    purple: 'text-purple-400',
   983→    orange: 'text-orange-400',
   984→    pink: 'text-pink-400',
   985→    cyan: 'text-cyan-400',
   986→    default: 'text-white/80',
   987→  };
   988→  
   989→  const bgColorClasses: Record<string, string> = {
   990→    red: 'bg-red-500/20 border-red-500/30',
   991→    green: 'bg-green-500/20 border-green-500/30',
   992→    blue: 'bg-blue-500/20 border-blue-500/30',
   993→    yellow: 'bg-yellow-500/20 border-yellow-500/30',
   994→    purple: 'bg-purple-500/20 border-purple-500/30',
   995→    orange: 'bg-orange-500/20 border-orange-500/30',
   996→    pink: 'bg-pink-500/20 border-pink-500/30',
   997→    cyan: 'bg-cyan-500/20 border-cyan-500/30',
   998→    default: 'bg-white/10 border-white/20',
   999→  };
  1000→  
  1001→  const progressColorClasses: Record<string, string> = {
  1002→    red: 'bg-red-500',
  1003→    green: 'bg-green-500',
  1004→    blue: 'bg-blue-500',
  1005→    yellow: 'bg-yellow-500',
  1006→    purple: 'bg-purple-500',
  1007→    orange: 'bg-orange-500',
  1008→    pink: 'bg-pink-500',
  1009→    cyan: 'bg-cyan-500',
  1010→    default: 'bg-white/50',
  1011→  };
  1012→  
  1013→  const textColor = colorClasses[color] || colorClasses.default;
  1014→  const bgColor = bgColorClasses[color] || bgColorClasses.default;
  1015→  const progressColor = progressColorClasses[color] || progressColorClasses.default;
  1016→  
  1017→  // Render based on style
  1018→  switch (style) {
  1019→    case 'progress': {
  1020→      const percentage = Math.max(0, Math.min(100, ((Number(value) - min) / (max - min)) * 100));
  1021→      return (
  1022→        <div className="space-y-1">
  1023→          <div className="flex items-center justify-between gap-2">
  1024→            <div className="flex items-center gap-1.5">
  1025→              {icon && <span className="text-sm">{icon}</span>}
  1026→              <span className="text-xs text-white/50">{attribute.name}</span>
  1027→            </div>
  1028→            <span className="text-xs font-medium text-white/80">
  1029→              {value}{unit && <span className="text-white/40 ml-0.5">{unit}</span>}
  1030→            </span>
  1031→          </div>
  1032→          <div className="w-full bg-white/10 rounded-full h-2">
  1033→            <div className={`h-full rounded-full transition-all ${progressColor}`} style={{ width: `${percentage}%` }} />
  1034→          </div>
  1035→        </div>
  1036→      );
  1037→    }
