     8→// This module implements the sprite priority system. The rules are:
     9→//
    10→// 1. TRIGGER SPRITE HAS ABSOLUTE PRIORITY
    11→//    - Once set, triggerSpriteUrl MUST NOT be cleared by:
    12→//      * startGenerationForCharacter()
    13→//      * endGenerationForCharacter()
    14→//      * State changes (talk/thinking/idle)
    15→//    - Only cleared by:
    16→//      * Timer expiration (returnToIdleMs > 0)
    17→//      * New trigger replacing it
    18→//      * User manual action
    19→//
    20→// 2. TIMER BEHAVIOR
    21→//    - returnToIdleMs = 0: Trigger persists indefinitely
    22→//    - returnToIdleMs > 0: Trigger clears after X ms
    23→//
    24→// 3. RETURN MODE
    25→//    - 'clear': Clear trigger, show state-based sprite (talk/thinking/idle)
    26→//    - 'idle': Clear trigger, show idle sprite
    27→//    - 'talk': Clear trigger, show talk sprite
    28→//    - 'thinking': Clear trigger, show thinking sprite
    29→//
    30→// See: /docs/SPRITE_PRIORITY_SYSTEM.md for full documentation

    65→// Active sound chain tracking
    66→const activeSoundChains = new Map<string, { timeouts: NodeJS.Timeout[]; audios: HTMLAudioElement[] }>();
    67→
    68→/**
    69→ * Play a sound chain for a character
    70→ * This is called directly when startSoundChain is invoked
    71→ */
    72→function playSoundChainForCharacter(
    73→  characterId: string,
    74→  chain: SoundChain,
    75→  soundTriggers: SoundTrigger[],
    76→  soundCollections: SoundCollection[],
    77→  globalVolume: number
    78→): void {
    79→  // Stop any existing chain for this character
    80→  stopSoundChainForCharacter(characterId);
    81→  
    82→  const chainState = { timeouts: [] as NodeJS.Timeout[], audios: [] as HTMLAudioElement[] };
    83→  activeSoundChains.set(characterId, chainState);
    84→  
    85→  const chainVolume = (chain.volume ?? 1) * globalVolume;
    86→  
    87→  console.log('[SpriteSlice] Playing sound chain:', {
    88→    characterId,
    89→    steps: chain.steps.length,
    90→    volume: chainVolume,
    91→  });
    92→  
    93→  chain.steps.forEach((step: SoundChainStep, index: number) => {
    94→    const timeout = setTimeout(() => {
    95→      playSoundChainStep(
    96→        characterId,
    97→        step,
    98→        soundTriggers,
    99→        soundCollections,
   100→        chainVolume
   101→      );
   102→    }, step.delayMs);
   103→    
   104→    chainState.timeouts.push(timeout);
   105→  });
   106→}

   108→/**
   109→ * Play a single sound chain step
   110→ */
   111→function playSoundChainStep(
   112→  characterId: string,
   113→  step: SoundChainStep,
   114→  soundTriggers: SoundTrigger[],
   115→  soundCollections: SoundCollection[],
   116→  chainVolume: number
   117→): void {
   118→  let soundUrl: string | null = null;
   119→  let volume = (step.volume ?? 1) * chainVolume;
   120→  
   121→  // Try sound trigger key first
   122→  if (step.soundTriggerKey) {
   123→    const trigger = soundTriggers.find(t => t.name === step.soundTriggerKey || t.id === step.soundTriggerKey);
   124→    if (trigger) {
   125→      const collection = soundCollections.find(c => c.name === trigger.collection);
   126→      if (collection && collection.files.length > 0) {
   127→        // Pick sound based on play mode
   128→        let soundFile: string;
   129→        if (trigger.playMode === 'random') {
   130→          soundFile = collection.files[Math.floor(Math.random() * collection.files.length)];
   131→        } else {
   132→          const index = (trigger.currentIndex || 0) % collection.files.length;
   133→          soundFile = collection.files[index];
   134→        }
   135→        soundUrl = soundFile;
   136→        volume *= trigger.volume;
   137→      }
   138→    }
   139→  }
   140→  // Fall back to direct URL
   141→  else if (step.soundUrl) {
   142→    soundUrl = step.soundUrl;
   143→  }
   144→  
   145→  if (!soundUrl) {
   146→    console.warn('[SpriteSlice] No sound URL for step:', step);
   147→    return;
   148→  }
   149→  
   150→  try {
   151→    const audio = new Audio(soundUrl);
   152→    audio.volume = Math.min(1, Math.max(0, volume));
   153→    
   154→    // Emit comic sound visual event
   155→    const stepName = step.soundTriggerKey || 'chain_sound';
   156→    emitComicSoundEvent(stepName, stepName, characterId);
   157→    
   158→    audio.play().catch(e => {
   159→      console.warn('[SpriteSlice] Audio play failed:', e);
   160→    });
   161→    
   162→    // Track the audio
   163→    const chainState = activeSoundChains.get(characterId);
   164→    if (chainState) {
   165→      chainState.audios.push(audio);
   166→    }
   167→    
   168→    audio.onended = () => {
   169→      const state = activeSoundChains.get(characterId);
   170→      if (state) {
   171→        const idx = state.audios.indexOf(audio);
   172→        if (idx > -1) state.audios.splice(idx, 1);
   173→      }
   174→    };
   175→  } catch (error) {
   176→    console.error('[SpriteSlice] Failed to play sound:', error);
   177→  }
   178→}

   180→/**
   181→ * Stop sound chain for a character
   182→ */
   183→function stopSoundChainForCharacter(characterId: string): void {
   184→  const chainState = activeSoundChains.get(characterId);
   185→  if (!chainState) return;
   186→  
   187→  // Clear all timeouts
   188→  chainState.timeouts.forEach(t => clearTimeout(t));
   189→  
   190→  // Stop all audio
   191→  chainState.audios.forEach(a => {
   192→    a.pause();
   193→    a.remove();
   194→  });
   195→  
   196→  activeSoundChains.delete(characterId);
   197→}
