  2782→/**
  2783→ * Quest Reward: Conditional Sprite Collection
  2784→ *
  2785→ * Activates a Trigger Collection that is in conditional mode.
  2786→ * Instead of a simple key-based trigger, this reward evaluates
  2787→ * the collection's ConditionalSpriteEntries against current attribute values,
  2788→ * selecting the sprite based on priority and conditions.
  2789→ *
  2790→ * Flow:
  2791→ * 1. Find TriggerCollection by collectionId
  2792→ * 2. If collection.conditionalMode = true:
  2793→ *    a. Sort conditionalEntries by priority (DESC)
  2794→ *    b. Evaluate each entry's conditions against sessionStats
  2795→ *    c. First matching entry → use its spriteId
  2796→ *    d. No match → use defaultSpriteId or principalSpriteId
  2797→ * 3. Apply sprite as trigger + schedule fallback
  2798→ */
  2799→export interface QuestRewardConditionalSpriteCollection {
  2800→  collectionId: string;              // TriggerCollection to activate (must have conditionalMode = true)
  2801→  targetMode: TriggerTargetMode;     // Who receives the sprite
  2802→  targetCharacterId?: string;        // For 'target' mode
  2803→  returnToIdleMs?: number;           // Time before returning to idle (0 = persist)
  2804→  fallbackMode?: TriggerFallbackMode; // What happens after trigger expires
  2805→}

  2807→// ============================================
  2808→// ACTIVATE SPRITE PACK REWARD
  2809→// ============================================
  2810→
  2811→/**
  2812→ * Quest Reward: Activate Sprite Pack
  2813→ *
  2814→ * Activates a SpritePackV2 directly. If the pack has conditionalMode enabled,
  2815→ * it evaluates each sprite's conditions (by priority) to pick the right sprite.
  2816→ * If no conditions match, uses the defaultSpriteId or falls back to behavior.
  2817→ *
  2818→ * Flow:
  2819→ * 1. Find SpritePackV2 by packId from character's spritePacksV2
  2820→ * 2. If pack.conditionalMode = true:
  2821→ *    a. Sort sprites with conditionalEnabled by priority (DESC)
  2822→ *    b. Evaluate each sprite's conditions against sessionStats
  2823→ *    c. First matching sprite → use it
  2824→ *    d. No match → use defaultSpriteId or first sprite
  2825→ * 3. If pack.conditionalMode = false:
  2826→ *    a. Use behavior-based resolution (principal/random/list)
  2827→ * 4. Apply sprite as trigger for the target character(s)
  2828→ * 5. Schedule return to idle if returnToIdleMs > 0
  2829→ */
  2830→export interface QuestRewardActivateSpritePack {
  2831→  packId: string;                      // SpritePackV2 ID to activate
  2832→  behavior?: 'principal' | 'random' | 'list';  // Behavior override (defaults to principal)
  2833→  principalSpriteId?: string;          // For 'principal' behavior
  2834→  targetMode: TriggerTargetMode;       // Who receives the sprite
  2835→  targetCharacterId?: string;          // For 'target' mode - which character to apply the pack to
  2836→  returnToIdleMs?: number;             // Time before returning to idle (0 = persist)
  2837→  fallbackMode?: TriggerFallbackMode;  // What happens after trigger expires
  2838→  fallbackPackId?: string;             // Fallback sprite pack to activate when this one expires (for 'self' mode)
  2839→  targetPackId?: string;               // Sprite pack to activate on target character (for 'target' mode)
  2840→}
