   183→interface TriggerCollectionEditorProps {
   184→  character: CharacterCard;
   185→  onChange: (updates: Partial<CharacterCard>) => void;
   186→}
   187→
   188→export function TriggerCollectionEditor({ 
   189→  character, 
   190→  onChange 
   191→}: TriggerCollectionEditorProps) {
   192→  // Get collections and packs
   193→  const triggerCollections: TriggerCollection[] = useMemo(() => {
   194→    return character.triggerCollections || [];
   195→  }, [character.triggerCollections]);
   196→
   197→  const spritePacksV2: SpritePackV2[] = useMemo(() => {
   198→    return character.spritePacksV2 || [];
   199→  }, [character.spritePacksV2]);
   200→
   201→  // Store actions for testing
   202→  const applyTriggerForCharacter = useTavernStore((state) => state.applyTriggerForCharacter);
   203→  const scheduleReturnToIdleForCharacter = useTavernStore((state) => state.scheduleReturnToIdleForCharacter);
   204→  const cancelReturnToIdleForCharacter = useTavernStore((state) => state.cancelReturnToIdleForCharacter);
   205→
   206→  // State
   207→  const [showEditor, setShowEditor] = useState(false);
   208→  const [editingCollection, setEditingCollection] = useState<TriggerCollection | null>(null);
   209→  const [testingCollectionId, setTestingCollectionId] = useState<string | null>(null);
   210→  const [expandedSections, setExpandedSections] = useState<Record<string, boolean>>({
   211→    keys: true,
   212→    behavior: true,
   213→    fallback: true,
   214→    chains: false,
   215→    sprites: false,
   216→  });
   217→
   218→  // Update collections
   219→  const updateCollections = (newCollections: TriggerCollection[]) => {
   220→    onChange({
   221→      triggerCollections: newCollections,
   222→    });
   223→  };
   224→
   225→  // Test trigger - Apply the trigger sprite to the character
   226→  const handleTestTrigger = (collection: TriggerCollection) => {
   227→    const pack = spritePacksV2.find(p => p.id === collection.packId);
   228→    if (!pack || pack.sprites.length === 0) {
   229→      logger.warn('Cannot test: no sprites in pack', { collectionId: collection.id });
   230→      return;
   231→    }
   232→
   233→    // Get the sprite to show based on behavior
   234→    let sprite;
   235→    switch (collection.collectionBehavior) {
   236→      case 'random':
   237→        const randomIndex = Math.floor(Math.random() * pack.sprites.length);
   238→        sprite = pack.sprites[randomIndex];
   239→        break;
   240→      case 'principal':
   241→      case 'list':
   242→      default:
   243→        sprite = collection.principalSpriteId 
   244→          ? pack.sprites.find(s => s.id === collection.principalSpriteId)
   245→          : pack.sprites[0];
   246→        break;
   247→    }
   248→
   249→    if (!sprite) {
   250→      logger.warn('Cannot test: no sprite found', { collectionId: collection.id });
   251→      return;
   252→    }
   253→
   254→    // Mark as testing
   255→    setTestingCollectionId(collection.id);
   256→    logger.info('Testing trigger', { 
   257→      collectionId: collection.id, 
   258→      spriteUrl: sprite.url,
   259→      spriteLabel: sprite.label 
   260→    });
   261→
   262→    // Apply the trigger sprite
   263→    applyTriggerForCharacter(character.id, {
   264→      spriteUrl: sprite.url,
   265→      spriteLabel: sprite.label,
   266→      returnToIdleMs: collection.fallbackDelayMs,
   267→      packId: collection.packId,
   268→      collectionId: collection.id,
   269→      useTimelineSounds: collection.useTimelineSounds ?? false,
   270→    });
   271→
   272→    // Note: Timeline sounds are now handled by useTimelineSpriteSounds hook
   273→    // which watches for triggerSpriteUrl changes and checks useTimelineSounds
   274→
   275→    // Schedule fallback if configured
   276→    if (collection.fallbackDelayMs > 0) {
   277→      // Get fallback sprite based on mode
   278→      let fallbackSpriteUrl: string | null = null;
   279→      let returnToMode: 'idle' | 'talk' | 'thinking' | 'clear' = 'idle';
   280→
   281→      if (collection.fallbackMode === 'custom_sprite' && collection.fallbackSpriteId) {
   282→        const fallbackSprite = pack.sprites.find(s => s.id === collection.fallbackSpriteId);
   283→        fallbackSpriteUrl = fallbackSprite?.url || null;
   284→        returnToMode = 'idle'; // Apply the custom sprite
   285→      } else if (collection.fallbackMode === 'collection_default') {
   286→        const principalSprite = collection.principalSpriteId
   287→          ? pack.sprites.find(s => s.id === collection.principalSpriteId)
   288→          : pack.sprites[0];
   289→        fallbackSpriteUrl = principalSprite?.url || null;
   290→        returnToMode = 'idle'; // Apply the collection default sprite
   291→      } else if (collection.fallbackMode === 'idle_collection') {
   292→        // For 'idle_collection', clear the trigger and let the normal state logic
   293→        // (idle state from State Collections V2) determine what to show
   294→        returnToMode = 'clear';
   295→        // fallbackSpriteUrl can be empty for 'clear' mode
   296→        fallbackSpriteUrl = '';
   297→      }
   298→
   299→      // Always schedule fallback when delay > 0
   300→      // For 'clear' mode, the empty string is fine (won't be used)
   301→      scheduleReturnToIdleForCharacter(
   302→        character.id,
   303→        sprite.url,
   304→        returnToMode,
   305→        fallbackSpriteUrl || '',
   306→        sprite.label,
   307→        collection.fallbackDelayMs
   308→      );
   309→
   310→      logger.info('Test trigger: fallback scheduled', {
   311→        collectionId: collection.id,
   312→        fallbackMode: collection.fallbackMode,
   313→        returnToMode,
   314→        fallbackDelayMs: collection.fallbackDelayMs,
   315→      });
   316→    }
   317→
   318→    // Clear testing indicator after a delay
   319→    setTimeout(() => {
   320→      setTestingCollectionId(null);
   321→    }, 1000);
   322→  };
   323→
   324→  // Create new collection
   325→  const handleCreateCollection = () => {
   326→    const now = new Date().toISOString();
   327→    const newCollection: TriggerCollection = {
   328→      id: uuidv4(),
   329→      name: 'Nueva Colección de Triggers',
   330→      active: true,
   331→      priority: 1,
   332→      packId: spritePacksV2[0]?.id || '',
   333→      collectionKey: '',
   334→      collectionKeys: [],
   335→      collectionKeyRequirePipes: true,
   336→      collectionKeyCaseSensitive: false,
   337→      collectionBehavior: 'principal',
   338→      principalSpriteId: undefined,
   339→      fallbackMode: 'idle_collection',
   340→      fallbackSpriteId: undefined,
   341→      fallbackDelayMs: 3000,
   342→      spriteChain: undefined,
   343→      useTimelineSounds: false,
   344→      cooldownMs: 1000,
   345→      spriteConfigs: {},
   346→      conditionalMode: false,
   347→      conditionalEntries: [],
   348→      defaultSpriteId: undefined,
   349→      createdAt: now,
   350→      updatedAt: now,
   351→    };
   352→
   353→    setEditingCollection(newCollection);
   354→    setShowEditor(true);
   355→  };
   356→
   357→  // Edit collection
   358→  const handleEditCollection = (collection: TriggerCollection) => {
   359→    setEditingCollection({ ...collection });
   360→    setShowEditor(true);
   361→  };
   362→
   363→  // Delete collection
   364→  const handleDeleteCollection = (id: string) => {
   365→    const collection = triggerCollections.find(c => c.id === id);
   366→    if (!collection) return;
   367→
   368→    if (!confirm(`¿Eliminar la colección "${collection.name}"?`)) return;
   369→
   370→    updateCollections(triggerCollections.filter(c => c.id !== id));
   371→    logger.info('Deleted trigger collection', { collectionId: id });
   372→  };
   373→
   374→  // Duplicate collection
   375→  const handleDuplicateCollection = (collection: TriggerCollection) => {
   376→    const now = new Date().toISOString();
   377→    const newCollection: TriggerCollection = {
   378→      ...collection,
   379→      id: uuidv4(),
   380→      name: `${collection.name} (copia)`,
   381→      createdAt: now,
   382→      updatedAt: now,
   383→    };
   384→
   385→    updateCollections([...triggerCollections, newCollection]);
   386→    logger.info('Duplicated trigger collection', { 
   387→      originalId: collection.id, 
   388→      newId: newCollection.id 
   389→    });
   390→  };
   391→
   392→  // Save collection
   393→  const handleSaveCollection = () => {
   394→    if (!editingCollection) return;
   395→
   396→    const now = new Date().toISOString();
   397→    const updatedCollection: TriggerCollection = {
   398→      ...editingCollection,
   399→      updatedAt: now,
   400→    };
   401→
   402→    const existingIndex = triggerCollections.findIndex(c => c.id === updatedCollection.id);
   403→    
   404→    if (existingIndex >= 0) {
   405→      // Update existing
   406→      const newCollections = [...triggerCollections];
   407→      newCollections[existingIndex] = updatedCollection;
   408→      updateCollections(newCollections);
   409→    } else {
   410→      // Add new
   411→      updatedCollection.createdAt = now;
   412→      updateCollections([...triggerCollections, updatedCollection]);
   413→    }
   414→
   415→    setEditingCollection(null);
   416→    setShowEditor(false);
   417→    logger.info('Saved trigger collection', { collectionId: updatedCollection.id });
   418→  };
   419→
   420→  // Toggle collection active
   421→  const handleToggleActive = (id: string, active: boolean) => {
   422→    updateCollections(triggerCollections.map(c =>
   423→      c.id === id ? { ...c, active, updatedAt: new Date().toISOString() } : c
   424→    ));
   425→  };
   426→
   427→  // Get pack name
   428→  const getPackName = (packId: string) => {
   429→    const pack = spritePacksV2.find(p => p.id === packId);
   430→    return pack?.name || 'Pack no encontrado';
   431→  };
   432→
   433→  // Get pack sprites
   434→  const getPackSprites = (packId: string) => {
   435→    const pack = spritePacksV2.find(p => p.id === packId);
   436→    return pack?.sprites || [];
   437→  };
   438→
   439→  // Format time
   440→  const formatTime = (ms: number) => {
   441→    if (ms <= 0) return 'Nunca';
   442→    if (ms < 1000) return `${ms}ms`;
   443→    const seconds = Math.floor(ms / 1000);
   444→    if (seconds < 60) return `${seconds}s`;
   445→    const minutes = Math.floor(seconds / 60);
   446→    const remainingSeconds = seconds % 60;
   447→    return remainingSeconds > 0 ? `${minutes}m ${remainingSeconds}s` : `${minutes}m`;
   448→  };
   449→
   450→  // Toggle section
   451→  const toggleSection = (section: string) => {
   452→    setExpandedSections(prev => ({
   453→      ...prev,
   454→      [section]: !prev[section],
   455→    }));
   456→  };
   457→
   458→  return (
   459→    <div className="space-y-4">
   460→      {/* Header */}
   461→      <div className="flex items-center justify-between">
   462→        <div>
   463→          <h4 className="text-sm font-medium flex items-center gap-2">
   464→            <Zap className="w-4 h-4 text-amber-500" />
   465→            Trigger Collections
   466→          </h4>
   467→          <p className="text-xs text-muted-foreground">
   468→            Colecciones de triggers con sistema de prioridades y cadenas.
   469→          </p>
   470→        </div>
   471→        <Button
   472→          size="sm"
   473→          className="h-8"
   474→          onClick={handleCreateCollection}
   475→          disabled={spritePacksV2.length === 0}
   476→        >
   477→          <Plus className="w-4 h-4 mr-1" />
   478→          Nueva Colección
   479→        </Button>
   480→      </div>
   481→
   482→      {/* Info Banner */}
   483→      <div className="text-xs bg-amber-500/10 border border-amber-500/20 rounded-lg p-3 space-y-2">
   484→        <div className="flex items-center gap-2 font-medium text-amber-600">
   485→          <Zap className="w-4 h-4" />
   486→          Sistema de Trigger Collections
   487→        </div>
   488→        <p className="text-muted-foreground">
   489→          Las colecciones de triggers usan <strong>prioridades</strong> para determinar qué trigger 
   490→          se ejecuta. Puedes activar por <strong>key de colección</strong> (general) o por 
   491→          <strong> keys individuales</strong> por sprite.
   492→        </p>
   493→        <div className="grid grid-cols-2 gap-2 mt-2">
   494→          <div className="p-2 bg-background/50 rounded border">
   495→            <div className="flex items-center gap-1 text-xs font-medium text-amber-600">
   496→              <Crown className="w-3 h-3" />
   497→              Prioridad
   498→            </div>
   499→            <p className="text-[10px] text-muted-foreground mt-0.5">
   500→              Mayor número = mayor prioridad
   501→            </p>
   502→          </div>
   503→          <div className="p-2 bg-background/50 rounded border">
   504→            <div className="flex items-center gap-1 text-xs font-medium text-blue-600">
   505→              <Play className="w-3 h-3" />
   506→              Cadenas
   507→            </div>
   508→            <p className="text-[10px] text-muted-foreground mt-0.5">
   509→              Secuencias de sprites/sonidos
   510→            </p>
   511→          </div>
   512→        </div>
   513→      </div>
   514→
   515→      {/* No packs warning */}
   516→      {spritePacksV2.length === 0 && (
   517→        <div className="text-center py-4 text-amber-600 bg-amber-500/5 border border-amber-500/20 rounded-lg">
   518→          <Package className="w-6 h-6 mx-auto mb-2" />
   519→          <p className="text-sm font-medium">No hay Sprite Packs creados</p>
   520→          <p className="text-xs mt-1">
   521→            Crea un Sprite Pack en la pestaña "Sprite Packs" antes de crear triggers.
   522→          </p>
   523→        </div>
   524→      )}
   525→
   526→      {/* Collections List */}
   527→      {triggerCollections.length > 0 ? (
   528→        <Accordion type="multiple" className="w-full space-y-2">
   529→          {triggerCollections
   530→            .sort((a, b) => b.priority - a.priority)
   531→            .map(collection => {
   532→              const pack = spritePacksV2.find(p => p.id === collection.packId);
   533→              const sprites = pack?.sprites || [];
   534→
   535→              return (
   536→                <AccordionItem
   537→                  key={collection.id}
   538→                  value={collection.id}
   539→                  className="border rounded-lg px-0"
   540→                >
   541→                  <AccordionTrigger className="px-4 py-3 hover:no-underline">
   542→                    <div className="flex items-center gap-3 w-full">
   543→                      <div className={cn(
   544→                        "p-1.5 rounded",
   545→                        collection.active ? "bg-amber-500/10" : "bg-muted"
   546→                      )}>
   547→                        <Zap className={cn(
   548→                          "w-4 h-4",
   549→                          collection.active ? "text-amber-500" : "text-muted-foreground"
   550→                        )} />
   551→                      </div>
   552→                      <div className="flex-1 text-left">
   553→                        <div className="flex items-center gap-2">
   554→                          <span className="text-sm font-medium">{collection.name}</span>
   555→                          <Badge variant="outline" className="text-xs">
   556→                            Prioridad: {collection.priority}
   557→                          </Badge>
   558→                        </div>
   559→                        <div className="flex items-center gap-2 text-xs text-muted-foreground">
   560→                          <Package className="w-3 h-3" />
   561→                          <span>{getPackName(collection.packId)}</span>
   562→                          <span>•</span>
   563→                          <span>{sprites.length} sprites</span>
   564→                          {collection.collectionKey && (
   565→                            <>
   566→                              <span>•</span>
   567→                              <code className="bg-muted px-1 rounded text-[10px]">
   568→                                {collection.collectionKeyRequirePipes ? `|${collection.collectionKey}|` : collection.collectionKey}
   569→                              </code>
   570→                            </>
   571→                          )}
   572→                        </div>
   573→                      </div>
   574→                      {!collection.active && (
   575→                        <Badge variant="secondary" className="text-xs">Inactivo</Badge>
   576→                      )}
   577→                      {collection.conditionalMode && (
   578→                        <Badge variant="secondary" className="text-xs bg-teal-500/10 text-teal-600">
   579→                          <GitBranch className="w-3 h-3 mr-1" />
   580→                          Condicional
   581→                        </Badge>
   582→                      )}
   583→                      {collection.conditionalMode && (collection.conditionalEntries?.length || 0) > 0 && (
   584→                        <Badge variant="outline" className="text-xs">
   585→                          {collection.conditionalEntries?.length} entradas
   586→                        </Badge>
   587→                      )}
   588→                    </div>
   589→                  </AccordionTrigger>
   590→                  <AccordionContent className="px-4 pb-4">
   591→                    <div className="space-y-3">

   786→function ConditionalSimulator({
   787→  collection,
   788→  character,
   789→  packSprites,
   790→}: ConditionalSimulatorProps) {
   791→  const [simValues, setSimValues] = useState<Record<string, number>>({});
   792→  const [showSimulator, setShowSimulator] = useState(false);
   793→
   794→  // Get unique attribute keys from all conditional entries
   795→  const uniqueAttributes = useMemo(() => {
   796→    const keys = new Set<string>();
   797→    (collection.conditionalEntries || []).forEach(entry => {
   798→      entry.conditions.forEach(cond => {
   799→        if (cond.attributeKey) keys.add(cond.attributeKey);
   800→      });
   801→    });
   802→    return Array.from(keys);
   803→  }, [collection.conditionalEntries]);
   804→
   805→  // Get attribute definitions from character stats config
   806→  const attributeDefs = useMemo(() => {
   807→    return character.statsConfig?.attributes || [];
   808→  }, [character.statsConfig?.attributes]);
   809→
   810→  // Get attribute display info
   811→  const getAttrInfo = (key: string) => {
   812→    const def = attributeDefs.find(a => a.key === key);
   813→    return {
   814→      name: def?.name || key,
   815→      min: def?.min ?? 0,
   816→      max: def?.max ?? 100,
   817→    };
   818→  };
   819→
   820→  // Get session stats from store
   821→  const sessionStats = useTavernStore((state) => {
   822→    const activeSessionId = state.activeSessionId;
   823→    if (!activeSessionId) return null;
   824→    const session = state.sessions.find(s => s.id === activeSessionId);
   825→    return session?.sessionStats ?? null;
   826→  });
   827→
   828→  // Build simulated session stats
   829→  const simulatedStats: SessionStats | null = useMemo(() => {
   830→    if (!sessionStats) return null;
   831→    // Override the character's attribute values with slider values
   832→    const charStats = sessionStats.characterStats?.[character.id];
   833→    if (!charStats) return sessionStats;
   834→
   835→    return {
   836→      ...sessionStats,
   837→      characterStats: {
   838→        ...sessionStats.characterStats,
   839→        [character.id]: {
   840→          ...charStats,
   841→          attributeValues: {
   842→            ...charStats.attributeValues,
   843→            ...simValues,
   844→          },
   845→        },
   846→      },
   847→    };
   848→  }, [sessionStats, character.id, simValues]);
   849→
   850→  // Evaluate which entry wins
   851→  const winner = useMemo(() => {
   852→    if (!simulatedStats || !collection.conditionalEntries?.length) return null;
   853→    return evaluateConditionalEntries(collection.conditionalEntries, simulatedStats, character.id);
   854→  }, [collection.conditionalEntries, simulatedStats, character.id]);
   855→
   856→  // Get session value for an attribute
   857→  const getSessionValue = (key: string): number | string | null => {
   858→    if (!sessionStats) return null;
   859→    return getAttributeValueFromStats(character.id, key, sessionStats);
   860→  };
   861→
   862→  if (uniqueAttributes.length === 0) {
   863→    return null;
   864→  }
   865→
   866→  return (
   867→    <div className="space-y-3 p-4 border rounded-lg bg-slate-500/5 border-slate-500/20">
   868→      <div
   869→        className="flex items-center justify-between cursor-pointer"
   870→        onClick={() => setShowSimulator(!showSimulator)}
   871→      >
   872→        <div>
   873→          <h4 className="text-sm font-medium flex items-center gap-2 text-slate-600">
   874→            <SlidersHorizontal className="w-4 h-4" />
   875→            Simulador de Condiciones
   876→          </h4>
   877→          <p className="text-xs text-muted-foreground mt-1">
   878→            Ajusta los valores para ver qué sprite se mostraría.
   879→          </p>
   880→        </div>
   881→        <Button variant="ghost" size="sm" className="h-7 w-7 p-0">
   882→          {showSimulator ? <ChevronUp className="w-4 h-4" /> : <ChevronDown className="w-4 h-4" />}
   883→        </Button>
   884→      </div>
   885→
   886→      {showSimulator && (
   887→        <div className="space-y-3">
   888→          {/* Attribute Sliders */}
   889→          {uniqueAttributes.map(key => {
   890→            const info = getAttrInfo(key);
   891→            const currentValue = simValues[key] ?? (getSessionValue(key) as number ?? info.min);
   892→            const sessionVal = getSessionValue(key);
   893→            return (
   894→              <div key={key} className="space-y-1">
   895→                <div className="flex items-center justify-between">
   896→                  <Label className="text-xs font-medium">{info.name}</Label>
   897→                  <div className="flex items-center gap-2">
   898→                    <Input
   899→                      type="number"
   900→                      value={currentValue}
   901→                      onChange={(e) => {
   902→                        setSimValues(prev => ({
   903→                          ...prev,
   904→                          [key]: parseFloat(e.target.value) || 0,
   905→                        }));
   906→                      }}
   907→                      className="h-6 w-16 text-xs text-center"
   908→                    />
   909→                    {sessionVal !== null && (
   910→                      <span className="text-[10px] text-muted-foreground">
   911→                        (sesión: {String(sessionVal)})
   912→                      </span>
   913→                    )}
   914→                  </div>
   915→                </div>
   916→                <input
   917→                  type="range"
   918→                  min={info.min}
   919→                  max={info.max}
   920→                  value={currentValue}
   921→                  onChange={(e) => {
   922→                    setSimValues(prev => ({
   923→                      ...prev,
   924→                      [key]: parseFloat(e.target.value),
   925→                    }));
   926→                  }}
   927→                  className="w-full h-1.5 bg-muted rounded-lg appearance-none cursor-pointer accent-teal-500"
   928→                />
   929→              </div>
   930→            );
   931→          })}
   932→
   933→          {/* Result */}
   934→          <div className="p-3 bg-background rounded-lg border border-teal-500/20">
   935→            {winner ? (
   936→              <div className="space-y-2">
   937→                <div className="flex items-center gap-2 text-sm font-medium text-teal-600">
   938→                  <CheckCircle className="w-4 h-4" />
   939→                  <span>Entrada activa: &quot;{winner.name}&quot; (P{winner.priority})</span>
   940→                </div>
   941→                <div className="flex items-center gap-2">
   942→                  {(() => {
   943→                    const sprite = packSprites.find(s => s.id === winner.spriteId);
   944→                    return sprite ? (
   945→                      <>
   946→                        <div className="w-10 h-10 rounded overflow-hidden bg-muted/50">
   947→                          <SpritePreview
   948→                            src={sprite.url}
   949→                            alt={sprite.label}
   950→                            className="w-full h-full"
   951→                            objectFit="contain"
   952→                          />
   953→                        </div>
   954→                        <span className="text-xs">{sprite.label}</span>
   955→                      </>
   956→                    ) : (
   957→                      <span className="text-xs text-muted-foreground">Sprite no encontrado</span>
   958→                    );
   959→                  })()}
   960→                </div>
   961→                {/* Show condition evaluation details */}
   962→                {winner.conditions.map((cond, i) => {
   963→                  const attrVal = simValues[cond.attributeKey] ?? getSessionValue(cond.attributeKey);
   964→                  const passes = simulatedStats
   965→                    ? evaluateStatConditions([cond], simulatedStats, character.id)
   966→                    : false;
   967→                  const operatorSymbol = cond.operator === '<=' ? '≤' : cond.operator === '>=' ? '≥' : cond.operator;
   968→                  return (
   969→                    <div key={i} className={cn(
   970→                      "text-xs flex items-center gap-1",
   971→                      passes ? "text-green-600" : "text-red-500"
   972→                    )}>
   973→                      {passes ? '✅' : '❌'}
   974→                      <span>{cond.attributeKey} {operatorSymbol} {String(cond.value)}</span>
   975→                      {cond.operator === 'between' && <span> a {cond.valueMax}</span>}
   976→                      <span className="text-muted-foreground">→ {String(attrVal ?? '?')} {operatorSymbol} {String(cond.value)}</span>
   977→                      <span>{passes ? '✓' : '✗'}</span>
   978→                    </div>
   979→                  );
   980→                })}
   981→              </div>
   982→            ) : (
   983→              <div className="space-y-1">
   984→                <div className="text-sm font-medium text-muted-foreground flex items-center gap-2">
   985→                  <Info className="w-4 h-4" />
   986→                  Ninguna condición coincide
   987→                </div>
   988→                {collection.defaultSpriteId ? (
   989→                  <div className="flex items-center gap-2">
   990→                    {(() => {
   991→                      const sprite = packSprites.find(s => s.id === collection.defaultSpriteId);
   992→                      return sprite ? (
   993→                        <>
   994→                          <div className="w-8 h-8 rounded overflow-hidden bg-muted/50">
   995→                            <SpritePreview
   996→                              src={sprite.url}
   997→                              alt={sprite.label}
   998→                              className="w-full h-full"
   999→                              objectFit="contain"
  1000→                            />
  1001→                          </div>
  1002→                          <span className="text-xs">Usando sprite por defecto: {sprite.label}</span>
  1003→                        </>
  1004→                      ) : (
  1005→                        <span className="text-xs text-muted-foreground">Sprite por defecto no encontrado</span>
  1006→                      );
  1007→                    })()}
  1008→                  </div>
  1009→                ) : (
  1010→                  <p className="text-xs text-muted-foreground">Se usará el comportamiento normal de la colección</p>
  1011→                )}
  1012→              </div>
  1013→            )}
  1014→          </div>
  1015→        </div>
  1016→      )}
  1017→    </div>
  1018→  );
  1019→}
