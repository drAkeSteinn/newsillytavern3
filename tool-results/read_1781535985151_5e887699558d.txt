    84→    84→/**
    85→    85→ * Build a HUD context section for prompt injection
    86→    86→ * 
    87→    87→ * Now resolves ALL keys including:
    88→    88→ * - Template variables: {{user}}, {{char}}, {{userpersona}}
    89→    89→ * - Stats keys: {{resistencia}}, {{habilidades}}, etc.
    90→    90→ */
    91→    91→export function buildHUDContextSection(
    92→    92→  contextConfig: HUDContextConfig,
    93→    93→  keyContext?: KeyResolutionContext
    94→    94→): PromptSection | null {
    95→    95→  if (!contextConfig.enabled || !contextConfig.content.trim()) {
    96→    96→    return null;
    97→    97→  }
    98→    98→
    99→    99→  // Resolve all keys in the context content
   100→   100→  let resolvedContent = contextConfig.content;
   101→   101→  if (keyContext) {
   102→   102→    resolvedContent = resolveAllKeys(contextConfig.content, keyContext);
   103→   103→  }
   104→   104→
   105→   105→  return {
   106→   106→    type: 'hud_context',
   107→   107→    label: 'HUD Context',
   108→   108→    content: resolvedContent,
   109→   109→    color: SECTION_COLORS.hud_context
   110→   110→  };
   111→   111→}
   112→   112→
   113→   113→/**
   114→   114→ * Inject HUD context at the specified position
   115→   115→ *
   116→   116→ * Positions:
   117→   117→ * 0 = After system prompt
   118→   118→ * 1 = After user message (after last user message)
   119→   119→ * 2 = Before user message (before last user message)
   120→   120→ * 3 = After assistant message (after last assistant message)
   121→   121→ * 4 = Before assistant message (before last assistant message)
   122→   122→ * 5 = At top of chat (before chat history)
   123→   123→ * 6 = At bottom of chat (after all messages)
   124→   124→ * 7 = After lorebook / Author's Note position
   125→   125→ */
   126→   126→export function injectHUDContextIntoMessages(
   127→   127→  messages: ChatApiMessage[],
   128→   128→  contextSection: PromptSection,
   129→   129→  position: number
   130→   130→): ChatApiMessage[] {
   131→   131→  if (!contextSection.content.trim()) {
   132→   132→    return messages;
   133→   133→  }
   134→   134→
   135→   135→  const contextContent = `[${contextSection.label}]\n${contextSection.content}`;
   136→   136→  const result: ChatApiMessage[] = [...messages];
   137→   137→
   138→   138→  switch (position) {
   139→   139→    case 0: // After system prompt
   140→   140→      // Find the first system/assistant message (the system prompt)
   141→   141→      const sysIdx = result.findIndex(m => m.role === 'assistant' || m.role === 'system');
   142→   142→      if (sysIdx >= 0) {
   143→   143→        result[sysIdx] = {
   144→   144→          ...result[sysIdx],
   145→   145→          content: result[sysIdx].content + '\n\n' + contextContent
   146→   146→        };
   147→   147→      }
   148→   148→      break;
   149→   149→
   150→   150→    case 1: // After user message
   151→   151→      // Find the last user message
   152→   152→      const lastUserIdx = result.map((m, i) => m.role === 'user' ? i : -1).filter(i => i >= 0).pop();
   153→   153→      if (lastUserIdx !== undefined && lastUserIdx >= 0) {
   154→   154→        result[lastUserIdx] = {
   155→   155→          ...result[lastUserIdx],
   156→   156→          content: result[lastUserIdx].content + '\n\n' + contextContent
   157→   157→        };
   158→   158→      }
   159→   159→      break;
   160→   160→
   161→   161→    case 2: // Before user message
   162→   162→      const lastUserIdx2 = result.map((m, i) => m.role === 'user' ? i : -1).filter(i => i >= 0).pop();
   163→   163→      if (lastUserIdx2 !== undefined && lastUserIdx2 >= 0) {
   164→   164→        result[lastUserIdx2] = {
   165→   165→          ...result[lastUserIdx2],
   166→   166→          content: contextContent + '\n\n' + result[lastUserIdx2].content
   167→   167→        };
   168→   168→      }
   169→   169→      break;
   170→   170→
   171→   171→    case 3: // After assistant message
   172→   172→      const lastAsstIdx = result.map((m, i) => m.role === 'assistant' ? i : -1).filter(i => i >= 0).pop();
   173→   173→      if (lastAsstIdx !== undefined && lastAsstIdx >= 0) {
   174→   174→        result[lastAsstIdx] = {
   175→   175→          ...result[lastAsstIdx],
   176→   176→          content: result[lastAsstIdx].content + '\n\n' + contextContent
   177→   177→        };
   178→   178→      }
   179→   179→      break;
   180→   180→
   181→   181→    case 4: // Before assistant message
   182→   182→      const lastAsstIdx4 = result.map((m, i) => m.role === 'assistant' ? i : -1).filter(i => i >= 0).pop();
   183→   183→      if (lastAsstIdx4 !== undefined && lastAsstIdx4 >= 0) {
   184→   184→        result[lastAsstIdx4] = {
   185→   185→          ...result[lastAsstIdx4],
   186→   186→          content: contextContent + '\n\n' + result[lastAsstIdx4].content
   187→   187→        };
   188→   188→      }
   189→   189→      break;
   190→   190→
   191→   191→    case 5: // At top of chat (after system, before first message)
   192→   192→      // Merge into the first system message to avoid breaking alternation
   193→   193→      {
   194→   194→        const sysMsgIdx5 = result.findIndex(m => m.role === 'system');
   195→   195→        if (sysMsgIdx5 >= 0) {
   196→   196→          result[sysMsgIdx5] = {
   197→   197→            ...result[sysMsgIdx5],
   198→   198→            content: result[sysMsgIdx5].content + '\n\n' + contextContent
   199→   199→          };
   200→   200→        }
   201→   201→      }
   202→   202→      break;
   203→   203→
   204→   204→    case 6: // At bottom of chat (after all messages)
   205→   205→      // Merge into the first system message to avoid breaking alternation
   206→   206→      {
   207→   207→        const sysMsgIdx6 = result.findIndex(m => m.role === 'system');
   208→   208→        if (sysMsgIdx6 >= 0) {
   209→   209→          result[sysMsgIdx6] = {
   210→   210→            ...result[sysMsgIdx6],
   211→   211→            content: result[sysMsgIdx6].content + '\n\n' + contextContent
   212→   212→          };
   213→   213→        }
   214→   214→      }
   215→   215→      break;
   216→   216→
   217→   217→    case 7: // After lorebook / Author's Note position
   218→   218→      // Merge into the first system message to avoid breaking alternation
   219→   219→      {
   220→   220→        const sysMsgIdx7 = result.findIndex(m => m.role === 'system' || m.role === 'assistant');
   221→   221→        if (sysMsgIdx7 >= 0) {
   222→   222→          result[sysMsgIdx7] = {
   223→   223→            ...result[sysMsgIdx7],
   224→   224→            content: result[sysMsgIdx7].content + '\n\n' + contextContent
   225→   225→          };
   226→   226→        }
   227→   227→      }
   228→   228→      break;
   229→   229→
   230→   230→    default:
   231→   231→      // Merge into the first system message to avoid breaking alternation
   232→   232→      {
   233→   233→        const sysMsgIdxDef = result.findIndex(m => m.role === 'system');
   234→   234→        if (sysMsgIdxDef >= 0) {
   235→   235→          result[sysMsgIdxDef] = {
   236→   236→            ...result[sysMsgIdxDef],
   237→   237→            content: result[sysMsgIdxDef].content + '\n\n' + contextContent
   238→   238→          };
   239→   239→        } else {
   240→   240→          result.push({
   241→   241→            role: 'system',
   242→   242→            content: contextContent
   243→   243→          });
   244→   244→        }
   245→   245→      }
   246→   246→      break;
   247→   247→  }
   248→   248→
   249→   249→  return result;
   250→   250→}

   252→   252→/**
   253→   253→ * Inject HUD context into prompt sections
   254→   254→ * This is for the prompt viewer display
   255→   255→ *
   256→   256→ * Positions:
   257→   257→ * 0 = After system prompt
   258→   258→ * 1 = After user message (not applicable to sections, uses chat_history)
   259→   259→ * 2 = Before user message (not applicable to sections, uses chat_history)
   260→   260→ * 3 = After assistant message (not applicable to sections, uses chat_history)
   261→   261→ * 4 = Before assistant message (not applicable to sections, uses chat_history)
   262→   262→ * 5 = At top of chat (before chat history)
   263→   263→ * 6 = At bottom of chat (after all sections)
   264→   264→ * 7 = After lorebook
   265→   265→ */
   266→   266→export function injectHUDContextIntoSections(
   267→   267→  sections: PromptSection[],
   268→   268→  contextSection: PromptSection,
   269→   269→  position: number
   270→   270→): PromptSection[] {
   271→   271→  if (!contextSection.content.trim()) {
   272→   272→    return sections;
   273→   273→  }
   274→   274→
   275→   275→  const result: PromptSection[] = [...sections];
   276→   276→
   277→   277→  switch (position) {
   278→   278→    case 0: // After system prompt
   279→   279→      const sysSectionIdx = result.findIndex(s => s.type === 'system');
   280→   280→      if (sysSectionIdx >= 0) {
   281→   281→        result.splice(sysSectionIdx + 1, 0, contextSection);
   282→   282→      } else {
   283→   283→        result.unshift(contextSection);
   284→   284→      }
   285→   285→      break;
   286→   286→
   287→   287→    case 1: // After user message - inject before chat history
   288→   288→    case 2: // Before user message - inject before chat history
   289→   289→    case 3: // After assistant message - inject before chat history
   290→   290→    case 4: // Before assistant message - inject before chat history
   291→   291→      // These positions apply to messages, for sections we inject before chat history
   292→   292→      const chatHistoryIdx = result.findIndex(s => s.type === 'chat_history');
   293→   293→      if (chatHistoryIdx >= 0) {
   294→   294→        result.splice(chatHistoryIdx, 0, contextSection);
   295→   295→      } else {
   296→   296→        result.push(contextSection);
   297→   297→      }
   298→   298→      break;
   299→   299→
   300→   300→    case 5: // At top of chat (before chat history)
   301→   301→      const chatIdx = result.findIndex(s => s.type === 'chat_history');
   302→   302→      if (chatIdx >= 0) {
   303→   303→        result.splice(chatIdx, 0, contextSection);
   304→   304→      } else {
   305→   305→        result.push(contextSection);
   306→   306→      }
   307→   307→      break;
   308→   308→
   309→   309→    case 7: // After lorebook
   310→   310→      const lorebookIdx = result.findIndex(s => s.type === 'lorebook');
   311→   311→      if (lorebookIdx >= 0) {
   312→   312→        result.splice(lorebookIdx + 1, 0, contextSection);
   313→   313→      } else {
   314→   314→        // If no lorebook, insert after system prompt
   315→   315→        const sysIdx = result.findIndex(s => s.type === 'system');
   316→   316→        if (sysIdx >= 0) {
   317→   317→          result.splice(sysIdx + 1, 0, contextSection);
   318→   318→        } else {
   319→   319→          result.unshift(contextSection);
   320→   320→        }
   321→   321→      }
   322→   322→      break;
   323→   323→
   324→   324→    case 6: // At bottom of chat
   325→   325→    default:
   326→   326→      result.push(contextSection);
   327→   327→      break;
   328→   328→  }
   329→   329→
   330→   330→  return result;
   331→   331→}

   334→   334→// Inventory V2 Section
   335→   335→// ============================================
   336→   336→
   337→   337→/**
   338→   338→ * Build an inventory section for prompt injection.
   339→   339→ *
   340→   340→ * Shows:
   341→   341→ * - Current currency
   342→   342→ * - Equipped items and their effects
   343→   343→ * - Active consumable effects with remaining turns
   344→   344→ * - Available (non-equipped) items in inventory
   345→   345→ *
   346→   346→ * Uses the `promptTemplate` from `inventorySettings` for formatting.
   347→   347→ * If no template is provided or the template is empty, a default format is used.
   348→   348→ *
   349→   349→ * Returns null if inventory data is empty or invalid.
   350→   350→ */
   351→   351→export function buildInventorySection(
   352→   352→  inventoryData: InventoryPromptData,
   353→   353→  keyContext?: KeyResolutionContext
   354→   354→): PromptSection | null {
   355→   355→  const {
   356→   356→    personaItems,
   357→   357→    equippedItems,
   358→   358→    sessionEquipment,
   359→   359→    activeEffects,
   360→   360→    currency,
   361→   361→    currencyName,
   362→   362→    currencyIcon,
   363→   363→    inventorySettings,
   364→   364→  } = inventoryData;
   365→   365→
   366→   366→  // If no items and no effects and no currency, skip section
   367→   367→  if (personaItems.length === 0 && activeEffects.length === 0 && currency === 0) {
   368→   368→    return null;
   369→   369→  }
   370→   370→
   371→   371→  // Use the template from inventory settings, or fall back to default
   372→   372→  const template = inventorySettings.promptTemplate?.trim() || `[Inventario Activo]
   373→   373→{{activeItems}}
   374→   374→
   375→   375→[Efectos Activos]
   376→   376→{{activeEffects}}
   377→   377→
   378→   378→[Divisa]
   379→   379→{{currency}}`;
   380→   380→
   381→   381→  // Build items list (all items in inventory)
   382→   382→  const itemLines = personaItems.map(({ entry, item }) => {
   383→   383→    const qty = entry.quantity > 1 ? ` x${entry.quantity}` : '';
   384→   384→    const eq = entry.equipped ? ' [Equipado]' : '';
   385→   385→    const effects = (item.attributeEffects && item.attributeEffects.length > 0)
   386→   386→      ? ` (${item.attributeEffects.map(e => `${e.operator}${e.value} ${e.attributeKey}`).join(', ')})`
   387→   387→      : '';
   388→   388→    return `- ${item.icon || ''} ${item.name}${qty}${eq}${effects}`;
   389→   389→  }).join('\n');
   390→   390→
   391→   391→  // Build active effects list
   392→   392→  const effectLines = activeEffects.map(e => {
   393→   393→    const turnsLeft = e.remainingTurns > 0 ? ` (${e.remainingTurns}/${e.totalTurns} turnos)` : '';
   394→   394→    // Use consumableEffect free text if available, otherwise fall back to attribute effects
   395→   395→    let effectDesc: string;
   396→   396→    if (e.consumableEffect) {
   397→   397→      effectDesc = e.consumableEffect;
   398→   398→    } else {
   399→   399→      effectDesc = e.effects.map(ef =>
   400→   400→        `${ef.operator}${ef.value} ${ef.attributeKey}${ef.targetId !== '__user__' ? ` → ${ef.targetName || ef.targetId}` : ''}`
   401→   401→      ).join(', ');
   402→   402→    }
   403→   403→    return `- ${e.itemName}: ${effectDesc}${turnsLeft}`;
   404→   404→  }).join('\n');
   405→   405→
   406→   406→  // Build equipped items list - prefer sessionEquipment (per-session), fallback to equippedItems (legacy)
   407→   407→  let equipLines = '';
   408→   408→  if (sessionEquipment && sessionEquipment.length > 0) {
   409→   409→    // Use session equipment data
   410→   410→    const items = personaItems.map(({ item }) => item);
   411→   411→    equipLines = sessionEquipment.map(eq => {
   412→   412→      const item = items.find(i => i.id === eq.itemId);
   413→   413→      const slotDef = inventorySettings.equipmentSlots?.find(s => s.id === eq.equippedSlotId);
   414→   414→      const slotLabel = slotDef?.name || eq.equippedSlotId;
   415→   415→      const effectText = eq.slotEffectText
   416→   416→        || item?.slotEffects?.find(se => se.slotId === eq.equippedSlotId)?.effectText
   417→   417→        || '';
   418→   418→      return `- ${item?.icon || ''} ${item?.name || '???'} [${slotLabel}]${effectText ? ` → ${effectText}` : ''}`;
   419→   419→    }).join('\n');
   420→   420→  } else if (equippedItems && equippedItems.length > 0) {
   421→   421→    // Fallback to legacy equipped items
   422→   422→    equipLines = equippedItems.map(({ item }) => {
   423→   423→      // Prefer slotEffects (V3) over attributeEffects (V2)
   424→   424→      let effects = '';
   425→   425→      if (item.slotEffects && item.slotEffects.length > 0) {
   426→   426→        effects = item.slotEffects.map(se => se.effectText).filter(Boolean).join('; ');
   427→   427→      } else if (item.attributeEffects && item.attributeEffects.length > 0) {
   428→   428→        effects = item.attributeEffects.map(e => `${e.operator}${e.value} ${e.attributeKey}`).join(', ');
   429→   429→      }
   430→   430→      const slotDef = inventorySettings.equipmentSlots?.find(s => s.id === item.slot);
   431→   431→      const slotLabel = slotDef?.name || item.slot || '';
   432→   432→      return `- ${item.icon || ''} ${item.name}${slotLabel ? ` [${slotLabel}]` : ''}${effects ? ` → ${effects}` : ''}`;
   433→   433→    }).join('\n');
   434→   434→  }
   435→   435→
   436→   436→  // Build currency line
   437→   437→  const currencyLine = `${currencyIcon || '💰'} ${currencyName || 'Divisa'}: ${currency}`;
   438→   438→
   439→   439→  // Apply template replacements
   440→   440→  let content = template
   441→   441→    .replace('{{activeItems}}', itemLines || 'Vacío')
   442→   442→    .replace('{{activeEffects}}', effectLines || 'Ninguno')
   443→   443→    .replace('{{equippedItems}}', equipLines || 'Ninguno')
   444→   444→    .replace('{{currency}}', currencyLine);
   445→   445→
   446→   446→  // Resolve all keys in the inventory content if keyContext is available
   447→   447→  if (keyContext) {
   448→   448→    content = resolveAllKeys(content, keyContext);
   449→   449→  }
   450→   450→
   451→   451→  return {
   452→   452→    type: 'inventory',
   453→   453→    label: 'Inventory',
   454→   454→    content,
   455→   455→    color: SECTION_COLORS.inventory
   456→   456→  };
   457→   457→}

   494→   494→/**
   495→   495→ * Build the system prompt from character data (SillyTavern style)
   496→   496→ *
   497→   497→ * Uses unified key resolution for ALL sections:
   498→   498→ * - Template variables: {{user}}, {{char}}, {{userpersona}}, etc.
   499→   499→ * - Stats keys: {{resistencia}}, {{habilidades}}, etc.
   500→   500→ * - Sound keys: {{sonidos}}
   501→   501→ */
   502→   502→export function buildSystemPrompt(
   503→   503→  character: CharacterCard,
   504→   504→  userName: string = 'User',
   505→   505→  persona?: Persona,
   506→   506→  lorebookPlan?: LorebookInjectionPlan | null,
   507→   507→  sessionStats?: SessionStats,
   508→   508→  allCharacters?: CharacterCard[],
   509→   509→  soundTriggers?: SoundTrigger[],
   510→   510→  soundSettings?: AppSettings['sound'],
   511→   511→  questTemplates?: QuestTemplate[],
   512→   512→  sessionQuests?: SessionQuestInstance[],
   513→   513→  questSettings?: QuestSettings,
   514→   514→  lorebookAttributeKeys?: Record<string, string>,
   515→   515→  inventoryData?: InventoryPromptData,
   516→   516→  lorebookEntryKeyMap?: Record<string, string>
   517→   517→): { prompt: string; sections: PromptSection[]; lorebookChatInjections: LorebookChatInjection[]; exampleMessages: ChatApiMessage[] } {
   518→   518→  const sections: PromptSection[] = [];
   519→   519→
   520→   520→  // Resolve stats for the persona FIRST (user attributes like {{resistencia}})
   521→   521→  // This must be done before character stats so personaResolvedStats is available
   522→   522→  let personaResolvedStats: ResolvedStats | null = null;
   523→   523→  if (persona?.statsConfig?.enabled && sessionStats) {
   524→   524→    personaResolvedStats = resolveStats({
   525→   525→      characterId: '__user__',
   526→   526→      statsConfig: persona.statsConfig,
   527→   527→      sessionStats,
   528→   528→      lorebookEntryKeys: lorebookEntryKeyMap,
   529→   529→    });
   530→   530→  }
   531→   531→
   532→   532→  // Resolve stats for this character (includes skills block with full key resolution)
   533→   533→  const resolvedStats = resolveStats({
   534→   534→    characterId: character.id,
   535→   535→    statsConfig: character.statsConfig,
   536→   536→    sessionStats,
   537→   537→    allCharacters,
   538→   538→    userName,
   539→   539→    characterName: character.name,
   540→   540→    questTemplates,
   541→   541→    personaDescription: persona?.description,
   542→   542→    personaResolvedStats,
   543→   543→    lorebookEntryKeys: lorebookEntryKeyMap,
   544→   544→  });
   545→   545→
   546→   546→  // Build outlet sections map from lorebook plan for {{outlet::name}} macro resolution
   547→   547→  const outletSections: Record<string, string> = {};
   548→   548→  if (lorebookPlan?.outletSections.length) {
   549→   549→    for (const outletSection of lorebookPlan.outletSections) {
   550→   550→      // Extract outlet name from label like "World Info (myOutlet)" → "myOutlet"
   551→   551→      const match = outletSection.label.match(/^World Info \((.+)\)$/);
   552→   552→      const outletName = match ? match[1] : outletSection.label;
   553→   553→      outletSections[outletName] = outletSection.content;
   554→   554→    }
   555→   555→  }
   556→   556→
   557→   557→  // Build unified key resolution context (includes quest data for {{activeQuests}}, outlet sections, lorebook attribute keys, lorebook entry keys, and inventory data for {{slots}})
   558→   558→  const keyContext = buildKeyResolutionContext(character, userName, persona, resolvedStats, sessionStats, soundTriggers, soundSettings, personaResolvedStats, questTemplates, sessionQuests, questSettings, outletSections, lorebookAttributeKeys, inventoryData ? {
   559→   559→    personaItems: inventoryData.personaItems,
   560→   560→    sessionEquipment: inventoryData.sessionEquipment || inventoryData.equippedItems?.flatMap(({ entry, item }) =>
   561→   561→      entry.equippedSlotId ? [{ itemId: item.id, equippedSlotId: entry.equippedSlotId, slotEffectText: item.slotEffects?.find(se => se.slotId === entry.equippedSlotId)?.effectText }] : []
   562→   562→    ) || [],
   563→   563→    activeEffects: inventoryData.activeEffects,
   564→   564→    currency: inventoryData.currency,
   565→   565→    currencyName: inventoryData.currencyName,
   566→   566→    currencyIcon: inventoryData.currencyIcon,
   567→   567→    inventorySettings: inventoryData.inventorySettings,
   568→   568→  } : undefined, lorebookEntryKeyMap);
   569→   569→
   570→   570→  // Main system instruction
   571→   571→  // If character has a custom system prompt, use it instead of the default
   572→   572→  const systemContent = character.systemPrompt?.trim()
   573→   573→    ? character.systemPrompt
   574→   574→    : `You are now in roleplay mode. You will act as ${character.name}.`;
   575→   575→
   576→   576→  sections.push({
   577→   577→    type: 'system',
   578→   578→    label: 'System Prompt',
   579→   579→    content: systemContent,
   580→   580→    color: SECTION_COLORS.system
   581→   581→  });
   582→   582→
   583→   583→  // Lorebook position 0: After system prompt
   584→   584→  if (lorebookPlan?.position0Section) {
   585→   585→    sections.push(lorebookPlan.position0Section);
   586→   586→  }
   587→   587→
   588→   588→  // NOTE: User's Persona section removed - use {{persona}} key in character sections instead
   589→   589→  // The persona content is now injected via the {{persona}} template key which can be
   590→   590→  // placed anywhere in the character's description, scenario, etc.
   591→   591→
   592→   592→  // NOTE: Inventory section removed - use {{slots}} key in character sections instead.
   593→   593→  // The {{slots}} key resolves slot equipment + active consumable effects and can be
   594→   594→  // placed anywhere in the character's description, scenario, systemPrompt, etc.
   595→   595→
   596→   596→  // Add character description
   597→   597→  if (character.description) {
   598→   598→    sections.push({
   599→   599→      type: 'character_description',
   600→   600→      label: 'Character Description',
   601→   601→      content: character.description,
   602→   602→      color: SECTION_COLORS.character_description
   603→   603→    });
   604→   604→  }
   605→   605→
   606→   606→  // Add personality
   607→   607→  if (character.personality) {
   608→   608→    sections.push({
   609→   609→      type: 'personality',
   610→   610→      label: 'Personality',
   611→   611→      content: character.personality,
   612→   612→      color: SECTION_COLORS.personality
   613→   613→    });
   614→   614→  }
   615→   615→
   616→   616→  // FASE 5: Add emotional state injection
   617→   617→  // When enabled, inject the character's current emotional state into the prompt
   618→   618→  // This gives the LLM awareness of the character's mood for consistent behavior
   619→   619→  if (character.emotionalConfig?.enabled && character.emotionalConfig.includeInPrompt) {
   620→   620→    const emotionalState = sessionStats?.characterStats?.[character.id]?.emotionalState;
   621→   621→    if (emotionalState) {
   622→   622→      const format = character.emotionalConfig.promptInjectionFormat || 'Estado emocional actual: {estado}';
   623→   623→      const emotionContent = format.replace('{estado}', emotionalState);
   624→   624→      sections.push({
   625→   625→        type: 'personality',
   626→   626→        label: 'Estado Emocional',
   627→   627→        content: emotionContent,
   628→   628→        color: 'bg-rose-100 text-rose-700 dark:bg-rose-900/30 dark:text-rose-300'
   629→   629→      });
   630→   630→    }
   631→   631→  }
   632→   632→
   633→   633→  // Add scenario
   634→   634→  if (character.scenario) {
   635→   635→    sections.push({
   636→   636→      type: 'scenario',
   637→   637→      label: 'Scenario',
   638→   638→      content: character.scenario,
   639→   639→      color: SECTION_COLORS.scenario
   640→   640→    });
   641→   641→  }
   642→   642→
   643→   643→  // Add character's note (user-defined instructions for this character)
   644→   644→  // NOTE: According to SillyTavern docs, Character Notes comes BEFORE Example Dialogue
   645→   645→  if (character.characterNote) {
   646→   646→    sections.push({
   647→   647→      type: 'character_note',
   648→   648→      label: "Character's Note",
   649→   649→      content: character.characterNote,
   650→   650→      color: SECTION_COLORS.character_note
   651→   651→    });
   652→   652→  }
   653→   653→
   654→   654→  // Example Dialogue: Format as numbered [EJEMPLO N] text section
   655→   655→  // Each <START> block becomes a numbered example with --- separators.
   656→   656→  // Supports both <START> and <START></START> tag formats.
   657→   657→  // This replaces the old approach of injecting as user/assistant chat messages,
   658→   658→  // making examples visible in the Prompt Viewer and clearly identifiable by the LLM.
   659→   659→  if (character.mesExample) {
   660→   660→    const exampleContent = processExampleDialogue(character.mesExample, userName, character.name);
   661→   661→    if (exampleContent) {
   662→   662→      sections.push({
   663→   663→        type: 'example_dialogue',
   664→   664→        label: 'EJEMPLOS DE MENSAJES',
   665→   665→        content: exampleContent,
   666→   666→        color: SECTION_COLORS.example_dialogue
   667→   667→      });
   668→   668→    }
   669→   669→  }
   670→   670→
   671→   671→  // Lorebook position 5: At top of chat (before chat history)
   672→   672→  if (lorebookPlan?.position5Section) {
   673→   673→    sections.push(lorebookPlan.position5Section);
   674→   674→  }
   675→   675→
   676→   676→  // Lorebook position 7: Outlets (custom positions)
   677→   677→  if (lorebookPlan?.outletSections.length) {
   678→   678→    sections.push(...lorebookPlan.outletSections);
   679→   679→  }
   680→   680→
   681→   681→  // Lorebook position 6: At bottom of chat (appended at end of system sections)
   682→   682→  if (lorebookPlan?.position6Section) {
   683→   683→    sections.push(lorebookPlan.position6Section);
   684→   684→  }
   685→   685→
   686→   686→  // Note: postHistoryInstructions should NOT be in system prompt
   687→   687→  // It must be injected AFTER the chat history in buildChatMessages
   688→   688→
   689→   689→  // ========================================
   690→   690→  // UNIFIED KEY RESOLUTION - Apply to ALL sections
   691→   691→  // ========================================
   692→   692→  // This resolves:
   693→   693→  // - Template variables: {{user}}, {{char}}, {{userpersona}}, etc.
   694→   694→  // - Stats keys: {{resistencia}}, {{habilidades}}, etc.
   695→   695→  // All in one place, consistently
   696→   696→  const processedSections = resolveSectionsKeys(sections, keyContext);
   697→   697→
   698→   698→  // Build the prompt string from processed sections
   699→   699→  const prompt = processedSections.map(s => `[${s.label}]\n${s.content}`).join('\n\n');
   700→   700→
   701→   701→  // Example dialogue is now included as a PromptSection above (visible in Prompt Viewer).
   702→   702→  // Return empty exampleMessages for backward compatibility with API routes.
   703→   703→  const exampleMessages: ChatApiMessage[] = [];
   704→   704→
   705→   705→  return { prompt, sections: processedSections, lorebookChatInjections: lorebookPlan?.chatInjections || [], exampleMessages };
   706→   706→}

   708→   708→/**
   709→   709→ * Build complete lorebook injection plan from active lorebooks and chat messages.
   710→   710→ * This replaces the old single-section approach with position-aware injection.
   711→   711→ * Also resolves attribute-type lorebook entries to their injection keys.
   712→   712→ */
   713→   713→export function buildLorebookSectionForPrompt(
   714→   714→  messages: ChatMessage[],
   715→   715→  lorebooks: Lorebook[],
   716→   716→  options?: LorebookInjectOptions,
   717→   717→  attributeContext?: LorebookAttributeContext
   718→   718→): { section: PromptSection | null; plan: LorebookInjectionPlan; lorebookAttributeKeys: Record<string, string>; lorebookEntryKeyMap: Record<string, string>; lorebookDebugEntries?: import('@/lib/lorebook/attribute-resolver').LorebookAttrDebugEntry[] } {
   719→   719→  // Resolve attribute-type entries to key→content map
   720→   720→  let lorebookAttributeKeys: Record<string, string> = {};
   721→   721→  let lorebookDebugEntries: import('@/lib/lorebook/attribute-resolver').LorebookAttrDebugEntry[] | undefined;
   722→   722→
   723→   723→  if (attributeContext) {
   724→   724→    const result = resolveLorebookAttributeKeys(lorebooks, attributeContext);
   725→   725→    lorebookAttributeKeys = result.keys;
   726→   726→    lorebookDebugEntries = result.debugEntries;
   727→   727→  }
   728→   728→
   729→   729→  // Build lorebook entry key map from traditional entries for {{key}} resolution
   730→   730→  // in action descriptions and other text fields
   731→   731→  const entryKeyMapResult = buildLorebookEntryKeyMap(lorebooks);
   732→   732→  const lorebookEntryKeyMap = entryKeyMapResult.keys;
   733→   733→
   734→   734→  // Build injection plan for traditional entries only (attribute entries are skipped by scanner)
   735→   735→  // Pass userName/charName for <START> dialogue formatting in lorebook entries
   736→   736→  const lorebookOptions = {
   737→   737→    ...options,
   738→   738→    userName: options?.userName,
   739→   739→    charName: options?.charName,
   740→   740→  };
   741→   741→  const plan = buildLorebookInjectionPlan(messages, lorebooks, lorebookOptions);
   742→   742→
   743→   743→  // Combine all system-level sections into one for backward compat with callers that only need a single section
   744→   744→  const allSystemSections = [
   745→   745→    plan.position0Section,
   746→   746→    plan.position5Section,
   747→   747→    plan.position6Section,
   748→   748→    ...plan.outletSections
   749→   749→  ].filter((s): s is PromptSection => s !== null);
   750→   750→
   751→   751→  const section = allSystemSections.length > 0
   752→   752→    ? {
   753→   753→        type: 'lorebook' as const,
   754→   754→        label: 'World Information',
   755→   755→        content: allSystemSections.map(s => s.content).join('\n\n'),
   756→   756→        color: 'bg-indigo-100 text-indigo-700 dark:bg-indigo-900/30 dark:text-indigo-300'
   757→   757→      }
   758→   758→    : null;
   759→   759→
   760→   760→  return { section, plan, lorebookAttributeKeys, lorebookEntryKeyMap, lorebookDebugEntries };
   761→   761→}

   764→   764→ * Build author's note section
   765→   765→ * This section is injected AFTER the chat history, BEFORE post-history instructions
   766→   766→ * 
   767→   767→ * According to SillyTavern docs, the order is:
   768→   768→ * 1. Chat History
   769→   769→ * 2. Author's Note (this)
   770→   770→ * 3. Post-History Instructions
   771→   771→ * 
   772→   772→ * @param authorNote - Raw author's note text (may contain {{keys}})

   776→   776→  authorNote: string | undefined,
   777→   777→  keyContext?: KeyResolutionContext
   778→   778→): PromptSection | null {
   779→   779→  if (!authorNote?.trim()) {
   780→   780→    return null;
   781→   781→  }
   782→   782→
   783→   783→  // Resolve all keys if keyContext is provided
   784→   784→  const resolvedContent = keyContext 
   785→   785→    ? resolveAllKeys(authorNote, keyContext)
   786→   786→    : authorNote;
   787→   787→
   788→   788→  return {
   789→   789→    type: 'author_note',
   790→   790→    label: "Author's Note",

   824→   824→/**
   825→   825→ * Build chat history sections for prompt viewer
   826→   826→ */
   827→   827→export function buildChatHistorySections(
   828→   828→  messages: ChatMessage[],
   829→   829→  characterName: string,
   830→   830→  userName: string
   831→   831→): PromptSection[] {
   832→   832→  const sections: PromptSection[] = [];
   833→   833→  // Exclude narrator messages from chat history display
   834→   834→  const visibleMessages = messages.filter(m => !m.isDeleted && !m.isNarratorMessage);
   835→   835→
   836→   836→  const historyParts: string[] = [];
   837→   837→  for (const msg of visibleMessages) {
   838→   838→    const name = msg.role === 'user' ? userName : characterName;
   839→   839→    historyParts.push(`${name}: ${msg.content}`);
   840→   840→  }
   841→   841→
   842→   842→  if (historyParts.length > 0) {
   843→   843→    sections.push({
   844→   844→      type: 'chat_history',
   845→   845→      label: 'Chat History',
   846→   846→      content: historyParts.join('\n\n'),
   847→   847→      color: SECTION_COLORS.chat_history
   848→   848→    });
   849→   849→  }
   850→   850→
   851→   851→  return sections;
   852→   852→}
   853→   853→
   854→   854→/**
   855→   855→ * Inject lorebook content at chat-level positions (1-4).
   856→   856→ * Merges content into existing messages to avoid breaking message alternation.
   857→   857→ *
   858→   858→ * Position mapping:
   859→   859→ * 1 = After last user message
   860→   860→ * 2 = Before last user message
   861→   861→ * 3 = After last assistant message
   862→   862→ * 4 = Before last assistant message
   863→   863→ */
   864→   864→function applyChatInjections(
   865→   865→  messages: ChatApiMessage[],
   866→   866→  injections: LorebookChatInjection[]
   867→   867→): void {
   868→   868→  for (const injection of injections) {
   869→   869→    const content = `\n\n[${injection.label}]\n${injection.content}`;
   870→   870→
   871→   871→    switch (injection.position) {
   872→   872→      case 1: { // After last user message
   873→   873→        const lastUserIdx = messages.map((m, i) => m.role === 'user' ? i : -1).filter(i => i >= 0).pop();
   874→   874→        if (lastUserIdx !== undefined && lastUserIdx >= 0) {
   875→   875→          messages[lastUserIdx] = {
   876→   876→            ...messages[lastUserIdx],
   877→   877→            content: messages[lastUserIdx].content + content
   878→   878→          };
   879→   879→        }
   880→   880→        break;
   881→   881→      }
   882→   882→      case 2: { // Before last user message
   883→   883→        const lastUserIdx2 = messages.map((m, i) => m.role === 'user' ? i : -1).filter(i => i >= 0).pop();
   884→   884→        if (lastUserIdx2 !== undefined && lastUserIdx2 >= 0) {
   885→   885→          messages[lastUserIdx2] = {
   886→   886→            ...messages[lastUserIdx2],
   887→   887→            content: content + messages[lastUserIdx2].content
   888→   888→          };
   889→   889→        }
   890→   890→        break;
   891→   891→      }
   892→   892→      case 3: { // After last assistant message
   893→   893→        const lastAsstIdx = messages.map((m, i) => m.role === 'assistant' ? i : -1).filter(i => i >= 0).pop();
   894→   894→        if (lastAsstIdx !== undefined && lastAsstIdx >= 0) {
   895→   895→          messages[lastAsstIdx] = {
   896→   896→            ...messages[lastAsstIdx],
   897→   897→            content: messages[lastAsstIdx].content + content
   898→   898→          };
   899→   899→        }
   900→   900→        break;
   901→   901→      }
   902→   902→      case 4: { // Before last assistant message
   903→   903→        const lastAsstIdx4 = messages.map((m, i) => m.role === 'assistant' ? i : -1).filter(i => i >= 0).pop();
   904→   904→        if (lastAsstIdx4 !== undefined && lastAsstIdx4 >= 0) {
   905→   905→          messages[lastAsstIdx4] = {
   906→   906→            ...messages[lastAsstIdx4],
   907→   907→            content: content + messages[lastAsstIdx4].content
   908→   908→          };
   909→   909→        }
   910→   910→        break;
   911→   911→      }
   912→   912→    }
   913→   913→  }
   914→   914→}
   915→   915→
   916→   916→/**
   917→   917→ * Build messages array for chat models
   918→   918→ * 
   919→   919→ * Order (SillyTavern style):
   920→   920→ * 1. System message (system prompt)
   921→   921→ * 2. Chat history
   922→   922→ * 3. Author's Note (injected AFTER chat history, as system message)
   923→   923→ * 4. Post-History Instructions (injected AFTER Author's Note, as system message)
   924→   924→ */
   925→   925→export function buildChatMessages(
   926→   926→  systemPrompt: string,
   927→   927→  messages: ChatMessage[],
   928→   928→  character: CharacterCard,
   929→   929→  userName: string = 'User',
   930→   930→  postHistoryInstructions?: string,
   931→   931→  authorNote?: string,
   932→   932→  useSystemRole: boolean = false,
   933→   933→  embeddingsContext?: string,  // embeddings injected before chat history
   934→   934→  lorebookChatInjections?: LorebookChatInjection[],  // positions 1-4: inject into specific messages
   935→   935→  exampleMessages?: ChatApiMessage[]  // SillyTavern-style example dialogue as chat messages
   936→   936→): ChatApiMessage[] {
   937→   937→  // =============================================
   938→   938→  // Step 1: Build all system content as ONE message
   939→   939→  // =============================================
   940→   940→  // Consolidate system prompt + embeddings + author note + post-history
   941→   941→  // into a single system message to avoid consecutive system messages
   942→   942→  // which break Jinja templates in models like LM Studio.
   943→   943→  const systemParts: string[] = [];
   944→   944→
   945→   945→  // Main system prompt
   946→   946→  if (systemPrompt.trim()) {
   947→   947→    systemParts.push(systemPrompt);
   948→   948→  }
   949→   949→
   950→   950→  // Embeddings Context
   951→   951→  if (embeddingsContext?.trim()) {
   952→   952→    systemParts.push(embeddingsContext);
   953→   953→  }
   954→   954→
   955→   955→  // Author's Note
   956→   956→  if (authorNote?.trim()) {
   957→   957→    systemParts.push(`[Author's Note]\n${authorNote}`);
   958→   958→  }
   959→   959→
   960→   960→  // Post-History Instructions
   961→   961→  if (postHistoryInstructions?.trim()) {
   962→   962→    systemParts.push(postHistoryInstructions);
   963→   963→  }
   964→   964→
   965→   965→  const chatMessages: ChatApiMessage[] = [];
   966→   966→
   967→   967→  // Single system message (or assistant if useSystemRole is false)
   968→   968→  if (systemParts.length > 0) {
   969→   969→    chatMessages.push({
   970→   970→      role: useSystemRole ? 'system' : 'assistant',
   971→   971→      content: systemParts.join('\n\n---\n\n')
   972→   972→    });
   973→   973→  }
   974→   974→
   975→   975→  // =============================================
   976→   976→  // Step 2: Inject example dialogue as chat messages (SillyTavern style)
   977→   977→  // =============================================
   978→   978→  // These go between the system message and the actual chat history
   979→   979→  // They serve as few-shot examples for the LLM
   980→   980→  if (exampleMessages && exampleMessages.length > 0) {
   981→   981→    chatMessages.push(...exampleMessages);
   982→   982→  }
   983→   983→
   984→   984→  // =============================================
   985→   985→  // Step 3: Build chat history with proper alternation
   986→   986→  // =============================================
   987→   987→  // Merge consecutive same-role messages and ensure
   988→   988→  // strict user/assistant alternation for OpenAI-compatible APIs.
   989→   989→  const visibleMessages = messages.filter(m => !m.isDeleted && !m.isNarratorMessage);
   990→   990→
   991→   991→  // First pass: merge consecutive same-role messages
   992→   992→  const mergedMessages: ChatApiMessage[] = [];
   993→   993→  for (const msg of visibleMessages) {
   994→   994→    const role = msg.role === 'user' ? 'user' : 'assistant';
   995→   995→    const last = mergedMessages[mergedMessages.length - 1];
   996→   996→    if (last && last.role === role) {
   997→   997→      // Merge into previous message
   998→   998→      last.content += '\n' + msg.content;
   999→   999→    } else {
  1000→  1000→      mergedMessages.push({ role, content: msg.content });
  1001→  1001→    }
  1002→  1002→  }
  1003→  1003→
  1004→  1004→  // Second pass: enforce user/assistant alternation
  1005→  1005→  // If the first chat message is 'assistant', prepend a synthetic user message
  1006→  1006→  if (mergedMessages.length > 0 && mergedMessages[0].role === 'assistant') {
  1007→  1007→    mergedMessages.unshift({ role: 'user', content: '*continúa*' });
  1008→  1008→  }
  1009→  1009→
  1010→  1010→  // If two same-role messages are still adjacent (shouldn't happen after merge,
  1011→  1011→  // but as safety), insert a bridging message
  1012→  1012→  const finalHistory: ChatApiMessage[] = [];
  1013→  1013→  for (const msg of mergedMessages) {
  1014→  1014→    const last = finalHistory[finalHistory.length - 1];
  1015→  1015→    if (last && last.role === msg.role) {
  1016→  1016→      // Insert bridging message
  1017→  1017→      const bridgeRole = msg.role === 'user' ? 'assistant' : 'user';
  1018→  1018→      finalHistory.push({ role: bridgeRole, content: '*continúa*' });
  1019→  1019→    }
  1020→  1020→    finalHistory.push(msg);
  1021→  1021→  }
  1022→  1022→
  1023→  1023→  // Bridge between example messages and chat history if needed
  1024→  1024→  // If the last example message has the same role as the first chat message,
  1025→  1025→  // insert a bridging message to maintain proper user/assistant alternation
  1026→  1026→  if (finalHistory.length > 0 && chatMessages.length > 0) {
  1027→  1027→    const lastExistingMsg = chatMessages[chatMessages.length - 1];
  1028→  1028→    const firstChatMsg = finalHistory[0];
  1029→  1029→    if (lastExistingMsg.role === firstChatMsg.role) {
  1030→  1030→      const bridgeRole = firstChatMsg.role === 'user' ? 'assistant' : 'user';
  1031→  1031→      chatMessages.push({ role: bridgeRole, content: '*continúa*' });
  1032→  1032→    }
  1033→  1033→  }
  1034→  1034→
  1035→  1035→  chatMessages.push(...finalHistory);
  1036→  1036→
  1037→  1037→  // =============================================
  1038→  1038→  // Step 4: Inject lorebook chat-level content (positions 1-4)
  1039→  1039→  // =============================================
  1040→  1040→  if (lorebookChatInjections?.length) {
  1041→  1041→    applyChatInjections(chatMessages, lorebookChatInjections);
  1042→  1042→  }
  1043→  1043→
  1044→  1044→  return chatMessages;
  1045→  1045→}

  1047→  1047→/**
  1048→  1048→ * Build prompt for completion-style APIs (Ollama, KoboldCPP, etc.)
  1049→  1049→ * 
  1050→  1050→ * Order (SillyTavern style):
  1051→  1051→ * 1. System prompt
  1052→  1052→ * 2. Chat history
  1053→  1053→ * 3. Author's Note
  1054→  1054→ * 4. Post-History Instructions
  1055→  1055→ * 5. Assistant prefix
  1056→  1056→ */
  1057→  1057→export function buildCompletionPrompt(config: CompletionPromptConfig): string {
  1058→  1058→  const { systemPrompt, messages, character, userName, postHistoryInstructions, authorNote, embeddingsContext, exampleMessages, allCharacters } = config;
  1059→  1059→  const parts: string[] = [];
  1060→  1060→
  1061→  1061→  parts.push(systemPrompt);
  1062→  1062→  parts.push('\n---\n');
  1063→  1063→
  1064→  1064→  // Embeddings Context - injected before chat history for recency primacy
  1065→  1065→  if (embeddingsContext?.trim()) {
  1066→  1066→    parts.push(embeddingsContext);
  1067→  1067→    parts.push('\n---\n');
  1068→  1068→  }
  1069→  1069→
  1070→  1070→  // Example dialogue is now included in the systemPrompt string (as [EJEMPLO] sections).
  1071→  1071→  // The exampleMessages parameter is kept for backward compatibility but will always be empty.
  1072→  1072→  // No separate injection needed here.
  1073→  1073→
  1074→  1074→  // Exclude narrator messages from prompt
  1075→  1075→  const visibleMessages = messages.filter(m => !m.isDeleted && !m.isNarratorMessage);
  1076→  1076→
  1077→  1077→  for (const msg of visibleMessages) {
  1078→  1078→    if (msg.role === 'user') {
  1079→  1079→      parts.push(`${userName}: ${msg.content}`);
  1080→  1080→    } else if (msg.role === 'assistant') {
  1081→  1081→      // Use actual speaker name from characterId (important for group chats where
  1082→  1082→      // multiple characters speak - each should be attributed to their own name)
  1083→  1083→      const speakerName = msg.characterId
  1084→  1084→        ? (allCharacters?.find(c => c.id === msg.characterId)?.name || character.name)
  1085→  1085→        : character.name;
  1086→  1086→      parts.push(`${speakerName}: ${msg.content}`);
  1087→  1087→    }
  1088→  1088→  }
  1089→  1089→
  1090→  1090→  // Author's Note - injected AFTER chat history, BEFORE post-history instructions
  1091→  1091→  if (authorNote?.trim()) {
  1092→  1092→    parts.push(`\n[Author's Note]\n${authorNote}`);
  1093→  1093→  }
  1094→  1094→
  1095→  1095→  // Post-History Instructions - injected AFTER Author's Note
  1096→  1096→  if (postHistoryInstructions) {
  1097→  1097→    parts.push(`\n${postHistoryInstructions}`);
  1098→  1098→  }
  1099→  1099→
  1100→  1100→  parts.push(`\n${character.name}:`);
  1101→  1101→
  1102→  1102→  return parts.join('\n');
  1103→  1103→}

  1352→  1352→  isForNarrator: boolean = false,

  1354→  1354→  lorebookChatInjections?: LorebookChatInjection[],  // positions 1-4: inject into specific messages
  1355→  1355→  exampleMessages?: ChatApiMessage[]  // SillyTavern-style example dialogue as chat messages
  1356→  1356→): GroupPromptBuildResult {
  1357→  1357→  // =============================================
  1358→  1358→  // Step 1: Build all system content as ONE message
  1359→  1359→  // =============================================
  1360→  1360→  const systemParts: string[] = [];
  1361→  1361→
  1362→  1362→  if (systemPrompt.trim()) {
  1363→  1363→    systemParts.push(systemPrompt);
  1364→  1364→  }
  1365→  1365→  if (embeddingsContext?.trim()) {
  1366→  1366→    systemParts.push(embeddingsContext);
  1367→  1367→  }
  1368→  1368→  if (authorNote?.trim()) {
  1369→  1369→    systemParts.push(`[Author's Note]\n${authorNote}`);
  1370→  1370→  }
  1371→  1371→  if (postHistoryInstructions?.trim()) {
  1372→  1372→    systemParts.push(postHistoryInstructions);
  1373→  1373→  }
  1374→  1374→
  1375→  1375→  const chatMessages: ChatApiMessage[] = [];
  1376→  1376→
  1377→  1377→  // Single system/assistant message
  1378→  1378→  if (systemParts.length > 0) {
  1379→  1379→    chatMessages.push({
  1380→  1380→      role: 'assistant',
  1381→  1381→      content: systemParts.join('\n\n---\n\n')
  1382→  1382→    });
  1383→  1383→  }
  1384→  1384→
  1385→  1385→  // =============================================
  1386→  1386→  // Step 2: Inject example dialogue as chat messages (SillyTavern style)
  1387→  1387→  // =============================================
  1388→  1388→  // These go between the system message and the actual chat history
  1389→  1389→  // They serve as few-shot examples for the LLM
  1390→  1390→  if (exampleMessages && exampleMessages.length > 0) {
  1391→  1391→    chatMessages.push(...exampleMessages);
  1392→  1392→  }
  1393→  1393→
  1394→  1394→  // =============================================
  1395→  1395→  // Step 3: Build chat history with proper alternation
  1396→  1396→  // =============================================
  1397→  1397→  const visibleMessages = isForNarrator
  1398→  1398→    ? messages.filter(m => !m.isDeleted)
  1399→  1399→    : messages.filter(m => !m.isDeleted && !m.isNarratorMessage);
  1400→  1400→
  1401→  1401→  const historyLines: string[] = [];
  1402→  1402→
  1403→  1403→  // Build history lines + API messages
  1404→  1404→  for (const msg of visibleMessages) {
  1405→  1405→    const speaker = msg.role === 'user' ? userName :
  1406→  1406→      (allCharacters.find(c => c.id === msg.characterId)?.name || 'Character');
  1407→  1407→    historyLines.push(`${speaker}: ${msg.content}`);
  1408→  1408→  }
  1409→  1409→
  1410→  1410→  // Build merged API messages with proper alternation
  1411→  1411→  // IMPORTANT: In group chats, multiple characters speak as 'assistant' role.
  1412→  1412→  // When merging consecutive assistant messages, we MUST include speaker names
  1413→  1413→  // so the LLM can distinguish which character said what.
  1414→  1414→  const mergedMessages: ChatApiMessage[] = [];
  1415→  1415→  for (const msg of visibleMessages) {
  1416→  1416→    const role = msg.role === 'user' ? 'user' : 'assistant';
  1417→  1417→    // Include speaker name for assistant messages so different characters' lines
  1418→  1418→    // are distinguishable even when merged into the same message
  1419→  1419→    const speakerName = msg.role === 'user' ? userName :
  1420→  1420→      (allCharacters.find(c => c.id === msg.characterId)?.name || character.name);
  1421→  1421→    const contentWithSpeaker = msg.role === 'user' ? msg.content : `${speakerName}: ${msg.content}`;
  1422→  1422→    const last = mergedMessages[mergedMessages.length - 1];
  1423→  1423→    if (last && last.role === role) {
  1424→  1424→      last.content += '\n' + contentWithSpeaker;
  1425→  1425→    } else {
  1426→  1426→      mergedMessages.push({ role, content: contentWithSpeaker });
  1427→  1427→    }
  1428→  1428→  }
  1429→  1429→
  1430→  1430→  // Add previous responses from this turn
  1431→  1431→  if (previousResponses && previousResponses.length > 0) {
  1432→  1432→    for (const resp of previousResponses) {
  1433→  1433→      historyLines.push(`${resp.characterName}: ${resp.content}`);
  1434→  1434→      const last = mergedMessages[mergedMessages.length - 1];
  1435→  1435→      const contentWithSpeaker = `${resp.characterName}: ${resp.content}`;
  1436→  1436→      if (last && last.role === 'assistant') {
  1437→  1437→        last.content += '\n' + contentWithSpeaker;
  1438→  1438→      } else {
  1439→  1439→        mergedMessages.push({ role: 'assistant', content: contentWithSpeaker });
  1440→  1440→      }
  1441→  1441→    }
  1442→  1442→  }
  1443→  1443→
  1444→  1444→  // Enforce alternation: if first is assistant, prepend synthetic user
  1445→  1445→  if (mergedMessages.length > 0 && mergedMessages[0].role === 'assistant') {
  1446→  1446→    mergedMessages.unshift({ role: 'user', content: '*continúa*' });
  1447→  1447→  }
  1448→  1448→
  1449→  1449→  // Safety: insert bridging messages if any same-role adjacency remains
  1450→  1450→  const finalHistory: ChatApiMessage[] = [];
  1451→  1451→  for (const msg of mergedMessages) {
  1452→  1452→    const last = finalHistory[finalHistory.length - 1];
  1453→  1453→    if (last && last.role === msg.role) {
  1454→  1454→      const bridgeRole = msg.role === 'user' ? 'assistant' : 'user';
  1455→  1455→      finalHistory.push({ role: bridgeRole, content: '*continúa*' });
  1456→  1456→    }
  1457→  1457→    finalHistory.push(msg);
  1458→  1458→  }
  1459→  1459→
  1460→  1460→  // Bridge between example messages and chat history if needed
  1461→  1461→  if (finalHistory.length > 0 && chatMessages.length > 0) {
  1462→  1462→    const lastExistingMsg = chatMessages[chatMessages.length - 1];
  1463→  1463→    const firstChatMsg = finalHistory[0];
  1464→  1464→    if (lastExistingMsg.role === firstChatMsg.role) {
  1465→  1465→      const bridgeRole = firstChatMsg.role === 'user' ? 'assistant' : 'user';
  1466→  1466→      chatMessages.push({ role: bridgeRole, content: '*continúa*' });
  1467→  1467→    }
  1468→  1468→  }
  1469→  1469→
  1470→  1470→  chatMessages.push(...finalHistory);
  1471→  1471→
  1472→  1472→  // Inject lorebook chat-level content (positions 1-4)
  1473→  1473→  if (lorebookChatInjections?.length) {
  1474→  1474→    applyChatInjections(chatMessages, lorebookChatInjections);
  1475→  1475→  }
  1476→  1476→
  1477→  1477→  // Build chat history section for prompt viewer
  1478→  1478→  let chatHistorySection: PromptSection | undefined;
  1479→  1479→  if (historyLines.length > 0) {
  1480→  1480→    chatHistorySection = {
  1481→  1481→      type: 'chat_history',
  1482→  1482→      label: 'Chat History',
  1483→  1483→      content: historyLines.join('\n\n'),
  1484→  1484→      color: SECTION_COLORS.chat_history
  1485→  1485→    };
  1486→  1486→  }
  1487→  1487→
  1488→  1488→  return {
  1489→  1489→    systemPrompt,
  1490→  1490→    sections: [],
  1491→  1491→    chatMessages,
  1492→  1492→    chatHistorySection
  1493→  1493→  };
  1494→  1494→}

  1598→  1598→export function buildMemorySection(memory: CharacterMemory, characterName: string): PromptSection | null {
  1599→  1599→  if (!memory.events.length && !memory.relationships.length && !memory.notes) {
  1600→  1600→    return null;
  1601→  1601→  }
  1602→  1602→
  1603→  1603→  const parts: string[] = [];
  1604→  1604→
  1605→  1605→  // Add events (sorted by importance, highest first)
  1606→  1606→  if (memory.events.length > 0) {
  1607→  1607→    parts.push(`[Eventos y hechos clave]`);
  1608→  1608→    const sortedEvents = [...memory.events].sort((a, b) => {
  1609→  1609→      // Support both old (0-1) and new (1-5) importance scales
  1610→  1610→      const impA = a.importance > 1 ? a.importance : Math.round(a.importance * 5);
  1611→  1611→      const impB = b.importance > 1 ? b.importance : Math.round(b.importance * 5);
  1612→  1612→      return impB - impA;
  1613→  1613→    });
  1614→  1614→    for (const event of sortedEvents) {
  1615→  1615→      // Support both old (0-1) and new (1-5) importance scales
  1616→  1616→      const normalizedImportance = event.importance > 1 ? event.importance : Math.round(event.importance * 5);
  1617→  1617→      const importance = normalizedImportance >= 4 ? '⭐' : '';
  1618→  1618→      parts.push(`${importance} ${event.content}`);
  1619→  1619→    }
  1620→  1620→  }
  1621→  1621→
  1622→  1622→  // Add relationships
  1623→  1623→  if (memory.relationships.length > 0) {
  1624→  1624→    parts.push(`\n[Relaciones]`);
  1625→  1625→    for (const rel of memory.relationships) {
  1626→  1626→      const sentiment = rel.sentiment > 50 ? '😊' : rel.sentiment < -50 ? '😞' : '😐';
  1627→  1627→      parts.push(`${sentiment} ${rel.targetName}: ${rel.relationship} (${rel.sentiment >= 0 ? '+' : ''}${rel.sentiment})`);
  1628→  1628→    }
  1629→  1629→  }
  1630→  1630→
  1631→  1631→  // Add notes
  1632→  1632→  if (memory.notes) {
  1633→  1633→    parts.push(`\n[Notas]\n${memory.notes}`);
  1634→  1634→  }
  1635→  1635→
  1636→  1636→  return {
  1637→  1637→    type: 'character_note',
  1638→  1638→    label: `Memoria de ${characterName}`,
  1639→  1639→    content: parts.join('\n'),
  1640→  1640→    color: SECTION_COLORS.memory
  1641→  1641→  };
  1642→  1642→}
