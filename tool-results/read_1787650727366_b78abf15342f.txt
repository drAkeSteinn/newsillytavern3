    94→/**
    95→ * Execute detected tool calls, send SSE events, and return results.
    96→ * Returns { newContent: combined display messages, shouldContinue: true if tools were executed }
    97→ */
    98→async function executeToolCallsAndContinue(
    99→  toolCalls: NativeToolCall[],
   100→  availableTools: Array<{ id: string; name: string; label: string; icon: string }>,
   101→  currentRound: number,
   102→  maxRounds: number,
   103→  character: CharacterCard,
   104→  sessionId: string,
   105→  userName: string,
   106→  controller: { enqueue: (chunk: string) => void },
   107→  sessionQuests?: SessionQuestInstance[],
   108→  questTemplates?: QuestTemplate[],
   109→  statsConfig?: CharacterStatsConfig,
   110→  sessionStats?: SessionStats,
   111→  allCharacters?: CharacterCard[],
   112→  characterMemory?: CharacterMemory,
   113→  lorebooks?: Lorebook[],
   114→): Promise<{ newContent: string; shouldContinue: boolean; toolResults: Array<{ success: boolean; displayMessage: string }>; questActivations: QuestActivation[]; toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> }> {
   115→  if (toolCalls.length === 0 || currentRound >= maxRounds) {
   116→    return { newContent: '', shouldContinue: false, toolResults: [], questActivations: [], toolsUsed: [] };
   117→  }
   118→
   119→  const toolResults: Array<{ success: boolean; displayMessage: string }> = [];
   120→  const questActivations: QuestActivation[] = [];
   121→  const toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> = [];
   122→  let allDisplayMessages = '';
   123→
   124→  for (let i = 0; i < toolCalls.length; i++) {
   125→    const tc = toolCalls[i];
   126→    const toolDef = availableTools.find(t => t.name === tc.name);
   127→    const callId = `${Date.now()}_${i}`;
   128→
   129→    // Send tool_call_start event
   130→    controller.enqueue(createSSEJSON({
   131→      type: 'tool_call_start',
   132→      toolName: tc.name,
   133→      toolLabel: toolDef?.label || tc.name,
   134→      toolIcon: toolDef?.icon || 'Wrench',
   135→      params: tc.arguments,
   136→      callId,
   137→    }));
   138→
   139→    console.log(`[Tools] Executing tool call: ${tc.name}`, tc.arguments);
   140→
   141→    // Execute the tool
   142→    const toolResult = await executeTool(
   143→      tc.name,
   144→      tc.arguments,
   145→      {
   146→        characterId: character.id,
   147→        characterName: character.name,
   148→        sessionId,
   149→        userName,
   150→        sessionQuests,
   151→        questTemplates,
   152→        statsConfig: character.statsConfig,
   153→        sessionStats,
   154→        allCharacters,
   155→        characterMemory,
   156→        lorebooks,
   157→      },
   158→    );
   159→
   160→    // Send tool_call_result event
   161→    controller.enqueue(createSSEJSON({
   162→      type: 'tool_call_result',
   163→      toolName: tc.name,
   164→      success: toolResult.success,
   165→      displayMessage: toolResult.displayMessage,
   166→      duration: toolResult.duration || 0,
   167→      callId,
   168→    }));
   169→
   170→    // Check for quest activation and send SSE event
   171→    if (toolResult.questActivation) {
   172→      const activation = toolResult.questActivation;
   173→      console.log(`[Tools] Quest activation from ${tc.name}:`, activation);
   174→      
   175→      controller.enqueue(createSSEJSON({
   176→        type: 'quest_activation',
   177→        toolName: tc.name,
   178→        activationType: activation.type,
   179→        key: activation.key,
   180→        metadata: activation.metadata,
   181→      }));
   182→      
   183→      questActivations.push(activation);
   184→    }
   185→
   186→    // Check for action/skill activation and send SSE event
   187→    if (toolResult.actionActivation) {
   188→      const action = toolResult.actionActivation;
   189→      console.log(`[Tools] Action activation from ${tc.name}:`, action.skillName);
   190→      
   191→      controller.enqueue(createSSEJSON({
   192→        type: 'action_activation',
   193→        toolName: tc.name,
   194→        skillId: action.skillId,
   195→        skillName: action.skillName,
   196→        skillDescription: action.skillDescription,
   197→        skillCompletedDescription: action.skillCompletedDescription,
   198→        activationCosts: action.activationCosts,
   199→        activationRewards: action.activationRewards,
   200→        characterId: action.characterId,
   201→      }));
   202→    }
   203→
   204→    // Check for relationship activation and send SSE event
   205→    if (toolResult.relationshipActivation) {
   206→      const rel = toolResult.relationshipActivation;
   207→      console.log(`[Tools] Relationship activation from ${tc.name}:`, rel.aName, '↔', rel.bName, rel.prevPoints, '→', rel.newPoints);
   208→
   209→      controller.enqueue(createSSEJSON({
   210→        type: 'relationship_activation',
   211→        toolName: tc.name,
   212→        aId: rel.aId,
   213→        aName: rel.aName,
   214→        bId: rel.bId,
   215→        bName: rel.bName,
   216→        prevPoints: rel.prevPoints,
   217→        newPoints: rel.newPoints,
   218→        reason: rel.reason,
   219→      }));
   220→    }
   221→
   222→    // Check for world-time activation and send SSE event
   223→    if (toolResult.timeActivation) {
   224→      const timeAct = toolResult.timeActivation;
   225→      console.log(`[Tools] Time activation from ${tc.name}:`, timeAct.type, timeAct.minutes || timeAct.hour || timeAct.season);
   226→
   227→      controller.enqueue(createSSEJSON({
   228→        type: 'time_activation',
   229→        toolName: tc.name,
   230→        activationType: timeAct.type,
   231→        minutes: timeAct.minutes,
   232→        hour: timeAct.hour,
   233→        minute: timeAct.minute,
   234→        season: timeAct.season,
   235→      }));
   236→    }
   237→
   238→    // Check for skill check activation and send SSE event
   239→    if (toolResult.checkActivation) {
   240→      const check = toolResult.checkActivation;
   241→      console.log(`[Tools] Skill check from ${tc.name}:`, check.statName, `${check.roll}${check.modifier >= 0 ? '+' : ''}${check.modifier}=${check.total} vs CD ${check.dc} → ${check.outcome}`);
   242→
   243→      controller.enqueue(createSSEJSON({
   244→        type: 'check_activation',
   245→        toolName: tc.name,
   246→        characterId: check.characterId,
   247→        characterName: check.characterName,
   248→        statName: check.statName,
   249→        statValue: check.statValue,
   250→        roll: check.roll,
   251→        modifier: check.modifier,
   252→        dc: check.dc,
   253→        total: check.total,
   254→        outcome: check.outcome,
   255→        outcomeLabel: check.outcomeLabel,
   256→        narrative: check.narrative,
   257→      }));
   258→    }
   259→
   260→    // Check for stat modification from modify_stat tool and send SSE event
   261→    if (toolResult.statActivation) {
   262→      const stat = toolResult.statActivation;
   263→      console.log(`[Tools] Stat activation from ${tc.name}:`, stat.attributeKey, stat.oldValue, '→', stat.newValue);
   264→      
   265→      controller.enqueue(createSSEJSON({
   266→        type: 'stat_activation',
   267→        toolName: tc.name,
   268→        characterId: stat.characterId,
   269→        attributeKey: stat.attributeKey,
   270→        attributeName: stat.attributeName,
   271→        attributeType: stat.attributeType,
   272→        oldValue: stat.oldValue,
   273→        newValue: stat.newValue,
   274→        reason: stat.reason,
   275→      }));
   276→    }
   277→
   278→    // Check for solicitud activation/completion and send SSE event
   279→    if (toolResult.solicitudActivation) {
   280→      const sol = toolResult.solicitudActivation;
   281→      console.log(`[Tools] Solicitud activation from ${tc.name}:`, sol.type, sol.solicitudKey);
   282→      
   283→      controller.enqueue(createSSEJSON({
   284→        type: 'solicitud_activation',
   285→        toolName: tc.name,
   286→        activationType: sol.type,
   287→        solicitudKey: sol.solicitudKey,
   288→        targetCharacterId: sol.targetCharacterId,
   289→        targetCharacterName: sol.targetCharacterName,
   290→        fromCharacterId: sol.fromCharacterId,
   291→        fromCharacterName: sol.fromCharacterName,
   292→        description: sol.description,
   293→        completionDescription: sol.completionDescription,
   294→        peticionKey: sol.peticionKey,
   295→      }));
   296→    }
   297→
   298→    // Check for memory activation (sync to client-side Character Memory)
   299→    if (toolResult.memoryActivation) {
   300→      const mem = toolResult.memoryActivation;
   301→      console.log(`[Tools] Memory activation from ${tc.name}:`, mem.type);
   302→      
   303→      controller.enqueue(createSSEJSON({
   304→        type: 'memory_activation',
   305→        toolName: tc.name,
   306→        activationType: mem.type,
   307→        characterId: mem.characterId,
   308→        eventData: mem.eventData,
   309→        relationshipData: mem.relationshipData,
   310→        noteContent: mem.noteContent,
   311→        deleteEventId: mem.deleteEventId,
   312→        deleteEmbeddingId: mem.deleteEmbeddingId,
   313→      }));
   314→    }
   315→
   316→    console.log(`[Tools] Tool ${tc.name}: success=${toolResult.success} duration=${toolResult.duration}ms`, toolResult.displayMessage);
   317→
   318→    toolResults.push({ success: toolResult.success, displayMessage: toolResult.displayMessage });
   319→    toolsUsed.push({
   320→      name: tc.name,
   321→      label: toolDef?.label || tc.name,
   322→      icon: toolDef?.icon || 'Wrench',
   323→      success: toolResult.success,
   324→    });
   325→    if (toolResult.displayMessage) {
   326→      allDisplayMessages += (allDisplayMessages ? '\n' : '') + toolResult.displayMessage;
   327→    }
   328→  }
   329→
   330→  // Return the display messages as content, tool results, and signal that a follow-up call is needed
   331→  return {
   332→    newContent: allDisplayMessages,
   333→    shouldContinue: true, // Always continue to get the LLM's natural response
   334→    toolResults,
   335→    questActivations,
   336→    toolsUsed,
   337→  };
   338→}

   340→/**
   341→ * POST /api/chat/proactive
   342→ *
   343→ * Generates a proactive message for a character using SSE streaming.
   344→ * Full feature parity with the stream route including tools, embeddings,
   345→ * key resolution, memory reinforcement, and extraction.
   346→ */
   347→export async function POST(request: NextRequest) {
   348→  try {
   349→    // Capture auth headers from Z.ai gateway for token resolution
   350→    const incomingXToken = request.headers.get('X-Token');
   351→    const fcSecurityToken = request.headers.get('x-fc-security-token');
   352→
   353→    const body = await request.json();
   354→
   355→    // Extract required fields from body
   356→    const {
   357→      character,
   358→      messages = [],
   359→      llmConfig,
   360→      userName = 'User',
   361→      persona,
   362→      proactiveConfig,
   363→      reason = 'timer_idle',
   364→      lastActivityAt,
   365→      isGroupChat = false,
   366→      // FASE 11 v2: tracking de índices usados para la rotación linear/random de casos.
   367→      usedCaseIndices: clientUsedCaseIndices = {},
   368→    } = body;
   369→
   370→    const clientUsedCaseIndicesTyped: UsedCaseIndices =
   371→      (clientUsedCaseIndices && typeof clientUsedCaseIndices === 'object')
   372→        ? clientUsedCaseIndices as UsedCaseIndices
   373→        : {};
   374→
   375→    // Validate required fields
   376→    if (!character?.name || !llmConfig?.provider) {
   377→      return createErrorResponse('Missing required fields: character, llmConfig', 400);
   378→    }
   379→
   380→    if (!proactiveConfig?.enabled) {
   381→      return createErrorResponse('Proactive messages are disabled for this character', 400);
   382→    }
   383→
   384→    if (!llmConfig) {
   385→      return createErrorResponse('No LLM configuration provided', 400);
   386→    }
   387→
   388→    // Validate API key for providers that require one
   389→    const providersRequiringApiKey = ['openai', 'vllm', 'lm-studio', 'custom', 'anthropic', 'grok'];
   390→    if (providersRequiringApiKey.includes(llmConfig.provider) && !llmConfig.apiKey) {
   391→      return createErrorResponse(
   392→        `API Key is required for ${llmConfig.provider} provider. Please configure it in Settings.`,
   393→        400
   394→      );
   395→    }
   396→
   397→    // Validate endpoint for providers that require one
   398→    const providersRequiringEndpoint = ['openai', 'vllm', 'lm-studio', 'custom', 'anthropic', 'ollama', 'text-generation-webui', 'koboldcpp'];
   399→    if (providersRequiringEndpoint.includes(llmConfig.provider) && !llmConfig.endpoint) {
   400→      return createErrorResponse(
   401→        `Endpoint URL is required for ${llmConfig.provider} provider. Please configure it in Settings.`,
   402→        400
   403→      );
   404→    }
   405→
   406→    // If using Z.ai provider, collect gateway-forwarded auth headers
   407→    let zaiRuntimeToken: string | undefined;
   408→    if (llmConfig.provider === 'z-ai') {
   409→      const gatewayToken = incomingXToken || fcSecurityToken || undefined;
   410→      if (gatewayToken) {
   411→        zaiRuntimeToken = gatewayToken;
   412→        console.log(`[Proactive Route] Z.ai runtime token available (${gatewayToken.length} chars, source: ${incomingXToken ? 'X-Token' : 'fc-security-token'})`);
   413→      } else {
   414→        console.log(`[Proactive Route] Z.ai: no gateway token available, using config file only`);
   415→      }
   416→    }
   417→
   418→    console.log(`[Proactive] Generating proactive message for "${character.name}" (reason: ${reason})`);
   419→
   420→    // Extract all additional fields from body (matching stream route)
   421→    const lorebooks: Lorebook[] = body.lorebooks || [];
   422→    const sessionStats: SessionStats | undefined = body.sessionStats;
   423→    const allCharacters: CharacterCard[] = body.allCharacters || [character];
   424→
   425→    // Extract HUD context from body
   426→    const hudContext: HUDContextConfig | undefined = body.hudContext;
   427→
   428→    // Extract Quest data
   429→    const questTemplates: QuestTemplate[] = body.questTemplates || [];
   430→    const sessionQuests: SessionQuestInstance[] = body.sessionQuests || [];
   431→    const questSettings: QuestSettings = {
   432→      ...DEFAULT_QUEST_SETTINGS,
   433→      ...(body.questSettings || {})
   434→    };
   435→
   436→    // Extract Sound data for {{sonidos}} key resolution
   437→    const soundTriggers: SoundTrigger[] = body.soundTriggers || [];
   438→    const soundSettings = body.settings?.sound;
   439→
   440→    // Extract summary for memory/context compression
   441→    const summary: SessionSummary | undefined = body.summary;
   442→
   443→    // Extract embeddings chat settings
   444→    const embeddingsChat: Partial<EmbeddingsChatSettings> = body.embeddingsChat || {};
   445→    const characterMemory: CharacterMemory | undefined = body.characterMemory;
   446→    const sessionId: string | undefined = body.sessionId;
   447→    const characterId: string | undefined = body.characterId;
   448→    const inventoryData: InventoryPromptData | undefined = body.inventoryData;
   449→
   450→    // Extract tools settings
   451→    const toolsSettings: ToolsSettings = {
   452→      enabled: body.toolsSettings?.enabled ?? true,
   453→      maxToolCallsPerTurn: body.toolsSettings?.maxToolCallsPerTurn ?? 2,
   454→      characterConfigs: body.toolsSettings?.characterConfigs || [],
   455→      usePromptBasedFallback: body.toolsSettings?.usePromptBasedFallback ?? false,
   456→      disabledTools: body.toolsSettings?.disabledTools || [],
   457→    };
   458→
   459→    // Create default character if none provided
   460→    const effectiveCharacter: CharacterCard = character || DEFAULT_CHARACTER;
   461→
   462→    // Get effective user name from persona or use provided userName
   463→    const effectiveUserName = getEffectiveUserName(persona, userName);
   464→
   465→    // Build context configuration from request or use defaults
   466→    const contextConfig: Partial<ContextConfig> = body.contextConfig || {};
   467→
   468→    // Apply sliding window to messages
   469→    const contextWindow = selectContextMessages(messages, llmConfig, contextConfig);
   470→
   471→    // Log context stats (for debugging)
   472→    const stats = getContextStats(messages);
   473→
   474→    // Process lorebooks and get matched entries
   475→    const { plan: lorebookPlan, lorebookAttributeKeys, lorebookEntryKeyMap, lorebookDebugEntries } = buildLorebookSectionForPrompt(
   476→      messages,
   477→      lorebooks,
   478→      {
   479→        scanDepth: (contextConfig as any).scanDepth,
   480→        userName: effectiveUserName,
   481→        charName: effectiveCharacter?.name,
   482→      },
   483→      { sessionStats, characterId: effectiveCharacter?.id, characters: allCharacters }
   484→    );
   485→
   486→    // ========================================
   487→    // DEBUG: Build lorebook debug data for SSE event
   488→    // ========================================
   489→    const lorebookDebugData = {
   490→      lorebookAttributeKeys,
   491→      debugEntries: lorebookDebugEntries,
   492→      availableStats: sessionStats
   493→        ? Object.fromEntries(
   494→            Object.entries(sessionStats.characterStats || {}).map(([charId, cs]) => [
   495→              charId,
   496→              cs?.attributeValues ? { ...cs.attributeValues } : '(no values)',
   497→            ])
   498→          )
   499→        : '(no sessionStats)',
   500→    };
   501→
   502→    // ========================================
   503→    // Embeddings Context Retrieval
   504→    // ========================================
   505→    // For proactive messages, use recent chat context as the search query
   506→    // since there's no new user message
   507→    let enrichedSearchQuery = '';
   508→    const searchContextDepth = embeddingsChat.searchContextDepth || 0;
   509→    if (searchContextDepth > 0) {
   510→      const recentHistory = messages
   511→        .filter(m => !m.isDeleted)
   512→        .slice(-(searchContextDepth * 2))
   513→        .map(m => m.content?.trim())
   514→        .filter(Boolean);
   515→      
   516→      if (recentHistory.length > 0) {
   517→        enrichedSearchQuery = recentHistory.join(' ');
   518→      }
   519→    } else {
   520→      // Use last few messages as search context
   521→      const recentHistory = messages
   522→        .filter(m => !m.isDeleted)
   523→        .slice(-4)
   524→        .map(m => m.content?.trim())
   525→        .filter(Boolean);
   526→      
   527→      if (recentHistory.length > 0) {
   528→        enrichedSearchQuery = recentHistory.join(' ');
   529→      }
   530→    }
   531→
   532→    // Safety: smart truncation based on embedding model's context window.
   533→    try {
   534→      const { MODEL_CONTEXT_LENGTHS, DEFAULT_CONTEXT_LENGTH, CHARS_PER_TOKEN } = await import('@/lib/embeddings/types');
   535→      const { loadConfig } = await import('@/lib/embeddings/config-persistence');
   536→      const embConfig = loadConfig();
   537→      const modelKey = embConfig.model || 'bge-m3:567m';
   538→      const modelCtx = MODEL_CONTEXT_LENGTHS[modelKey]
   539→        || MODEL_CONTEXT_LENGTHS[modelKey.split(':')[0]]
   540→        || DEFAULT_CONTEXT_LENGTH;
   541→      const maxQueryChars = Math.floor(modelCtx * 0.75 * CHARS_PER_TOKEN);
   542→
   543→      if (enrichedSearchQuery.length > maxQueryChars) {
   544→        // For proactive: keep the most recent part (end of the string)
   545→        enrichedSearchQuery = enrichedSearchQuery.slice(-maxQueryChars);
   546→        console.warn(
   547→          `[Proactive Route] Search query trimmed to ${enrichedSearchQuery.length} chars ` +
   548→          `(model: ${modelKey}, context: ${modelCtx} tokens)`
   549→        );
   550→      }
   551→    } catch { /* fallback: Ollama client handles truncation */ }
   552→
   553→    // Retrieve relevant embeddings based on enriched query and settings
   554→    // Pass Character Memory events for deduplication (avoid duplicate memory in prompt)
   555→    const existingMemoryEvents = characterMemory?.events?.map(e => ({
   556→      content: e.content,
   557→      importance: e.importance,
   558→    }));
   559→
   560→    // Extract last assistant message for bidirectional search
   561→    const lastAssistantMsg = messages
   562→      .filter(m => !m.isDeleted && m.role === 'assistant')
   563→      .pop()?.content;
   564→
   565→    const embeddingsResult = await retrieveEmbeddingsContext(
   566→      enrichedSearchQuery,
   567→      characterId || effectiveCharacter.id,
   568→      sessionId,
   569→      embeddingsChat,
   570→      undefined, // groupId
   571→      existingMemoryEvents, // for deduplication
   572→      lastAssistantMsg, // bidirectional search with last assistant message
   573→    );
