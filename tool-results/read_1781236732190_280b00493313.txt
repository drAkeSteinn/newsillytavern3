    94→
    95→export function CharacterEditor({ characterId, open, onClose }: CharacterEditorProps) {
    96→  // Use individual selectors to avoid re-rendering on unrelated store changes
    97→  const addCharacter = useTavernStore((s) => s.addCharacter);
    98→  const updateCharacter = useTavernStore((s) => s.updateCharacter);
    99→  const getCharacterById = useTavernStore((s) => s.getCharacterById);
   100→  const characters = useTavernStore((s) => s.characters);
   101→  const personas = useTavernStore((s) => s.personas);
   102→  const activePersonaId = useTavernStore((s) => s.activePersonaId);
   103→  const questTemplates = useTavernStore((s) => s.questTemplates);
   104→
   105→  // Active tab state
   106→  const [activeTab, setActiveTab] = useState('info');
   107→
   108→  // Global TTS config for voice panel (OmniVoice provider awareness)
   109→  const [globalTTSConfig, setGlobalTTSConfig] = useState<TTSWebUIConfig | null>(null);
   110→
   111→  const loadTTSConfig = useCallback(async () => {
   112→    try {
   113→      const response = await fetch('/api/tts/config');
   114→      const data = await response.json();
   115→      if (data.success && data.config?.tts) {
   116→        setGlobalTTSConfig(data.config.tts);
   117→      }
   118→    } catch {
   119→      // Silently fail — TTS config is optional for voice panel
   120→    }
   121→  }, []);
   122→
   123→  useEffect(() => {
   124→    if (open) {
   125→      loadTTSConfig();
   126→    }
   127→  }, [open, loadTTSConfig]);
   128→
   129→  // Reset tab when opening
   130→  useEffect(() => {
   131→    if (open) {
   132→      setActiveTab('info');
   133→    }
   134→  }, [open]);
   135→
   136→  // Escape key handler
   137→  const onCloseRef = useRef(onClose);
   138→  onCloseRef.current = onClose;
   139→
   140→  useEffect(() => {
   141→    if (!open) return;
   142→    const handleKeyDown = (e: KeyboardEvent) => {
   143→      if (e.key === 'Escape') {
   144→        e.preventDefault();
   145→        onCloseRef.current();
   146→      }
   147→    };
   148→    document.addEventListener('keydown', handleKeyDown);
   149→    return () => document.removeEventListener('keydown', handleKeyDown);
   150→  }, [open]);
   151→
   152→  // Get active persona
   153→  const activePersona = personas.find(p => p.id === activePersonaId);
   154→
   155→  // Get all characters except the one being edited (for target selection in invitations)
   156→  // Also include the active persona if it has solicitudes configured
   157→  const allCharacters = useMemo(() => {
   158→    const result = characters
   159→      .filter(c => c.id !== characterId)
   160→      .map(c => ({
   161→        id: c.id,
   162→        name: c.name,
   163→        solicitudDefinitions: c.statsConfig?.solicitudDefinitions || []
   164→      }));
   165→    
   166→    // Add active persona if it has solicitudes configured
   167→    if (activePersona?.statsConfig?.enabled && 
   168→        (activePersona.statsConfig.solicitudDefinitions?.length || 0) > 0) {
   169→      result.push({
   170→        id: '__user__',
   171→        name: activePersona.name || 'Usuario',
   172→        solicitudDefinitions: activePersona.statsConfig.solicitudDefinitions || []
   173→      });
   174→    }
   175→    
   176→    return result;
   177→  }, [characters, characterId, activePersona]);
   178→
   179→  // Build available targets for target_attribute rewards: other characters + persona
   180→  const availableTargets = useMemo(() => {
   181→    const targets: Array<{
   182→      id: string;
   183→      name: string;
   184→      attributes: Array<{ key: string; name: string; type: 'number' | 'keyword' | 'text'; min?: number; max?: number }>;
   185→      spritePacks?: Array<{ id: string; name: string; conditionalMode?: boolean; spriteCount: number }>;
   186→    }> = [];
   187→    // Helper: map attributes filtering out those without a valid key
   188→    const mapAttrs = (attrs: Array<{ key: string; name: string; type: 'number' | 'keyword' | 'text'; min?: number; max?: number }>) =>
   189→      attrs.filter(a => a.key && a.key.trim() !== '').map(a => ({
   190→        key: a.key,
   191→        name: a.name,
   192→        type: a.type,
   193→        min: a.min,
   194→        max: a.max,
   195→      }));
   196→
   197→    // Helper: map sprite packs for a character
   198→    const mapPacks = (packs: Array<{ id: string; name: string; conditionalMode?: boolean; sprites: unknown[] }>) =>
   199→      packs.map(p => ({ id: p.id, name: p.name, conditionalMode: p.conditionalMode, spriteCount: p.sprites.length }));
   200→
   201→    // Add OTHER characters with attributes (exclude the one being edited)
   202→    characters.forEach(c => {
   203→      const attrs = mapAttrs(c.statsConfig?.attributes || []);
   204→      if (c.id !== characterId && c.statsConfig?.enabled && attrs.length > 0) {
   205→        targets.push({
   206→          id: c.id,
   207→          name: c.name,
   208→          attributes: attrs,
   209→          spritePacks: mapPacks(c.spritePacksV2 || []),
   210→        });
   211→      }
   212→    });
   213→    // Add active persona with attributes
   214→    if (activePersona?.statsConfig?.enabled) {
   215→      const attrs = mapAttrs(activePersona.statsConfig.attributes || []);
   216→      if (attrs.length > 0) {
   217→        targets.push({
   218→          id: '__user__',
   219→          name: activePersona.name || 'Persona',
   220→          attributes: attrs,
   221→          spritePacks: mapPacks(activePersona.spritePacksV2 || []),
   222→        });
   223→      }
   224→    }
   225→    return targets;
   226→  }, [characters, characterId, activePersona]);
   227→
   228→  // Initialize character data based on characterId
   229→  const getInitialCharacter = () => {
   230→    if (characterId) {
   231→      const existing = getCharacterById(characterId);
   232→      if (existing) {
   233→        return existing;
   234→      }
   235→    }
   236→    return defaultCharacter;
   237→  };
   238→
   239→  const [character, setCharacter] = useState(getInitialCharacter);
   240→  const [newTag, setNewTag] = useState('');
   241→  const [uploading, setUploading] = useState(false);
   242→  const fileInputRef = useRef<HTMLInputElement>(null);
   243→
   244→  const handleSave = () => {
   245→    if (!character.name.trim()) {
   246→      alert('El nombre del personaje es requerido');
   247→      return;
   248→    }
   249→
   250→    if (characterId) {
   251→      updateCharacter(characterId, character);
   252→    } else {
   253→      addCharacter(character);
   254→    }
   255→    onClose();
   256→  };
   257→
   258→  const handleAvatarUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   259→    const file = e.target.files?.[0];
   260→    if (!file) return;
   261→
   262→    // Validate file size (max 5MB)
   263→    if (file.size > 5 * 1024 * 1024) {
   264→      alert('La imagen es muy grande. El tamaño máximo es 5MB.');
   265→      return;
   266→    }
   267→
   268→    // Validate file type
   269→    const allowedTypes = ['image/jpeg', 'image/png', 'image/gif', 'image/webp'];
   270→    if (!allowedTypes.includes(file.type)) {
   271→      alert('Formato no soportado. Usa JPEG, PNG, GIF o WebP.');
   272→      return;
   273→    }
   274→
   275→    setUploading(true);
   276→
   277→    try {
   278→      // Upload via API to save as file
   279→      const formData = new FormData();
   280→      formData.append('file', file);
   281→      formData.append('type', 'avatar');
   282→
   283→      const res = await fetch('/api/upload', {
   284→        method: 'POST',
   285→        body: formData,
   286→      });
   287→
   288→      const data = await res.json();
   289→
   290→      if (res.ok && data.success) {
   291→        setCharacter(prev => ({ ...prev, avatar: data.url }));
   292→      } else {
   293→        alert(data.error || 'Error al subir la imagen');
   294→      }
   295→    } catch (error) {
   296→      editorLogger.error('Upload error', { error });
   297→      alert('Error de conexión al subir la imagen');
   298→    } finally {
   299→      setUploading(false);
   300→      if (fileInputRef.current) {
   301→        fileInputRef.current.value = '';
   302→      }
   303→    }
   304→  };
   305→
   306→  const handleAddTag = () => {
   307→    if (newTag.trim() && !character.tags.includes(newTag.trim())) {
   308→      setCharacter(prev => ({
   309→        ...prev,
   310→        tags: [...prev.tags, newTag.trim()]
   311→      }));
   312→      setNewTag('');
   313→    }
   314→  };
   315→
   316→  const handleRemoveTag = (tag: string) => {
   317→    setCharacter(prev => ({
   318→      ...prev,
   319→      tags: prev.tags.filter(t => t !== tag)
   320→    }));
   321→  };
   322→
   323→  // Tab content renderers
   324→  const renderInfoTab = () => (
   325→    <div className="space-y-6">
   326→      {/* Banner */}
   327→      <div className="flex items-center gap-3 p-3 bg-gradient-to-r from-blue-500/10 to-indigo-500/10 border border-blue-500/20 rounded-lg">
   328→        <Palette className="w-4 h-4 text-blue-500 shrink-0" />
   329→        <p className="text-xs text-muted-foreground">
   330→          Configura <strong>avatar</strong>, <strong>información básica</strong> y <strong>asignaciones</strong> del personaje.
   331→        </p>
   332→      </div>
   333→
   334→      {/* ═══ Section: Identidad ═══ */}
   335→      <div className="rounded-lg border bg-card p-5">
   336→        <div className="flex items-center gap-2 mb-4">
   337→          <div className="flex items-center justify-center w-6 h-6 rounded-md bg-blue-500/10">
   338→            <Palette className="w-3.5 h-3.5 text-blue-500" />
   339→          </div>
   340→          <h3 className="text-sm font-semibold">Identidad</h3>
   341→        </div>
   342→
   343→        <div className="flex gap-6 items-start">
   344→          {/* Avatar */}
   345→          <div className="flex-shrink-0">
   346→            <Label className="text-xs font-medium mb-2 block">Avatar</Label>
   347→            <Tooltip>
   348→              <TooltipTrigger asChild>
   349→                <div 
   350→                  className={cn(
   351→                    "w-32 h-32 rounded-xl overflow-hidden bg-muted border-2 border-dashed border-muted-foreground/25 flex items-center justify-center transition-colors",
   352→                    !uploading && "cursor-pointer hover:border-primary/50 hover:bg-muted/50"
   353→                  )}
   354→                  onClick={() => !uploading && fileInputRef.current?.click()}
   355→                >
   356→                  {uploading ? (
   357→                    <div className="text-center text-muted-foreground">
   358→                      <Loader2 className="w-7 h-7 mx-auto animate-spin" />
   359→                      <span className="text-[10px] mt-1 block">Subiendo...</span>
   360→                    </div>
   361→                  ) : character.avatar ? (
   362→                    <img 
   363→                      src={character.avatar} 
   364→                      alt={character.name || 'Avatar'}
   365→                      className="w-full h-full object-cover"
   366→                    />
   367→                  ) : (
   368→                    <div className="text-center text-muted-foreground">
   369→                      <Camera className="w-8 h-8 mx-auto mb-1 opacity-50" />
   370→                      <span className="text-[10px]">Subir avatar</span>
   371→                    </div>
   372→                  )}
   373→                </div>
   374→              </TooltipTrigger>
   375→              <TooltipContent>
   376→                <p>Haz clic para subir avatar (máx 5MB)</p>
   377→              </TooltipContent>
   378→            </Tooltip>
   379→            <input
   380→              type="file"
   381→              ref={fileInputRef}
   382→              className="hidden"
   383→              accept="image/*"
   384→              onChange={handleAvatarUpload}
   385→              disabled={uploading}
   386→            />
   387→          </div>
   388→
   389→          {/* Name + Tags */}
   390→          <div className="flex-1 min-w-0 space-y-4">
   391→            {/* Name field */}
   392→            <div>
   393→              <div className="flex items-center gap-1.5 mb-1.5">
   394→                <Label htmlFor="name" className="text-sm font-medium">Nombre *</Label>
   395→                <Tooltip>
   396→                  <TooltipTrigger asChild>
   397→                    <HelpCircle className="w-3.5 h-3.5 text-muted-foreground cursor-help" />
   398→                  </TooltipTrigger>
   399→                  <TooltipContent className="max-w-xs">
   400→                    <p>El nombre del personaje que se mostrará en el chat.</p>
   401→                  </TooltipContent>
   402→                </Tooltip>
   403→              </div>
   404→              <Input
   405→                id="name"
   406→                value={character.name}
   407→                onChange={(e) => setCharacter(prev => ({ ...prev, name: e.target.value }))}
   408→                placeholder="Nombre del personaje"
   409→                className="h-10"
   410→              />
   411→            </div>
   412→
   413→            {/* Tags input */}
   414→            <div>
   415→              <div className="flex items-center gap-1.5 mb-1.5">
   416→                <Label className="text-sm font-medium">Etiquetas</Label>
   417→                <Tooltip>
   418→                  <TooltipTrigger asChild>
   419→                    <HelpCircle className="w-3.5 h-3.5 text-muted-foreground cursor-help" />
   420→                  </TooltipTrigger>
   421→                  <TooltipContent className="max-w-xs">
   422→                    <p>Tags para organizar y buscar personajes.</p>
   423→                  </TooltipContent>
   424→                </Tooltip>
   425→              </div>
   426→              <div className="flex gap-2">
   427→                <Input
   428→                  value={newTag}
   429→                  onChange={(e) => setNewTag(e.target.value)}
   430→                  placeholder="Agregar etiqueta..."
   431→                  onKeyDown={(e) => e.key === 'Enter' && handleAddTag()}
   432→                  className="h-9 flex-1"
   433→                />
   434→                <Button variant="outline" size="sm" className="h-9 px-4" onClick={handleAddTag}>
   435→                  <Plus className="w-3.5 h-3.5 mr-1" />
   436→                  Agregar
   437→                </Button>
   438→              </div>
   439→              {character.tags.length > 0 && (
   440→                <div className="flex flex-wrap gap-1.5 mt-2">
   441→                  {character.tags.map((tag) => (
   442→                    <Badge key={tag} variant="secondary" className="gap-1 text-xs py-1 px-2.5">
   443→                      {tag}
   444→                      <button
   445→                        type="button"
   446→                        className="ml-0.5 rounded-full p-0.5 hover:bg-destructive/20 hover:text-destructive transition-colors"
   447→                        onClick={(e) => { e.stopPropagation(); handleRemoveTag(tag); }}
   448→                      >
   449→                        <X className="w-3 h-3" />
   450→                      </button>
   451→                    </Badge>
   452→                  ))}
   453→                </div>
   454→              )}
   455→            </div>
   456→          </div>
   457→        </div>
   458→      </div>
   459→
   460→      {/* ═══ Section: Asignaciones ═══ */}
   461→      <div className="rounded-lg border bg-card p-5">
   462→        <div className="flex items-center gap-2 mb-4">
   463→          <div className="flex items-center justify-center w-6 h-6 rounded-md bg-amber-500/10">
   464→            <Package className="w-3.5 h-3.5 text-amber-500" />
   465→          </div>
   466→          <h3 className="text-sm font-semibold">Asignaciones</h3>
   467→          <span className="text-xs text-muted-foreground ml-1">— Recursos vinculados a este personaje</span>
   468→        </div>
   469→
   470→        <div className="grid grid-cols-1 md:grid-cols-2 gap-5">
   471→          {/* HUD Selector */}
   472→          <div className="space-y-2">
   473→            <div className="flex items-center gap-2">
   474→              <Layers className="w-4 h-4 text-blue-500" />
   475→              <Label className="text-sm font-medium">HUD</Label>
   476→              <Tooltip>
   477→                <TooltipTrigger asChild>
   478→                  <HelpCircle className="w-3.5 h-3.5 text-muted-foreground cursor-help" />
   479→                </TooltipTrigger>
   480→                <TooltipContent className="max-w-xs">
   481→                  <p>Selecciona un template de HUD para mostrar estadísticas del personaje.</p>
   482→                </TooltipContent>
   483→              </Tooltip>
   484→            </div>
   485→            <HUDSelector
   486→              value={character.hudTemplateId}
   487→              onChange={(hudTemplateId) => setCharacter(prev => ({ ...prev, hudTemplateId }))}
   488→              placeholder="Sin HUD"
   489→            />
   490→          </div>
   491→
   492→          {/* Sprite Collection Selector */}
   493→          <div className="space-y-2">
   494→            <div className="flex items-center gap-2">
   495→              <ImageIcon className="w-4 h-4 text-green-500" />
   496→              <Label className="text-sm font-medium">Colección de Sprites</Label>
   497→              <Tooltip>
   498→                <TooltipTrigger asChild>
   499→                  <HelpCircle className="w-3.5 h-3.5 text-muted-foreground cursor-help" />
   500→                </TooltipTrigger>
   501→                <TooltipContent className="max-w-xs">
   502→                  <p>Selecciona una colección de sprites para este personaje.</p>
   503→                </TooltipContent>
   504→              </Tooltip>
   505→            </div>
   506→            <SpriteCollectionSelector
   507→              value={character.spriteConfig?.collection}
   508→              onChange={(collectionName) => setCharacter(prev => ({ 
   509→                ...prev, 
   510→                spriteConfig: { 
   511→                  ...prev.spriteConfig, 
   512→                  enabled: true,
   513→                  collection: collectionName,
   514→                  sprites: prev.spriteConfig?.sprites || {},
   515→                  stateCollections: prev.spriteConfig?.stateCollections || {}
   516→                } 
   517→              }))}
   518→              placeholder="Sin sprites"
   519→            />
   520→          </div>
   521→
   522→          {/* Lorebook Selector */}
   523→          <div className="space-y-2">
   524→            <div className="flex items-center gap-2">
   525→              <BookOpen className="w-4 h-4 text-amber-500" />
   526→              <Label className="text-sm font-medium">Lorebooks</Label>
   527→              <Tooltip>
   528→                <TooltipTrigger asChild>
   529→                  <HelpCircle className="w-3.5 h-3.5 text-muted-foreground cursor-help" />
   530→                </TooltipTrigger>
   531→                <TooltipContent className="max-w-xs">
   532→                  <p>Lorebooks que se inyectarán en el prompt al chatear con este personaje.</p>
   533→                </TooltipContent>
   534→              </Tooltip>
   535→            </div>
   536→            <LorebookSelector
   537→              value={character.lorebookIds}
   538→              onChange={(lorebookIds) => setCharacter(prev => ({ ...prev, lorebookIds }))}
   539→              placeholder="Sin lorebooks asignados"
   540→            />
   541→          </div>
   542→
   543→          {/* Quest Templates Selector */}
   544→          <div className="space-y-2">
   545→            <div className="flex items-center gap-2">
   546→              <ScrollText className="w-4 h-4 text-purple-500" />
   547→              <Label className="text-sm font-medium">Misiones</Label>
   548→              <Tooltip>
   549→                <TooltipTrigger asChild>
   550→                  <HelpCircle className="w-3.5 h-3.5 text-muted-foreground cursor-help" />
   551→                </TooltipTrigger>
   552→                <TooltipContent className="max-w-xs">
   553→                  <p>Templates de misiones disponibles para este personaje.</p>
   554→                </TooltipContent>
   555→              </Tooltip>
   556→            </div>
   557→            <QuestSelector
