    66→// ============================================
    67→// Sprite List Rotation Tracker
    68→// Tracks rotation indices per character per state
    69→// Avoids modifying character data (no persistence noise)
    70→// ============================================
    71→const _spriteRotationIndex = new Map<string, Map<SpriteState, number>>();
    72→
    73→function getRotationIndex(characterId: string, state: SpriteState): number {
    74→  let charMap = _spriteRotationIndex.get(characterId);
    75→  if (!charMap) {
    76→    charMap = new Map<SpriteState, number>();
    77→    _spriteRotationIndex.set(characterId, charMap);
    78→  }
    79→  return charMap.get(state) ?? 0;
    80→}
    81→
    82→function advanceRotationIndex(characterId: string, state: SpriteState): number {
    83→  let charMap = _spriteRotationIndex.get(characterId);
    84→  if (!charMap) {
    85→    charMap = new Map<SpriteState, number>();
    86→    _spriteRotationIndex.set(characterId, charMap);
    87→  }
    88→  const current = charMap.get(state) ?? 0;
    89→  const next = current + 1;
    90→  charMap.set(state, next);
    91→  return next;
    92→}

    94→// Helper: resolve a sprite from a pack given behavior config
    95→function resolveSpriteFromPack(
    96→  pack: SpritePackV2,
    97→  behavior: 'principal' | 'random' | 'list',
    98→  principalSpriteId: string | undefined,
    99→  characterId: string,
   100→  state: SpriteState
   101→): { url: string | null; label: string | null } {
   102→  if (pack.sprites.length === 0) return { url: null, label: null };
   103→
   104→  switch (behavior) {
   105→    case 'principal':
   106→      // Use principal sprite if specified, otherwise first sprite
   107→      if (principalSpriteId) {
   108→        const principal = pack.sprites.find(s => s.id === principalSpriteId);
   109→        if (principal) return { url: principal.url, label: principal.label };
   110→      }
   111→      return { url: pack.sprites[0].url, label: pack.sprites[0].label };
   112→
   113→    case 'random':
   114→      // Random selection
   115→      const randomIndex = Math.floor(Math.random() * pack.sprites.length);
   116→      const randomSprite = pack.sprites[randomIndex];
   117→      return { url: randomSprite.url, label: randomSprite.label };
   118→
   119→    case 'list': {
   120→      // Rotate through sprites using the rotation tracker
   121→      const index = getRotationIndex(characterId, state);
   122→      const safeIndex = index % pack.sprites.length;
   123→      const listSprite = pack.sprites[safeIndex];
   124→      return { url: listSprite.url, label: listSprite.label };
   125→    }
   126→
   127→    default:
   128→      return { url: pack.sprites[0].url, label: pack.sprites[0].label };
   129→  }
   130→}

   132→// Get sprite from State Collection V2 (uses sprite packs)
   133→// Now supports conditional mode at the pack level
   134→function getSpriteFromStateCollectionV2(
   135→  state: SpriteState,
   136→  stateCollectionsV2: StateCollectionV2[],
   137→  spritePacksV2: SpritePackV2[],
   138→  characterId: string,
   139→  sessionStats?: SessionStats | null
   140→): { url: string | null; label: string | null } {
   141→  const stateCollection = stateCollectionsV2.find(c => c.state === state);
   142→  if (!stateCollection) return { url: null, label: null };
   143→
   144→  const pack = spritePacksV2.find(p => p.id === stateCollection.packId);
   145→  if (!pack || pack.sprites.length === 0) return { url: null, label: null };
   146→
   147→  // ============================================
   148→  // Evaluate pack-level conditional sprites (if pack has conditionalMode)
   149→  // ============================================
   150→  if (pack.conditionalMode) {
   151→    const winningSprite = evaluatePackConditionalSprites(
   152→      pack.sprites,
   153→      sessionStats,
   154→      characterId
   155→    );
   156→
   157→    if (winningSprite) {
   158→      return { url: winningSprite.url, label: winningSprite.label };
   159→    }
   160→
   161→    // No conditional sprite matched - try defaultSpriteId
   162→    if (pack.defaultSpriteId) {
   163→      const defaultSprite = pack.sprites.find(s => s.id === pack.defaultSpriteId);
   164→      if (defaultSprite) {
   165→        return { url: defaultSprite.url, label: defaultSprite.label };
   166→      }
   167→    }
   168→
   169→    // Also check for sprites marked as isDefault
   170→    const markedDefault = pack.sprites.find(s => s.isDefault);
   171→    if (markedDefault) {
   172→      return { url: markedDefault.url, label: markedDefault.label };
   173→    }
   174→
   175→    // Fall through to behavior-based resolution
   176→  }
   177→
   178→  // ============================================
   179→  // Default: use the state collection's behavior
   180→  // ============================================
   181→  return resolveSpriteFromPack(
   182→    pack,
   183→    stateCollection.behavior,
   184→    stateCollection.principalSpriteId,
   185→    characterId,
   186→    state
   187→  );
   188→}

   190→// Get the appropriate sprite URL based on state and config
   191→// Priority: V2 State Collections > Avatar (legacy system removed)
   192→function getSpriteUrl(
   193→  state: SpriteState,
   194→  avatarUrl?: string,
   195→  character?: CharacterCard,
   196→  characterId?: string,
   197→  sessionStats?: SessionStats | null
   198→): { url: string; label: string | null } {
   199→  const charId = characterId || '';
   200→  // Try V2 state collections (new system)
   201→  // Check for non-empty arrays (empty arrays should not be considered as "configured")
   202→  const hasV2Collections = character?.stateCollectionsV2 && character.stateCollectionsV2.length > 0;
   203→  const hasV2Packs = character?.spritePacksV2 && character.spritePacksV2.length > 0;
   204→  
   205→  if (hasV2Collections && hasV2Packs) {
   206→    const v2Result = getSpriteFromStateCollectionV2(
   207→      state,
   208→      character.stateCollectionsV2,
   209→      character.spritePacksV2,
   210→      charId,
   211→      sessionStats
   212→    );
   213→    if (v2Result.url) {
   214→      return v2Result;
   215→    }
   216→  }
   217→
   218→  // Additional fallback for talk and thinking states - try V2 idle collection
   219→  if (state === 'talk' || state === 'thinking') {
   220→    if (hasV2Collections && hasV2Packs) {
   221→      const v2IdleResult = getSpriteFromStateCollectionV2(
   222→        'idle',
   223→        character!.stateCollectionsV2,
   224→        character!.spritePacksV2,
   225→        charId,
   226→        sessionStats
   227→      );
   228→      if (v2IdleResult.url) {
   229→        return v2IdleResult;
   230→      }
   231→    }
   232→  }
   233→
   234→  // Final fallback to avatar
   235→  if (avatarUrl) {
   236→    return { url: avatarUrl, label: 'avatar' };
   237→  }
   238→
   239→  return { url: '', label: null };
   240→}

   242→// Format milliseconds to readable time
   243→function formatTime(ms: number): string {
   244→  if (ms <= 0) return '0s';
   245→  const seconds = Math.floor(ms / 1000);
   246→  if (seconds < 60) return `${seconds}s`;
   247→  const minutes = Math.floor(seconds / 60);
   248→  const remainingSeconds = seconds % 60;
   249→  return `${minutes}m ${remainingSeconds}s`;
   250→}

   252→export function CharacterSprite({ 
   253→  characterId, 
   254→  characterName, 
   255→  avatarUrl,
   256→  character,
   257→  isStreaming = false,
   258→  hasContent = false,
   259→  isTTSPlaying = false,
   260→  onSettingsChange 
   261→}: CharacterSpriteProps) {
   262→  const [settings, setSettings] = useState<SpriteSettings>(() => {
   263→    if (typeof window !== 'undefined') {
   264→      const saved = localStorage.getItem(`sprite-${characterId}`);
   265→      if (saved) {
   266→        try {
   267→          return { ...DEFAULT_SPRITE_SETTINGS, ...JSON.parse(saved) };
   268→        } catch {
   269→          return DEFAULT_SPRITE_SETTINGS;
   270→        }
   271→      }
   272→    }
   273→    return DEFAULT_SPRITE_SETTINGS;
   274→  });
   275→
   276→  const [isDragging, setIsDragging] = useState(false);
   277→  const [isResizing, setIsResizing] = useState(false);
   278→  const [showControls, setShowControls] = useState(false);
   279→  const [countdown, setCountdown] = useState<number>(0);
   280→  
   281→  const spriteRef = useRef<HTMLDivElement>(null);
   282→  const dragStartRef = useRef({ x: 0, y: 0, settingsX: 0, settingsY: 0 });
   283→  const resizeStartRef = useRef({ x: 0, y: 0, width: 0, height: 0 });
   284→
   285→  // ============================================
   286→  // UNIFIED SYSTEM: Get sprite state from store
   287→  // ============================================
   288→  // 
   289→  // ⚠️ SPRITE PRIORITY SYSTEM - DO NOT MODIFY ⚠️
   290→  // 
   291→  // The sprite system follows a STRICT priority order:
   292→  // 
   293→  // 1. TRIGGER SPRITE (HIGHEST PRIORITY)
   294→  //    - Activated by sprite triggers from AI response
   295→  //    - Once active, CANNOT be overridden by talk/thinking/idle
   296→  //    - Only changes when:
   297→  //      a) Timer expires (returnToIdleMs > 0)
   298→  //      b) Another trigger replaces it
   299→  //      c) User manually clears it
   300→  //
   301→  // 2. STATE COLLECTION V2 SPRITES
   302→  //    - Talk sprite (streaming with content)
   303→  //    - Thinking sprite (generating without content)
   304→  //    - Idle sprite (default state)
   305→  //    - Only shown when NO trigger sprite is active
   306→  //
   307→  // 3. AVATAR (lowest priority)
   308→  //    - Character avatar as final fallback
   309→  //
   310→  // See: /docs/SPRITE_PRIORITY_SYSTEM.md
   311→  //
   312→  // Use proper Zustand selectors to ensure reactivity
   313→  const characterSpriteStates = useTavernStore((state) => state.characterSpriteStates);
   314→  const isLocked = useTavernStore((state) => state.spriteLock.active);
   315→  const lockState = useTavernStore((state) => state.spriteLock);
   316→  const getReturnToIdleCountdownForCharacter = useTavernStore((state) => state.getReturnToIdleCountdownForCharacter);
   317→  const cancelReturnToIdleForCharacter = useTavernStore((state) => state.cancelReturnToIdleForCharacter);
   318→  const executeReturnToIdleForCharacter = useTavernStore((state) => state.executeReturnToIdleForCharacter);
   319→  const clearSpriteLock = useTavernStore((state) => state.clearSpriteLock);
   320→  const setSpriteStateForCharacter = useTavernStore((state) => state.setSpriteStateForCharacter);
   321→
   322→  // Get sessionStats from the active session for conditional sprite evaluation
   323→  const sessionStats = useTavernStore((state) => {
   324→    const activeSessionId = state.activeSessionId;
   325→    if (!activeSessionId) return null;
   326→    const session = state.sessions.find(s => s.id === activeSessionId);
   327→    return session?.sessionStats ?? null;
   328→  });
   329→  
   330→  // Get per-character sprite state from the unified store
   331→  // This is now reactive because characterSpriteStates is a proper selector
   332→  const charSpriteState = characterSpriteStates[characterId] || createDefaultCharacterState();
   333→  
   334→  // Debug logging for trigger sprite state
   335→  useEffect(() => {
   336→    if (charSpriteState.triggerSpriteUrl) {
   337→      console.log('[CharacterSprite] Trigger sprite active:', {
   338→        characterId,
   339→        triggerSpriteUrl: charSpriteState.triggerSpriteUrl,
   340→        triggerSpriteLabel: charSpriteState.triggerSpriteLabel,
   341→        isStreaming,
   342→        hasContent,
   343→      });
   344→    }
   345→  }, [charSpriteState.triggerSpriteUrl, charSpriteState.triggerSpriteLabel, characterId, isStreaming, hasContent]);
   346→  
   347→  // Check if sprite is locked (global state)
   348→  const spriteLockActive = isLocked && lockState.until === 0 ? true : (isLocked && Date.now() < lockState.until);
   349→  
   350→  // Get return to idle state for this character
   351→  const isReturnToIdleScheduled = charSpriteState.returnToIdle.active;
   352→  
   353→  // Determine the effective sprite state:
   354→  // Priority: Trigger sprite > TTS Talk > Streaming Thinking > Store State
   355→  // 
   356→  // When TTS is active:
   357→  //   - During streaming → 'thinking' (not talk, since the character is "thinking" while generating)
   358→  //   - During TTS playback → 'talk' (the character is "speaking")
   359→  //   - After TTS ends → 'idle'
   360→  // 
   361→  // If a trigger sprite is active, it always takes absolute priority.
   362→  const effectiveSpriteState = charSpriteState.triggerSpriteUrl
   363→    ? 'idle' // Trigger active, use trigger sprite (handled below)
   364→    : isTTSPlaying
   365→      ? 'talk'                    // TTS is playing → character is speaking
   366→      : isStreaming
   367→        ? 'thinking'               // Generating → thinking (not talk, even with content)
   368→        : charSpriteState.spriteState;
   369→
   370→  // ============================================
   371→  // LIST MODE ROTATION: Advance index on state transition
   372→  // When effectiveSpriteState transitions to 'talk' or 'thinking',
   373→  // advance the rotation index so the NEXT sprite in the list is shown.
   374→  // ============================================
   375→  const prevEffectiveStateRef = useRef<SpriteState>(effectiveSpriteState);
   376→  useEffect(() => {
   377→    const prev = prevEffectiveStateRef.current;
   378→    prevEffectiveStateRef.current = effectiveSpriteState;
   379→
   380→    // Only advance when transitioning INTO talk or thinking (not from one to the other)
   381→    if (prev !== 'talk' && prev !== 'thinking' && 
   382→        (effectiveSpriteState === 'talk' || effectiveSpriteState === 'thinking')) {
   383→      // Check if the target state collection uses 'list' behavior
   384→      const stateCollections = character?.stateCollectionsV2;
   385→      const collection = stateCollections?.find(c => c.state === effectiveSpriteState);
   386→      if (collection?.behavior === 'list') {
   387→        advanceRotationIndex(characterId, effectiveSpriteState);
   388→      }
   389→    }
   390→  }, [effectiveSpriteState, character?.stateCollectionsV2, characterId]);
   391→
   392→  // ============================================
   393→  // TTS ↔ SPRITE STATE SYNC
   394→  // When TTS playback ends, return the sprite to idle state
   395→  // (only if no trigger is active and not currently streaming)
   396→  // ============================================
   397→  const prevTTSToStateRef = useRef(isTTSPlaying);
   398→  useEffect(() => {
   399→    const wasPlaying = prevTTSToStateRef.current;
   400→    prevTTSToStateRef.current = isTTSPlaying;
   401→
   402→    // TTS just stopped playing → return to idle
   403→    if (wasPlaying && !isTTSPlaying && !isStreaming) {
   404→      const currentState = characterSpriteStates[characterId];
   405→      // Only update if no trigger is active (trigger has absolute priority)
   406→      if (currentState && !currentState.triggerSpriteUrl) {
   407→        setSpriteStateForCharacter(characterId, 'idle');
   408→      }
   409→    }
   410→  }, [isTTSPlaying, isStreaming, characterId, characterSpriteStates, setSpriteStateForCharacter]);
   411→
   412→  // Update countdown for return to idle
   413→  // PERF FIX: Changed from 100ms to 1000ms — the countdown is in seconds, so 1Hz
   414→  // is sufficient and avoids 10 React state updates per second per visible character.
   415→  useEffect(() => {
   416→    if (!isReturnToIdleScheduled) {
   417→      setCountdown(0);
   418→      return;
   419→    }
   420→
   421→    const updateCountdown = () => {
   422→      const remaining = getReturnToIdleCountdownForCharacter(characterId);
   423→      setCountdown(remaining);
   424→    };
   425→
   426→    updateCountdown();
   427→    const interval = setInterval(updateCountdown, 1000);
   428→
   429→    return () => clearInterval(interval);
   430→  }, [isReturnToIdleScheduled, getReturnToIdleCountdownForCharacter, characterId]);
   431→
   432→  // Determine the current sprite URL
   433→  // PRIORITY:
   434→  // 1. Trigger sprite from store (highest priority)
   435→  // 2. State collection V2 (if available)
   436→  // 3. Avatar (lowest priority)
   437→  const spriteResult = charSpriteState.triggerSpriteUrl 
   438→    ? { url: charSpriteState.triggerSpriteUrl, label: charSpriteState.triggerSpriteLabel }
   439→    : getSpriteUrl(
   440→        effectiveSpriteState,
   441→        avatarUrl,
   442→        character,
   443→        characterId,
   444→        sessionStats
   445→      );
   446→  const currentSpriteUrl = spriteResult.url;
   447→  const currentSpriteLabel = spriteResult.label;
   448→
   449→  // FASE 7: Determine the active transition config
   450→  // If a trigger sprite is active, use the trigger's transition config
   451→  // Otherwise, use the character's default transition for state-based changes
   452→  const activeTransition: SpriteTransitionConfig | undefined = charSpriteState.triggerSpriteUrl
   453→    ? (charSpriteState.triggerTransition ?? undefined)
   454→    : (charSpriteState.defaultTransition ?? character?.defaultTransition ?? undefined);
   455→
   456→  // Debug: Log when sprite data changes
   457→  useEffect(() => {
   458→    console.log('[CharacterSprite] Sprite data:', {
   459→      characterId,
   460→      hasCharacter: !!character,
   461→      spritePacksV2Count: character?.spritePacksV2?.length ?? 'undefined',
   462→      stateCollectionsV2Count: character?.stateCollectionsV2?.length ?? 'undefined',
   463→      currentSpriteUrl: currentSpriteUrl?.substring(0, 50),
   464→      avatarUrl: avatarUrl?.substring(0, 50),
   465→    });
   466→  }, [characterId, character?.spritePacksV2, character?.stateCollectionsV2, currentSpriteUrl, avatarUrl]);
   467→
   468→  // Save settings to localStorage
   469→  useEffect(() => {
   470→    localStorage.setItem(`sprite-${characterId}`, JSON.stringify(settings));
   471→    onSettingsChange?.(settings);
   472→  }, [settings, characterId, onSettingsChange]);
   473→
   474→  // Drag handlers - click anywhere on sprite to drag
   475→  const handleDragStart = (e: React.MouseEvent) => {
   476→    // Don't start drag if clicking on controls
   477→    if ((e.target as HTMLElement).closest('.sprite-controls')) return;
   478→    if (isResizing) return;
   479→    
   480→    e.preventDefault();
   481→    e.stopPropagation();
   482→    setIsDragging(true);
   483→    dragStartRef.current = {
   484→      x: e.clientX,
   485→      y: e.clientY,
   486→      settingsX: settings.x,
   487→      settingsY: settings.y
   488→    };
   489→  };
   490→
   491→  useEffect(() => {
   492→    if (!isDragging) return;
   493→
   494→    const handleDragMove = (e: MouseEvent) => {
   495→      const container = spriteRef.current?.parentElement;
   496→      if (!container) return;
   497→
   498→      const rect = container.getBoundingClientRect();
   499→      const deltaX = ((e.clientX - dragStartRef.current.x) / rect.width) * 100;
   500→      const deltaY = ((e.clientY - dragStartRef.current.y) / rect.height) * 100;
   501→
   502→      // Calculate new position
   503→      let newX = dragStartRef.current.settingsX + deltaX;
   504→      // Invert deltaY because we use 'bottom' positioning
   505→      // Dragging up (negative deltaY) should increase Y (move sprite up)
   506→      let newY = dragStartRef.current.settingsY - deltaY;
   507→
   508→      // Constrain X: sprite center must stay within screen
   509→      const halfWidth = settings.width / 2;
   510→      newX = Math.max(halfWidth, Math.min(100 - halfWidth, newX));
   511→
   512→      // Constrain Y: sprite bottom must stay within screen
   513→      // y = 0 means at bottom, higher values move it up
   514→      // Maximum Y should not exceed (100 - height) to keep sprite visible
   515→      const maxY = 100 - settings.height;
   516→      newY = Math.max(0, Math.min(maxY, newY));
   517→
   518→      setSettings(prev => ({ ...prev, x: newX, y: newY }));
   519→    };
   520→
   521→    const handleDragEnd = () => {
   522→      setIsDragging(false);
   523→    };
   524→
   525→    document.addEventListener('mousemove', handleDragMove);
   526→    document.addEventListener('mouseup', handleDragEnd);
   527→
   528→    return () => {
   529→      document.removeEventListener('mousemove', handleDragMove);
   530→      document.removeEventListener('mouseup', handleDragEnd);
   531→    };
   532→  }, [isDragging, settings.width, settings.height]);
   533→
   534→  // Resize handlers
   535→  const handleResizeStart = (e: React.MouseEvent) => {
   536→    e.preventDefault();
   537→    e.stopPropagation();
   538→    setIsResizing(true);
   539→    resizeStartRef.current = {
   540→      x: e.clientX,
   541→      y: e.clientY,
   542→      width: settings.width,
   543→      height: settings.height
   544→    };
   545→  };
   546→
   547→  useEffect(() => {
   548→    if (!isResizing) return;
   549→
   550→    const handleResizeMove = (e: MouseEvent) => {
   551→      const container = spriteRef.current?.parentElement;
   552→      if (!container) return;
   553→
   554→      const rect = container.getBoundingClientRect();
   555→      const deltaX = ((e.clientX - resizeStartRef.current.x) / rect.width) * 100;
   556→      const deltaY = ((e.clientY - resizeStartRef.current.y) / rect.height) * 100;
