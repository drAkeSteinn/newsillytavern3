    98→/**
    99→ * Execute detected tool calls, send SSE events, and return results.
   100→ * Returns { newContent: combined display messages, shouldContinue: true if tools were executed }
   101→ */
   102→async function executeToolCallsAndContinue(
   103→  toolCalls: NativeToolCall[],
   104→  availableTools: Array<{ id: string; name: string; label: string; icon: string }>,
   105→  currentRound: number,
   106→  maxRounds: number,
   107→  character: CharacterCard,
   108→  sessionId: string,
   109→  userName: string,
   110→  controller: { enqueue: (chunk: string) => void },
   111→  sessionQuests?: SessionQuestInstance[],
   112→  questTemplates?: QuestTemplate[],
   113→  statsConfig?: CharacterStatsConfig,
   114→  sessionStats?: SessionStats,
   115→  allCharacters?: CharacterCard[],
   116→  lorebooks?: Lorebook[],
   117→): Promise<{ newContent: string; shouldContinue: boolean; toolResults: Array<{ success: boolean; displayMessage: string }>; questActivations: QuestActivation[]; toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> }> {
   118→  if (toolCalls.length === 0 || currentRound >= maxRounds) {
   119→    return { newContent: '', shouldContinue: false, toolResults: [], questActivations: [], toolsUsed: [] };
   120→  }
   121→
   122→  const toolResults: Array<{ success: boolean; displayMessage: string }> = [];
   123→  const questActivations: QuestActivation[] = [];
   124→  const toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> = [];
   125→  let allDisplayMessages = '';
   126→
   127→  for (let i = 0; i < toolCalls.length; i++) {
   128→    const tc = toolCalls[i];
   129→    const toolDef = availableTools.find(t => t.name === tc.name);
   130→    const callId = `${Date.now()}_${i}`;
   131→
   132→    // Send tool_call_start event
   133→    controller.enqueue(createSSEJSON({
   134→      type: 'tool_call_start',
   135→      toolName: tc.name,
   136→      toolLabel: toolDef?.label || tc.name,
   137→      toolIcon: toolDef?.icon || 'Wrench',
   138→      params: tc.arguments,
   139→      callId,
   140→    }));
   141→
   142→    console.log(`[Tools] Executing tool call: ${tc.name}`, tc.arguments);
   143→
   144→    // Execute the tool
   145→    const toolResult = await executeTool(
   146→      tc.name,
   147→      tc.arguments,
   148→      {
   149→        characterId: character.id,
   150→        characterName: character.name,
   151→        sessionId,
   152→        userName,
   153→        sessionQuests,
   154→        questTemplates,
   155→        statsConfig: character.statsConfig,
   156→        sessionStats,
   157→        allCharacters,
   158→        lorebooks,
   159→        character,  // FASE 12: needed by manage_wardrobe tool
   160→      },
   161→    );
   162→
   163→    // Send tool_call_result event
   164→    controller.enqueue(createSSEJSON({
   165→      type: 'tool_call_result',
   166→      toolName: tc.name,
   167→      success: toolResult.success,
   168→      displayMessage: toolResult.displayMessage,
   169→      duration: toolResult.duration || 0,
   170→      callId,
   171→    }));
   172→
   173→    // Check for quest activation and send SSE event
   174→    if (toolResult.questActivation) {
   175→      const activation = toolResult.questActivation;
   176→      console.log(`[Tools] Quest activation from ${tc.name}:`, activation);
   177→      
   178→      controller.enqueue(createSSEJSON({
   179→        type: 'quest_activation',
   180→        toolName: tc.name,
   181→        activationType: activation.type,
   182→        key: activation.key,
   183→        metadata: activation.metadata,
   184→      }));
   185→      
   186→      questActivations.push(activation);
   187→    }
   188→
   189→    // Check for action/skill activation and send SSE event
   190→    if (toolResult.actionActivation) {
   191→      const action = toolResult.actionActivation;
   192→      console.log(`[Tools] Action activation from ${tc.name}:`, action.skillName);
   193→      
   194→      controller.enqueue(createSSEJSON({
   195→        type: 'action_activation',
   196→        toolName: tc.name,
   197→        skillId: action.skillId,
   198→        skillName: action.skillName,
   199→        skillDescription: action.skillDescription,
   200→        skillCompletedDescription: action.skillCompletedDescription,
   201→        activationCosts: action.activationCosts,
   202→        activationRewards: action.activationRewards,
   203→        characterId: action.characterId,
   204→      }));
   205→    }
   206→
   207→    // Check for relationship activation and send SSE event
   208→    if (toolResult.relationshipActivation) {
   209→      const rel = toolResult.relationshipActivation;
   210→      console.log(`[Tools] Relationship activation from ${tc.name}:`, rel.aName, '↔', rel.bName, rel.prevPoints, '→', rel.newPoints);
   211→
   212→      controller.enqueue(createSSEJSON({
   213→        type: 'relationship_activation',
   214→        toolName: tc.name,
   215→        aId: rel.aId,
   216→        aName: rel.aName,
   217→        bId: rel.bId,
   218→        bName: rel.bName,
   219→        prevPoints: rel.prevPoints,
   220→        newPoints: rel.newPoints,
   221→        reason: rel.reason,
   222→      }));
   223→    }
   224→
   225→    // Check for world-time activation and send SSE event
   226→    if (toolResult.timeActivation) {
   227→      const timeAct = toolResult.timeActivation;
   228→      console.log(`[Tools] Time activation from ${tc.name}:`, timeAct.type, timeAct.minutes || timeAct.hour || timeAct.season);
   229→
   230→      controller.enqueue(createSSEJSON({
   231→        type: 'time_activation',
   232→        toolName: tc.name,
   233→        activationType: timeAct.type,
   234→        minutes: timeAct.minutes,
   235→        hour: timeAct.hour,
   236→        minute: timeAct.minute,
   237→        season: timeAct.season,
   238→      }));
   239→    }
   240→
   241→    // Check for skill check activation and send SSE event
   242→    if (toolResult.checkActivation) {
   243→      const check = toolResult.checkActivation;
   244→      console.log(`[Tools] Skill check from ${tc.name}:`, check.statName, `${check.roll}${check.modifier >= 0 ? '+' : ''}${check.modifier}=${check.total} vs CD ${check.dc} → ${check.outcome}`);
   245→
   246→      controller.enqueue(createSSEJSON({
   247→        type: 'check_activation',
   248→        toolName: tc.name,
   249→        characterId: check.characterId,
   250→        characterName: check.characterName,
   251→        statName: check.statName,
   252→        statValue: check.statValue,
   253→        roll: check.roll,
   254→        modifier: check.modifier,
   255→        dc: check.dc,
   256→        total: check.total,
   257→        outcome: check.outcome,
   258→        outcomeLabel: check.outcomeLabel,
   259→        narrative: check.narrative,
   260→      }));
   261→    }
   262→
   263→    // Check for stat modification from modify_stat tool and send SSE event
   264→    if (toolResult.statActivation) {
   265→      const stat = toolResult.statActivation;
   266→      console.log(`[Tools] Stat activation from ${tc.name}:`, stat.attributeKey, stat.oldValue, '→', stat.newValue);
   267→
   268→      controller.enqueue(createSSEJSON({
   269→        type: 'stat_activation',
   270→        toolName: tc.name,
   271→        characterId: stat.characterId,
   272→        attributeKey: stat.attributeKey,
   273→        attributeName: stat.attributeName,
   274→        attributeType: stat.attributeType,
   275→        oldValue: stat.oldValue,
   276→        newValue: stat.newValue,
   277→        reason: stat.reason,
   278→      }));
   279→
   280→      // Keep the working sessionStats in sync so SUBSEQUENT tool calls read
   281→      // up-to-date values. Without this, every tool round received the same
   282→      // request-start snapshot: two modify_stat("+10") calls both computed
   283→      // base+10 (the second overwrote the first), and a following check_stat
   284→      // reported the pre-modification value, misleading the LLM.
   285→      // The object is per-request (parsed from this request's body), so
   286→      // in-place mutation is safe and visible to the outer multi-round loop,
   287→      // which passes the same reference.
   288→      if (sessionStats) {
   289→        if (!sessionStats.characterStats) sessionStats.characterStats = {};
   290→        if (!sessionStats.characterStats[stat.characterId]) {
   291→          sessionStats.characterStats[stat.characterId] = {
   292→            attributeValues: {},
   293→            lastUpdated: {},
   294→          } as any;
   295→        }
   296→        const charStats = sessionStats.characterStats[stat.characterId];
   297→        charStats.attributeValues = charStats.attributeValues || {};
   298→        charStats.attributeValues[stat.attributeKey] = stat.newValue;
   299→        charStats.lastUpdated = charStats.lastUpdated || {};
   300→        charStats.lastUpdated[stat.attributeKey] = Date.now();
   301→        sessionStats.lastModified = Date.now();
   302→      }
   303→    }
   304→
   305→    // Check for solicitud activation/completion and send SSE event
   306→    if (toolResult.solicitudActivation) {
   307→      const sol = toolResult.solicitudActivation;
   308→      console.log(`[Tools] Solicitud activation from ${tc.name}:`, sol.type, sol.solicitudKey);
   309→      
   310→      controller.enqueue(createSSEJSON({
   311→        type: 'solicitud_activation',
   312→        toolName: tc.name,
   313→        activationType: sol.type,
   314→        solicitudKey: sol.solicitudKey,
   315→        targetCharacterId: sol.targetCharacterId,
   316→        targetCharacterName: sol.targetCharacterName,
   317→        fromCharacterId: sol.fromCharacterId,
   318→        fromCharacterName: sol.fromCharacterName,
   319→        description: sol.description,
   320→        completionDescription: sol.completionDescription,
   321→        peticionKey: sol.peticionKey,
   322→      }));
   323→    }
   324→
   325→
   326→    // Check for wardrobe activation and send SSE event (GUARDARROPA V2)
   327→    if (toolResult.wardrobeActivation) {
   328→      const wa = toolResult.wardrobeActivation;
   329→      console.log(`[Tools] Wardrobe activation from ${tc.name}:`, wa.action, wa.previousOutfitId ?? '(none)', '→', wa.outfitId ?? '(default)', wa.outfitName);
   330→
   331→      controller.enqueue(createSSEJSON({
   332→        type: 'wardrobe_activation',
   333→        toolName: tc.name,
   334→        characterId: wa.characterId,
   335→        action: wa.action,
   336→        outfitId: wa.outfitId,
   337→        outfitName: wa.outfitName,
   338→        outfitDescription: wa.outfitDescription,
   339→        previousOutfitId: wa.previousOutfitId,
   340→        changed: wa.changed,
   341→        reason: wa.reason,
   342→      }));
   343→    }
   344→
   345→    // Check for scenario activation and send SSE event (ESCENARIO V2)
   346→    if (toolResult.scenarioActivation) {
   347→      const sa = toolResult.scenarioActivation;
   348→      console.log(`[Tools] Scenario activation from ${tc.name}:`, sa.action, sa.previousLocationId ?? '(none)', '→', sa.locationId ?? '(default)', sa.locationName);
   349→
   350→      controller.enqueue(createSSEJSON({
   351→        type: 'scenario_activation',
   352→        toolName: tc.name,
   353→        characterId: sa.characterId,
   354→        action: sa.action,
   355→        locationId: sa.locationId,
   356→        locationName: sa.locationName,
   357→        locationDescription: sa.locationDescription,
   358→        previousLocationId: sa.previousLocationId,
   359→        changed: sa.changed,
   360→        reason: sa.reason,
   361→      }));
   362→    }
   363→
   364→    console.log(`[Tools] Tool ${tc.name}: success=${toolResult.success} duration=${toolResult.duration}ms`, toolResult.displayMessage);
   365→
   366→    toolResults.push({ success: toolResult.success, displayMessage: toolResult.displayMessage });
   367→    toolsUsed.push({
   368→      name: tc.name,
   369→      label: toolDef?.label || tc.name,
   370→      icon: toolDef?.icon || 'Wrench',
   371→      success: toolResult.success,
   372→      // Build a meaningful tooltip summary so the user understands what changed
   373→      details: summarizeToolResult(toolResult, tc.arguments),
   374→      displayMessage: toolResult.displayMessage,
   375→    });
   376→    if (toolResult.displayMessage) {
   377→      allDisplayMessages += (allDisplayMessages ? '\n' : '') + toolResult.displayMessage;
   378→    }
   379→  }
   380→
   381→  // Return the display messages as content, tool results, and signal that a follow-up call is needed
   382→  return {
   383→    newContent: allDisplayMessages,
   384→    shouldContinue: true, // Always continue to get the LLM's natural response
   385→    toolResults,
   386→    questActivations,
   387→    toolsUsed,
   388→  };
   389→}

   391→export async function POST(request: NextRequest) {
   392→  try {
   393→    // Capture auth headers from Z.ai gateway for token resolution
   394→    // Only use actual JWT tokens - session IDs are NOT valid X-Tokens
   395→    const incomingXToken = request.headers.get('X-Token');
   396→    const fcSecurityToken = request.headers.get('x-fc-security-token');
   397→
   398→    const body = await request.json();
   399→
   400→    // Validate request (automatically detects request type)
   401→    const validation = validateRequest(null, body);
   402→    if (!validation.success) {
   403→      return createErrorResponse(validation.error, 400);
   404→    }
   405→
   406→    const {
   407→      message,
   408→      character,
   409→      messages = [],
   410→      llmConfig,
   411→      userName = 'User',
   412→      persona,
   413→      sessionStats,
   414→      allCharacters, // All characters for peticiones/solicitudes resolution
   415→    } = validation.data;
   416→
   417→    // Extract lorebooks from body (not validated by validation.ts)
   418→    const lorebooks: Lorebook[] = body.lorebooks || [];
   419→
   420→    // Extract HUD context from body
   421→    const hudContext: HUDContextConfig | undefined = body.hudContext;
   422→
   423→    // Extract Quest data for pre-LLM integration (NEW FORMAT)
   424→    const questTemplates: QuestTemplate[] = body.questTemplates || [];
   425→    const sessionQuests: SessionQuestInstance[] = body.sessionQuests || [];
   426→    const questSettings: QuestSettings = {
   427→      ...DEFAULT_QUEST_SETTINGS,
   428→      ...(body.questSettings || {})
   429→    };
   430→
   431→    // Extract Sound data for {{sonidos}} key resolution
   432→    const soundTriggers: SoundTrigger[] = body.soundTriggers || [];
   433→    const soundSettings = body.settings?.sound;
   434→
   435→    // Extract summary for memory/context compression (single summary from session)
   436→    const summary: SessionSummary | undefined = body.summary;
   437→    
   438→    // Extract embeddings chat settings for automatic context retrieval
   439→    const embeddingsChat: Partial<EmbeddingsChatSettings> = body.embeddingsChat || {};
   440→    const sessionId: string | undefined = body.sessionId;
   441→    const characterId: string | undefined = body.characterId;
   442→    
   443→    // (Memory V2: client-side CharacterMemory mirror removed — memories are
   444→    // managed exclusively in the V2 store.)
   445→    
   446→    // Extract tools settings for tool/action system (native tool calling only)
   447→    const toolsSettings: ToolsSettings = {
   448→      enabled: body.toolsSettings?.enabled ?? true,
   449→      maxToolCallsPerTurn: body.toolsSettings?.maxToolCallsPerTurn ?? 4,
   450→      characterConfigs: body.toolsSettings?.characterConfigs || [],
   451→      usePromptBasedFallback: body.toolsSettings?.usePromptBasedFallback ?? false,
   452→      disabledTools: body.toolsSettings?.disabledTools || [],
   453→    };
   454→
   455→    // Extract inventory data for Inventory V2 system
   456→    const inventoryData: InventoryPromptData | undefined = body.inventoryData;
   457→
   458→    if (!llmConfig) {
   459→      return createErrorResponse('No LLM configuration provided', 400);
   460→    }
   461→
   462→    // Validate API key for providers that require one
   463→    const providersRequiringApiKey = ['openai', 'vllm', 'lm-studio', 'custom', 'anthropic', 'grok'];
   464→    if (providersRequiringApiKey.includes(llmConfig.provider) && !llmConfig.apiKey) {
   465→      return createErrorResponse(
   466→        `API Key is required for ${llmConfig.provider} provider. Please configure it in Settings.`,
   467→        400
   468→      );
   469→    }
   470→
   471→    // Validate endpoint for providers that require one
   472→    const providersRequiringEndpoint = ['openai', 'vllm', 'lm-studio', 'custom', 'anthropic', 'ollama', 'text-generation-webui', 'koboldcpp'];
   473→    if (providersRequiringEndpoint.includes(llmConfig.provider) && !llmConfig.endpoint) {
   474→      return createErrorResponse(
   475→        `Endpoint URL is required for ${llmConfig.provider} provider. Please configure it in Settings.`,
   476→        400
   477→      );
   478→    }
   479→
   480→    // If using Z.ai provider, collect gateway-forwarded auth headers
   481→    // The SDK v0.0.17+ reads config from /etc/.z-ai-config and uses the "token" field for X-Token.
   482→    // Gateway headers are passed as runtime overrides to the provider.
   483→    let zaiRuntimeToken: string | undefined;
   484→    if (llmConfig.provider === 'z-ai') {
   485→      // Only use actual auth tokens (not session IDs)
   486→      const gatewayToken = incomingXToken || fcSecurityToken || undefined;
   487→      if (gatewayToken) {
   488→        zaiRuntimeToken = gatewayToken;
   489→        console.log(`[Stream Route] Z.ai runtime token available (${gatewayToken.length} chars, source: ${incomingXToken ? 'X-Token' : 'fc-security-token'})`);
   490→      } else {
   491→        console.log(`[Stream Route] Z.ai: no gateway token available, using config file only`);
   492→      }
   493→    }
   494→
   495→    // Sanitize user message
   496→    const sanitizedMessage = sanitizeInput(message);
   497→
   498→    // Create default character if none provided
   499→    const effectiveCharacter: CharacterCard = character || DEFAULT_CHARACTER;
   500→
   501→    // Get effective user name from persona or use provided userName
   502→    const effectiveUserName = getEffectiveUserName(persona, userName);
   503→
   504→    // Build context configuration from request or use defaults
   505→    const contextConfig: Partial<ContextConfig> = body.contextConfig || {};
   506→
   507→    // Apply sliding window to messages
   508→    const contextWindow = selectContextMessages(messages, llmConfig, contextConfig);
   509→
   510→    // Log context stats (for debugging)
   511→    const stats = getContextStats(messages);
   512→
   513→    // Process lorebooks and get matched entries
   514→    const { plan: lorebookPlan, lorebookAttributeKeys, lorebookEntryKeyMap, lorebookDebugEntries } = buildLorebookSectionForPrompt(
   515→      messages,
   516→      lorebooks,
   517→      {
   518→        scanDepth: contextConfig.scanDepth,
   519→        userName: effectiveUserName,
   520→        charName: effectiveCharacter?.name,
   521→      },
   522→      { sessionStats, characterId: effectiveCharacter?.id, characters: allCharacters }
   523→    );
   524→
   525→    // ========================================
   526→    // DEBUG: Build lorebook debug data for SSE event
   527→    // ========================================
   528→    const lorebookDebugData = {
   529→      lorebookAttributeKeys,
   530→      debugEntries: lorebookDebugEntries,
   531→      availableStats: sessionStats
   532→        ? Object.fromEntries(
   533→            Object.entries(sessionStats.characterStats || {}).map(([charId, cs]) => [
   534→              charId,
   535→              cs?.attributeValues ? { ...cs.attributeValues } : '(no values)',
   536→            ])
   537→          )
   538→        : '(no sessionStats)',
   539→    };
   540→
   541→    // ========================================
   542→    // Embeddings Context Retrieval
   543→    // ========================================
   544→    // Enrich search query with recent context for better semantic matching
   545→    const searchContextDepth = embeddingsChat.searchContextDepth || 0;
   546→    let enrichedSearchQuery = sanitizedMessage;
   547→    if (searchContextDepth > 0) {
   548→      // Collect recent messages from history for context enrichment
   549→      const recentHistory = messages
   550→        .filter(m => !m.isDeleted)
   551→        .slice(-(searchContextDepth * 2 + 1)) // user+assistant pairs
   552→        .map(m => m.content?.trim())
   553→        .filter(Boolean)
   554→        .slice(0, -1); // exclude current message (already in sanitizedMessage)
   555→      
   556→      if (recentHistory.length > 0) {
   557→        enrichedSearchQuery = recentHistory.join(' ') + ' ' + sanitizedMessage;
   558→      }
   559→    }
   560→
   561→    // Smart truncation: limit query to the embedding model's context window.
   562→    // Prioritize the user's CURRENT message over history when truncating.
   563→    // The Ollama client also handles truncation, but this avoids sending huge payloads.
   564→    try {
