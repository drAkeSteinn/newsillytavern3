   169→/**
   170→ * Clamp a numeric value to the attribute's min/max bounds
   171→ * Returns the clamped value, or the original value if not numeric or no bounds defined
   172→ */
   173→function clampAttributeValue(
   174→  value: number | string,
   175→  attributeDef: AttributeDefinition | undefined
   176→): number | string {
   177→  // Only clamp numeric values
   178→  if (typeof value !== 'number') {
   179→    return value;
   180→  }
   181→
   182→  // If no attribute definition, return as-is
   183→  if (!attributeDef) {
   184→    return value;
   185→  }
   186→
   187→  let clampedValue = value;
   188→
   189→  // Apply min bound - handle both number and string min values
   190→  if (attributeDef.min !== undefined) {
   191→    const minVal = typeof attributeDef.min === 'number'
   192→      ? attributeDef.min
   193→      : parseFloat(String(attributeDef.min));
   194→
   195→    if (!isNaN(minVal)) {
   196→      clampedValue = Math.max(clampedValue, minVal);
   197→    }
   198→  }
   199→
   200→  // Apply max bound - handle both number and string max values
   201→  if (attributeDef.max !== undefined) {
   202→    const maxVal = typeof attributeDef.max === 'number'
   203→      ? attributeDef.max
   204→      : parseFloat(String(attributeDef.max));
   205→
   206→    if (!isNaN(maxVal)) {
   207→      clampedValue = Math.min(clampedValue, maxVal);
   208→    }
   209→  }
   210→
   211→  return clampedValue;
   212→}

   214→/**
   215→ * Create default character stats from config
   216→ */
   217→function createDefaultCharacterStats(
   218→  statsConfig?: CharacterStatsConfig
   219→): CharacterSessionStats {
   220→  const attributeValues: Record<string, number | string> = {};
   221→  const lastUpdated: Record<string, number> = {};
   222→  const now = Date.now();
   223→  
   224→  if (statsConfig?.attributes) {
   225→    for (const attr of statsConfig.attributes) {
   226→      attributeValues[attr.key] = attr.defaultValue;
   227→      lastUpdated[attr.key] = now;
   228→    }
   229→  }
   230→  
   231→  return {
   232→    attributeValues,
   233→    lastUpdated,
   234→    changeLog: [],
   235→  };
   236→}

   238→/**
   239→ * Add entry to change log
   240→ */
   241→function addChangeLogEntry(
   242→  stats: CharacterSessionStats,
   243→  attribute: AttributeDefinition | undefined,
   244→  attributeKey: string,
   245→  oldValue: number | string | undefined,
   246→  newValue: number | string,
   247→  reason: StatChangeLogEntry['reason']
   248→): void {
   249→  if (!stats.changeLog) {
   250→    stats.changeLog = [];
   251→  }
   252→  
   253→  stats.changeLog.push({
   254→    attributeId: attribute?.id || attributeKey,
   255→    attributeKey,
   256→    attributeName: attribute?.name || attributeKey,
   257→    oldValue: oldValue ?? '',
   258→    newValue,
   259→    reason,
   260→    timestamp: Date.now(),
   261→  });
   262→  
   263→  // Keep only last 100 entries
   264→  if (stats.changeLog.length > 100) {
   265→    stats.changeLog = stats.changeLog.slice(-100);
   266→  }
   267→}

   293→

   296→      const stats = createDefaultCharacterStats(char.statsConfig);
   297→      // FASE 5: Initialize emotional state if emotional config is present
   298→      if (char.emotionalConfig?.enabled && char.emotionalConfig.initialState) {
   299→        stats.emotionalState = char.emotionalConfig.initialState;
   300→        stats.emotionalStateLastEval = now;
   301→        stats.emotionalStateTurnCount = 0;
   302→        // Also set as attribute value for {{emocion}} key resolution
   303→        stats.attributeValues['emocion'] = char.emotionalConfig.initialState;
   304→      }
   305→      characterStats[char.id] = stats;
   306→    }
   307→    
   308→    const newSessionStats: SessionStats = {
   309→      characterStats,
   310→      solicitudes: {
   311→        characterSolicitudes: {},
   312→        lastModified: now,
   313→      },
   314→      initialized: true,
   315→      lastModified: now,
   316→      // Initialize timer state
   317→      lastTimerUpdate: now,
   318→      keywordCycleIndex: {},
   319→    };
   320→    
   321→    // Update session with new stats
   322→    set((state: any) => ({
   323→      sessions: state.sessions.map((s: any) =>
   324→        s.id === sessionId 
   325→          ? { ...s, sessionStats: newSessionStats, updatedAt: new Date().toISOString() }
   326→          : s
   327→      ),
   328→    }));
   329→  },
   330→
   331→  updateCharacterStat: (sessionId, characterId, attributeKey, value, reason = 'manual') => {
   332→    // Default result
   333→    const defaultResult: UpdateCharacterStatResult = {
   334→      oldValue: undefined,
   335→      newValue: typeof value === 'number' ? value : value,
   336→      clamped: false,
   337→      thresholdsReached: [],
   338→    };
   339→
   340→    // Read current state
   341→    const state = get();
   342→    const sessions = state.sessions as Array<{ 
   343→      id: string; 
   344→      sessionStats?: SessionStats;
   345→      characterId?: string;
   346→      groupId?: string;
   347→    }>;
   348→    const sessionIndex = sessions.findIndex(s => s.id === sessionId);
   349→    
   350→    if (sessionIndex === -1) return defaultResult;
   351→    
   352→    const session = sessions[sessionIndex];
   353→    let sessionStats = session.sessionStats;
   354→    
   355→    // Auto-initialize sessionStats if missing
   356→    if (!sessionStats) {
   357→      sessionStats = {
   358→        characterStats: {},
   359→        solicitudes: {
   360→          characterSolicitudes: {},
   361→          lastModified: Date.now(),
   362→        },
   363→        ultimo_objetivo_completado: undefined,
   364→        ultima_solicitud_completada: undefined,
   365→        ultima_solicitud_realizada: undefined,
   366→        ultima_accion_realizada: undefined,
   367→        initialized: true,
   368→        lastModified: Date.now(),
   369→      };
   370→    }
   371→    
   372→    // Auto-initialize character stats if missing
   373→    if (!sessionStats.characterStats[characterId]) {
   374→      let statsConfig: CharacterStatsConfig | undefined;
   375→      if (characterId === '__user__') {
   376→        // Look up active persona's statsConfig for __user__
   377→        const activePersonaId = (state as any).activePersonaId;
   378→        const personas: any[] = (state as any).personas || [];
   379→        const activePersona = personas.find((p: any) => p.id === activePersonaId);
   380→        statsConfig = activePersona?.statsConfig;
   381→      } else {
   382→        const character = state.characters.find((c: any) => c.id === characterId);
   383→        statsConfig = character?.statsConfig;
   384→      }
   385→      let newStats = createDefaultCharacterStats(statsConfig);
   386→      // For persona (__user__), also initialize equipment slot values
   387→      if (characterId === '__user__') {
   388→        const equipmentSlots = (state as any).inventorySettings?.equipmentSlots;
   389→        if (equipmentSlots && equipmentSlots.length > 0) {
   390→          for (const slot of equipmentSlots) {
   391→            if (!(slot.key in newStats.attributeValues)) {
   392→              newStats.attributeValues[slot.key] = '';
   393→              newStats.lastUpdated[slot.key] = Date.now();
   394→            }
   395→          }
   396→        }
   397→      }
   398→      sessionStats = {
   399→        ...sessionStats,
   400→        characterStats: {
   401→          ...sessionStats.characterStats,
   402→          [characterId]: newStats,
   403→        },
   404→      };
   405→    }
   406→    
   407→    const stats = sessionStats.characterStats[characterId];
   408→    if (!stats) return defaultResult;
   409→
   410→    const oldValue = stats.attributeValues[attributeKey];
   411→
   412→    // Find attribute definition for logging and clamping
   413→    let attributeDef: AttributeDefinition | undefined;
   414→    if (characterId === '__user__') {
   415→      // Look up active persona's statsConfig for __user__
   416→      const activePersonaId = (state as any).activePersonaId;
   417→      const personas: any[] = (state as any).personas || [];
   418→      const activePersona = personas.find((p: any) => p.id === activePersonaId);
   419→      attributeDef = activePersona?.statsConfig?.attributes?.find(
   420→        (a: AttributeDefinition) => a.key === attributeKey
   421→      );
   422→    } else {
   423→      const character = state.characters.find((c: any) => c.id === characterId);
   424→      attributeDef = character?.statsConfig?.attributes?.find(
   425→        (a: AttributeDefinition) => a.key === attributeKey
   426→      );
   427→    }
   428→
   429→    // Clamp value to min/max bounds
   430→    const clampedValue = clampAttributeValue(value, attributeDef);
   431→    const clamped = clampedValue !== value;
   432→
   433→    // Log if clamping occurred
   434→    if (clamped) {
   435→      console.log(`[StatsSlice] Clamped ${attributeKey}: ${value} → ${clampedValue} (min: ${attributeDef?.min}, max: ${attributeDef?.max})`);
   436→    }
   437→
   438→    // Detect threshold reached
   439→    const thresholdsReached: ThresholdReachedInfo[] = [];
   440→    
   441→    if (attributeDef) {
   442→      // V2: Evaluate new thresholdEffects (flexible conditions with priority)
   443→      if (attributeDef.thresholdEffects && attributeDef.thresholdEffects.length > 0) {
   444→        // Build a temporary session stats with the new value for evaluation
   445→        const tempSessionStats: SessionStats = {
   446→          ...sessionStats,
   447→          characterStats: {
   448→            ...sessionStats.characterStats,
   449→            [characterId]: {
   450→              ...stats,
   451→              attributeValues: {
   452→                ...stats.attributeValues,
   453→                [attributeKey]: clampedValue,
   454→              },
   455→            },
   456→          },
   457→        };
   458→        
   459→        const matchingEffects = evaluateThresholdEffects(
   460→          attributeDef.thresholdEffects,
   461→          tempSessionStats,
   462→          characterId
   463→        );
   464→        
   465→        for (const effect of matchingEffects) {
   466→          thresholdsReached.push({
   467→            attributeKey: attributeDef.key,
   468→            attributeName: attributeDef.name,
   469→            thresholdType: 'custom',
   470→            effectName: effect.name,
   471→            effectId: effect.id,
   472→            priority: effect.priority,
   473→            rewards: effect.rewards,
   474→          });
   475→          console.log(`[StatsSlice] Threshold effect "${effect.name}" triggered for ${attributeDef.name} (priority: ${effect.priority})`);
   476→        }
   477→      }
   478→      
   479→      // Legacy: Check old onMinReached/onMaxReached (only if no thresholdEffects defined)
   480→      if (!attributeDef.thresholdEffects || attributeDef.thresholdEffects.length === 0) {
   481→        if (typeof clampedValue === 'number') {
   482→          // Check if reached minimum
   483→          if (attributeDef.min !== undefined && clampedValue === attributeDef.min) {
   484→            if (attributeDef.onMinReached?.enabled && attributeDef.onMinReached.rewards.length > 0) {
   485→              thresholdsReached.push({
   486→                attributeKey: attributeDef.key,
   487→                attributeName: attributeDef.name,
   488→                thresholdType: 'min',
   489→                thresholdValue: attributeDef.min,
   490→                rewards: attributeDef.onMinReached.rewards,
   491→              });
   492→              console.log(`[StatsSlice] Threshold reached: ${attributeDef.name} hit minimum (${attributeDef.min})`);
   493→            }
   494→          }
   495→          
   496→          // Check if reached maximum
   497→          if (attributeDef.max !== undefined && clampedValue === attributeDef.max) {
   498→            if (attributeDef.onMaxReached?.enabled && attributeDef.onMaxReached.rewards.length > 0) {
   499→              thresholdsReached.push({
   500→                attributeKey: attributeDef.key,
   501→                attributeName: attributeDef.name,
   502→                thresholdType: 'max',
   503→                thresholdValue: attributeDef.max,
   504→                rewards: attributeDef.onMaxReached.rewards,
   505→              });
   506→              console.log(`[StatsSlice] Threshold reached: ${attributeDef.name} hit maximum (${attributeDef.max})`);
   507→            }
   508→          }
   509→        }
   510→      }
   511→    }
   512→
   513→    // Build the updated state
   514→    const updatedCharacterStats = {
   515→      ...sessionStats.characterStats,
   516→      [characterId]: {
   517→        ...stats,
   518→        attributeValues: {
   519→          ...stats.attributeValues,
   520→          [attributeKey]: clampedValue,
   521→        },
   522→        lastUpdated: {
   523→          ...stats.lastUpdated,
   524→          [attributeKey]: Date.now(),
   525→        },
   526→      },
   527→    };
   528→
   529→    // Add to change log
   530→    addChangeLogEntry(
   531→      updatedCharacterStats[characterId],
   532→      attributeDef,
   533→      attributeKey,
   534→      oldValue,
   535→      clampedValue,
   536→      reason
   537→    );
   538→    
   539→    const newSessionStats: SessionStats = {
   540→      ...sessionStats,
   541→      characterStats: updatedCharacterStats,
   542→      lastModified: Date.now(),
   543→    };
   544→    
   545→    // Update the store
   546→    set({
   547→      sessions: state.sessions.map((s: any) =>
   548→        s.id === sessionId
   549→          ? { 
   550→              ...s, 
   551→              sessionStats: newSessionStats,
   552→              updatedAt: new Date().toISOString() 
   553→            }
   554→          : s
   555→      ),
   556→    });
   557→
   558→    // Sync currency changes to persona (Inventory V2)
   559→    // When currency is updated via quest rewards/triggers on __user__,
   560→    // also update persona.currency so the InventoryPanel/HUD stay in sync
   561→    if (characterId === '__user__' && attributeKey === 'currency' && typeof clampedValue === 'number') {
   562→      try {
   563→        const activePersonaId = (get() as any).activePersonaId;
   564→        if (activePersonaId) {
   565→          (get() as any).updatePersona?.(activePersonaId, { currency: clampedValue });
   566→        }
   567→      } catch { /* non-critical sync */ }
   568→    }
   569→
   570→    // Refresh objective visibility since attribute conditions may have changed
   571→    try {
   572→      (get() as any).refreshAllObjectiveVisibility?.(sessionId);
   573→    } catch { /* non-critical */ }
   574→
   575→    // Refresh activation conditions since attribute conditions may have changed
   576→    try {
   577→      (get() as any).refreshAllActivationConditions?.(sessionId);
   578→    } catch { /* non-critical */ }
   579→
   580→    // Return result with threshold info
   581→    return {
   582→      oldValue,
   583→      newValue: clampedValue,
   584→      clamped,
   585→      thresholdsReached,
   586→    };
   587→  },
   588→
   589→  batchUpdateCharacterStats: (sessionId, characterId, updates, reason = 'llm_detection') => {
   590→    set((state: any) => {
   591→      const sessions = state.sessions as Array<{
   592→        id: string; 
   593→        sessionStats?: SessionStats;
   594→      }>;
   595→      const sessionIndex = sessions.findIndex(s => s.id === sessionId);
   596→      
   597→      if (sessionIndex === -1) return state;
   598→      
   599→      const session = sessions[sessionIndex];
   600→      let sessionStats = session.sessionStats;
   601→      
   602→      // Auto-initialize sessionStats if missing (includes event fields reset)
   603→      if (!sessionStats) {
   604→        sessionStats = {
   605→          characterStats: {},
   606→          solicitudes: {
   607→            characterSolicitudes: {},
   608→            lastModified: Date.now(),
   609→          },
   610→          ultimo_objetivo_completado: undefined,
   611→          ultima_solicitud_completada: undefined,
   612→          ultima_solicitud_realizada: undefined,
   613→          ultima_accion_realizada: undefined,
   614→          initialized: true,
   615→          lastModified: Date.now(),
   616→        };
   617→      }
   618→      
   619→      // Auto-initialize character stats if missing
   620→      if (!sessionStats.characterStats[characterId]) {
   621→        let statsConfig: CharacterStatsConfig | undefined;
   622→        if (characterId === '__user__') {
   623→          // Look up active persona's statsConfig for __user__
   624→          const activePersonaId = (state as any).activePersonaId;
   625→          const personas: any[] = (state as any).personas || [];
   626→          const activePersona = personas.find((p: any) => p.id === activePersonaId);
   627→          statsConfig = activePersona?.statsConfig;
   628→        } else {
   629→          const character = state.characters.find((c: any) => c.id === characterId);
   630→          statsConfig = character?.statsConfig;
   631→        }
   632→        let autoStats = createDefaultCharacterStats(statsConfig);
   633→        // For persona (__user__), also initialize equipment slot values
   634→        if (characterId === '__user__') {
   635→          const equipmentSlots = (state as any).inventorySettings?.equipmentSlots;
   636→          if (equipmentSlots && equipmentSlots.length > 0) {
   637→            for (const slot of equipmentSlots) {
   638→              if (!(slot.key in autoStats.attributeValues)) {
   639→                autoStats.attributeValues[slot.key] = '';
   640→                autoStats.lastUpdated[slot.key] = Date.now();
   641→              }
   642→            }
   643→          }
   644→        }
   645→        sessionStats = {
   646→          ...sessionStats,
   647→          characterStats: {
   648→            ...sessionStats.characterStats,
   649→            [characterId]: autoStats,
   650→          },
   651→        };
   652→      }
   653→      
   654→      const stats = sessionStats.characterStats[characterId];
   655→      if (!stats) return state;
   656→      
   657→      // Look up statsConfig for the character (or persona for __user__)
   658→      let statsConfig: CharacterStatsConfig | undefined;
   659→      if (characterId === '__user__') {
   660→        const activePersonaId = (state as any).activePersonaId;
   661→        const personas: any[] = (state as any).personas || [];
   662→        const activePersona = personas.find((p: any) => p.id === activePersonaId);
   663→        statsConfig = activePersona?.statsConfig;
   664→      } else {
   665→        const character = state.characters.find((c: any) => c.id === characterId);
   666→        statsConfig = character?.statsConfig;
   667→      }
   668→      const now = Date.now();
   669→      
   670→      // Apply all updates
   671→      const newAttributeValues = { ...stats.attributeValues };
   672→      const newLastUpdated = { ...stats.lastUpdated };
   673→      const newChangeLog = [...(stats.changeLog || [])];
   674→
   675→      for (const update of updates) {
   676→        const oldValue = newAttributeValues[update.attributeKey];
   677→        const attributeDef = statsConfig?.attributes?.find(
   678→          (a: AttributeDefinition) => a.key === update.attributeKey
   679→        );
   680→
   681→        // Clamp value to min/max bounds
   682→        const clampedValue = clampAttributeValue(update.value, attributeDef);
   683→
   684→        // Log if clamping occurred
   685→        if (clampedValue !== update.value) {
   686→          console.log(`[StatsSlice] Clamped ${update.attributeKey}: ${update.value} → ${clampedValue} (min: ${attributeDef?.min}, max: ${attributeDef?.max})`);
   687→        }
   688→
   689→        newAttributeValues[update.attributeKey] = clampedValue;
   690→        newLastUpdated[update.attributeKey] = now;
   691→
   692→        newChangeLog.push({
   693→          attributeId: attributeDef?.id || update.attributeKey,
   694→          attributeKey: update.attributeKey,
   695→          attributeName: attributeDef?.name || update.attributeKey,
   696→          oldValue: oldValue ?? '',
   697→          newValue: clampedValue,
   698→          reason,
   699→          timestamp: now,
   700→        });
   701→      }
   702→      
   703→      // Keep only last 100 entries
   704→      const trimmedChangeLog = newChangeLog.slice(-100);
   705→      
   706→      const updatedCharacterStats = {
   707→        ...sessionStats.characterStats,
   708→        [characterId]: {
   709→          ...stats,
   710→          attributeValues: newAttributeValues,
   711→          lastUpdated: newLastUpdated,
   712→          changeLog: trimmedChangeLog,
   713→        },
   714→      };
   715→      
   716→      const newSessionStats: SessionStats = {
   717→        ...sessionStats,
   718→        characterStats: updatedCharacterStats,
   719→        lastModified: now,
   720→      };
   721→      
   722→      return {
   723→        sessions: state.sessions.map((s: any) =>
   724→          s.id === sessionId
   725→            ? { 
   726→                ...s, 
   727→                sessionStats: newSessionStats,
   728→                updatedAt: new Date().toISOString() 
   729→              }
   730→            : s
   731→        ),
   732→      };
   733→    });
   734→
   735→    // Refresh objective visibility since attribute conditions may have changed
   736→    try {
   737→      (get() as any).refreshAllObjectiveVisibility?.(sessionId);
   738→    } catch { /* non-critical */ }
   739→
   740→    // Refresh activation conditions since attribute conditions may have changed
   741→    try {
   742→      (get() as any).refreshAllActivationConditions?.(sessionId);
   743→    } catch { /* non-critical */ }
   744→  },

  1720→/**
  1721→ * Evaluate a single requirement against current stats
  1722→ * Supports target requirements (checking attributes of other characters or persona)
  1723→ */
  1724→export function evaluateRequirement(
  1725→  requirement: StatRequirement,
  1726→  attributeValues: Record<string, number | string>,
  1727→  sessionStats?: SessionStats | null
  1728→): boolean {
  1729→  let currentValue: number | string | undefined;
  1730→
  1731→  if (requirement.targetCharacterId) {
  1732→    // Target mode - look up attribute from another character or persona
  1733→    currentValue = sessionStats?.characterStats?.[requirement.targetCharacterId]?.attributeValues?.[requirement.attributeKey];
  1734→  } else {
  1735→    // Self mode - use own attribute values
  1736→    currentValue = attributeValues[requirement.attributeKey];
  1737→  }
  1738→
  1739→  if (currentValue === undefined) return false;
  1740→
  1741→  const currentNum = typeof currentValue === 'number' ? currentValue : parseFloat(currentValue);
  1742→  const valueNum = typeof requirement.value === 'number' ? requirement.value : parseFloat(requirement.value);
  1743→
  1744→  // String comparison for text operators
  1745→  const isTextOperator = requirement.operator === 'contains' || requirement.operator === 'not_contains';
  1746→  if (isTextOperator || isNaN(currentNum) || isNaN(valueNum)) {
  1747→    const currentStr = String(currentValue).toLowerCase();
  1748→    const valueStr = String(requirement.value).toLowerCase();
  1749→
  1750→    switch (requirement.operator) {
  1751→      case '==': return currentStr === valueStr;
  1752→      case '!=': return currentStr !== valueStr;
  1753→      case 'contains': return currentStr.includes(valueStr);
  1754→      case 'not_contains': return !currentStr.includes(valueStr);
  1755→      default: return false;
  1756→    }
  1757→  }
  1758→
  1759→  switch (requirement.operator) {
  1760→    case '<': return currentNum < valueNum;
  1761→    case '<=': return currentNum <= valueNum;
  1762→    case '>': return currentNum > valueNum;
  1763→    case '>=': return currentNum >= valueNum;
  1764→    case '==': return currentNum === valueNum;
  1765→    case '!=': return currentNum !== valueNum;
  1766→    case 'between': {
  1767→      const maxNum = typeof requirement.valueMax === 'number'
  1768→        ? requirement.valueMax
  1769→        : parseFloat(requirement.valueMax?.toString() || '0');
  1770→      return currentNum >= valueNum && currentNum <= maxNum;
  1771→    }
  1772→    default: return false;
  1773→  }
  1774→}
  1775→
  1776→/**
  1777→ * Evaluate all requirements (supports AND and OR logic)
  1778→ * Default is AND logic for backward compatibility
  1779→ */
  1780→export function evaluateRequirements(
  1781→  requirements: StatRequirement[],
  1782→  attributeValues: Record<string, number | string>,
  1783→  sessionStats?: SessionStats | null,
  1784→  operator?: 'AND' | 'OR'
  1785→): boolean {
  1786→  if (!requirements || requirements.length === 0) return true;
  1787→  const logicFn = operator === 'OR' ? requirements.some.bind(requirements) : requirements.every.bind(requirements);
  1788→  return logicFn(req => evaluateRequirement(req, attributeValues, sessionStats));
  1789→}
