     1→// ============================================
     2→// Use Timeline Sprite Sounds Hook
     3→// ============================================
     4→//
     5→// This hook connects sprite display (idle, trigger, talk, thinking)
     6→// with timeline sound AND haptic playback.
     7→//
     8→// When a sprite is displayed in the chat scene, it:
     9→// 1. Extracts the collection name from the sprite URL
    10→// 2. Loads the metadata.json from that collection
    11→// 3. Finds the sprite's timeline configuration
    12→// 4. Plays sounds at keyframe times (via requestAnimationFrame loop)
    13→// 5. Plays haptic patterns via HSP (Handy Server Pattern)
    14→//
    15→// HSP vs HDSP approach:
    16→// - HSP preloads ALL points into the device buffer before playing
    17→// - The device handles timing, interpolation, and looping natively
    18→// - This eliminates network latency, loop wraparound jumps, and erratic velocity
    19→// - Sound tracks still use requestAnimationFrame for precise keyframe triggering
    20→//
    21→// Supports: idle sprites, trigger sprites, WEBP/GIF/WebM
    22→// ============================================

    75→// ============================================
    76→// Audio Cache
    77→// ============================================
    78→// PERF FIX: Added LRU eviction to prevent unbounded memory growth.
    79→// Previously, every unique sound URL created a new HTMLAudioElement that was
    80→// cached forever (~1-5MB per decoded buffer). Now we keep at most MAX_CACHE_SIZE
    81→// entries and evict the oldest when the cache is full.
    82→
    83→const MAX_AUDIO_CACHE_SIZE = 20;
    84→const audioCache = new Map<string, HTMLAudioElement>();
    85→
    86→function getAudio(url: string): HTMLAudioElement {
    87→  if (audioCache.has(url)) {
    88→    // Move to end (most recently used) by re-inserting
    89→    const audio = audioCache.get(url)!;
    90→    audioCache.delete(url);
    91→    audioCache.set(url, audio);
    92→    return audio;
    93→  }
    94→
    95→  // Evict oldest entry if cache is full (Map preserves insertion order, so first = oldest)
    96→  if (audioCache.size >= MAX_AUDIO_CACHE_SIZE) {
    97→    const oldestKey = audioCache.keys().next().value;
    98→    if (oldestKey) {
    99→      const oldAudio = audioCache.get(oldestKey);
   100→      // Release the audio resource
   101→      if (oldAudio) {
   102→        oldAudio.pause();
   103→        oldAudio.src = '';
   104→      }
   105→      audioCache.delete(oldestKey);
   106→    }
   107→  }
   108→
   109→  const audio = new Audio(url);
   110→  audio.load();
   111→  audioCache.set(url, audio);
   112→  return audioCache.get(url)!;
   113→}

   115→/** Clear the entire audio cache — call on session change to free memory */
   116→export function clearAudioCache() {
   117→  for (const [, audio] of audioCache) {
   118→    audio.pause();
   119→    audio.src = '';
   120→  }
   121→  audioCache.clear();
   122→}

   191→// HSP Pattern Playback (via REST API proxy)
   192→// ============================================
   193→//
   194→// These functions handle HSP pattern playback directly via the
   195→// /api/handy/hsp/* endpoints. They don't use the React hook
   196→// because they're called from non-React contexts (global functions).
   197→//
   198→// Flow:
   199→//   1. Get server time for sync
   200→//   2. Set device mode to HSP (mode 4)
   201→//   3. Set slider stroke to full range
   202→//   4. HSP setup (assign stream ID)
   203→//   5. HSP add (send points in batches)
   204→//   6. HSP play (with server_time, loop flag)
   205→//   7. On stop: HSP stop → HSP flush → center device
   206→
   207→let serverTimeOffset: number = 0;
   208→
   209→/**
   210→ * NTP-style server time sync with multiple samples and outlier removal.
   211→ * Matches the approach in use-haptic-playback.ts.
   212→ */
   213→const TIMELINE_SYNC_SAMPLES = 8;
   214→const TIMELINE_SYNC_OUTLIERS = 2;
   215→
   216→async function syncServerTime(): Promise<void> {
   217→  const samples: Array<{ offset: number; rtd: number }> = [];
   218→
   219→  for (let i = 0; i < TIMELINE_SYNC_SAMPLES; i++) {
   220→    try {
   221→      const startLocal = Date.now();
   222→      const response = await fetch('/api/handy/servertime', {
   223→        method: 'GET',
   224→        signal: AbortSignal.timeout(5000),
   225→      });
   226→      const endLocal = Date.now();
   227→      const data = await response.json();
   228→      // Handy API v3 /servertime returns: { result: { server_time: number } }
   229→      const serverTime = data?.result?.server_time ?? data?.result ?? data;
   230→
   231→      if (typeof serverTime === 'number' && serverTime > 0) {
   232→        const rtd = endLocal - startLocal;
   233→        const estimatedLocalAtServer = startLocal + rtd / 2;
   234→        const offset = serverTime - estimatedLocalAtServer;
   235→        samples.push({ offset, rtd });
   236→      }
   237→    } catch {
   238→      // Skip failed samples
   239→    }
   240→  }
   241→
   242→  if (samples.length > 0) {
   243→    // Sort by round-trip delay and remove worst outliers
   244→    samples.sort((a, b) => a.rtd - b.rtd);
   245→    const trimmed = samples.slice(0, Math.max(1, samples.length - TIMELINE_SYNC_OUTLIERS));
   246→
   247→    let offsetAccum = 0;
   248→    for (const s of trimmed) {
   249→      offsetAccum += s.offset;
   250→    }
   251→    serverTimeOffset = offsetAccum / trimmed.length;
   252→  }
   253→  // Keep existing offset if all samples failed
   254→}
   255→
   256→function estimateServerTime(): number {
   257→  // Per Handy API v3 reference: estimate = Math.round(Date.now() + offset)
   258→  return Math.round(Date.now() + serverTimeOffset);
   259→}

   298→/**
   299→ * Start HSP pattern playback for a timeline's haptic tracks.
   300→ * Converts keyframes to HSP points, loads them into the device, and starts playback.
   301→ */
   302→async function startHspPatternPlayback(
   303→  timeline: SpriteTimelineData,
   304→): Promise<boolean> {
   305→  const config = readHandyConfig();
   306→  if (!config || !readHapticEnabled()) return false;
   307→
   308→  const { appId, connectionKey } = config;
   309→  const inverted = readInverted();
   310→
   311→  // Find haptic tracks
   312→  const hapticTracks = timeline.tracks.filter(
   313→    (t) => t.type === 'haptic' && !t.muted && t.enabled && t.keyframes.length > 0
   314→  );
   315→
   316→  if (hapticTracks.length === 0) return false;
   317→
   318→  // Convert haptic keyframes to HSP points
   319→  // For multiple haptic tracks, merge them (use the first active track)
   320→  // In practice, there's usually only one haptic track per timeline
   321→  const track = hapticTracks[0];
   322→  const keyframes = track.keyframes.map((kf) => ({
   323→    time: kf.time,
   324→    value: kf.value as HapticKeyframeValue,
   325→    interpolation: kf.interpolation,
   326→  }));
   327→
   328→  const points = generateHspPattern(keyframes, timeline.duration || 5000, timeline.loop);
   329→
   330→  if (points.length === 0) return false;
   331→
   332→  // Apply inversion to all points
   333→  const adjustedPoints = inverted
   334→    ? points.map((p) => ({ t: p.t, x: 100 - p.x }))
   335→    : points;
   336→
   337→  console.log('[Timeline-HSP] 🎮 Starting HSP pattern playback:', {
   338→    points: adjustedPoints.length,
   339→    duration: timeline.duration,
   340→    loop: timeline.loop,
   341→    firstPoint: adjustedPoints[0],
   342→    lastPoint: adjustedPoints[adjustedPoints.length - 1],
   343→  });
   344→
   345→  try {
   346→    // 1. Sync server time (multiple samples with outlier removal)
   347→    await syncServerTime();
   348→
   349→    // 2. Set device mode to HSP (mode 4)
   350→    const modeSet = await setHandyMode(4);
   351→    if (!modeSet) {
   352→      console.warn('[Timeline-HSP] ⚠️ Failed to switch to HSP mode');
   353→      return false;
   354→    }
   355→
   356→    // 3. Set slider stroke to full range
   357→    await setSliderStroke(0, 1.0);
   358→
   359→    // 4. HSP setup
   360→    const streamId = Math.floor(Math.random() * 1024);
   361→    const setupResponse = await fetch('/api/handy/hsp/setup', {
   362→      method: 'PUT',
   363→      headers: { 'Content-Type': 'application/json' },
   364→      body: JSON.stringify({ appId, connectionKey, stream_id: streamId }),
   365→      signal: AbortSignal.timeout(5000),
   366→    });
   367→
   368→    if (!setupResponse.ok) {
   369→      console.error('[Timeline-HSP] ❌ HSP setup failed:', setupResponse.status);
   370→      return false;
   371→    }
   372→
   373→    // 5. HSP add - send initial batch
   374→    const initialBatch = adjustedPoints.slice(0, 10);
   375→    const addResponse = await fetch('/api/handy/hsp/add', {
   376→      method: 'PUT',
   377→      headers: { 'Content-Type': 'application/json' },
   378→      body: JSON.stringify({
   379→        appId,
   380→        connectionKey,
   381→        flush: false,
   382→        points: initialBatch,
   383→        tail_point_stream_index: initialBatch.length - 1,
   384→      }),
   385→      signal: AbortSignal.timeout(5000),
   386→    });
   387→
   388→    if (!addResponse.ok) {
   389→      console.error('[Timeline-HSP] ❌ HSP add (initial) failed');
   390→      return false;
   391→    }
   392→
   393→    // 6. HSP play - use current estimated server time directly
   394→    // (matching the reference implementation approach)
   395→    const playServerTime = estimateServerTime();
   396→    const playResponse = await fetch('/api/handy/hsp/play', {
   397→      method: 'PUT',
   398→      headers: { 'Content-Type': 'application/json' },
   399→      body: JSON.stringify({
   400→        appId,
   401→        connectionKey,
   402→        server_time: playServerTime,
   403→        start_time: 0,
   404→        loop: timeline.loop,
   405→        playback_rate: 1,
   406→      }),
   407→      signal: AbortSignal.timeout(5000),
   408→    });
   409→
   410→    if (!playResponse.ok) {
   411→      console.error('[Timeline-HSP] ❌ HSP play failed');
   412→      return false;
   413→    }
   414→
   415→    // 7. Send remaining points in batches
   416→    if (adjustedPoints.length > 10) {
   417→      const remaining = adjustedPoints.slice(10);
   418→      const batchCount = Math.ceil(remaining.length / 100);
   419→      let sentCount = 10;
   420→
   421→      for (let i = 0; i < batchCount; i++) {
   422→        const batchPoints = remaining.slice(i * 100, (i + 1) * 100);
   423→        sentCount += batchPoints.length;
   424→
   425→        try {
   426→          await fetch('/api/handy/hsp/add', {
   427→            method: 'PUT',
   428→            headers: { 'Content-Type': 'application/json' },
   429→            body: JSON.stringify({
   430→              appId,
   431→              connectionKey,
   432→              flush: false,
   433→              points: batchPoints,
   434→              tail_point_stream_index: sentCount - 1,
   435→            }),
   436→            signal: AbortSignal.timeout(5000),
   437→          });
   438→        } catch {
   439→          console.warn(`[Timeline-HSP] ⚠️ Error sending batch ${i + 1}/${batchCount}`);
   440→        }
   441→      }
   442→    }
   443→
   444→    console.log('[Timeline-HSP] ✅ HSP pattern playback started successfully');
   445→    return true;
   446→
   447→  } catch (err) {
   448→    console.error('[Timeline-HSP] ❌ Error:', err);
   449→    return false;
   450→  }
   451→}

   453→/**
   454→ * Stop HSP pattern playback and return device to center.
   455→ */
   456→async function stopHspPatternPlayback(): Promise<void> {
   457→  const config = readHandyConfig();
   458→  if (!config) return;
   459→
   460→  const { appId, connectionKey } = config;
   461→  const inverted = readInverted();
   462→
   463→  try {
   464→    // 1. Stop HSP
   465→    await fetch('/api/handy/hsp/stop', {
   466→      method: 'PUT',
   467→      headers: { 'Content-Type': 'application/json' },
   468→      body: JSON.stringify({ appId, connectionKey }),
   469→      signal: AbortSignal.timeout(5000),
   470→    }).catch(() => {});
   471→
   472→    // 2. Flush buffer
   473→    await fetch('/api/handy/hsp/flush', {
   474→      method: 'PUT',
   475→      headers: { 'Content-Type': 'application/json' },
   476→      body: JSON.stringify({ appId, connectionKey }),
   477→      signal: AbortSignal.timeout(5000),
   478→    }).catch(() => {});
   479→
   480→    // 3. Switch to HDSP mode and center device
   481→    await setHandyMode(2);
   482→
   483→    const centerPos = inverted ? 0.5 : 0.5;
   484→    await fetch('/api/handy/hdsp/xpvp', {
   485→      method: 'PUT',
   486→      headers: { 'Content-Type': 'application/json' },
   487→      body: JSON.stringify({
   488→        appId,
   489→        connectionKey,
   490→        xp: centerPos,
   491→        vp: 0.3,
   492→        stop_on_target: true,
   493→        immediate_rsp: true,
   494→      }),
   495→    }).catch(() => {});
   496→
   497→    // 4. Stop HAMP
   498→    setTimeout(() => {
   499→      fetch('/api/handy/hamp/stop', {
   500→        method: 'PUT',
   501→        headers: { 'Content-Type': 'application/json' },
   502→        body: JSON.stringify({ appId, connectionKey }),
   503→      }).catch(() => {});
   504→    }, 300);
   505→
   506→  } catch (err) {
   507→    console.error('[Timeline-HSP] ❌ Error stopping:', err);
   508→  }
   509→}

   549→
   550→async function playSoundFromTrigger(
   551→  trigger: SoundTrigger,
   552→  collections: SoundCollection[],
   553→  volume: number = 1,
   554→  characterId?: string
   555→): Promise<HTMLAudioElement | null> {
   556→  if (isGlobalMuted()) return null;
   557→  const collection = collections.find(c => c.name === trigger.collection);
   558→  if (!collection || !collection.files || collection.files.length === 0) return null;
   559→
   560→  let soundFile: string;
   561→  if (trigger.playMode === 'random') {
   562→    soundFile = collection.files[Math.floor(Math.random() * collection.files.length)];
   563→  } else {
   564→    const index = trigger.currentIndex || 0;
   565→    soundFile = collection.files[index % collection.files.length];
   566→  }
   567→
   568→  try {
   569→    const baseAudio = getAudio(soundFile);
   570→    const audioClone = baseAudio.cloneNode() as HTMLAudioElement;
   571→    audioClone.volume = volume * (trigger.volume || 1);
   572→    audioClone.currentTime = 0;
   573→    await audioClone.play().catch(() => {});
   574→
   575→    // Emit comic sound visual event for timeline trigger sounds
   576→    emitComicSoundEvent(trigger.name, trigger.keywords[0] || 'timeline_sound', characterId);
   577→
   578→    return audioClone;
   579→  } catch {
   580→    return null;
   581→  }
   582→}

   607→// ============================================
   608→// Timeline Sound Player
   609→// ============================================
   610→
   611→function playSoundsAtTime(
   612→  timeline: SpriteTimelineData,
   613→  currentTime: number,
   614→  soundTriggers: SoundTrigger[],
   615→  soundCollections: SoundCollection[],
   616→  active: ActiveTimeline,
   617→  toleranceMs: number = 150
   618→): void {
   619→  if (isGlobalMuted()) return;
   620→  const globalVolume = timeline.globalVolume ?? 1;
   621→
   622→  for (const track of timeline.tracks) {
   623→    if (track.muted || !track.enabled) continue;
   624→
   625→    // Only process sound tracks
   626→    const isSoundTrack = track.type === 'sound' || (
   627→      track.type === 'sprite' && track.keyframes.some(kf => {
   628→        const val = kf.value as unknown as Record<string, unknown>;
   629→        return val?.soundTriggerId || val?.play;
   630→      })
   631→    );
   632→
   633→    if (!isSoundTrack) continue;
   634→
   635→    const trackId = track.id;
   636→
   637→    for (const keyframe of track.keyframes) {
   638→      const keyframeId = keyframe.id;
   639→      const keyframeTime = keyframe.time;
   640→
   641→      const isCrossing = currentTime >= keyframeTime && currentTime < keyframeTime + toleranceMs;
   642→
   643→      if (isCrossing && !active.triggeredKeyframes.has(keyframeId)) {
   644→        active.triggeredKeyframes.add(keyframeId);
   645→
   646→        const soundValue = keyframe.value as {
   647→          soundTriggerId?: string;
   648→          soundTriggerName?: string;
   649→          soundUrl?: string;
   650→          volume?: number;
   651→          play?: boolean;
   652→          stop?: boolean;
   653→        };
   654→
   655→        if (soundValue.play) {
   656→          (async () => {
   657→            let audioEl: HTMLAudioElement | null = null;
   658→
   659→            if (soundValue.soundTriggerId) {
   660→              let trigger = soundTriggers.find(t => t.id === soundValue.soundTriggerId);
   661→
   662→              if (!trigger && soundValue.soundTriggerName) {
   663→                trigger = soundTriggers.find(t =>
   664→                  t.name.toLowerCase() === soundValue.soundTriggerName!.toLowerCase()
   665→                );
   666→              }
   667→
   668→              if (trigger) {
   669→                audioEl = await playSoundFromTrigger(
   670→                  trigger,
   671→                  soundCollections,
   672→                  (soundValue.volume || 1) * globalVolume,
   673→                  active.characterId
   674→                );
   675→              }
   676→            }
   677→            else if (soundValue.soundUrl) {
   678→              audioEl = await playSoundFromUrl(
   679→                soundValue.soundUrl,
   680→                (soundValue.volume || 1) * globalVolume,
   681→                active.characterId
   682→              );
   683→            }
   684→
   685→            if (audioEl) {
   686→              const trackAudios = active.activeAudios.get(trackId) || [];
   687→              trackAudios.push(audioEl);
   688→              active.activeAudios.set(trackId, trackAudios);
   689→              audioEl.onended = () => {
   690→                const idx = trackAudios.indexOf(audioEl!);
   691→                if (idx > -1) trackAudios.splice(idx, 1);
   692→              };
   693→            }
   694→          })();
   695→        }
   696→
   697→        if (soundValue.stop) {
   698→          const trackAudios = active.activeAudios.get(trackId) || [];
   699→          for (const audio of trackAudios) {
   700→            audio.pause();
   701→            audio.remove();
   702→          }
   703→          trackAudios.length = 0;
   704→        }
   705→      }
   706→
   707→      // Reset trigger for keyframes we've passed (for looping)
   708→      if (currentTime < keyframeTime - toleranceMs) {
   709→        active.triggeredKeyframes.delete(keyframeId);
   710→      }
   711→    }
   712→  }
   713→}
   714→
   715→// ============================================
   716→// Loop Checker - Only for Sound Tracks
   717→// ============================================
   718→// HSP handles haptic playback natively - no need for requestAnimationFrame
   719→// haptic updates. This loop only triggers sound keyframes.
   720→
   721→function startLoopChecker() {
   722→  if (loopCheckerRunning) return;
   723→
   724→  loopCheckerRunning = true;
   725→
   726→  const check = () => {
   727→    if (activeTimelines.size === 0) {
   728→      loopCheckerRunning = false;
   729→      loopAnimationId = null;
   730→      return;
   731→    }
   732→
   733→    // If globally muted, keep the loop running but skip processing
   734→    if (isGlobalMuted()) {
   735→      loopAnimationId = requestAnimationFrame(check);
   736→      return;
   737→    }
   738→
   739→    const now = Date.now();
   740→
   741→    for (const [, active] of activeTimelines) {
   742→      // Skip paused timelines
   743→      if (active.pauseElapsed !== null) continue;
   744→
   745→      // Only process sound tracks via the loop checker
   746→      const hasSoundTracks = active.timelineData.tracks.some(t =>
   747→        !t.muted && t.enabled && (t.type === 'sound' || (
   748→          t.type === 'sprite' && t.keyframes.some(kf => {
   749→            const val = kf.value as unknown as Record<string, unknown>;
   750→            return val?.soundTriggerId || val?.play;
   751→          })
   752→        ))
   753→      );
   754→
   755→      if (hasSoundTracks) {
   756→        const elapsed = now - active.startTime;
   757→        const currentTime = elapsed % active.duration;
   758→        playSoundsAtTime(
   759→          active.timelineData,
   760→          currentTime,
   761→          active.soundTriggers,
   762→          active.soundCollections,
   763→          active
   764→        );
   765→      }
   766→    }
   767→
   768→    loopAnimationId = requestAnimationFrame(check);
   769→  };
   770→
   771→  loopAnimationId = requestAnimationFrame(check);
   772→}

  1037→function startTimeline(
  1038→  characterId: string,
  1039→  spriteUrl: string,
  1040→  timeline: SpriteTimelineData,
  1041→  soundTriggers: SoundTrigger[],
  1042→  soundCollections: SoundCollection[]
  1043→): void {
  1044→  // Stop any existing timeline for this character
  1045→  stopTimeline(characterId);
  1046→
  1047→  // Check if there are any playable tracks
  1048→  const hasSoundTracks = timeline.tracks.some(t => !t.muted && t.enabled && (
  1049→    t.type === 'sound' || (
  1050→      t.type === 'sprite' && t.keyframes.some(kf => {
  1051→        const val = kf.value as unknown as Record<string, unknown>;
  1052→        return val?.soundTriggerId || val?.play;
  1053→      })
  1054→    )
  1055→  ));
  1056→
  1057→  const hasHapticTracks = readHapticEnabled() && timeline.tracks.some(
  1058→    t => t.type === 'haptic' && !t.muted && t.enabled && t.keyframes.length > 0
  1059→  );
  1060→
  1061→  if (!hasSoundTracks && !hasHapticTracks) return;
  1062→
  1063→  const active: ActiveTimeline = {
  1064→    spriteUrl,
  1065→    startTime: Date.now(),
  1066→    duration: timeline.duration || 5000,
  1067→    loop: timeline.loop,
  1068→    activeAudios: new Map(),
  1069→    triggeredKeyframes: new Set(),
  1070→    timelineData: timeline,
  1071→    soundTriggers,
  1072→    soundCollections,
  1073→    characterId,
  1074→    hspPlaying: false,
  1075→    pauseElapsed: null,
  1076→  };
  1077→
  1078→  // If globally muted, start in paused state (will be resumed when unmuted)
  1079→  if (isGlobalMuted()) {
  1080→    active.pauseElapsed = 0; // Paused at the beginning
  1081→    activeTimelines.set(characterId, active);
  1082→
  1083→    console.log('[Timeline] ⏸ Started (paused, global mute active) for', characterId.substring(0, 8), {
  1084→      duration: timeline.duration,
  1085→      sounds: hasSoundTracks,
  1086→      haptic: hasHapticTracks,
  1087→      url: spriteUrl.split('/').pop(),
  1088→    });
  1089→    return;
  1090→  }
  1091→
  1092→  activeTimelines.set(characterId, active);
  1093→
  1094→  console.log('[Timeline] ▶️ Started for', characterId.substring(0, 8), {
  1095→    duration: timeline.duration,
  1096→    sounds: hasSoundTracks,
  1097→    haptic: hasHapticTracks,
  1098→    url: spriteUrl.split('/').pop(),
  1099→  });
  1100→
  1101→  // Start the loop checker for sound tracks
  1102→  if (hasSoundTracks) {
  1103→    startLoopChecker();
  1104→
  1105→    // Play initial sounds at time 0
  1106→    playSoundsAtTime(timeline, 0, soundTriggers, soundCollections, active);
  1107→  }
  1108→
  1109→  // Start HSP pattern playback for haptic tracks
  1110→  // Use async IIFE to properly await stop before starting new session
  1111→  if (hasHapticTracks) {
  1112→    (async () => {
  1113→      // Ensure any previous HSP playback is fully stopped before starting new one
  1114→      // This prevents race conditions where two HSP sessions run simultaneously
  1115→      await stopHspPatternPlayback().catch(() => {});
  1116→
  1117→      const success = await startHspPatternPlayback(timeline);
  1118→      if (success) {
  1119→        const tl = activeTimelines.get(characterId);
  1120→        if (tl) tl.hspPlaying = true;
  1121→        console.log('[Timeline] 🎮 HSP pattern playback started');
  1122→      } else {
  1123→        console.warn('[Timeline] ⚠️ HSP pattern playback failed to start');
  1124→      }
  1125→    })();
  1126→  }
  1127→}

  1133→function computeIdleSpriteUrl(
  1134→  spriteState: string,
  1135→  character: CharacterCard | undefined,
  1136→  characterId: string,
  1137→): string | null {
  1138→  if (!character) return null;
  1139→
  1140→  const hasV2Collections = !!(character.stateCollectionsV2 && character.stateCollectionsV2.length > 0);
  1141→  const hasV2Packs = !!(character.spritePacksV2 && character.spritePacksV2.length > 0);
  1142→
  1143→  if (!hasV2Collections || !hasV2Packs) return null;
  1144→
  1145→  const stateCollection = character.stateCollectionsV2!.find(
  1146→    (c: StateCollectionV2) => c.state === spriteState
  1147→  );
  1148→  if (!stateCollection) return null;
  1149→
  1150→  const pack = character.spritePacksV2!.find(
  1151→    (p: SpritePackV2) => p.id === stateCollection.packId
  1152→  );
  1153→  if (!pack || pack.sprites.length === 0) return null;
  1154→
  1155→  switch (stateCollection.behavior) {
  1156→    case 'principal': {
  1157→      if (stateCollection.principalSpriteId) {
  1158→        const principal = pack.sprites.find(s => s.id === stateCollection.principalSpriteId);
  1159→        if (principal) return principal.url;
  1160→      }
  1161→      return pack.sprites[0]?.url || null;
  1162→    }
  1163→    case 'random':
  1164→    case 'list': {
  1165→      return pack.sprites[0]?.url || null;
  1166→    }
  1167→    default:
  1168→      return pack.sprites[0]?.url || null;
  1169→  }
  1170→}

  1176→export function useTimelineSpriteSounds() {
  1177→  const characterSpriteStates = useTavernStore((state) => state.characterSpriteStates);
  1178→  const soundTriggers = useTavernStore((state) => state.soundTriggers ?? []);
  1179→  const soundCollections = useTavernStore((state) => state.soundCollections ?? []);
  1180→  const characters = useTavernStore((state) => state.characters ?? []);
  1181→
  1182→  const characterMap = useRef<Map<string, CharacterCard>>(new Map());
  1183→  useEffect(() => {
  1184→    characterMap.current = new Map(characters.map(c => [c.id, c]));
  1185→  }, [characters]);
  1186→
  1187→  const prevSpriteUrlsRef = useRef<Record<string, string>>({});
  1188→
  1189→  // Start/stop timeline when sprite changes
  1190→  useEffect(() => {
  1191→    const currentSpriteUrls: Record<string, string> = {};
  1192→
  1193→    for (const [characterId, charState] of Object.entries(characterSpriteStates)) {
  1194→      let effectiveUrl = charState.triggerSpriteUrl || '';
  1195→
  1196→      if (!effectiveUrl) {
  1197→        const character = characterMap.current.get(characterId);
  1198→        effectiveUrl = computeIdleSpriteUrl(
  1199→          charState.spriteState,
  1200→          character,
  1201→          characterId
  1202→        ) || '';
  1203→      }
  1204→
  1205→      currentSpriteUrls[characterId] = effectiveUrl;
  1206→
  1207→      const prevUrl = prevSpriteUrlsRef.current[characterId];
  1208→
  1209→      if (effectiveUrl && effectiveUrl !== prevUrl) {
  1210→        if (charState.triggerSpriteUrl && !charState.useTimelineSounds) {
  1211→          continue;
  1212→        }
  1213→
  1214→        const collectionName = extractCollectionFromUrl(effectiveUrl);
  1215→        const filename = extractFilenameFromUrl(effectiveUrl);
  1216→
  1217→        if (!collectionName || !filename) continue;
  1218→
  1219→        (async () => {
  1220→          const metadata = await loadCollectionMetadata(collectionName);
  1221→          if (!metadata) return;
  1222→
  1223→          const spriteMeta = metadata.sprites[filename];
  1224→          if (!spriteMeta?.timeline?.tracks) return;
  1225→
  1226→          startTimeline(
  1227→            characterId,
  1228→            effectiveUrl,
  1229→            spriteMeta.timeline,
  1230→            soundTriggers,
  1231→            soundCollections
  1232→          );
  1233→        })();
  1234→      }
  1235→
  1236→      if (!effectiveUrl && prevUrl) {
  1237→        stopTimeline(characterId);
  1238→      }
  1239→    }
  1240→
  1241→    prevSpriteUrlsRef.current = currentSpriteUrls;
  1242→  }, [characterSpriteStates, soundTriggers, soundCollections]);
  1243→
  1244→  // Cleanup on unmount
