   169→   169→/**
   170→   170→ * Clamp a numeric value to the attribute's min/max bounds
   171→   171→ * Returns the clamped value, or the original value if not numeric or no bounds defined
   172→   172→ */
   173→   173→function clampAttributeValue(
   174→   174→  value: number | string,
   175→   175→  attributeDef: AttributeDefinition | undefined
   176→   176→): number | string {
   177→   177→  // Only clamp numeric values
   178→   178→  if (typeof value !== 'number') {
   179→   179→    return value;
   180→   180→  }
   181→   181→
   182→   182→  // If no attribute definition, return as-is
   183→   183→  if (!attributeDef) {
   184→   184→    return value;
   185→   185→  }
   186→   186→
   187→   187→  let clampedValue = value;
   188→   188→
   189→   189→  // Apply min bound - handle both number and string min values
   190→   190→  if (attributeDef.min !== undefined) {
   191→   191→    const minVal = typeof attributeDef.min === 'number'
   192→   192→      ? attributeDef.min
   193→   193→      : parseFloat(String(attributeDef.min));
   194→   194→
   195→   195→    if (!isNaN(minVal)) {
   196→   196→      clampedValue = Math.max(clampedValue, minVal);
   197→   197→    }
   198→   198→  }
   199→   199→
   200→   200→  // Apply max bound - handle both number and string max values
   201→   201→  if (attributeDef.max !== undefined) {
   202→   202→    const maxVal = typeof attributeDef.max === 'number'
   203→   203→      ? attributeDef.max
   204→   204→      : parseFloat(String(attributeDef.max));
   205→   205→
   206→   206→    if (!isNaN(maxVal)) {
   207→   207→      clampedValue = Math.min(clampedValue, maxVal);
   208→   208→    }
   209→   209→  }
   210→   210→
   211→   211→  return clampedValue;
   212→   212→}

   214→   214→/**
   215→   215→ * Create default character stats from config
   216→   216→ */
   217→   217→function createDefaultCharacterStats(
   218→   218→  statsConfig?: CharacterStatsConfig
   219→   219→): CharacterSessionStats {
   220→   220→  const attributeValues: Record<string, number | string> = {};
   221→   221→  const lastUpdated: Record<string, number> = {};
   222→   222→  const now = Date.now();
   223→   223→  
   224→   224→  if (statsConfig?.attributes) {
   225→   225→    for (const attr of statsConfig.attributes) {
   226→   226→      attributeValues[attr.key] = attr.defaultValue;
   227→   227→      lastUpdated[attr.key] = now;
   228→   228→    }
   229→   229→  }
   230→   230→  
   231→   231→  return {
   232→   232→    attributeValues,
   233→   233→    lastUpdated,
   234→   234→    changeLog: [],
   235→   235→  };
   236→   236→}

   239→   239→ * Add entry to change log
   240→   240→ */
   241→   241→function addChangeLogEntry(
   242→   242→  stats: CharacterSessionStats,
   243→   243→  attribute: AttributeDefinition | undefined,
   244→   244→  attributeKey: string,
   245→   245→  oldValue: number | string | undefined,
   246→   246→  newValue: number | string,
   247→   247→  reason: StatChangeLogEntry['reason']
   248→   248→): void {
   249→   249→  if (!stats.changeLog) {
   250→   250→    stats.changeLog = [];
   251→   251→  }
   252→   252→  
   253→   253→  stats.changeLog.push({
   254→   254→    attributeId: attribute?.id || attributeKey,
   255→   255→    attributeKey,
   256→   256→    attributeName: attribute?.name || attributeKey,
   257→   257→    oldValue: oldValue ?? '',
   258→   258→    newValue,
   259→   259→    reason,
   260→   260→    timestamp: Date.now(),
   261→   261→  });
   262→   262→  
   263→   263→  // Keep only last 100 entries
   264→   264→  if (stats.changeLog.length > 100) {
   265→   265→    stats.changeLog = stats.changeLog.slice(-100);
   266→   266→  }
   267→   267→}

   273→   273→export const createStatsSlice = (set: any, get: any): StatsSlice => ({
   274→   274→  // Initial State
   275→   275→  sessionStats: null,
   276→   276→
   277→   277→  // ============================================
   278→   278→  // Session Stats Actions

   281→   281→  initializeSessionStats: (sessionId, characters) => {
   282→   282→    const state = get();
   283→   283→    const sessions = state.sessions as Array<{ id: string; sessionStats?: SessionStats }>;
   284→   284→    const session = sessions.find(s => s.id === sessionId);
   285→   285→    
   286→   286→    if (!session) return;
   287→   287→    
   288→   288→    // Check if already initialized
   289→   289→    if (session.sessionStats?.initialized) return;
   290→   290→    
   291→   291→    const now = Date.now();
   292→   292→    const characterStats: Record<string, CharacterSessionStats> = {};
   293→   293→    
   294→   294→    // Initialize stats for each character
   295→   295→    for (const char of characters) {
   296→   296→      const stats = createDefaultCharacterStats(char.statsConfig);
   297→   297→      // FASE 5: Initialize emotional state if emotional config is present
   298→   298→      if (char.emotionalConfig?.enabled && char.emotionalConfig.initialState) {
   299→   299→        stats.emotionalState = char.emotionalConfig.initialState;
   300→   300→        stats.emotionalStateLastEval = now;
   301→   301→        stats.emotionalStateTurnCount = 0;
   302→   302→        // Also set as attribute value for {{emocion}} key resolution
   303→   303→        stats.attributeValues['emocion'] = char.emotionalConfig.initialState;
   304→   304→      }
   305→   305→      characterStats[char.id] = stats;
   306→   306→    }
   307→   307→    
   308→   308→    const newSessionStats: SessionStats = {
   309→   309→      characterStats,
   310→   310→      solicitudes: {
   311→   311→        characterSolicitudes: {},
   312→   312→        lastModified: now,
   313→   313→      },
   314→   314→      initialized: true,
   315→   315→      lastModified: now,
   316→   316→      // Initialize timer state
   317→   317→      lastTimerUpdate: now,
   318→   318→      keywordCycleIndex: {},
   319→   319→    };
   320→   320→    
   321→   321→    // Update session with new stats
   322→   322→    set((state: any) => ({
   323→   323→      sessions: state.sessions.map((s: any) =>
   324→   324→        s.id === sessionId 
   325→   325→          ? { ...s, sessionStats: newSessionStats, updatedAt: new Date().toISOString() }
   326→   326→          : s
   327→   327→      ),
   328→   328→    }));
   329→   329→  },
   330→   330→
   331→   331→  updateCharacterStat: (sessionId, characterId, attributeKey, value, reason = 'manual') => {
   332→   332→    // Default result
   333→   333→    const defaultResult: UpdateCharacterStatResult = {
   334→   334→      oldValue: undefined,
   335→   335→      newValue: typeof value === 'number' ? value : value,
   336→   336→      clamped: false,
   337→   337→      thresholdsReached: [],
   338→   338→    };
   339→   339→
   340→   340→    // Read current state
   341→   341→    const state = get();
   342→   342→    const sessions = state.sessions as Array<{ 
   343→   343→      id: string; 
   344→   344→      sessionStats?: SessionStats;
   345→   345→      characterId?: string;
   346→   346→      groupId?: string;
   347→   347→    }>;
   348→   348→    const sessionIndex = sessions.findIndex(s => s.id === sessionId);
   349→   349→    
   350→   350→    if (sessionIndex === -1) return defaultResult;
   351→   351→    
   352→   352→    const session = sessions[sessionIndex];
   353→   353→    let sessionStats = session.sessionStats;
   354→   354→    
   355→   355→    // Auto-initialize sessionStats if missing
   356→   356→    if (!sessionStats) {
   357→   357→      sessionStats = {
   358→   358→        characterStats: {},
   359→   359→        solicitudes: {
   360→   360→          characterSolicitudes: {},
   361→   361→          lastModified: Date.now(),
   362→   362→        },
   363→   363→        ultimo_objetivo_completado: undefined,
   364→   364→        ultima_solicitud_completada: undefined,
   365→   365→        ultima_solicitud_realizada: undefined,
   366→   366→        ultima_accion_realizada: undefined,
   367→   367→        initialized: true,
   368→   368→        lastModified: Date.now(),
   369→   369→      };
   370→   370→    }
   371→   371→    
   372→   372→    // Auto-initialize character stats if missing
   373→   373→    if (!sessionStats.characterStats[characterId]) {
   374→   374→      let statsConfig: CharacterStatsConfig | undefined;
   375→   375→      if (characterId === '__user__') {
   376→   376→        // Look up active persona's statsConfig for __user__
   377→   377→        const activePersonaId = (state as any).activePersonaId;
   378→   378→        const personas: any[] = (state as any).personas || [];
   379→   379→        const activePersona = personas.find((p: any) => p.id === activePersonaId);
   380→   380→        statsConfig = activePersona?.statsConfig;
   381→   381→      } else {
   382→   382→        const character = state.characters.find((c: any) => c.id === characterId);
   383→   383→        statsConfig = character?.statsConfig;
   384→   384→      }
   385→   385→      let newStats = createDefaultCharacterStats(statsConfig);
   386→   386→      // For persona (__user__), also initialize equipment slot values
   387→   387→      if (characterId === '__user__') {
   388→   388→        const equipmentSlots = (state as any).inventorySettings?.equipmentSlots;
   389→   389→        if (equipmentSlots && equipmentSlots.length > 0) {
   390→   390→          for (const slot of equipmentSlots) {
   391→   391→            if (!(slot.key in newStats.attributeValues)) {
   392→   392→              newStats.attributeValues[slot.key] = '';
   393→   393→              newStats.lastUpdated[slot.key] = Date.now();
   394→   394→            }
   395→   395→          }
   396→   396→        }
   397→   397→      }
   398→   398→      sessionStats = {
   399→   399→        ...sessionStats,
   400→   400→        characterStats: {
   401→   401→          ...sessionStats.characterStats,
   402→   402→          [characterId]: newStats,
   403→   403→        },
   404→   404→      };
   405→   405→    }
   406→   406→    
   407→   407→    const stats = sessionStats.characterStats[characterId];
   408→   408→    if (!stats) return defaultResult;
   409→   409→
   410→   410→    const oldValue = stats.attributeValues[attributeKey];
   411→   411→
   412→   412→    // Find attribute definition for logging and clamping
   413→   413→    let attributeDef: AttributeDefinition | undefined;
   414→   414→    if (characterId === '__user__') {
   415→   415→      // Look up active persona's statsConfig for __user__
   416→   416→      const activePersonaId = (state as any).activePersonaId;
   417→   417→      const personas: any[] = (state as any).personas || [];
   418→   418→      const activePersona = personas.find((p: any) => p.id === activePersonaId);
   419→   419→      attributeDef = activePersona?.statsConfig?.attributes?.find(
   420→   420→        (a: AttributeDefinition) => a.key === attributeKey
   421→   421→      );
   422→   422→    } else {
   423→   423→      const character = state.characters.find((c: any) => c.id === characterId);
   424→   424→      attributeDef = character?.statsConfig?.attributes?.find(
   425→   425→        (a: AttributeDefinition) => a.key === attributeKey
   426→   426→      );
   427→   427→    }
   428→   428→
   429→   429→    // Clamp value to min/max bounds
   430→   430→    const clampedValue = clampAttributeValue(value, attributeDef);
   431→   431→    const clamped = clampedValue !== value;
   432→   432→
   433→   433→    // Log if clamping occurred
   434→   434→    if (clamped) {
   435→   435→      console.log(`[StatsSlice] Clamped ${attributeKey}: ${value} → ${clampedValue} (min: ${attributeDef?.min}, max: ${attributeDef?.max})`);
   436→   436→    }
   437→   437→
   438→   438→    // Detect threshold reached
   439→   439→    const thresholdsReached: ThresholdReachedInfo[] = [];
   440→   440→    
   441→   441→    if (attributeDef) {
   442→   442→      // V2: Evaluate new thresholdEffects (flexible conditions with priority)
   443→   443→      if (attributeDef.thresholdEffects && attributeDef.thresholdEffects.length > 0) {
   444→   444→        // Build a temporary session stats with the new value for evaluation
   445→   445→        const tempSessionStats: SessionStats = {
   446→   446→          ...sessionStats,
   447→   447→          characterStats: {
   448→   448→            ...sessionStats.characterStats,
   449→   449→            [characterId]: {
   450→   450→              ...stats,
   451→   451→              attributeValues: {
   452→   452→                ...stats.attributeValues,
   453→   453→                [attributeKey]: clampedValue,
   454→   454→              },
   455→   455→            },
   456→   456→          },
   457→   457→        };
   458→   458→        
   459→   459→        const matchingEffects = evaluateThresholdEffects(
   460→   460→          attributeDef.thresholdEffects,
   461→   461→          tempSessionStats,
   462→   462→          characterId
   463→   463→        );
   464→   464→        
   465→   465→        for (const effect of matchingEffects) {
   466→   466→          thresholdsReached.push({
   467→   467→            attributeKey: attributeDef.key,
   468→   468→            attributeName: attributeDef.name,
   469→   469→            thresholdType: 'custom',
   470→   470→            effectName: effect.name,
   471→   471→            effectId: effect.id,
   472→   472→            priority: effect.priority,
   473→   473→            rewards: effect.rewards,
   474→   474→          });
   475→   475→          console.log(`[StatsSlice] Threshold effect "${effect.name}" triggered for ${attributeDef.name} (priority: ${effect.priority})`);
   476→   476→        }
   477→   477→      }
   478→   478→      
   479→   479→      // Legacy: Check old onMinReached/onMaxReached (only if no thresholdEffects defined)
   480→   480→      if (!attributeDef.thresholdEffects || attributeDef.thresholdEffects.length === 0) {
   481→   481→        if (typeof clampedValue === 'number') {
   482→   482→          // Check if reached minimum
   483→   483→          if (attributeDef.min !== undefined && clampedValue === attributeDef.min) {
   484→   484→            if (attributeDef.onMinReached?.enabled && attributeDef.onMinReached.rewards.length > 0) {
   485→   485→              thresholdsReached.push({
   486→   486→                attributeKey: attributeDef.key,
   487→   487→                attributeName: attributeDef.name,
   488→   488→                thresholdType: 'min',
   489→   489→                thresholdValue: attributeDef.min,
   490→   490→                rewards: attributeDef.onMinReached.rewards,
   491→   491→              });
   492→   492→              console.log(`[StatsSlice] Threshold reached: ${attributeDef.name} hit minimum (${attributeDef.min})`);
   493→   493→            }
   494→   494→          }
   495→   495→          
   496→   496→          // Check if reached maximum
   497→   497→          if (attributeDef.max !== undefined && clampedValue === attributeDef.max) {
   498→   498→            if (attributeDef.onMaxReached?.enabled && attributeDef.onMaxReached.rewards.length > 0) {
   499→   499→              thresholdsReached.push({
   500→   500→                attributeKey: attributeDef.key,
   501→   501→                attributeName: attributeDef.name,
   502→   502→                thresholdType: 'max',
   503→   503→                thresholdValue: attributeDef.max,
   504→   504→                rewards: attributeDef.onMaxReached.rewards,
   505→   505→              });
   506→   506→              console.log(`[StatsSlice] Threshold reached: ${attributeDef.name} hit maximum (${attributeDef.max})`);
   507→   507→            }
   508→   508→          }
   509→   509→        }
   510→   510→      }
   511→   511→    }
   512→   512→
   513→   513→    // Build the updated state
   514→   514→    const updatedCharacterStats = {
   515→   515→      ...sessionStats.characterStats,
   516→   516→      [characterId]: {
   517→   517→        ...stats,
   518→   518→        attributeValues: {
   519→   519→          ...stats.attributeValues,

  1720→  1720→/**
  1721→  1721→ * Evaluate a single requirement against current stats
  1722→  1722→ * Supports target requirements (checking attributes of other characters or persona)
  1723→  1723→ */
  1724→  1724→export function evaluateRequirement(
  1725→  1725→  requirement: StatRequirement,
  1726→  1726→  attributeValues: Record<string, number | string>,
  1727→  1727→  sessionStats?: SessionStats | null
  1728→  1728→): boolean {
  1729→  1729→  let currentValue: number | string | undefined;
  1730→  1730→
  1731→  1731→  if (requirement.targetCharacterId) {
  1732→  1732→    // Target mode - look up attribute from another character or persona
  1733→  1733→    currentValue = sessionStats?.characterStats?.[requirement.targetCharacterId]?.attributeValues?.[requirement.attributeKey];
  1734→  1734→  } else {
  1735→  1735→    // Self mode - use own attribute values
  1736→  1736→    currentValue = attributeValues[requirement.attributeKey];
  1737→  1737→  }
  1738→  1738→
  1739→  1739→  if (currentValue === undefined) return false;
  1740→  1740→
  1741→  1741→  const currentNum = typeof currentValue === 'number' ? currentValue : parseFloat(currentValue);
  1742→  1742→  const valueNum = typeof requirement.value === 'number' ? requirement.value : parseFloat(requirement.value);
  1743→  1743→
  1744→  1744→  // String comparison for text operators
  1745→  1745→  const isTextOperator = requirement.operator === 'contains' || requirement.operator === 'not_contains';
  1746→  1746→  if (isTextOperator || isNaN(currentNum) || isNaN(valueNum)) {
  1747→  1747→    const currentStr = String(currentValue).toLowerCase();
  1748→  1748→    const valueStr = String(requirement.value).toLowerCase();
  1749→  1749→
  1750→  1750→    switch (requirement.operator) {
  1751→  1751→      case '==': return currentStr === valueStr;
  1752→  1752→      case '!=': return currentStr !== valueStr;
  1753→  1753→      case 'contains': return currentStr.includes(valueStr);
  1754→  1754→      case 'not_contains': return !currentStr.includes(valueStr);
  1755→  1755→      default: return false;
  1756→  1756→    }
  1757→  1757→  }
  1758→  1758→
  1759→  1759→  switch (requirement.operator) {
  1760→  1760→    case '<': return currentNum < valueNum;
  1761→  1761→    case '<=': return currentNum <= valueNum;
  1762→  1762→    case '>': return currentNum > valueNum;
  1763→  1763→    case '>=': return currentNum >= valueNum;
  1764→  1764→    case '==': return currentNum === valueNum;
  1765→  1765→    case '!=': return currentNum !== valueNum;
  1766→  1766→    case 'between': {
  1767→  1767→      const maxNum = typeof requirement.valueMax === 'number'
  1768→  1768→        ? requirement.valueMax
  1769→  1769→        : parseFloat(requirement.valueMax?.toString() || '0');
  1770→  1770→      return currentNum >= valueNum && currentNum <= maxNum;
  1771→  1771→    }
  1772→  1772→    default: return false;
  1773→  1773→  }
  1774→  1774→}

  1776→  1776→/**
  1777→  1777→ * Evaluate all requirements (supports AND and OR logic)
  1778→  1778→ * Default is AND logic for backward compatibility
  1779→  1779→ */
  1780→  1780→export function evaluateRequirements(
  1781→  1781→  requirements: StatRequirement[],
  1782→  1782→  attributeValues: Record<string, number | string>,
  1783→  1783→  sessionStats?: SessionStats | null,
  1784→  1784→  operator?: 'AND' | 'OR'
  1785→  1785→): boolean {
  1786→  1786→  if (!requirements || requirements.length === 0) return true;
  1787→  1787→  const logicFn = operator === 'OR' ? requirements.some.bind(requirements) : requirements.every.bind(requirements);
  1788→  1788→  return logicFn(req => evaluateRequirement(req, attributeValues, sessionStats));
  1789→  1789→}
