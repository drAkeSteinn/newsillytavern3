   144→const ALL_TYPES: V2Type[] = ['evento', 'hecho', 'preferencia', 'relacion', 'nota', 'resumen_escena'];
   145→
   146→function fmtDate(iso: string): string {
   147→  if (!iso) return '';
   148→  const d = new Date(iso);
   149→  if (isNaN(d.getTime())) return iso.slice(0, 10);
   150→  return d.toLocaleDateString('es-ES', { day: '2-digit', month: '2-digit', year: 'numeric' });
   151→}

   156→  /** Personaje propietario de los recuerdos. Opcional en modo grupo. */
   157→  charId?: string;

   166→  refreshSignal?: number;
   167→  /** Notifica el nº de registros activos (para badges externos) */

   171→
   172→interface RecordFormState {
   173→  id: string;              // '' = crear
   174→  type: V2Type;
   175→  content: string;
   176→  importance: number;
   177→  eventDate: string;       // YYYY-MM-DD
   178→  subject: string;
   179→}
   180→
   181→const EMPTY_FORM: RecordFormState = {
   182→  id: '',
   183→  type: 'evento',
   184→  content: '',
   185→  importance: 0.6,
   186→  eventDate: '',
   187→  subject: 'pareja',
   188→};
   189→
   190→export function MemoryV2Manager({
   191→  charId,
   192→  groupId,
   193→  charName,
   194→  userName = 'el usuario',
   195→  compact = false,
   196→  refreshSignal,
   197→  onCountChange,
   198→  className,
   199→}: MemoryV2ManagerProps) {
   200→  const { toast } = useToast();
   201→  const isGroupScope = !charId && !!groupId;
   202→
   203→  const [records, setRecords] = useState<V2Record[]>([]);
   204→  const [health, setHealth] = useState<V2HealthData | null>(null);
   205→  const [loading, setLoading] = useState(false);
   206→  const [initialized, setInitialized] = useState(false);
   207→
   208→  // Filtros
   209→  const [typeFilter, setTypeFilter] = useState<'todos' | V2Type>('todos');
   210→  const [sourceFilter, setSourceFilter] = useState<'todas' | 'curada' | 'auto'>('todas');
   211→  const [showArchived, setShowArchived] = useState(false);
   212→  const [expandedTypes, setExpandedTypes] = useState<Record<string, boolean>>({});
   213→
   214→  // Diálogos
   215→  const [formOpen, setFormOpen] = useState(false);
   216→  const [form, setForm] = useState<RecordFormState>(EMPTY_FORM);
   217→  const [saving, setSaving] = useState(false);
   218→  const [pendingHardDelete, setPendingHardDelete] = useState<V2Record | null>(null);
   219→
   220→  // Búsqueda debug
   221→  const [searchQuery, setSearchQuery] = useState('');
   222→  const [searchResults, setSearchResults] = useState<any[] | null>(null);
   223→  const [searching, setSearching] = useState(false);
   224→
   225→  // ============ Data ============
   226→
   227→  const load = useCallback(async () => {
   228→    if (!charId && !groupId) return;
   229→    const scopeQuery = charId
   230→      ? `charId=${encodeURIComponent(charId)}`
   231→      : `groupId=${encodeURIComponent(groupId!)}`;
   232→    setLoading(true);
   233→    try {
   234→      const [recRes, healthRes] = await Promise.all([
   235→        fetch(`/api/memory/v2?action=records&${scopeQuery}&limit=300&includeArchived=${showArchived ? 1 : 0}`),
   236→        fetch('/api/memory/v2?action=health'),
   237→      ]);
   238→      const recData = await recRes.json();
   239→      const healthData = await healthRes.json();
   240→      if (recData.success) {
   241→        setRecords(recData.records || []);
   242→        onCountChange?.(recData.total ?? recData.records?.length ?? 0);
   243→      }
   244→      if (healthData.success) setHealth(healthData.health);
   245→    } catch {
   246→      // silencioso — el health mostrará el error
   247→    } finally {
   248→      setLoading(false);
   249→      setInitialized(true);
   250→    }
   251→  }, [charId, groupId, showArchived, onCountChange]);
   252→
   253→  useEffect(() => { load(); }, [load]);
   254→
   255→  // Reload only when the parent bumps the signal (not on every load identity change)
   256→  const loadRef = useRef(load);
   257→  loadRef.current = load;
   258→  useEffect(() => {
   259→    if (refreshSignal !== undefined && refreshSignal > 0) loadRef.current();
   260→  }, [refreshSignal]);
   261→
   262→  // ============ Filtros (cliente) ============
   263→
   264→  const filtered = useMemo(() => {
   265→    return records.filter(r => {
   266→      if (typeFilter !== 'todos' && r.type !== typeFilter) return false;
   267→      if (sourceFilter !== 'todas' && r.source !== sourceFilter) return false;
   268→      return true;
   269→    });
   270→  }, [records, typeFilter, sourceFilter]);
   271→
   272→  const grouped = useMemo(() => {
   273→    const g: Record<string, V2Record[]> = {};
   274→    for (const r of filtered) (g[r.type] ||= []).push(r);
   275→    // ordenar dentro de cada grupo: curada primero, luego fecha desc
   276→    for (const t of Object.keys(g)) {
   277→      g[t].sort((a, b) => {
   278→        if (a.source !== b.source) return a.source === 'curada' ? -1 : 1;
   279→        return (b.eventDate || b.createdAt || '').localeCompare(a.eventDate || a.createdAt || '');
   280→      });
   281→    }
   282→    return g;
   283→  }, [filtered]);
   284→
   285→  const activeCount = records.filter(r => !r.supersededBy).length;
   286→  const archivedCount = records.length - activeCount;
   287→
   288→  // ============ Acciones ============
   289→
   290→  const openCreate = () => {
   291→    setForm({ ...EMPTY_FORM, eventDate: new Date().toISOString().slice(0, 10) });
   292→    setFormOpen(true);
   293→  };
   294→
   295→  const openEdit = (r: V2Record) => {
   296→    setForm({
   297→      id: r.id,
   298→      type: r.type,
   299→      content: r.content,
   300→      importance: r.importance,
   301→      eventDate: (r.eventDate || '').slice(0, 10),
   302→      subject: r.subject || 'pareja',
   303→    });
   304→    setFormOpen(true);
   305→  };
   306→
   307→  const saveForm = async () => {
   308→    if (!form.content.trim()) {
   309→      toast({ title: 'Falta el contenido', description: 'Escribe el texto del recuerdo.', variant: 'destructive' });
   310→      return;
   311→    }
   312→    setSaving(true);
   313→    try {
   314→      const url = '/api/memory/v2';
   315→      const payload = form.id
   316→        ? { action: 'edit', id: form.id, content: form.content, type: form.type, importance: form.importance, eventDate: form.eventDate, subject: form.subject }
   317→        : isGroupScope
   318→          ? { action: 'add', groupId, type: form.type, content: form.content, importance: form.importance, eventDate: form.eventDate, subject: form.subject }
   319→          : { action: 'add', charId, type: form.type, content: form.content, importance: form.importance, eventDate: form.eventDate, subject: form.subject };
   320→      const res = await fetch(url, {
   321→        method: 'POST',
   322→        headers: { 'Content-Type': 'application/json' },
   323→        body: JSON.stringify(payload),
   324→      });
   325→      const data = await res.json();
   326→      if (!data.success) throw new Error(data.error || 'Error al guardar');
   327→      toast({
   328→        title: form.id ? 'Recuerdo actualizado' : 'Recuerdo creado',
   329→        description: form.id
   330→          ? 'El vector se regeneró con el nuevo texto.'
   331→          : 'Se guardó como recuerdo fijado: SIEMPRE entrará en el prompt.',
   332→      });
   333→      setFormOpen(false);
   334→      load();
   335→    } catch (err: any) {
   336→      toast({ title: 'Error al guardar', description: err?.message || String(err), variant: 'destructive' });
   337→    } finally {
   338→      setSaving(false);
   339→    }
   340→  };
   341→
   342→  const archiveRecord = async (r: V2Record) => {
   343→    try {
   344→      const res = await fetch('/api/memory/v2', {
   345→        method: 'POST',
   346→        headers: { 'Content-Type': 'application/json' },
   347→        body: JSON.stringify({ action: 'supersede', id: r.id }),
   348→      });
   349→      const data = await res.json();
   350→      if (!data.success) throw new Error(data.error || 'Error');
   351→      toast({ title: 'Recuerdo archivado', description: 'Ya no se inyecta. Queda en el historial (activa "Ver archivados").' });
   352→      load();
   353→    } catch (err: any) {
   354→      toast({ title: 'Error al archivar', description: err?.message || String(err), variant: 'destructive' });
   355→    }
   356→  };
   357→
   358→  const hardDelete = async (r: V2Record) => {
   359→    try {
   360→      const res = await fetch('/api/memory/v2', {
   361→        method: 'POST',
   362→        headers: { 'Content-Type': 'application/json' },
   363→        body: JSON.stringify({ action: 'hard-delete', id: r.id }),
   364→      });
   365→      const data = await res.json();
   366→      if (!data.success) throw new Error(data.error || 'Error');
   367→      toast({ title: 'Recuerdo eliminado', description: 'Eliminado definitivamente del almacén.' });
   368→      setPendingHardDelete(null);
   369→      load();
   370→    } catch (err: any) {
   371→      toast({ title: 'Error al eliminar', description: err?.message || String(err), variant: 'destructive' });
   372→    }
   373→  };
   374→
   375→  const runSearch = async () => {
   376→    if (!searchQuery.trim() || (!charId && !groupId)) return;
   377→    setSearching(true);
   378→    setSearchResults(null);
   379→    try {
   380→      const scopeQuery = charId
   381→        ? `charId=${encodeURIComponent(charId)}`
   382→        : `groupId=${encodeURIComponent(groupId!)}`;
   383→      const res = await fetch(`/api/memory/v2?action=search&${scopeQuery}&query=${encodeURIComponent(searchQuery)}&limit=10`);
   384→      const data = await res.json();
   385→      if (data.success) setSearchResults(data.results || []);
   386→      else throw new Error(data.error);
   387→    } catch (err: any) {
   388→      toast({ title: 'Error en la búsqueda', description: err?.message || String(err), variant: 'destructive' });
   389→    } finally {
   390→      setSearching(false);
   391→    }
   392→  };
   393→
   394→  // ============ Render helpers ============
   395→
   396→  const backendBadge = () => {
   397→    if (!health) return null;
   398→    if (health.backend === 'lancedb') {
   399→      return <Badge variant="outline" className="text-[9px] font-normal">LanceDB · {health.dimension}d</Badge>;
   400→    }
   401→    return (
   402→      <Badge variant="outline" className="text-[9px] font-normal text-amber-600 dark:text-amber-400">
   403→        JSON fallback{health.ollama?.reachable === false ? ' · sin Ollama' : ''}
   404→      </Badge>
   405→    );
   406→  };
   407→
   408→  const renderRecord = (r: V2Record) => {
   409→    const meta = TYPE_META[r.type];
   410→    return (
   411→      <div key={r.id} className={cn('group rounded-md bg-white/5 p-2', r.supersededBy && 'opacity-50')}>
   412→        <div className="flex items-center gap-1.5 mb-0.5">
   413→          {r.source === 'curada' ? (
   414→            <Badge variant="secondary" className="text-[9px] px-1 py-0 gap-0.5">
   415→              <Pin className="w-2 h-2" /> fijado
   416→            </Badge>
   417→          ) : (
   418→            <Badge variant="outline" className="text-[9px] px-1 py-0 font-normal">auto</Badge>
   419→          )}
   420→          {r.supersededBy && (
   421→            <Badge variant="outline" className="text-[9px] px-1 py-0 font-normal">
   422→              {r.supersededBy === '__deleted__' ? 'eliminado' : 'reemplazado'}
   423→            </Badge>
   424→          )}
   425→          <span className="text-[9px] text-muted-foreground inline-flex items-center gap-0.5">
   426→            <CalendarDays className="w-2.5 h-2.5" /> {fmtDate(r.eventDate || r.createdAt)}
   427→          </span>
   428→          <span className="text-[9px] text-muted-foreground" title="Importancia">★ {(r.importance ?? 0).toFixed(1)}</span>
   429→          <span className="text-[9px] text-muted-foreground" title="Accesos / calor">🔥 {r.heat?.toFixed?.(0) ?? r.heat}</span>
   430→          <div className="ml-auto flex items-center gap-0.5 opacity-0 group-hover:opacity-100 transition-opacity">
   431→            {!r.supersededBy && (
   432→              <>
   433→                <Button size="sm" variant="ghost" className="h-6 w-6 p-0" title="Editar" onClick={() => openEdit(r)}>
   434→                  <Pencil className="w-3 h-3" />
   435→                </Button>
   436→                <Button size="sm" variant="ghost" className="h-6 w-6 p-0" title="Archivar (deja de inyectarse, se conserva en historial)" onClick={() => archiveRecord(r)}>
   437→                  <Archive className="w-3 h-3" />
   438→                </Button>
   439→              </>
   440→            )}
   441→            <Button size="sm" variant="ghost" className="h-6 w-6 p-0 text-red-500 hover:text-red-600" title="Eliminar definitivamente" onClick={() => setPendingHardDelete(r)}>
   442→              <Trash2 className="w-3 h-3" />
   443→            </Button>
   444→          </div>
   445→        </div>
   446→        <p className={cn('text-xs text-foreground/90', r.supersededBy && 'line-through decoration-muted-foreground/50')}>{r.content}</p>
   447→        {r.supersededBy && r.supersededBy !== '__deleted__' && (
   448→          <p className="text-[9px] text-muted-foreground mt-0.5">→ reemplazado por {r.supersededBy}</p>
   449→        )}
   450→        {!compact && meta && (
   451→          <p className="text-[9px] text-muted-foreground/70 mt-1">{meta.block}</p>
   452→        )}
   453→      </div>
   454→    );
   455→  };
   456→
   457→  const recordsList = (
   458→    <div className="space-y-2">
   459→      {loading && !initialized ? (
   460→        <div className="flex items-center justify-center py-4 text-muted-foreground text-xs">
   461→          <Loader2 className="w-4 h-4 mr-2 animate-spin" /> Cargando recuerdos…
   462→        </div>
   463→      ) : filtered.length === 0 ? (
   464→        <p className="text-xs text-muted-foreground">
   465→          {records.length === 0
   466→            ? 'Sin recuerdos todavía. Se crean automáticamente al chatear (extracción) o manualmente con "+ Agregar".'
   467→            : 'Ningún recuerdo coincide con los filtros actuales.'}
   468→        </p>
   469→      ) : (
   470→        <div className={cn('space-y-2 overflow-y-auto custom-scrollbar', compact ? 'max-h-96' : 'max-h-[420px]')}>
   471→          {ALL_TYPES.filter(t => grouped[t]?.length).map(t => {
   472→            const meta = TYPE_META[t];
   473→            const isExpanded = expandedTypes[t] !== false; // default open
   474→            return (
   475→              <div key={t} className="space-y-1">
   476→                <button
   477→                  className="flex items-center gap-1.5 px-1 py-0.5 w-full text-left hover:opacity-80"
   478→                  onClick={() => setExpandedTypes(prev => ({ ...prev, [t]: !isExpanded }))}
   479→                >
   480→                  {isExpanded ? <ChevronDown className="w-3 h-3 text-muted-foreground" /> : <ChevronRight className="w-3 h-3 text-muted-foreground" />}
   481→                  <div className={cn('w-1.5 h-1.5 rounded-full', meta.dot)} />
   482→                  <span className="text-xs font-medium">{meta.plural}</span>
   483→                  <span className="text-xs text-muted-foreground">({grouped[t].length})</span>
   484→                </button>
   485→                {isExpanded && grouped[t].map(renderRecord)}
   486→              </div>
   487→            );
   488→          })}
   489→        </div>
   490→      )}
   491→    </div>
   492→  );
   493→
   494→  const searchPanel = !compact && (
   495→    <div className="space-y-1.5 pt-1 border-t border-border/40">
   496→      <Label className="text-xs text-muted-foreground">Simular recuperación (con reranking real)</Label>
   497→      <div className="flex gap-1.5">
   498→        <Input
   499→          value={searchQuery}
   500→          onChange={e => setSearchQuery(e.target.value)}
   501→          onKeyDown={e => { if (e.key === 'Enter') runSearch(); }}
   502→          placeholder={`Ej: ¿qué sientes hacia ${userName}? · ¿recuerdas aquella noche?`}
   503→          className="h-8 text-xs"
   504→        />
   505→        <Button size="sm" variant="outline" className="h-8 px-2.5" onClick={runSearch} disabled={searching || !searchQuery.trim()}>
   506→          {searching ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Search className="w-3.5 h-3.5" />}
   507→        </Button>
   508→      </div>
   509→      {searchResults && (
   510→        <div className="space-y-1 max-h-48 overflow-y-auto custom-scrollbar pt-1">
   511→          {searchResults.length === 0 ? (
   512→            <p className="text-xs text-muted-foreground">Sin resultados relevantes (el LLM admitiría no recordar).</p>
   513→          ) : (
   514→            searchResults.map((r: any) => (
   515→              <div key={r.id} className="rounded bg-white/5 p-1.5 flex items-start gap-2">
   516→                <Badge variant="outline" className="text-[9px] px-1 py-0 font-mono shrink-0">{(r.score ?? 0).toFixed(2)}</Badge>
   517→                <div className="min-w-0">
   518→                  <p className="text-xs text-foreground/90">{r.content}</p>
   519→                  <p className="text-[9px] text-muted-foreground">{TYPE_META[r.type as V2Type]?.label} · cos {r.cosine?.toFixed?.(2)}</p>
   520→                </div>
   521→              </div>
   522→            ))
   523→          )}
   524→        </div>
   525→      )}
   526→    </div>
   527→  );
   528→
   529→  const docsSection = !compact && (
   530→    <Accordion type="single" collapsible className="pt-1 border-t border-border/40">
   531→      <AccordionItem value="docs" className="border-0">
   532→        <AccordionTrigger className="text-xs py-2 hover:no-underline">
   533→          <span className="inline-flex items-center gap-1.5">
   534→            <HelpCircle className="w-3.5 h-3.5 text-violet-400" />
   535→            ¿Cómo funciona la Memoria V2?
   536→          </span>
   537→        </AccordionTrigger>
   538→        <AccordionContent className="text-xs text-muted-foreground space-y-3 pt-1">
   539→          <div>
   540→            <p className="font-medium text-foreground">Qué es</p>
   541→            <p>
   542→              Una única memoria a largo plazo por personaje. Cada recuerdo es un registro con tipo, fecha absoluta,
   543→              importancia y origen. Al chatear, el sistema inyecta al LLM tres bloques organizados:
   544→            </p>
   545→            <ul className="list-disc ml-4 mt-1 space-y-0.5">
   546→              <li><span className="font-mono text-[10px] text-foreground">[HECHOS SOBRE {userName}]</span> — hechos, preferencias y notas.</li>
   547→              <li><span className="font-mono text-[10px] text-foreground">[EVENTOS ANTERIORES]</span> — eventos y resúmenes de escena, con su fecha.</li>
   548→              <li><span className="font-mono text-[10px] text-foreground">[ESTADO DE LA RELACIÓN]</span> — el vínculo emocional actual.</li>
   549→            </ul>
   550→          </div>
   551→          <div>
   552→            <p className="font-medium text-foreground">Recuerdos fijados vs automáticos</p>
   553→            <p>
   554→              <span className="text-foreground">Fijados (curada)</span>: los que creas/editas aquí. Siempre entran al prompt,
   555→              nunca decaen ni se suprimen. <span className="text-foreground">Automáticos</span>: los extrae el LLM mientras
   556→              chateas (eventos, hechos, cambios de relación) y se recuperan por relevancia cuando algo del mensaje
   557→              del usuario los conecta (búsqueda vectorial + reranking con fecha, importancia y calor).
   558→            </p>
   559→          </div>
   560→          <div>
   561→            <p className="font-medium text-foreground">Cómo escribir un buen recuerdo</p>
   562→            <ul className="list-disc ml-4 mt-1 space-y-0.5">
   563→              <li>En 3ª persona y con los nombres reales: «Aitana besó a Marcos…», nunca «el usuario».</li>
   564→              <li>Autocontenido: alguien que no lea el chat debe entenderlo.</li>
   565→              <li>Con fecha real del evento (hoy por defecto). El LLM la usa para situarse en el tiempo.</li>
   566→              <li>Importancia: 0.9-1 promesas/límites/confesiones · 0.6-0.8 eventos significativos · 0.3-0.5 detalles.</li>
   567→            </ul>
   568→          </div>
   569→          <div>
   570→            <p className="font-medium text-foreground">Archivar vs eliminar</p>
   571→            <p>
   572→              <span className="text-foreground">Archivar</span> quita el recuerdo del prompt pero conserva el historial
   573→              (puedes verlo con «Archivados»). <span className="text-foreground">Eliminar</span> lo borra para siempre.
   574→              La extracción automática también archiva cuando el personaje «actualiza» un recuerdo, dejando trazabilidad.
   575→            </p>
   576→          </div>
   577→          <div>
   578→            <p className="font-medium text-foreground">¿Y los Resúmenes y el Conocimiento?</p>
   579→            <p>
   580→              Los <span className="text-foreground">Resúmenes</span> comprimen mensajes antiguos para que el contexto no
   581→              crezca (ahorro de tokens): son independientes de la memoria semántica. El{' '}
   582→              <span className="text-foreground">Conocimiento</span> (archivos que subes para RAG) se indexa en la misma
   583→              base vectorial y entra al prompt como [CONTEXTO RELEVANTE]. Esta tienda (Memoria V2) es el ÚNICO
   584→              sistema de recuerdos del personaje: no existe memoria clásica paralela.
   585→            </p>
   586→          </div>
   587→          {health?.error && (
   588→            <div className="rounded bg-amber-500/10 border border-amber-500/30 p-2">
   589→              <p className="text-amber-600 dark:text-amber-400 text-[11px]">
   590→                Estado: {health.error}. Los recuerdos se guardan en JSON y la búsqueda usa coincidencia léxica;
   591→                al conectar Ollama se migran a vectores automáticamente.
   592→              </p>
   593→            </div>
   594→          )}
   595→        </AccordionContent>
   596→      </AccordionItem>
   597→    </Accordion>
   598→  );
   599→
   600→  const body = (
   601→    <div className="space-y-2">
   602→      {/* Toolbar: filtros */}
   603→      <div className="flex flex-wrap items-center gap-1.5">
   604→        <Select value={typeFilter} onValueChange={v => setTypeFilter(v as 'todos' | V2Type)}>
   605→          <SelectTrigger className="h-7 w-[150px] text-xs">
   606→            <SelectValue />
   607→          </SelectTrigger>
   608→          <SelectContent>
   609→            <SelectItem value="todos">Todos los tipos</SelectItem>
   610→            {ALL_TYPES.map(t => <SelectItem key={t} value={t}>{TYPE_META[t].plural}</SelectItem>)}
   611→          </SelectContent>
   612→        </Select>
   613→        <Select value={sourceFilter} onValueChange={v => setSourceFilter(v as 'todas' | 'curada' | 'auto')}>
   614→          <SelectTrigger className="h-7 w-[130px] text-xs">
   615→            <SelectValue />
   616→          </SelectTrigger>
   617→          <SelectContent>
   618→            <SelectItem value="todas">Fijados + auto</SelectItem>
   619→            <SelectItem value="curada">Solo fijados</SelectItem>
   620→            <SelectItem value="auto">Solo automáticos</SelectItem>
   621→          </SelectContent>
   622→        </Select>
   623→        {!compact && (
   624→          <div className="flex items-center gap-1.5">
   625→            <Switch id={`arch-${charId || groupId || 'mgr'}`} checked={showArchived} onCheckedChange={setShowArchived} className="scale-90" />
   626→            <Label htmlFor={`arch-${charId || groupId || 'mgr'}`} className="text-xs text-muted-foreground cursor-pointer">
   627→              Archivados{archivedCount > 0 ? ` (${archivedCount})` : ''}
   628→            </Label>
   629→          </div>
   630→        )}
   631→        <div className="ml-auto flex items-center gap-1">
   632→          <Button size="sm" variant="ghost" className="h-7 w-7 p-0" title="Recargar" onClick={load} disabled={loading}>
   633→            <RefreshCw className={cn('w-3.5 h-3.5', loading && 'animate-spin')} />
   634→          </Button>
   635→          <Button size="sm" className="h-7 gap-1 text-xs" onClick={openCreate}>
   636→            <Plus className="w-3.5 h-3.5" /> Agregar
   637→          </Button>
   638→        </div>
   639→      </div>
   640→
   641→      {recordsList}
   642→      {searchPanel}
   643→      {docsSection}
   644→    </div>
   645→  );
   646→
   647→  return (
   648→    <div className={className}>
   649→      {!compact ? (
   650→        <Card className="border-violet-500/20">
   651→          <CardHeader className="pb-2">
   652→            <div className="flex items-center gap-2 flex-wrap">
   653→              <DatabaseZap className="w-4 h-4 text-violet-500" />
   654→              <CardTitle className="text-sm">{isGroupScope ? `Memoria del grupo · ${charName}` : `Memoria de ${charName}`}</CardTitle>
   655→              <Badge variant="secondary" className="text-xs">{activeCount} activos</Badge>
   656→              {backendBadge()}
   657→              {health?.model && (
   658→                <span className="text-[9px] text-muted-foreground">{health.model}</span>
   659→              )}
   660→            </div>
   661→          </CardHeader>
   662→          <CardContent>{body}</CardContent>
   663→        </Card>
   664→      ) : (
   665→        body
   666→      )}
   667→
   668→      {/* Dialogo crear/editar */}
   669→      <Dialog open={formOpen} onOpenChange={setFormOpen}>
   670→        <DialogContent className="sm:max-w-md">
   671→          <DialogHeader>
   672→            <DialogTitle className="text-base">{form.id ? 'Editar recuerdo' : 'Nuevo recuerdo'}</DialogTitle>
   673→            <DialogDescription className="text-xs">
   674→              {form.id
   675→                ? 'El cambio se guarda en el mismo registro y el vector se regenera.'
   676→                : 'Se creará como recuerdo FIJADO: siempre se inyecta en el prompt y nunca decae.'}
   677→            </DialogDescription>
   678→          </DialogHeader>
   679→          <div className="space-y-3 py-1">
   680→            <div className="grid grid-cols-2 gap-3">
   681→              <div className="space-y-1.5">
   682→                <Label className="text-xs">Tipo</Label>
   683→                <Select value={form.type} onValueChange={v => setForm(f => ({ ...f, type: v as V2Type }))}>
   684→                  <SelectTrigger className="h-8 text-xs"><SelectValue /></SelectTrigger>
   685→                  <SelectContent>
   686→                    {ALL_TYPES.map(t => <SelectItem key={t} value={t}>{TYPE_META[t].label}</SelectItem>)}
   687→                  </SelectContent>
   688→                </Select>
   689→              </div>
   690→              <div className="space-y-1.5">
   691→                <Label className="text-xs">Fecha del evento</Label>
   692→                <Input
   693→                  type="date"
   694→                  value={form.eventDate}
   695→                  onChange={e => setForm(f => ({ ...f, eventDate: e.target.value }))}
   696→                  className="h-8 text-xs"
   697→                />
   698→              </div>
   699→            </div>
   700→            <div className="space-y-1.5">
   701→              <Label className="text-xs">Recuerdo</Label>
   702→              <Textarea
   703→                value={form.content}
   704→                onChange={e => setForm(f => ({ ...f, content: e.target.value }))}
   705→                placeholder={isGroupScope
   706→                  ? `Ej: El grupo acordó viajar a la capital al amanecer`
   707→                  : `Ej: ${charName} prometió a ${userName} no volver a mentirle después de lo del bar`}
   708→                className="text-xs min-h-[80px] resize-y"
   709→                maxLength={480}
   710→              />
   711→              <p className="text-[10px] text-muted-foreground">
   712→                {isGroupScope ? 'Recuerdo GRUPAL: lo verán todos los personajes del grupo. ' : ''}
   713→                {TYPE_META[form.type]?.help} · {form.content.length}/480
   714→              </p>
   715→            </div>
   716→            <div className="space-y-1.5">
   717→              <div className="flex items-center justify-between">
   718→                <Label className="text-xs">Importancia</Label>
   719→                <span className="text-xs text-muted-foreground">{form.importance.toFixed(1)}</span>
   720→              </div>
   721→              <Slider
   722→                value={[form.importance]}
   723→                onValueChange={([v]) => setForm(f => ({ ...f, importance: v }))}
   724→                min={0}
   725→                max={1}
   726→                step={0.1}
   727→              />
   728→              <div className="flex justify-between text-[9px] text-muted-foreground">
   729→                <span>detalle menor</span><span>evento normal</span><span>crítico</span>
   730→              </div>
   731→            </div>
   732→          </div>
   733→          <DialogFooter>
   734→            <Button variant="outline" size="sm" onClick={() => setFormOpen(false)} disabled={saving}>Cancelar</Button>
   735→            <Button size="sm" onClick={saveForm} disabled={saving || !form.content.trim()}>
   736→              {saving && <Loader2 className="w-3.5 h-3.5 mr-1 animate-spin" />}
   737→              {form.id ? 'Guardar cambios' : 'Crear recuerdo'}
   738→            </Button>
   739→          </DialogFooter>
   740→        </DialogContent>
   741→      </Dialog>
   742→
   743→      {/* Confirmación de borrado definitivo */}
   744→      <AlertDialog open={!!pendingHardDelete} onOpenChange={o => !o && setPendingHardDelete(null)}>
   745→        <AlertDialogContent>
   746→          <AlertDialogHeader>
   747→            <AlertDialogTitle className="text-base">¿Eliminar definitivamente?</AlertDialogTitle>
   748→            <AlertDialogDescription className="text-xs">
   749→              Este recuerdo se borrará del almacén sin dejar historial:
   750→              «{pendingHardDelete?.content?.slice(0, 120)}».
   751→              Si prefieres conservarlo fuera del prompt, usa Archivar.
   752→            </AlertDialogDescription>
   753→          </AlertDialogHeader>
   754→          <AlertDialogFooter>
   755→            <AlertDialogCancel className="text-xs">Cancelar</AlertDialogCancel>
   756→            <AlertDialogAction
   757→              className="bg-red-600 hover:bg-red-700 text-white text-xs"
   758→              onClick={() => pendingHardDelete && hardDelete(pendingHardDelete)}
   759→            >
   760→              Eliminar
   761→            </AlertDialogAction>
   762→          </AlertDialogFooter>
   763→        </AlertDialogContent>
   764→      </AlertDialog>
   765→    </div>
   766→  );
   767→}
