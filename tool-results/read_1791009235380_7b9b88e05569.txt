   181→    status: 'checking',

   185→  const [omniVoiceProfiles, setOmniVoiceProfiles] = useState<OmniVoiceProfile[]>([]);

   221→      const response = await fetch('/api/tts/config');

   224→        const loadedTts = { ...DEFAULT_TTS_CONFIG, ...data.config.tts };
   225→        setTtsConfig(loadedTts);
   226→        setAsrConfig(data.config.asr);
   227→        if (data.config.kws) setKwsConfig(data.config.kws);
   228→        if (data.config.vad) setVadConfig(data.config.vad);
   229→      }
   230→    } catch (error) {
   231→      console.warn('[TTS Settings] Failed to load TTS config:', error);
   232→    }
   233→  };
   234→
   235→  // Save configuration
   236→  const saveConfig = async () => {
   237→    setIsSaving(true);
   238→    try {
   239→      const response = await fetch('/api/tts/config', {
   240→        method: 'POST',
   241→        headers: { 'Content-Type': 'application/json' },
   242→        body: JSON.stringify({ 
   243→          tts: ttsConfig, 
   244→          asr: asrConfig,
   245→          kws: kwsConfig,
   246→          vad: vadConfig,
   247→        }),
   248→      });

   250→      if (data.success) {
   251→        setHasChanges(false);
   252→      }
   253→    } catch (error) {
   254→      console.warn('[TTS Settings] Failed to save TTS config:', error);

   256→      setIsSaving(false);
   257→    }
   258→  };

   265→      const response = await fetch(`/api/tts/speech?endpoint=${encodeURIComponent(ttsConfig.baseUrl)}&provider=${ttsConfig.provider}`);
   266→      const data = await response.json();
   267→
   268→      setServiceStatus({
   269→        status: data.status,
   270→        endpoint: ttsConfig.baseUrl,
   271→        error: data.error,
   272→      });
   273→    } catch (error) {
   274→      setServiceStatus({
   275→        status: 'offline',
   276→        endpoint: ttsConfig.baseUrl,
   277→        error: error instanceof Error ? error.message : 'Cannot connect',

   279→    }
   280→  }, [ttsConfig.baseUrl, ttsConfig.provider]);
   281→

   286→      const data = await response.json();

   288→        setAvailableVoices(data.voices);
   289→      } else {
   290→        setAvailableVoices([]);
   291→      }

   293→      if (data.engines && Array.isArray(data.engines)) {
   294→        setOmniVoiceEngines(data.engines);
   295→      } else {
   296→        setOmniVoiceEngines([]);
   297→      }
   298→    } catch (error) {
   299→      console.warn('[TTS Settings] Failed to load available voices:', error);
   300→      setAvailableVoices([]);
   301→      setOmniVoiceEngines([]);
   302→    }
   303→  };

   305→  // Load OmniVoice voice profiles

   309→      const response = await fetch(`/api/tts/omnivoice/profiles?endpoint=${encodeURIComponent(ttsConfig.baseUrl)}`);
   310→      const data = await response.json();
   311→      if (data.success && data.profiles) {
   312→        setOmniVoiceProfiles(data.profiles);
   313→      } else {
   314→        setOmniVoiceProfiles([]);
   315→      }
   316→    } catch (error) {
   317→      console.warn('[TTS Settings] Failed to load OmniVoice profiles:', error);
   318→      setOmniVoiceProfiles([]);
   319→    } finally {
   320→      setIsLoadingProfiles(false);
   321→    }
   322→  };

   325→  const loadOmniVoiceArchetypes = async () => {
   326→    setIsLoadingArchetypes(true);

   332→      } else {
   333→        setOmniVoiceArchetypes([]);

   335→    } catch (error) {
   336→      console.warn('[TTS Settings] Failed to load OmniVoice archetypes:', error);
   337→      setOmniVoiceArchetypes([]);
   338→    } finally {
   339→      setIsLoadingArchetypes(false);
   340→    }
   341→  };
   342→

   344→  const applyArchetype = async (archetypeId: string, archetypeName: string) => {
   345→    try {
   346→      const response = await fetch('/api/tts/omnivoice/archetypes/use', {
   347→        method: 'POST',
   348→        headers: { 'Content-Type': 'application/json' },
   349→        body: JSON.stringify({
   350→          endpoint: ttsConfig.baseUrl,

   352→          name: archetypeName,

   357→        // Refresh profiles after creating one from archetype
   358→        await loadOmniVoiceProfiles();
   359→        await loadAvailableVoices();
   360→      }

   362→      console.warn('[TTS Settings] Failed to use archetype:', error);
   363→    }
   364→  };
   365→
   366→  // Update TTS config and mark as changed
   367→  const updateTtsConfig = (updates: Partial<TTSWebUIConfig>) => {
   368→    setTtsConfig(prev => ({ ...prev, ...updates }));
   369→    setHasChanges(true);
   370→  };
   371→

   375→    setTtsConfig(prev => ({
   376→      ...prev,
   377→      provider: newProvider,
   378→      baseUrl: providerConfig.defaultUrl,
   379→      model: providerConfig.defaultModel,
   380→    }));
   381→    setHasChanges(true);
   382→  };
   383→
   384→  // Update ASR config and mark as changed
   385→  const updateAsrConfig = (updates: Partial<ASRConfig>) => {
   386→    setAsrConfig(prev => ({ ...prev, ...updates }));
   387→    setHasChanges(true);
   388→  };
   389→
   390→  // Update KWS config and mark as changed
   391→  const updateKwsConfig = (updates: Partial<WakeWordConfig>) => {
   392→    setKwsConfig(prev => ({ ...prev, ...updates }));
   393→    setHasChanges(true);
   394→  };
   395→
   396→  // Update VAD config and mark as changed
   397→  const updateVadConfig = (updates: Partial<VADConfig>) => {
   398→    setVadConfig(prev => ({ ...prev, ...updates }));
   399→    setHasChanges(true);
   400→  };
   401→
   402→  // Test TTS
   403→  const handleTestTTS = async () => {
   404→    if (!testText.trim()) return;
   405→
   406→    setIsLoading(true);
   407→    try {
   408→      const response = await fetch('/api/tts/speech', {
   409→        method: 'POST',
   410→        headers: { 'Content-Type': 'application/json' },
   411→        body: JSON.stringify({
   412→          text: testText,
   413→          model: ttsConfig.model,
   414→          voice: ttsConfig.defaultVoice,
   415→          speed: ttsConfig.speed,
   416→          response_format: ttsConfig.responseFormat,
   417→          language: ttsConfig.language,
   418→          endpoint: ttsConfig.baseUrl,
   419→          provider: ttsConfig.provider,
   420→          // TTS-WebUI specific
   421→          exaggeration: ttsConfig.exaggeration,
   422→          cfg_weight: ttsConfig.cfgWeight,
   423→          temperature: ttsConfig.temperature,
   424→          // OmniVoice specific
   425→          voiceDesign: ttsConfig.voiceDesign,
   426→          instruct: ttsConfig.instruct,
   427→        }),
   428→      });
   429→
   430→      const data = await response.json();
   431→
   432→      if (!response.ok) {
   433→        const errorMsg = data.error || `Error del servidor (${response.status})`;
   434→        console.warn('[TTS] Test error:', errorMsg);
   435→        alert(`Error TTS: ${errorMsg}`);
   436→      } else if (data.success && data.audio) {
   437→        // Create audio blob URL
   438→        const audioBlob = base64ToBlob(data.audio, `audio/${data.format}`);
   439→        const url = URL.createObjectURL(audioBlob);
   440→        setAudioUrl(url);
   441→
   442→        // Play audio — handle autoplay policy gracefully
   443→        const audio = new Audio(url);
   444→        setIsPlaying(true);
   445→        audio.play().catch((error) => {
   446→          const errorMsg = error?.message || String(error);
   447→          if (errorMsg.includes("user didn't interact") || 
   448→              errorMsg.includes('NotAllowedError') ||
   449→              errorMsg.includes('play() failed')) {
   450→            console.warn('[TTS] Autoplay blocked — click the Play button to hear the test');
   451→            setIsPlaying(false);
   452→          } else {
   453→            console.warn('[TTS] Playback error:', error);
   454→            setIsPlaying(false);
   455→          }
   456→        });
   457→        audio.onended = () => setIsPlaying(false);
   458→      } else {
   459→        console.warn('[TTS] Test error:', data.error);
   460→        alert(`Error TTS: ${data.error}`);
   461→      }
   462→    } catch (error) {
   463→      console.warn('[TTS] Failed to test TTS:', error);
   464→      alert(`Error: ${error instanceof Error ? error.message : 'Unknown error'}`);
   465→    } finally {
   466→      setIsLoading(false);
   467→    }
   468→  };
   469→
   470→  // Stop playback
   471→  const handleStopPlayback = () => {
   472→    if (audioUrl) {
   473→      URL.revokeObjectURL(audioUrl);
   474→      setAudioUrl(null);
   475→    }
   476→    setIsPlaying(false);
   477→  };
   478→
   479→  // Helper: base64 to blob
   480→  const base64ToBlob = (base64: string, mimeType: string) => {
   481→    const byteCharacters = atob(base64);
   482→    const byteArrays: BlobPart[] = [];
   483→
   484→    for (let offset = 0; offset < byteCharacters.length; offset += 512) {
   485→      const slice = byteCharacters.slice(offset, offset + 512);
   486→      const byteNumbers = new Array(slice.length);
   487→      for (let i = 0; i < slice.length; i++) {
   488→        byteNumbers[i] = slice.charCodeAt(i);
   489→      }
   490→      const byteArray = new Uint8Array(byteNumbers);
   491→      byteArrays.push(byteArray as BlobPart);
   492→    }
   493→
   494→    return new Blob(byteArrays, { type: mimeType });
   495→  };

   504→        'border-yellow-500/30 bg-yellow-500/5'
   505→      )}>
   506→        <CardContent className="p-4">
   507→          <div className="flex items-center justify-between">
   508→            <div className="flex items-center gap-3">
   509→              {serviceStatus.status === 'checking' ? (
   510→                <Loader2 className="w-5 h-5 animate-spin text-yellow-500" />
   511→              ) : serviceStatus.status === 'online' ? (
   512→                <CheckCircle className="w-5 h-5 text-green-500" />
   513→              ) : (
   514→                <AlertCircle className="w-5 h-5 text-red-500" />
   515→              )}
   516→              <div>
   517→                <p className="font-medium">
   518→                  {PROVIDER_CONFIGS[ttsConfig.provider].name} {serviceStatus.status === 'online' ? 'Conectado' : serviceStatus.status === 'checking' ? 'Verificando...' : 'Desconectado'}
   519→                </p>
   520→                <p className="text-xs text-muted-foreground">{serviceStatus.endpoint}</p>
   521→              </div>
   522→            </div>
   523→            <Button
   524→              variant="outline"
   525→              size="sm"
   526→              onClick={checkServiceStatus}
   527→              disabled={serviceStatus.status === 'checking'}
   528→            >
   529→              <RefreshCw className={cn('w-4 h-4 mr-2', serviceStatus.status === 'checking' && 'animate-spin')} />
   530→              Verificar
   531→            </Button>
   532→          </div>
   533→          {serviceStatus.error && (
   534→            <p className="mt-2 text-xs text-red-500">{serviceStatus.error}</p>
   535→          )}
   536→        </CardContent>
   537→      </Card>
   538→
   539→      {/* Save Button */}
   540→      {hasChanges && (
   541→        <Button onClick={saveConfig} disabled={isSaving} className="w-full">
   542→          {isSaving ? (
   543→            <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   544→          ) : (
   545→            <Save className="w-4 h-4 mr-2" />
   546→          )}
   547→          Guardar Configuración
   548→        </Button>
   549→      )}
   550→
   551→      <Tabs defaultValue="kws" className="w-full">
   552→        <TabsList className="grid w-full grid-cols-3">
   553→          <TabsTrigger value="kws" className="gap-2">
   554→            <Ear className="w-4 h-4" />
   555→            Voz (KWS)
   556→          </TabsTrigger>
   557→          <TabsTrigger value="tts" className="gap-2">
   558→            <Volume2 className="w-4 h-4" />
   559→            TTS
   560→          </TabsTrigger>
   561→          <TabsTrigger value="voices" className="gap-2">
   562→            <Music className="w-4 h-4" />
   563→            Voces
   564→          </TabsTrigger>
   565→        </TabsList>
   566→
   567→        {/* TTS Tab */}
   568→        <TabsContent value="tts" className="space-y-4 mt-4">
   569→          <Card>
