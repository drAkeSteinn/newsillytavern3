   824→  importance: number;

   835→  resumen_escena: 'Resúmenes de escena',

   838→function MemoryV2Tab() {
   839→  const settings = useTavernStore((s) => (s.settings as any)?.embeddingsChat) || DEFAULT_EMBEDDINGS_CHAT;
   840→  const updateSettings = useTavernStore((s) => s.updateSettings);
   841→  const activeCharacterId = useTavernStore((s) => s.activeCharacterId);
   842→  const activeCharacter = useTavernStore((s) => s.characters)?.find((c: any) => c.id === activeCharacterId);
   843→
   844→  const [health, setHealth] = useState<V2HealthData | null>(null);
   845→  const [records, setRecords] = useState<V2RecordLite[]>([]);
   846→  const [loading, setLoading] = useState(false);
   847→  const [migrating, setMigrating] = useState(false);
   848→  const [reiniting, setReiniting] = useState(false);
   849→  const [searchQuery, setSearchQuery] = useState('');
   850→  const [searchResults, setSearchResults] = useState<V2RecordLite[] | null>(null);
   851→
   852→  const updateEmbeddingsChat = (updates: Record<string, unknown>) => {
   853→    updateSettings({
   854→      embeddingsChat: { ...settings, ...updates },
   855→    } as any);
   856→  };
   857→
   858→  const loadHealth = useCallback(async () => {
   859→    setLoading(true);
   860→    try {
   861→      const res = await fetch('/api/memory/v2?action=health');
   862→      const data = await res.json();
   863→      if (data.success) setHealth(data.health);
   864→    } catch {
   865→      /* silent */
   866→    } finally {
   867→      setLoading(false);
   868→    }
   869→  }, []);
   870→
   871→  const loadRecords = useCallback(async () => {
   872→    if (!activeCharacterId) return;
   873→    try {
   874→      const res = await fetch(`/api/memory/v2?action=records&charId=${encodeURIComponent(activeCharacterId)}&limit=200`);
   875→      const data = await res.json();
   876→      if (data.success) setRecords(data.records || []);
   877→    } catch {
   878→      /* silent */
   879→    }
   880→  }, [activeCharacterId]);
   881→
   882→  // Initial load
   883→  useEffect(() => {
   884→    loadHealth();
   885→    loadRecords();
   886→  }, [loadHealth, loadRecords]);
   887→
   888→  const runMigration = async () => {
   889→    setMigrating(true);
   890→    try {
   891→      const res = await fetch('/api/memory/v2', {
   892→        method: 'POST',
   893→        headers: { 'Content-Type': 'application/json' },
   894→        body: JSON.stringify({ action: 'migrate', charId: activeCharacterId || undefined }),
   895→      });
   896→      const data = await res.json();
   897→      if (data.success) {
   898→        toast.success(`Migración completada: ${data.migrated} nuevos, ${data.skipped} ya existentes`);
   899→        loadHealth();
   900→        loadRecords();
   901→      } else {
   902→        toast.error(data.error || 'Migración fallida');
   903→      }
   904→    } catch {
   905→      toast.error('Error de red durante la migración');
   906→    } finally {
   907→      setMigrating(false);
   908→    }
   909→  };
   910→
   911→  const runReinit = async () => {
   912→    setReiniting(true);
   913→    try {
   914→      const res = await fetch('/api/memory/v2', {
   915→        method: 'POST',
   916→        headers: { 'Content-Type': 'application/json' },
   917→        body: JSON.stringify({ action: 'reinit' }),
   918→      });
   919→      const data = await res.json();
   920→      if (data.success) {
   921→        setHealth(data.health);
   922→        toast.success(`Store reinicializado (backend: ${data.health?.backend || '?'})`);
   923→        loadRecords();
   924→      }
   925→    } catch {
   926→      toast.error('Error reinicializando el store');
   927→    } finally {
   928→      setReiniting(false);

   939→        setSearchResults(data.results || []);
   940→      }
   941→    } catch {
   942→      /* silent */
   943→    } finally {
   944→      setLoading(false);
   945→    }
