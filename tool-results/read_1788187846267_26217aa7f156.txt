    45→// Tool Badge with rich tooltip
    46→// ============================================
    47→// Shows the tool icon + label, and on hover displays a meaningful tooltip
    48→// explaining WHAT changed (e.g., "Lujuria: 55 → 70 (+15)") instead of just

    51→const TOOL_ICON_MAP: Record<string, typeof Wrench> = {
    52→  Dices,
    53→  Brain: Zap,
    54→  Globe,
    55→  CloudSun,
    56→  Bell,
    57→  Search: Globe,
    58→  WebSearch: Globe,
    59→  Calculator: Zap,
    60→  Weather: CloudSun,
    61→  Timer: Bell,
    62→  Pencil,
    63→  Crown,
    64→  Heart,
    65→  Flame,
    66→  Sparkles,
    67→  Wrench,
    68→};
    69→
    70→function getToolBadgeIcon(iconName?: string): typeof Wrench {
    71→  if (!iconName) return Wrench;
    72→  return TOOL_ICON_MAP[iconName] ?? Wrench;
    73→}
    74→
    75→/** Build a rich tooltip content for a tool usage */
    76→function ToolBadgeWithTooltip({ tool, index }: { tool: ToolUsedInfo; index: number }) {
    77→  const ToolIcon = getToolBadgeIcon(tool.icon);
    78→  const isSuccess = tool.success !== false;
    79→  const hasDetails = !!tool.details;
    80→  const hasDisplayMessage = !!tool.displayMessage;
    81→
    82→  // Build the tooltip content: title + details + display message
    83→  const tooltipContent = (
    84→    <div className="max-w-xs space-y-1.5 text-left">
    85→      {/* Title row */}
    86→      <div className="flex items-center gap-1.5 font-semibold">
    87→        {React.createElement(ToolIcon, { className: 'w-3 h-3 shrink-0' })}
    88→        <span>{tool.label}</span>
    89→        {isSuccess ? (
    90→          <span className="text-emerald-400 text-[10px]">✓</span>
    91→        ) : (
    92→          <span className="text-red-400 text-[10px]">✗</span>
    93→        )}
    94→      </div>
    95→
    96→      {/* Meaningful details: what changed */}
    97→      {hasDetails && (
    98→        <div className="border-t border-border/50 pt-1">
    99→          {tool.details!.split('\n').map((line, i) => (
   100→            <p key={i} className="text-[11px] leading-snug text-muted-foreground whitespace-pre-wrap break-words">
   101→              {line}
   102→            </p>
   103→          ))}
   104→        </div>
   105→      )}
   106→
   107→      {/* Full display message (collapsible, only if different from details) */}
   108→      {hasDisplayMessage && tool.displayMessage !== tool.details && (
   109→        <details className="border-t border-border/50 pt-1">
   110→          <summary className="text-[10px] text-muted-foreground/70 cursor-pointer hover:text-muted-foreground">
   111→            Ver mensaje completo
   112→          </summary>
   113→          <pre className="mt-1 text-[10px] leading-tight text-muted-foreground/80 whitespace-pre-wrap break-words font-mono">
   114→            {tool.displayMessage}
   115→          </pre>
   116→        </details>
   117→      )}
   118→
   119→      {!hasDetails && !hasDisplayMessage && (
   120→        <p className="text-[11px] text-muted-foreground/70 italic">
   121→          {isSuccess ? 'Herramienta ejecutada' : 'La herramienta falló'}
   122→        </p>
   123→      )}
   124→    </div>
   125→  );
   126→
   127→  return (
   128→    <Tooltip key={`${tool.name}-${index}`}>
   129→      <TooltipTrigger asChild>
   130→        <Badge
   131→          variant="outline"
   132→          className={cn(
   133→            'text-[10px] py-0 h-4 px-1 gap-0.5 cursor-help',
   134→            isSuccess
   135→              ? 'border-green-500/30 text-green-400/70 hover:border-green-500/50 hover:text-green-400'
   136→              : 'border-red-500/30 text-red-400/70 hover:border-red-500/50 hover:text-red-400',
   137→          )}
   138→        >
   139→          {React.createElement(ToolIcon, { className: 'w-2.5 h-2.5' })}
   140→          <span className="truncate max-w-[60px]">{tool.label}</span>
   141→        </Badge>
   142→      </TooltipTrigger>
   143→      <TooltipContent side="top" className="max-w-xs p-2">
   144→        {tooltipContent}
   145→      </TooltipContent>
   146→    </Tooltip>
   147→  );
   148→}

   615→  // Get bubble styles
   616→  const bubbleStyle = getBubbleStyle();
   617→
   618→  return (
   619→    <div
   620→      className={cn(
   621→        'group flex gap-3',
   622→        getEntryAnimation(),

   664→
   665→      {/* Message Content */}
   666→      <div className={cn(
   667→        'flex-1',
   668→        isUser ? 'items-end' : 'items-start',
   669→        'flex flex-col'
   670→      )}
   671→      style={{ maxWidth: `${safeAppearance.bubbles.maxWidth}%` }}
   672→      >
   673→        {/* Name and Timestamp - Above the bubble */}

   675→          'flex items-center gap-2 mb-1 flex-wrap',

   678→          {isNarrator && !isUser && <Ghost className="w-3.5 h-3.5 text-violet-400" />}

   693→            {displayName}

   705→          {/* Tool badges - show on right side for user, left for characters */}

   718→        {/* Message Bubble */}
   719→        <div

   721→            bubbleStyle.className,
   722→            isCompact ? 'px-3 py-2' : 'px-4 py-3',
   723→            isNarrator && 'italic'
   724→          )}
   725→          style={{

   730→        >
   731→          {isEditing ? (
   732→            <div className="space-y-2 min-w-[200px]">
