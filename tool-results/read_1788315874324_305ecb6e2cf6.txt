    88→/**
    89→ * Build a HUD context section for prompt injection
    90→ * 
    91→ * Now resolves ALL keys including:
    92→ * - Template variables: {{user}}, {{char}}, {{userpersona}}
    93→ * - Stats keys: {{resistencia}}, {{habilidades}}, etc.
    94→ */
    95→export function buildHUDContextSection(
    96→  contextConfig: HUDContextConfig,
    97→  keyContext?: KeyResolutionContext
    98→): PromptSection | null {
    99→  if (!contextConfig.enabled || !contextConfig.content.trim()) {
   100→    return null;
   101→  }
   102→
   103→  // Resolve all keys in the context content
   104→  let resolvedContent = contextConfig.content;
   105→  if (keyContext) {
   106→    resolvedContent = resolveAllKeys(contextConfig.content, keyContext);
   107→  }
   108→
   109→  return {
   110→    type: 'hud_context',
   111→    label: 'HUD Context',
   112→    content: resolvedContent,
   113→    color: SECTION_COLORS.hud_context
   114→  };
   115→}

   117→/**
   118→ * Inject HUD context at the specified position
   119→ *
   120→ * Positions:
   121→ * 0 = After system prompt
   122→ * 1 = After user message (after last user message)
   123→ * 2 = Before user message (before last user message)
   124→ * 3 = After assistant message (after last assistant message)
   125→ * 4 = Before assistant message (before last assistant message)
   126→ * 5 = At top of chat (before chat history)
   127→ * 6 = At bottom of chat (after all messages)
   128→ * 7 = After lorebook / Author's Note position
   129→ */
   130→export function injectHUDContextIntoMessages(
   131→  messages: ChatApiMessage[],
   132→  contextSection: PromptSection,
   133→  position: number
   134→): ChatApiMessage[] {
   135→  if (!contextSection.content.trim()) {
   136→    return messages;
   137→  }
   138→
   139→  const contextContent = `[${contextSection.label}]\n${contextSection.content}`;
   140→  const result: ChatApiMessage[] = [...messages];
   141→
   142→  switch (position) {
   143→    case 0: // After system prompt
   144→      // Find the first system/assistant message (the system prompt)
   145→      const sysIdx = result.findIndex(m => m.role === 'assistant' || m.role === 'system');
   146→      if (sysIdx >= 0) {
   147→        result[sysIdx] = {
   148→          ...result[sysIdx],
   149→          content: result[sysIdx].content + '\n\n' + contextContent
   150→        };
   151→      }
   152→      break;
   153→
   154→    case 1: // After user message
   155→      // Find the last user message
   156→      const lastUserIdx = result.map((m, i) => m.role === 'user' ? i : -1).filter(i => i >= 0).pop();
   157→      if (lastUserIdx !== undefined && lastUserIdx >= 0) {
   158→        result[lastUserIdx] = {
   159→          ...result[lastUserIdx],
   160→          content: result[lastUserIdx].content + '\n\n' + contextContent
   161→        };
   162→      }
   163→      break;
   164→
   165→    case 2: // Before user message
   166→      const lastUserIdx2 = result.map((m, i) => m.role === 'user' ? i : -1).filter(i => i >= 0).pop();
   167→      if (lastUserIdx2 !== undefined && lastUserIdx2 >= 0) {
   168→        result[lastUserIdx2] = {
   169→          ...result[lastUserIdx2],
   170→          content: contextContent + '\n\n' + result[lastUserIdx2].content
   171→        };
   172→      }
   173→      break;
   174→
   175→    case 3: // After assistant message
   176→      const lastAsstIdx = result.map((m, i) => m.role === 'assistant' ? i : -1).filter(i => i >= 0).pop();
   177→      if (lastAsstIdx !== undefined && lastAsstIdx >= 0) {
   178→        result[lastAsstIdx] = {
   179→          ...result[lastAsstIdx],
   180→          content: result[lastAsstIdx].content + '\n\n' + contextContent
   181→        };
   182→      }
   183→      break;
   184→
   185→    case 4: // Before assistant message
   186→      const lastAsstIdx4 = result.map((m, i) => m.role === 'assistant' ? i : -1).filter(i => i >= 0).pop();
   187→      if (lastAsstIdx4 !== undefined && lastAsstIdx4 >= 0) {
   188→        result[lastAsstIdx4] = {
   189→          ...result[lastAsstIdx4],
   190→          content: contextContent + '\n\n' + result[lastAsstIdx4].content
   191→        };
   192→      }
   193→      break;
   194→
   195→    case 5: // At top of chat (after system, before first message)
   196→      // Merge into the first system message to avoid breaking alternation
   197→      {
   198→        const sysMsgIdx5 = result.findIndex(m => m.role === 'system');
   199→        if (sysMsgIdx5 >= 0) {
   200→          result[sysMsgIdx5] = {
   201→            ...result[sysMsgIdx5],
   202→            content: result[sysMsgIdx5].content + '\n\n' + contextContent
   203→          };
   204→        }
   205→      }
   206→      break;
   207→
   208→    case 6: // At bottom of chat (after all messages)
   209→      // Merge into the first system message to avoid breaking alternation
   210→      {
   211→        const sysMsgIdx6 = result.findIndex(m => m.role === 'system');
   212→        if (sysMsgIdx6 >= 0) {
   213→          result[sysMsgIdx6] = {
   214→            ...result[sysMsgIdx6],
   215→            content: result[sysMsgIdx6].content + '\n\n' + contextContent
   216→          };
   217→        }
   218→      }
   219→      break;
   220→
   221→    case 7: // After lorebook / Author's Note position
   222→      // Merge into the first system message to avoid breaking alternation
   223→      {
   224→        const sysMsgIdx7 = result.findIndex(m => m.role === 'system' || m.role === 'assistant');
   225→        if (sysMsgIdx7 >= 0) {
   226→          result[sysMsgIdx7] = {
   227→            ...result[sysMsgIdx7],
   228→            content: result[sysMsgIdx7].content + '\n\n' + contextContent
   229→          };
   230→        }
   231→      }
   232→      break;
   233→
   234→    default:
   235→      // Merge into the first system message to avoid breaking alternation
   236→      {
   237→        const sysMsgIdxDef = result.findIndex(m => m.role === 'system');
   238→        if (sysMsgIdxDef >= 0) {
   239→          result[sysMsgIdxDef] = {
   240→            ...result[sysMsgIdxDef],
   241→            content: result[sysMsgIdxDef].content + '\n\n' + contextContent
   242→          };
   243→        } else {
   244→          result.push({
   245→            role: 'system',
   246→            content: contextContent
   247→          });
   248→        }
   249→      }
   250→      break;
   251→  }
   252→
   253→  return result;
   254→}

   256→/**
   257→ * Inject HUD context into prompt sections
   258→ * This is for the prompt viewer display
   259→ *
   260→ * Positions:
   261→ * 0 = After system prompt
   262→ * 1 = After user message (not applicable to sections, uses chat_history)
   263→ * 2 = Before user message (not applicable to sections, uses chat_history)
   264→ * 3 = After assistant message (not applicable to sections, uses chat_history)
   265→ * 4 = Before assistant message (not applicable to sections, uses chat_history)
   266→ * 5 = At top of chat (before chat history)
   267→ * 6 = At bottom of chat (after all sections)
   268→ * 7 = After lorebook
   269→ */
   270→export function injectHUDContextIntoSections(
   271→  sections: PromptSection[],
   272→  contextSection: PromptSection,
   273→  position: number
   274→): PromptSection[] {
   275→  if (!contextSection.content.trim()) {
   276→    return sections;
   277→  }
   278→
   279→  const result: PromptSection[] = [...sections];
   280→
   281→  switch (position) {
   282→    case 0: // After system prompt
   283→      const sysSectionIdx = result.findIndex(s => s.type === 'system');
   284→      if (sysSectionIdx >= 0) {
   285→        result.splice(sysSectionIdx + 1, 0, contextSection);
   286→      } else {
   287→        result.unshift(contextSection);
   288→      }
   289→      break;
   290→
   291→    case 1: // After user message - inject before chat history
   292→    case 2: // Before user message - inject before chat history
   293→    case 3: // After assistant message - inject before chat history
   294→    case 4: // Before assistant message - inject before chat history
   295→      // These positions apply to messages, for sections we inject before chat history
   296→      const chatHistoryIdx = result.findIndex(s => s.type === 'chat_history');
   297→      if (chatHistoryIdx >= 0) {
   298→        result.splice(chatHistoryIdx, 0, contextSection);
   299→      } else {
   300→        result.push(contextSection);
   301→      }
   302→      break;
   303→
   304→    case 5: // At top of chat (before chat history)
   305→      const chatIdx = result.findIndex(s => s.type === 'chat_history');
   306→      if (chatIdx >= 0) {
   307→        result.splice(chatIdx, 0, contextSection);
   308→      } else {
   309→        result.push(contextSection);
   310→      }
   311→      break;
   312→
   313→    case 7: // After lorebook
   314→      const lorebookIdx = result.findIndex(s => s.type === 'lorebook');
   315→      if (lorebookIdx >= 0) {
   316→        result.splice(lorebookIdx + 1, 0, contextSection);
   317→      } else {
   318→        // If no lorebook, insert after system prompt
   319→        const sysIdx = result.findIndex(s => s.type === 'system');
   320→        if (sysIdx >= 0) {
   321→          result.splice(sysIdx + 1, 0, contextSection);
   322→        } else {
   323→          result.unshift(contextSection);
   324→        }
   325→      }
   326→      break;
   327→
   328→    case 6: // At bottom of chat
   329→    default:
   330→      result.push(contextSection);
   331→      break;
   332→  }
   333→
   334→  return result;
   335→}

   341→/**
   342→ * Build an inventory section for prompt injection.
   343→ *
   344→ * Shows:
   345→ * - Current currency
   346→ * - Equipped items and their effects
   347→ * - Active consumable effects with remaining turns
   348→ * - Available (non-equipped) items in inventory
   349→ *
   350→ * Uses the `promptTemplate` from `inventorySettings` for formatting.
   351→ * If no template is provided or the template is empty, a default format is used.
   352→ *
   353→ * Returns null if inventory data is empty or invalid.
   354→ */
   355→export function buildInventorySection(
   356→  inventoryData: InventoryPromptData,
   357→  keyContext?: KeyResolutionContext
   358→): PromptSection | null {
   359→  const {
   360→    personaItems,
   361→    equippedItems,
   362→    sessionEquipment,
   363→    activeEffects,
   364→    currency,
   365→    currencyName,
   366→    currencyIcon,
   367→    inventorySettings,
   368→  } = inventoryData;
   369→
   370→  // If no items and no effects and no currency, skip section
   371→  if (personaItems.length === 0 && activeEffects.length === 0 && currency === 0) {
   372→    return null;
   373→  }
   374→
   375→  // Use the template from inventory settings, or fall back to default
   376→  const template = inventorySettings.promptTemplate?.trim() || `[Inventario Activo]
   377→{{activeItems}}
   378→
   379→[Efectos Activos]
   380→{{activeEffects}}
   381→
   382→[Divisa]
   383→{{currency}}`;
   384→
   385→  // Build items list (all items in inventory)
   386→  const itemLines = personaItems.map(({ entry, item }) => {
   387→    const qty = entry.quantity > 1 ? ` x${entry.quantity}` : '';
   388→    const eq = entry.equipped ? ' [Equipado]' : '';
   389→    const effects = (item.attributeEffects && item.attributeEffects.length > 0)
   390→      ? ` (${item.attributeEffects.map(e => `${e.operator}${e.value} ${e.attributeKey}`).join(', ')})`
   391→      : '';
   392→    return `- ${item.icon || ''} ${item.name}${qty}${eq}${effects}`;
   393→  }).join('\n');
   394→
   395→  // Build active effects list
   396→  const effectLines = activeEffects.map(e => {
   397→    const turnsLeft = e.remainingTurns > 0 ? ` (${e.remainingTurns}/${e.totalTurns} turnos)` : '';
   398→    // Use consumableEffect free text if available, otherwise fall back to attribute effects
   399→    let effectDesc: string;
   400→    if (e.consumableEffect) {
   401→      effectDesc = e.consumableEffect;
   402→    } else {
   403→      effectDesc = e.effects.map(ef =>
   404→        `${ef.operator}${ef.value} ${ef.attributeKey}${ef.targetId !== '__user__' ? ` → ${ef.targetName || ef.targetId}` : ''}`
   405→      ).join(', ');
   406→    }
   407→    return `- ${e.itemName}: ${effectDesc}${turnsLeft}`;
   408→  }).join('\n');
   409→
   410→  // Build equipped items list - prefer sessionEquipment (per-session), fallback to equippedItems (legacy)
   411→  let equipLines = '';
   412→  if (sessionEquipment && sessionEquipment.length > 0) {
   413→    // Use session equipment data
   414→    const items = personaItems.map(({ item }) => item);
   415→    equipLines = sessionEquipment.map(eq => {
   416→      const item = items.find(i => i.id === eq.itemId);
   417→      const slotDef = inventorySettings.equipmentSlots?.find(s => s.id === eq.equippedSlotId);
   418→      const slotLabel = slotDef?.name || eq.equippedSlotId;
   419→      const effectText = eq.slotEffectText
   420→        || item?.slotEffects?.find(se => se.slotId === eq.equippedSlotId)?.effectText
   421→        || '';
   422→      return `- ${item?.icon || ''} ${item?.name || '???'} [${slotLabel}]${effectText ? ` → ${effectText}` : ''}`;
   423→    }).join('\n');
   424→  } else if (equippedItems && equippedItems.length > 0) {
   425→    // Fallback to legacy equipped items
   426→    equipLines = equippedItems.map(({ item }) => {
   427→      // Prefer slotEffects (V3) over attributeEffects (V2)
   428→      let effects = '';
   429→      if (item.slotEffects && item.slotEffects.length > 0) {
   430→        effects = item.slotEffects.map(se => se.effectText).filter(Boolean).join('; ');
   431→      } else if (item.attributeEffects && item.attributeEffects.length > 0) {
   432→        effects = item.attributeEffects.map(e => `${e.operator}${e.value} ${e.attributeKey}`).join(', ');
   433→      }
   434→      const slotDef = inventorySettings.equipmentSlots?.find(s => s.id === item.slot);
   435→      const slotLabel = slotDef?.name || item.slot || '';
   436→      return `- ${item.icon || ''} ${item.name}${slotLabel ? ` [${slotLabel}]` : ''}${effects ? ` → ${effects}` : ''}`;
   437→    }).join('\n');
   438→  }
   439→
   440→  // Build currency line
   441→  const currencyLine = `${currencyIcon || '💰'} ${currencyName || 'Divisa'}: ${currency}`;
   442→
   443→  // Apply template replacements
   444→  let content = template
   445→    .replace('{{activeItems}}', itemLines || 'Vacío')
   446→    .replace('{{activeEffects}}', effectLines || 'Ninguno')
   447→    .replace('{{equippedItems}}', equipLines || 'Ninguno')
   448→    .replace('{{currency}}', currencyLine);
   449→
   450→  // Resolve all keys in the inventory content if keyContext is available
   451→  if (keyContext) {
   452→    content = resolveAllKeys(content, keyContext);
   453→  }
   454→
   455→  return {
   456→    type: 'inventory',
   457→    label: 'Inventory',
   458→    content,
   459→    color: SECTION_COLORS.inventory
   460→  };
   461→}

   498→/**
   499→ * Build the system prompt from character data (SillyTavern style)
   500→ *
   501→ * Uses unified key resolution for ALL sections:
   502→ * - Template variables: {{user}}, {{char}}, {{userpersona}}, etc.
   503→ * - Stats keys: {{resistencia}}, {{habilidades}}, etc.
   504→ * - Sound keys: {{sonidos}}
   505→ */
   506→export function buildSystemPrompt(
   507→  character: CharacterCard,
   508→  userName: string = 'User',
   509→  persona?: Persona,
   510→  lorebookPlan?: LorebookInjectionPlan | null,
   511→  sessionStats?: SessionStats,
   512→  allCharacters?: CharacterCard[],
   513→  soundTriggers?: SoundTrigger[],
   514→  soundSettings?: AppSettings['sound'],
   515→  questTemplates?: QuestTemplate[],
   516→  sessionQuests?: SessionQuestInstance[],
   517→  questSettings?: QuestSettings,
   518→  lorebookAttributeKeys?: Record<string, string>,
   519→  inventoryData?: InventoryPromptData,
   520→  lorebookEntryKeyMap?: Record<string, string>
   521→): { prompt: string; sections: PromptSection[]; lorebookChatInjections: LorebookChatInjection[]; exampleMessages: ChatApiMessage[] } {
   522→  const sections: PromptSection[] = [];
   523→
   524→  // Resolve stats for the persona FIRST (user attributes like {{resistencia}})
   525→  // This must be done before character stats so personaResolvedStats is available
   526→  let personaResolvedStats: ResolvedStats | null = null;
   527→  if (persona?.statsConfig?.enabled && sessionStats) {
   528→    personaResolvedStats = resolveStats({
   529→      characterId: '__user__',
   530→      statsConfig: persona.statsConfig,
   531→      sessionStats,
   532→      lorebookEntryKeys: lorebookEntryKeyMap,
   533→    });
   534→  }
   535→
   536→  // Resolve stats for this character (includes skills block with full key resolution)
   537→  const resolvedStats = resolveStats({
   538→    characterId: character.id,
   539→    statsConfig: character.statsConfig,
   540→    sessionStats,
   541→    allCharacters,
   542→    userName,
   543→    characterName: character.name,
   544→    questTemplates,
   545→    personaDescription: persona?.description,
   546→    personaResolvedStats,
   547→    lorebookEntryKeys: lorebookEntryKeyMap,
   548→  });
   549→
   550→  // Build outlet sections map from lorebook plan for {{outlet::name}} macro resolution
   551→  const outletSections: Record<string, string> = {};
   552→  if (lorebookPlan?.outletSections.length) {
   553→    for (const outletSection of lorebookPlan.outletSections) {
   554→      // Extract outlet name from label like "World Info (myOutlet)" → "myOutlet"
   555→      const match = outletSection.label.match(/^World Info \((.+)\)$/);
   556→      const outletName = match ? match[1] : outletSection.label;
   557→      outletSections[outletName] = outletSection.content;
   558→    }
   559→  }
   560→

   960→/**
   961→ * Inject lorebook content at chat-level positions (1-4).
   962→ * Merges content into existing messages to avoid breaking message alternation.
   963→ *
   964→ * Position mapping:
   965→ * 1 = After last user message
   966→ * 2 = Before last user message
   967→ * 3 = After last assistant message
   968→ * 4 = Before last assistant message
   969→ */
   970→function applyChatInjections(
   971→  messages: ChatApiMessage[],
   972→  injections: LorebookChatInjection[]
   973→): void {
   974→  for (const injection of injections) {
   975→    const content = `\n\n[${injection.label}]\n${injection.content}`;
   976→
   977→    switch (injection.position) {
   978→      case 1: { // After last user message
   979→        const lastUserIdx = messages.map((m, i) => m.role === 'user' ? i : -1).filter(i => i >= 0).pop();
   980→        if (lastUserIdx !== undefined && lastUserIdx >= 0) {
   981→          messages[lastUserIdx] = {
   982→            ...messages[lastUserIdx],
   983→            content: messages[lastUserIdx].content + content
   984→          };
   985→        }
   986→        break;
   987→      }
   988→      case 2: { // Before last user message
   989→        const lastUserIdx2 = messages.map((m, i) => m.role === 'user' ? i : -1).filter(i => i >= 0).pop();
   990→        if (lastUserIdx2 !== undefined && lastUserIdx2 >= 0) {
   991→          messages[lastUserIdx2] = {
   992→            ...messages[lastUserIdx2],
   993→            content: content + messages[lastUserIdx2].content
   994→          };
   995→        }
   996→        break;
   997→      }
   998→      case 3: { // After last assistant message
   999→        const lastAsstIdx = messages.map((m, i) => m.role === 'assistant' ? i : -1).filter(i => i >= 0).pop();
  1000→        if (lastAsstIdx !== undefined && lastAsstIdx >= 0) {
  1001→          messages[lastAsstIdx] = {
  1002→            ...messages[lastAsstIdx],
  1003→            content: messages[lastAsstIdx].content + content
  1004→          };
  1005→        }
  1006→        break;
  1007→      }
  1008→      case 4: { // Before last assistant message
  1009→        const lastAsstIdx4 = messages.map((m, i) => m.role === 'assistant' ? i : -1).filter(i => i >= 0).pop();
  1010→        if (lastAsstIdx4 !== undefined && lastAsstIdx4 >= 0) {
  1011→          messages[lastAsstIdx4] = {
  1012→            ...messages[lastAsstIdx4],
  1013→            content: content + messages[lastAsstIdx4].content
  1014→          };
  1015→        }
  1016→        break;
  1017→      }
  1018→    }
  1019→  }
  1020→}

  1022→/**
  1023→ * Build messages array for chat models
  1024→ * 
  1025→ * Order (SillyTavern style):
  1026→ * 1. System message (system prompt)
  1027→ * 2. Chat history
  1028→ * 3. Author's Note (injected AFTER chat history, as system message)
  1029→ * 4. Post-History Instructions (injected AFTER Author's Note, as system message)
  1030→ */
  1031→export function buildChatMessages(
  1032→  systemPrompt: string,
  1033→  messages: ChatMessage[],
  1034→  character: CharacterCard,
  1035→  userName: string = 'User',
  1036→  postHistoryInstructions?: string,
  1037→  authorNote?: string,
  1038→  useSystemRole: boolean = false,
  1039→  embeddingsContext?: string,  // embeddings injected before chat history
  1040→  lorebookChatInjections?: LorebookChatInjection[],  // positions 1-4: inject into specific messages
  1041→  exampleMessages?: ChatApiMessage[]  // SillyTavern-style example dialogue as chat messages
  1042→): ChatApiMessage[] {
  1043→  // =============================================
  1044→  // Step 1: Build all system content as ONE message
  1045→  // =============================================
  1046→  // Consolidate system prompt + embeddings + author note + post-history
  1047→  // into a single system message to avoid consecutive system messages
  1048→  // which break Jinja templates in models like LM Studio.
  1049→  const systemParts: string[] = [];
  1050→
  1051→  // Main system prompt
  1052→  if (systemPrompt.trim()) {
  1053→    systemParts.push(systemPrompt);
  1054→  }
  1055→
  1056→  // Embeddings Context
  1057→  if (embeddingsContext?.trim()) {
  1058→    systemParts.push(embeddingsContext);
  1059→  }
  1060→
  1061→  // Author's Note
  1062→  if (authorNote?.trim()) {
  1063→    systemParts.push(`[Author's Note]\n${authorNote}`);
  1064→  }
  1065→
  1066→  // Post-History Instructions
  1067→  if (postHistoryInstructions?.trim()) {
  1068→    systemParts.push(postHistoryInstructions);
  1069→  }
  1070→
  1071→  const chatMessages: ChatApiMessage[] = [];
  1072→
  1073→  // Single system message (or assistant if useSystemRole is false)
  1074→  if (systemParts.length > 0) {
  1075→    chatMessages.push({
  1076→      role: useSystemRole ? 'system' : 'assistant',
  1077→      content: systemParts.join('\n\n---\n\n')
  1078→    });
  1079→  }
  1080→
  1081→  // =============================================
  1082→  // Step 2: Inject example dialogue as chat messages (SillyTavern style)
  1083→  // =============================================
  1084→  // These go between the system message and the actual chat history
  1085→  // They serve as few-shot examples for the LLM
  1086→  if (exampleMessages && exampleMessages.length > 0) {
  1087→    chatMessages.push(...exampleMessages);
  1088→  }
  1089→
  1090→  // =============================================
  1091→  // Step 3: Build chat history with proper alternation
  1092→  // =============================================
  1093→  // Merge consecutive same-role messages and ensure
  1094→  // strict user/assistant alternation for OpenAI-compatible APIs.
  1095→  const visibleMessages = messages.filter(m => !m.isDeleted && !m.isNarratorMessage);
  1096→
  1097→  // First pass: merge consecutive same-role messages
  1098→  const mergedMessages: ChatApiMessage[] = [];
  1099→  for (const msg of visibleMessages) {
  1100→    const role = msg.role === 'user' ? 'user' : 'assistant';
  1101→    const last = mergedMessages[mergedMessages.length - 1];
  1102→    if (last && last.role === role) {
  1103→      // Merge into previous message
  1104→      last.content += '\n' + msg.content;
  1105→    } else {
  1106→      mergedMessages.push({ role, content: msg.content });
  1107→    }
  1108→  }
  1109→
  1110→  // Second pass: enforce user/assistant alternation
  1111→  // If the first chat message is 'assistant', prepend a synthetic user message
  1112→  if (mergedMessages.length > 0 && mergedMessages[0].role === 'assistant') {
  1113→    mergedMessages.unshift({ role: 'user', content: '*continúa*' });
  1114→  }
  1115→
  1116→  // If two same-role messages are still adjacent (shouldn't happen after merge,
  1117→  // but as safety), insert a bridging message
  1118→  const finalHistory: ChatApiMessage[] = [];
  1119→  for (const msg of mergedMessages) {
  1120→    const last = finalHistory[finalHistory.length - 1];
  1121→    if (last && last.role === msg.role) {
  1122→      // Insert bridging message
  1123→      const bridgeRole = msg.role === 'user' ? 'assistant' : 'user';
  1124→      finalHistory.push({ role: bridgeRole, content: '*continúa*' });
  1125→    }
  1126→    finalHistory.push(msg);
  1127→  }
  1128→
  1129→  // Bridge between example messages and chat history if needed
  1130→  // If the last example message has the same role as the first chat message,
  1131→  // insert a bridging message to maintain proper user/assistant alternation
  1132→  if (finalHistory.length > 0 && chatMessages.length > 0) {
  1133→    const lastExistingMsg = chatMessages[chatMessages.length - 1];
  1134→    const firstChatMsg = finalHistory[0];
  1135→    if (lastExistingMsg.role === firstChatMsg.role) {
  1136→      const bridgeRole = firstChatMsg.role === 'user' ? 'assistant' : 'user';
  1137→      chatMessages.push({ role: bridgeRole, content: '*continúa*' });
  1138→    }
  1139→  }
  1140→
  1141→  chatMessages.push(...finalHistory);
  1142→
  1143→  // =============================================
  1144→  // Step 4: Inject lorebook chat-level content (positions 1-4)
  1145→  // =============================================
  1146→  if (lorebookChatInjections?.length) {
  1147→    applyChatInjections(chatMessages, lorebookChatInjections);
  1148→  }
  1149→
  1150→  return chatMessages;
  1151→}

  1153→/**
  1154→ * Build prompt for completion-style APIs (Ollama, KoboldCPP, etc.)
  1155→ * 
  1156→ * Order (SillyTavern style):
  1157→ * 1. System prompt
  1158→ * 2. Chat history
  1159→ * 3. Author's Note
  1160→ * 4. Post-History Instructions
  1161→ * 5. Assistant prefix
  1162→ */
  1163→export function buildCompletionPrompt(config: CompletionPromptConfig): string {
  1164→  const { systemPrompt, messages, character, userName, postHistoryInstructions, authorNote, embeddingsContext, exampleMessages, allCharacters } = config;
  1165→  const parts: string[] = [];
  1166→
  1167→  parts.push(systemPrompt);
  1168→  parts.push('\n---\n');
  1169→
  1170→  // Embeddings Context - injected before chat history for recency primacy
  1171→  if (embeddingsContext?.trim()) {
  1172→    parts.push(embeddingsContext);
  1173→    parts.push('\n---\n');
  1174→  }
  1175→
  1176→  // Example dialogue is now included in the systemPrompt string (as [EJEMPLO] sections).
  1177→  // The exampleMessages parameter is kept for backward compatibility but will always be empty.
  1178→  // No separate injection needed here.
  1179→
  1180→  // Exclude narrator messages from prompt
  1181→  const visibleMessages = messages.filter(m => !m.isDeleted && !m.isNarratorMessage);
  1182→
  1183→  for (const msg of visibleMessages) {
  1184→    if (msg.role === 'user') {
  1185→      parts.push(`${userName}: ${msg.content}`);
  1186→    } else if (msg.role === 'assistant') {
  1187→      // Use actual speaker name from characterId (important for group chats where
  1188→      // multiple characters speak - each should be attributed to their own name)
  1189→      const speakerName = msg.characterId
  1190→        ? (allCharacters?.find(c => c.id === msg.characterId)?.name || character.name)
  1191→        : character.name;
  1192→      parts.push(`${speakerName}: ${msg.content}`);
  1193→    }
  1194→  }
  1195→
  1196→  // Author's Note - injected AFTER chat history, BEFORE post-history instructions
  1197→  if (authorNote?.trim()) {
  1198→    parts.push(`\n[Author's Note]\n${authorNote}`);
  1199→  }
  1200→
  1201→  // Post-History Instructions - injected AFTER Author's Note
  1202→  if (postHistoryInstructions) {
  1203→    parts.push(`\n${postHistoryInstructions}`);
  1204→  }
  1205→
  1206→  parts.push(`\n${character.name}:`);
  1207→
  1208→  return parts.join('\n');
  1209→}

  1559→ */
  1560→export function buildGroupChatMessages(
  1561→  systemPrompt: string,
  1562→  messages: ChatMessage[],
  1563→  character: CharacterCard,
  1564→  allCharacters: CharacterCard[],
  1565→  userName: string = 'User',
  1566→  previousResponses?: Array<{ characterName: string; content: string }>,
  1567→  postHistoryInstructions?: string,
  1568→  authorNote?: string,
  1569→  isForNarrator: boolean = false,
  1570→  embeddingsContext?: string,  // embeddings injected before chat history
  1571→  lorebookChatInjections?: LorebookChatInjection[],  // positions 1-4: inject into specific messages
  1572→  exampleMessages?: ChatApiMessage[]  // SillyTavern-style example dialogue as chat messages
  1573→): GroupPromptBuildResult {
  1574→  // =============================================
  1575→  // Step 1: Build all system content as ONE message
  1576→  // =============================================
  1577→  const systemParts: string[] = [];
  1578→
  1579→  if (systemPrompt.trim()) {
  1580→    systemParts.push(systemPrompt);
  1581→  }
  1582→  if (embeddingsContext?.trim()) {
  1583→    systemParts.push(embeddingsContext);
  1584→  }
  1585→  if (authorNote?.trim()) {
  1586→    systemParts.push(`[Author's Note]\n${authorNote}`);
  1587→  }
  1588→  if (postHistoryInstructions?.trim()) {
  1589→    systemParts.push(postHistoryInstructions);
  1590→  }
  1591→
  1592→  const chatMessages: ChatApiMessage[] = [];
  1593→
  1594→  // Single system/assistant message
  1595→  if (systemParts.length > 0) {
  1596→    chatMessages.push({
  1597→      role: 'assistant',
  1598→      content: systemParts.join('\n\n---\n\n')
  1599→    });
  1600→  }
  1601→
  1602→  // =============================================
  1603→  // Step 2: Inject example dialogue as chat messages (SillyTavern style)
  1604→  // =============================================
  1605→  // These go between the system message and the actual chat history
  1606→  // They serve as few-shot examples for the LLM
  1607→  if (exampleMessages && exampleMessages.length > 0) {
  1608→    chatMessages.push(...exampleMessages);
  1609→  }
  1610→
  1611→  // =============================================
  1612→  // Step 3: Build chat history with proper alternation
  1613→  // =============================================
  1614→  const visibleMessages = isForNarrator
  1615→    ? messages.filter(m => !m.isDeleted)
  1616→    : messages.filter(m => !m.isDeleted && !m.isNarratorMessage);
  1617→
  1618→  const historyLines: string[] = [];
  1619→
  1620→  // Build history lines + API messages
  1621→  for (const msg of visibleMessages) {
  1622→    const speaker = msg.role === 'user' ? userName :
  1623→      (allCharacters.find(c => c.id === msg.characterId)?.name || 'Character');
  1624→    historyLines.push(`${speaker}: ${msg.content}`);
  1625→  }
  1626→
  1627→  // Build merged API messages with proper alternation
  1628→  // IMPORTANT: In group chats, multiple characters speak as 'assistant' role.
  1629→  // When merging consecutive assistant messages, we MUST include speaker names
  1630→  // so the LLM can distinguish which character said what.
  1631→  const mergedMessages: ChatApiMessage[] = [];
  1632→  for (const msg of visibleMessages) {
  1633→    const role = msg.role === 'user' ? 'user' : 'assistant';
  1634→    // Include speaker name for assistant messages so different characters' lines
  1635→    // are distinguishable even when merged into the same message
  1636→    const speakerName = msg.role === 'user' ? userName :
  1637→      (allCharacters.find(c => c.id === msg.characterId)?.name || character.name);
  1638→    const contentWithSpeaker = msg.role === 'user' ? msg.content : `${speakerName}: ${msg.content}`;
  1639→    const last = mergedMessages[mergedMessages.length - 1];
  1640→    if (last && last.role === role) {
  1641→      last.content += '\n' + contentWithSpeaker;
  1642→    } else {
  1643→      mergedMessages.push({ role, content: contentWithSpeaker });
  1644→    }
  1645→  }
  1646→
  1647→  // Add previous responses from this turn
  1648→  if (previousResponses && previousResponses.length > 0) {
  1649→    for (const resp of previousResponses) {
  1650→      historyLines.push(`${resp.characterName}: ${resp.content}`);
  1651→      const last = mergedMessages[mergedMessages.length - 1];
  1652→      const contentWithSpeaker = `${resp.characterName}: ${resp.content}`;
  1653→      if (last && last.role === 'assistant') {
  1654→        last.content += '\n' + contentWithSpeaker;
  1655→      } else {
  1656→        mergedMessages.push({ role: 'assistant', content: contentWithSpeaker });
  1657→      }
  1658→    }
  1659→  }
  1660→
  1661→  // Enforce alternation: if first is assistant, prepend synthetic user
  1662→  if (mergedMessages.length > 0 && mergedMessages[0].role === 'assistant') {
  1663→    mergedMessages.unshift({ role: 'user', content: '*continúa*' });
  1664→  }
  1665→
  1666→  // Safety: insert bridging messages if any same-role adjacency remains
  1667→  const finalHistory: ChatApiMessage[] = [];
  1668→  for (const msg of mergedMessages) {
  1669→    const last = finalHistory[finalHistory.length - 1];
  1670→    if (last && last.role === msg.role) {
  1671→      const bridgeRole = msg.role === 'user' ? 'assistant' : 'user';
  1672→      finalHistory.push({ role: bridgeRole, content: '*continúa*' });
  1673→    }
  1674→    finalHistory.push(msg);
  1675→  }
  1676→
  1677→  // Bridge between example messages and chat history if needed
  1678→  if (finalHistory.length > 0 && chatMessages.length > 0) {
  1679→    const lastExistingMsg = chatMessages[chatMessages.length - 1];
  1680→    const firstChatMsg = finalHistory[0];
  1681→    if (lastExistingMsg.role === firstChatMsg.role) {
  1682→      const bridgeRole = firstChatMsg.role === 'user' ? 'assistant' : 'user';
  1683→      chatMessages.push({ role: bridgeRole, content: '*continúa*' });
  1684→    }
  1685→  }
  1686→
  1687→  chatMessages.push(...finalHistory);
  1688→
  1689→  // Inject lorebook chat-level content (positions 1-4)
  1690→  if (lorebookChatInjections?.length) {
  1691→    applyChatInjections(chatMessages, lorebookChatInjections);
  1692→  }
  1693→
  1694→  // Build chat history section for prompt viewer
  1695→  let chatHistorySection: PromptSection | undefined;
  1696→  if (historyLines.length > 0) {
  1697→    chatHistorySection = {
  1698→      type: 'chat_history',
  1699→      label: 'Chat History',
  1700→      content: historyLines.join('\n\n'),
  1701→      color: SECTION_COLORS.chat_history
  1702→    };
  1703→  }
  1704→
  1705→  return {
  1706→    systemPrompt,
  1707→    sections: [],
  1708→    chatMessages,
  1709→    chatHistorySection
  1710→  };
  1711→}

  1812→/**
  1813→ * Build character memory section
  1814→ *
  1815→ * EVENTS ARE CAPPED: only the top `maxEvents` (default 20) most important +
  1816→ * most recent events are injected. Without this cap, the section grows
  1817→ * unboundedly as automatic extraction adds events every few turns, making
  1818→ * each request larger than the last.
  1819→ *
  1820→ * Sort priority: importance (desc) → timestamp (desc, most recent first).
  1821→ */
  1822→export function buildMemorySection(
  1823→  memory: CharacterMemory,
  1824→  characterName: string,
  1825→  maxEvents: number = 20,
  1826→): PromptSection | null {
  1827→  if (!memory.events.length && !memory.relationships.length && !memory.notes) {
  1828→    return null;
  1829→  }
  1830→
  1831→  // Normalize importance (support both old 0-1 and new 1-5 scales)
  1832→  const normalizeImportance = (imp: number) => (imp > 1 ? imp : Math.round(imp * 5));
  1833→
  1834→  const parts: string[] = [];
  1835→
  1836→  // Add events (capped: top N by importance, then recency)
  1837→  if (memory.events.length > 0) {
  1838→    const eventsCap = Math.max(2, Math.floor(maxEvents));
  1839→    const sortedEvents = [...memory.events].sort((a, b) => {
  1840→      const impA = normalizeImportance(a.importance);
  1841→      const impB = normalizeImportance(b.importance);
  1842→      if (impB !== impA) return impB - impA;
  1843→      // Same importance → most recent first
  1844→      return (b.timestamp || '').localeCompare(a.timestamp || '');
  1845→    });
  1846→    const includedEvents = sortedEvents.slice(0, eventsCap);
  1847→    const omitted = sortedEvents.length - includedEvents.length;
  1848→
  1849→    parts.push(`[Eventos y hechos clave]`);
  1850→    for (const event of includedEvents) {
  1851→      const normalizedImportance = normalizeImportance(event.importance);
  1852→      const importance = normalizedImportance >= 4 ? '⭐' : '';
  1853→      parts.push(`${importance} ${event.content}`);
  1854→    }
  1855→    if (omitted > 0) {
  1856→      parts.push(`(... ${omitted} eventos de menor relevancia omitidos)`);
  1857→    }
  1858→  }
  1859→
  1860→  // Add relationships (capped at 12, most extreme sentiment first)
  1861→  if (memory.relationships.length > 0) {
  1862→    parts.push(`\n[Relaciones]`);
  1863→    const sortedRels = [...memory.relationships].sort((a, b) => Math.abs(b.sentiment) - Math.abs(a.sentiment));
  1864→    const includedRels = sortedRels.slice(0, 12);
  1865→    for (const rel of includedRels) {
  1866→      const sentiment = rel.sentiment > 50 ? '😊' : rel.sentiment < -50 ? '😞' : '😐';
  1867→      parts.push(`${sentiment} ${rel.targetName}: ${rel.relationship} (${rel.sentiment >= 0 ? '+' : ''}${rel.sentiment})`);
  1868→    }
  1869→  }
  1870→
  1871→  // Add notes
  1872→  if (memory.notes) {
  1873→    parts.push(`\n[Notas]\n${memory.notes}`);
  1874→  }
  1875→
  1876→  return {
  1877→    type: 'character_note',
  1878→    label: `Memoria de ${characterName}`,
  1879→    content: parts.join('\n'),
  1880→    color: SECTION_COLORS.memory
  1881→  };
  1882→}
