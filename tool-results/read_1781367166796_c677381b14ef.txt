    80→const COMPARATOR_LABELS: Record<AttributeComparator, string> = {
    81→  '==': 'Igual a (=)',
    82→  '!=': 'Diferente de (!=)',
    83→  '<': 'Menor que (<)',
    84→  '<=': 'Menor o igual (<=)',

    86→  '>=': 'Mayor o igual (>=)',
    87→  'contains': 'Contiene (sin mayúsc.)',
    88→  'not_contains': 'No contiene (sin mayúsc.)',
    89→};
    90→
    91→// Operators recommended for numeric attributes
    92→const NUMERIC_COMPARATORS: AttributeComparator[] = [
    93→  '==', '!=', '<', '<=', '>', '>=',
    94→];
    95→
    96→// Operators recommended for text/keyword attributes
    97→const TEXT_COMPARATORS: AttributeComparator[] = [
    98→  '==', '!=', 'contains', 'not_contains',
    99→];
   100→
   101→/**
   102→ * Returns the appropriate comparators based on attribute type.
   103→ * Falls back to showing all if type is unknown.
   104→ */
   105→function getComparatorsForType(attrType?: string): AttributeComparator[] {
   106→  if (attrType === 'number') return NUMERIC_COMPARATORS;
   107→  if (attrType === 'keyword' || attrType === 'text') return TEXT_COMPARATORS;
   108→  // Unknown type — show all
   109→  return Object.keys(COMPARATOR_LABELS) as AttributeComparator[];
   110→}

   571→                                    <span className="text-[10px] text-muted-foreground truncate">
   572→                                      {entry.key.slice(0, 5).join(', ')}{entry.key.length > 5 ? '...' : ''}
   573→                                    </span>
   574→                                  </div>
   575→                                ) : null}
   576→                              </div>
   577→                              <div className="flex items-center gap-1.5 text-xs text-muted-foreground shrink-0">
   578→                                {entry.entryType === 'attribute' ? (
   579→                                  <Badge variant="outline" className="text-[9px] px-1 py-0 h-4 bg-orange-500/10 text-orange-600 border-orange-500/20">
   580→                                    {entry.attributeConfig?.injectionKey ? `{{${entry.attributeConfig.injectionKey}}}` : 'Atributo'}
   581→                                  </Badge>
   582→                                ) : (
   583→                                  <span className="bg-muted px-1.5 py-0.5 rounded">
   584→                                    {entry.key.length} claves
   585→                                  </span>
   586→                                )}
   587→                                {!entry.disable && entry.entryType !== 'attribute' && (
   588→                                  <Badge variant="outline" className="text-[9px] px-1 py-0 h-4 bg-green-500/10 text-green-600 border-green-500/20">
   589→                                    {POSITION_LABELS[entry.position] || `Pos.${entry.position}`}
   590→                                  </Badge>
   591→                                )}
   592→                              </div>
   593→                              <ChevronRight className="w-4 h-4 text-muted-foreground/50 group-hover:text-muted-foreground transition-colors" />
   594→                            </button>
   595→                          ))}
   596→                        </div>
   597→                      )}
   598→                    </div>
   599→                  </div>
   600→                </div>
   601→              </div>
   602→            </div>
   603→          ) : (
   604→            <div className="flex-1 flex items-center justify-center border rounded-lg text-muted-foreground">
   605→              <div className="text-center">
   606→                <BookOpen className="w-12 h-12 mx-auto mb-2 opacity-50" />
   607→                <p>Selecciona un lorebook para editar</p>
   608→              </div>
   609→            </div>
   610→          )}
   611→        </div>
   612→      )}
   613→
   614→      {/* Import Dialog */}
   615→      <Dialog open={importDialogOpen} onOpenChange={setImportDialogOpen}>
   616→        <DialogContent>
   617→          <DialogHeader>
   618→            <DialogTitle>Importar Lorebook</DialogTitle>
   619→          </DialogHeader>
   620→          {importError ? (
   621→            <div className="p-4 rounded-lg bg-destructive/10 text-destructive flex items-center gap-2">
   622→              <AlertCircle className="w-5 h-5" />
   623→              <div>
   624→                <p className="font-medium">Error al importar</p>
   625→                <p className="text-sm">{importError}</p>
   626→              </div>
   627→            </div>
   628→          ) : (
   629→            <div className="space-y-4">
   630→              <div>
   631→                <Label>Nombre del Lorebook</Label>
   632→                <Input
   633→                  value={importName}
   634→                  onChange={(e) => setImportName(e.target.value)}
   635→                  placeholder="Mi Lorebook"
   636→                  className="mt-1"
   637→                />
   638→              </div>
   639→              <div>
   640→                <Label>Descripción (opcional)</Label>
   641→                <Textarea
   642→                  value={importDescription}
   643→                  onChange={(e) => setImportDescription(e.target.value)}
   644→                  placeholder="Descripción del lorebook..."
   645→                  rows={3}
   646→                  className="mt-1"
   647→                />
   648→              </div>
   649→              {pendingImport && (
   650→                <div className="p-3 rounded-lg bg-muted text-sm">
   651→                  <p>
   652→                    <strong>{Object.keys(pendingImport.entries).length}</strong> entradas detectadas
   653→                  </p>
   654→                </div>
   655→              )}
   656→            </div>
   657→          )}
   658→          <DialogFooter>
   659→            <Button variant="outline" onClick={() => setImportDialogOpen(false)}>
   660→              Cancelar
   661→            </Button>
   662→            {!importError && (
   663→              <Button onClick={handleConfirmImport} disabled={!importName.trim()}>
   664→                Importar
   665→              </Button>
   666→            )}
   667→          </DialogFooter>
   668→        </DialogContent>
   669→      </Dialog>
   670→    </div>
   671→  );
   672→}

   674→// ============================================
   675→// Entry Editor Component
   676→// ============================================
   677→
   678→interface LorebookEntryEditorProps {
   679→  entry: LorebookEntry;
   680→  lorebookId: string;
   681→  characters: Array<{ id: string; name: string; statsConfig?: { attributes?: Array<{ key: string; name: string }> } }>;
   682→  activePersona?: { id: string; name: string; statsConfig?: { attributes?: Array<{ key: string; name: string }> } } | null;
   683→  onUpdate: (updates: Partial<LorebookEntry>) => void;
   684→  onDelete: () => void;
   685→  onDuplicate: () => void;
   686→}
   687→
   688→function LorebookEntryEditor({
   689→  entry,
   690→  characters,
   691→  activePersona,
   692→  onUpdate,
   693→  onDelete,
   694→  onDuplicate
   695→}: LorebookEntryEditorProps) {
   696→  const isAttribute = entry.entryType === 'attribute';
   697→  const config = entry.attributeConfig;
   698→
   699→  const parseKeywords = (value: string): string[] => {
   700→    return value
   701→      .split(',')
   702→      .map(k => k.trim())
   703→      .filter(k => k.length > 0);
   704→  };
   705→
   706→  const formatKeywords = (keywords: string[]): string => {
   707→    return keywords.join(', ');
   708→  };
   709→
   710→  // Get available attribute definitions based on selected character
   711→  const availableAttributes = useMemo(() => {
   712→    if (!isAttribute || !config) return [];
   713→
   714→    if (config.characterId === '__user__') {
   715→      return activePersona?.statsConfig?.attributes || [];
   716→    }
   717→
   718→    if (config.characterId === '__char__') {
   719→      // Return empty — will be resolved at runtime
   720→      return [];
   721→    }
   722→
   723→    const char = characters.find(c => c.id === config.characterId);
   724→    return char?.statsConfig?.attributes || [];
   725→  }, [isAttribute, config, characters, activePersona]);
   726→
   727→  // Determine the type of the currently selected attribute and filter comparators
   728→  const selectedAttributeType = useMemo(() => {
   729→    if (!config?.attributeKey || availableAttributes.length === 0) return undefined;
   730→    const attr = availableAttributes.find(a => a.key === config.attributeKey);
   731→    return attr?.type;
   732→  }, [config, availableAttributes]);
   733→
   734→  const filteredComparators = useMemo(() => {
   735→    return getComparatorsForType(selectedAttributeType);
   736→  }, [selectedAttributeType]);
   737→
   738→  // Auto-fix operator when switching attribute type (e.g., was using '<' on text attr)
   739→  const sanitizeOperator = useCallback((operator: AttributeComparator): AttributeComparator => {
   740→    if (filteredComparators.includes(operator)) return operator;
   741→    // Reset to a safe default for the type
   742→    return selectedAttributeType === 'number' ? '>' : '==';
   743→  }, [filteredComparators, selectedAttributeType]);
   744→
   745→  // Handle entry type change
   746→  const handleEntryTypeChange = (type: LorebookEntryType) => {
   747→    if (type === 'attribute') {
   748→      // Initialize default attribute config with injectionKey
   749→      const newConfig: LorebookAttributeConfig = {
   750→        characterId: '__user__',
   751→        attributeKey: '',
   752→        injectionKey: '',
   753→        mode: 'static',
   754→        staticCondition: { operator: '>', value: 0 },
   755→        dynamicConditions: [],
   756→      };
   757→      onUpdate({
   758→        entryType: 'attribute',
   759→        attributeConfig: newConfig,
   760→        constant: false,
   761→        key: [],
   762→        keysecondary: [],
   763→      });
   764→    } else {
   765→      // Switch to traditional
   766→      onUpdate({
   767→        entryType: 'traditional',
   768→        attributeConfig: undefined,
   769→      });
   770→    }
   771→  };
   772→
   773→  // Handle attribute config changes
   774→  const updateAttributeConfig = (partial: Partial<LorebookAttributeConfig>) => {
   775→    const currentConfig = config || {
   776→      characterId: '__user__',
   777→      attributeKey: '',
   778→      injectionKey: '',
   779→      mode: 'static' as AttributeEntryMode,
   780→      staticCondition: { operator: '>' as AttributeComparator, value: 0 },
   781→    };
   782→    onUpdate({ attributeConfig: { ...currentConfig, ...partial } });
   783→  };
   784→
   785→  // Handle mode change
   786→  const handleModeChange = (mode: AttributeEntryMode) => {
   787→    // Use a type-appropriate default operator
   788→    const defaultOp = selectedAttributeType === 'number' ? '>' : '==';
   789→    if (mode === 'static') {
   790→      updateAttributeConfig({
   791→        mode: 'static',
   792→        staticCondition: { operator: defaultOp, value: selectedAttributeType === 'number' ? 0 : '' },
   793→        dynamicConditions: undefined,
   794→        fallbackContent: undefined,
   795→      });
   796→    } else {
   797→      updateAttributeConfig({
   798→        mode: 'dynamic',
   799→        staticCondition: undefined,
   800→        dynamicConditions: [
   801→          { id: simpleId(), operator: defaultOp, value: selectedAttributeType === 'number' ? 0 : '', content: '' }
   802→        ],
   803→        fallbackContent: '',
   804→      });
   805→    }
   806→  };
   807→
   808→  // Handle static condition changes
   809→  const updateStaticCondition = (operator: AttributeComparator, value: number | string) => {
   810→    updateAttributeConfig({ staticCondition: { operator, value } });
   811→  };
   812→
   813→  // Handle dynamic condition changes
   814→  const updateDynamicCondition = (condId: string, partial: Partial<{ operator: AttributeComparator; value: number | string; content: string }>) => {
   815→    const conds = config?.dynamicConditions || [];
   816→    updateAttributeConfig({
   817→      dynamicConditions: conds.map(c => c.id === condId ? { ...c, ...partial } : c),
   818→    });
   819→  };
   820→
   821→  const addDynamicCondition = () => {
   822→    const conds = config?.dynamicConditions || [];
   823→    updateAttributeConfig({
   824→      dynamicConditions: [...conds, { id: simpleId(), operator: '>' as AttributeComparator, value: 0, content: '' }],
   825→    });
   826→  };
   827→
   828→  const removeDynamicCondition = (condId: string) => {
   829→    const conds = (config?.dynamicConditions || []).filter(c => c.id !== condId);
   830→    updateAttributeConfig({ dynamicConditions: conds.length > 0 ? conds : [{ id: simpleId(), operator: '>', value: 0, content: '' }] });
   831→  };
   832→
   833→  const moveDynamicCondition = (condId: string, direction: 'up' | 'down') => {
   834→    const conds = config?.dynamicConditions;
   835→    if (!conds || conds.length < 2) return;
   836→    const idx = conds.findIndex(c => c.id === condId);
   837→    if (idx === -1) return;
   838→    const newIdx = direction === 'up' ? idx - 1 : idx + 1;
   839→    if (newIdx < 0 || newIdx >= conds.length) return;
   840→    const swapped = [...conds];
   841→    [swapped[idx], swapped[newIdx]] = [swapped[newIdx], swapped[idx]];
   842→    updateAttributeConfig({ dynamicConditions: swapped });
   843→  };
   844→
   845→  return (
   846→    <div className="space-y-3 pt-1">
   847→      {/* Row 1: Title, Entry Type, Active Toggle */}
   848→      <div className="grid grid-cols-[1fr_auto_auto_auto] gap-3">
   849→        <div>
   850→          <Label className="text-xs">Título</Label>
   851→          <Input
   852→            value={entry.comment}
   853→            onChange={(e) => onUpdate({ comment: e.target.value })}
   854→            className="mt-1 h-7 text-sm"
   855→          />
   856→        </div>
   857→        <div>
   858→          <Label className="text-xs">Tipo</Label>
   859→          <Select
   860→            value={entry.entryType || 'traditional'}
   861→            onValueChange={(v) => handleEntryTypeChange(v as LorebookEntryType)}
   862→          >
   863→            <SelectTrigger className="mt-1 h-7 text-xs w-[130px]">
   864→              <SelectValue />
   865→            </SelectTrigger>
   866→            <SelectContent>
   867→              <SelectItem value="traditional" className="text-xs">
   868→                <span className="flex items-center gap-1.5">
   869→                  <Key className="w-3 h-3" />
   870→                  Tradicional
   871→                </span>
   872→              </SelectItem>
   873→              <SelectItem value="attribute" className="text-xs">
   874→                <span className="flex items-center gap-1.5">
   875→                  <Sword className="w-3 h-3" />
   876→                  Atributo
   877→                </span>
   878→              </SelectItem>
   879→            </SelectContent>
   880→          </Select>
   881→        </div>
   882→        <div className="flex items-end pb-1">
   883→          <label className="flex items-center gap-1.5 text-xs cursor-pointer">
   884→            <Switch
   885→              checked={!entry.disable}
   886→              onCheckedChange={(checked) => onUpdate({ disable: !checked })}
   887→            />
   888→            Activo
   889→          </label>
   890→        </div>
   891→      </div>
   892→
   893→      {/* ============================================ */}
   894→      {/* TRADITIONAL ENTRY FIELDS                   */}
   895→      {/* ============================================ */}
   896→      {!isAttribute && (
   897→        <>
   898→          {/* Primary Keys */}
   899→          <div className="space-y-1.5">
   900→            <div className="flex items-center gap-1.5">
   901→              <Key className="w-3 h-3 text-primary" />
   902→              <Label className="text-xs">Palabras Clave Primarias</Label>
   903→            </div>
   904→            <Input
   905→              value={formatKeywords(entry.key)}
   906→              onChange={(e) => onUpdate({ key: parseKeywords(e.target.value) })}
   907→              placeholder="palabra1, palabra2... o /regex/i"
   908→              className="h-7 text-sm"
   909→            />
   910→            <p className="text-[10px] text-muted-foreground">
   911→              Soporta regex: /patrón/flags (ej: /(?:clima|lluvia)/i)
   912→            </p>
   913→          </div>
   914→
   915→          {/* Secondary Keys */}
   916→          <div className="space-y-1.5">
   917→            <div className="flex items-center gap-1.5">
   918→              <Key className="w-3 h-3 text-muted-foreground" />
   919→              <Label className="text-xs">Palabras Clave Secundarias</Label>
   920→            </div>
   921→            <Input
   922→              value={formatKeywords(entry.keysecondary)}
   923→              onChange={(e) => onUpdate({ keysecondary: parseKeywords(e.target.value) })}
   924→              placeholder="adicional1, adicional2..."
   925→              className="h-7 text-sm"
   926→            />
   927→          </div>
   928→        </>
   929→      )}
   930→
   931→      {/* ============================================ */}
   932→      {/* ATTRIBUTE ENTRY FIELDS                     */}
   933→      {/* ============================================ */}
   934→      {isAttribute && config && (
   935→        <div className="space-y-3 p-3 border rounded-lg bg-muted/30">
   936→          <div className="flex items-center gap-2 text-xs font-medium text-orange-600 dark:text-orange-400">
   937→            <Sword className="w-4 h-4" />
   938→            Configuración de Atributo
   939→          </div>
   940→
   941→          {/* Injection Key */}
   942→          <div className="space-y-1.5">
   943→            <div className="flex items-center gap-1.5">
   944→              <Key className="w-3 h-3 text-orange-500" />
   945→              <Label className="text-xs">Key de Inyección</Label>
   946→            </div>
   947→            <Input
   948→              value={config.injectionKey || ''}
   949→              onChange={(e) => updateAttributeConfig({ injectionKey: e.target.value.replace(/[^a-zA-Z0-9_]/g, '') })}
   950→              placeholder="estadoHeroe"
   951→              className="h-7 text-sm font-mono"
   952→            />
   953→            <p className="text-[10px] text-muted-foreground">
   954→              Usa <code className="font-mono">{'{{' + (config.injectionKey || 'miKey') + '}}'}</code> en la descripción, system prompt, etc. para inyectar este contenido.
   955→            </p>
   956→          </div>
   957→
   958→          {/* Character selector */}
   959→          <div className="grid grid-cols-2 gap-3">
   960→            <div>
   961→              <Label className="text-xs">Personaje</Label>
   962→              <Select
   963→                value={config.characterId}
   964→                onValueChange={(v) => updateAttributeConfig({ characterId: v, attributeKey: '' })}
   965→              >
   966→                <SelectTrigger className="mt-1 h-7 text-xs">
   967→                  <SelectValue placeholder="Seleccionar..." />
   968→                </SelectTrigger>
   969→                <SelectContent>
   970→                  <SelectItem value="__user__" className="text-xs">
   971→                    <span className="flex items-center gap-1.5">
   972→                      <User className="w-3 h-3" />
   973→                      Usuario (Persona)
   974→                    </span>
   975→                  </SelectItem>
   976→                  <SelectItem value="__char__" className="text-xs">
   977→                    <span className="flex items-center gap-1.5">
   978→                      <Users className="w-3 h-3" />
   979→                      Personaje actual
   980→                    </span>
   981→                  </SelectItem>
   982→                  {characters.map(c => (
   983→                    <SelectItem key={c.id} value={c.id} className="text-xs">
   984→                      {c.name}
   985→                    </SelectItem>
   986→                  ))}
   987→                </SelectContent>
   988→              </Select>
   989→            </div>
   990→
   991→            {/* Attribute selector */}
   992→            <div>
   993→              <Label className="text-xs">Atributo</Label>
   994→              <Select
   995→                value={config.attributeKey}
   996→                onValueChange={(v) => updateAttributeConfig({ attributeKey: v })}
   997→              >
   998→                <SelectTrigger className="mt-1 h-7 text-xs">
   999→                  <SelectValue placeholder="Seleccionar..." />
  1000→                </SelectTrigger>
  1001→                <SelectContent>
  1002→                  {availableAttributes.length === 0 && (
  1003→                    <div className="px-2 py-1.5 text-xs text-muted-foreground">
  1004→                      Sin atributos definidos
  1005→                    </div>
  1006→                  )}
  1007→                  {availableAttributes.map(attr => (
  1008→                    <SelectItem key={attr.key} value={attr.key} className="text-xs">
  1009→                      {attr.name || attr.key}
  1010→                    </SelectItem>
  1011→                  ))}
  1012→                </SelectContent>
  1013→              </Select>
  1014→            </div>
  1015→          </div>
  1016→
  1017→          {/* Comparison mode */}
  1018→          <div>
  1019→            <Label className="text-xs">Modo de comparación</Label>
  1020→            <div className="flex gap-2 mt-1">
  1021→              <Button
  1022→                size="sm"
  1023→                variant={config.mode === 'static' ? 'default' : 'outline'}
  1024→                className="h-7 text-xs"
  1025→                onClick={() => handleModeChange('static')}
  1026→              >
  1027→                Estático
  1028→              </Button>
  1029→              <Button
  1030→                size="sm"
  1031→                variant={config.mode === 'dynamic' ? 'default' : 'outline'}
  1032→                className="h-7 text-xs"
  1033→                onClick={() => handleModeChange('dynamic')}
  1034→              >
  1035→                Dinámico
  1036→              </Button>
  1037→            </div>
  1038→          </div>
  1039→
  1040→          {/* Static condition */}
  1041→          {config.mode === 'static' && config.staticCondition && (
  1042→            <div className="space-y-2 p-2.5 border rounded-md bg-background">
  1043→              <Label className="text-xs font-medium">Condición</Label>
  1044→              <p className="text-[10px] text-muted-foreground">
  1045→                Si se cumple, se inyecta el contenido de abajo
  1046→              </p>
  1047→              <div className="grid grid-cols-[1fr_auto_1fr] gap-2 items-end">
  1048→                <div>
  1049→                  <Label className="text-[10px]">
  1050→                    Comparador
  1051→                    {selectedAttributeType && (
  1052→                      <span className="ml-1 text-muted-foreground">
  1053→                        ({selectedAttributeType === 'number' ? 'numérico' : 'texto'})
  1054→                      </span>
  1055→                    )}
  1056→                  </Label>
  1057→                  <Select
  1058→                    value={sanitizeOperator(config.staticCondition.operator)}
  1059→                    onValueChange={(v) => updateStaticCondition(v as AttributeComparator, config.staticCondition!.value)}
  1060→                  >
  1061→                    <SelectTrigger className="mt-0.5 h-7 text-xs">
  1062→                      <SelectValue />
  1063→                    </SelectTrigger>
  1064→                    <SelectContent>
  1065→                      {filteredComparators.map((comp) => (
  1066→                        <SelectItem key={comp} value={comp} className="text-xs">
  1067→                          {COMPARATOR_LABELS[comp]}
  1068→                        </SelectItem>
  1069→                      ))}
  1070→                    </SelectContent>
  1071→                  </Select>
  1072→                </div>
  1073→                <div className="pb-1 text-xs text-muted-foreground font-medium">
  1074→                  valor
  1075→                </div>
  1076→                <div>
  1077→                  <Label className="text-[10px]">Valor</Label>
  1078→                  <Input
  1079→                    value={String(config.staticCondition.value)}
  1080→                    onChange={(e) => {
  1081→                      const v = e.target.value;
  1082→                      const num = parseFloat(v);
  1083→                      updateStaticCondition(config.staticCondition!.operator, isNaN(num) ? v : num);
  1084→                    }}
  1085→                    placeholder="0"
  1086→                    className="mt-0.5 h-7 text-xs"
  1087→                  />
  1088→                </div>
  1089→              </div>
  1090→            </div>
  1091→          )}
  1092→
  1093→          {/* Dynamic conditions */}
  1094→          {config.mode === 'dynamic' && config.dynamicConditions && (
  1095→            <div className="space-y-2">
  1096→              <div className="flex items-center justify-between">
  1097→                <Label className="text-xs font-medium">Condiciones</Label>
  1098→                <Button
  1099→                  size="sm"
  1100→                  variant="outline"
  1101→                  className="h-6 text-[10px]"
  1102→                  onClick={addDynamicCondition}
  1103→                >
  1104→                  <Plus className="w-3 h-3 mr-1" />
  1105→                  Agregar condición
  1106→                </Button>
  1107→              </div>
  1108→              <p className="text-[10px] text-muted-foreground">
  1109→                Cada condición puede coincidir simultáneamente. Los contenidos se concatenan por prioridad.
  1110→              </p>
  1111→
  1112→              {/* Resolution mode selector */}
  1113→              <div className="flex items-center gap-2 p-2 border rounded-md bg-muted/20">
  1114→                <Label className="text-[10px] whitespace-nowrap">Resolución:</Label>
  1115→                <div className="flex gap-1">
  1116→                  <Button
  1117→                    size="sm"
  1118→                    variant={(!config.dynamicResolution || config.dynamicResolution === 'concat-all') ? 'default' : 'outline'}
  1119→                    className="h-6 text-[10px] px-2"
  1120→                    onClick={() => updateAttributeConfig({ dynamicResolution: 'concat-all' })}
