    98→/**
    99→ * Execute detected tool calls, send SSE events, and return results.
   100→ * Returns { newContent: combined display messages, shouldContinue: true if tools were executed }
   101→ */
   102→async function executeToolCallsAndContinue(
   103→  toolCalls: NativeToolCall[],
   104→  availableTools: Array<{ id: string; name: string; label: string; icon: string }>,
   105→  currentRound: number,
   106→  maxRounds: number,
   107→  character: CharacterCard,
   108→  sessionId: string,
   109→  userName: string,
   110→  controller: { enqueue: (chunk: string) => void },
   111→  sessionQuests?: SessionQuestInstance[],
   112→  questTemplates?: QuestTemplate[],
   113→  statsConfig?: CharacterStatsConfig,
   114→  sessionStats?: SessionStats,
   115→  allCharacters?: CharacterCard[],
   116→  characterMemory?: CharacterMemory,
   117→  lorebooks?: Lorebook[],
   118→): Promise<{ newContent: string; shouldContinue: boolean; toolResults: Array<{ success: boolean; displayMessage: string }>; questActivations: QuestActivation[]; toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> }> {
   119→  if (toolCalls.length === 0 || currentRound >= maxRounds) {
   120→    return { newContent: '', shouldContinue: false, toolResults: [], questActivations: [], toolsUsed: [] };
   121→  }
   122→
   123→  const toolResults: Array<{ success: boolean; displayMessage: string }> = [];
   124→  const questActivations: QuestActivation[] = [];
   125→  const toolsUsed: Array<{ name: string; label: string; icon: string; success: boolean }> = [];
   126→  let allDisplayMessages = '';
   127→
   128→  for (let i = 0; i < toolCalls.length; i++) {
   129→    const tc = toolCalls[i];
   130→    const toolDef = availableTools.find(t => t.name === tc.name);
   131→    const callId = `${Date.now()}_${i}`;
   132→
   133→    // Send tool_call_start event
   134→    controller.enqueue(createSSEJSON({
   135→      type: 'tool_call_start',
   136→      toolName: tc.name,
   137→      toolLabel: toolDef?.label || tc.name,
   138→      toolIcon: toolDef?.icon || 'Wrench',
   139→      params: tc.arguments,
   140→      callId,
   141→    }));
   142→
   143→    console.log(`[Tools] Executing tool call: ${tc.name}`, tc.arguments);
   144→
   145→    // Execute the tool
   146→    const toolResult = await executeTool(
   147→      tc.name,
   148→      tc.arguments,
   149→      {
   150→        characterId: character.id,
   151→        characterName: character.name,
   152→        sessionId,
   153→        userName,
   154→        sessionQuests,
   155→        questTemplates,
   156→        statsConfig: character.statsConfig,
   157→        sessionStats,
   158→        allCharacters,
   159→        characterMemory,
   160→        lorebooks,
   161→      },
   162→    );
   163→
   164→    // Send tool_call_result event
   165→    controller.enqueue(createSSEJSON({
   166→      type: 'tool_call_result',
   167→      toolName: tc.name,
   168→      success: toolResult.success,
   169→      displayMessage: toolResult.displayMessage,
   170→      duration: toolResult.duration || 0,
   171→      callId,
   172→    }));
   173→
   174→    // Check for quest activation and send SSE event
   175→    if (toolResult.questActivation) {
   176→      const activation = toolResult.questActivation;
   177→      console.log(`[Tools] Quest activation from ${tc.name}:`, activation);
   178→      
   179→      controller.enqueue(createSSEJSON({
   180→        type: 'quest_activation',
   181→        toolName: tc.name,
   182→        activationType: activation.type,
   183→        key: activation.key,
   184→        metadata: activation.metadata,
   185→      }));
   186→      
   187→      questActivations.push(activation);
   188→    }
   189→
   190→    // Check for action/skill activation and send SSE event
   191→    if (toolResult.actionActivation) {
   192→      const action = toolResult.actionActivation;
   193→      console.log(`[Tools] Action activation from ${tc.name}:`, action.skillName);
   194→      
   195→      controller.enqueue(createSSEJSON({
   196→        type: 'action_activation',
   197→        toolName: tc.name,
   198→        skillId: action.skillId,
   199→        skillName: action.skillName,
   200→        skillDescription: action.skillDescription,
   201→        skillCompletedDescription: action.skillCompletedDescription,
   202→        activationCosts: action.activationCosts,
   203→        activationRewards: action.activationRewards,
   204→        characterId: action.characterId,
   205→      }));
   206→    }
   207→
   208→    // Check for relationship activation and send SSE event
   209→    if (toolResult.relationshipActivation) {
   210→      const rel = toolResult.relationshipActivation;
   211→      console.log(`[Tools] Relationship activation from ${tc.name}:`, rel.aName, '↔', rel.bName, rel.prevPoints, '→', rel.newPoints);
   212→
   213→      controller.enqueue(createSSEJSON({
   214→        type: 'relationship_activation',
   215→        toolName: tc.name,
   216→        aId: rel.aId,
   217→        aName: rel.aName,
   218→        bId: rel.bId,
   219→        bName: rel.bName,
   220→        prevPoints: rel.prevPoints,
   221→        newPoints: rel.newPoints,
   222→        reason: rel.reason,
   223→      }));
   224→    }
   225→
   226→    // Check for world-time activation and send SSE event
   227→    if (toolResult.timeActivation) {
   228→      const timeAct = toolResult.timeActivation;
   229→      console.log(`[Tools] Time activation from ${tc.name}:`, timeAct.type, timeAct.minutes || timeAct.hour || timeAct.season);
   230→
   231→      controller.enqueue(createSSEJSON({
   232→        type: 'time_activation',
   233→        toolName: tc.name,
   234→        activationType: timeAct.type,
   235→        minutes: timeAct.minutes,
   236→        hour: timeAct.hour,
   237→        minute: timeAct.minute,
   238→        season: timeAct.season,
   239→      }));
   240→    }
   241→
   242→    // Check for skill check activation and send SSE event
   243→    if (toolResult.checkActivation) {
   244→      const check = toolResult.checkActivation;
   245→      console.log(`[Tools] Skill check from ${tc.name}:`, check.statName, `${check.roll}${check.modifier >= 0 ? '+' : ''}${check.modifier}=${check.total} vs CD ${check.dc} → ${check.outcome}`);
   246→
   247→      controller.enqueue(createSSEJSON({
   248→        type: 'check_activation',
   249→        toolName: tc.name,
   250→        characterId: check.characterId,
   251→        characterName: check.characterName,
   252→        statName: check.statName,
   253→        statValue: check.statValue,
   254→        roll: check.roll,
   255→        modifier: check.modifier,
   256→        dc: check.dc,
   257→        total: check.total,
   258→        outcome: check.outcome,
   259→        outcomeLabel: check.outcomeLabel,
   260→        narrative: check.narrative,
   261→      }));
   262→    }
   263→
   264→    // Check for stat modification from modify_stat tool and send SSE event
   265→    if (toolResult.statActivation) {
   266→      const stat = toolResult.statActivation;
   267→      console.log(`[Tools] Stat activation from ${tc.name}:`, stat.attributeKey, stat.oldValue, '→', stat.newValue);
   268→      
   269→      controller.enqueue(createSSEJSON({
   270→        type: 'stat_activation',
   271→        toolName: tc.name,
   272→        characterId: stat.characterId,
   273→        attributeKey: stat.attributeKey,
   274→        attributeName: stat.attributeName,
   275→        attributeType: stat.attributeType,
   276→        oldValue: stat.oldValue,
   277→        newValue: stat.newValue,
   278→        reason: stat.reason,
   279→      }));
   280→    }
   281→
   282→    // Check for solicitud activation/completion and send SSE event
   283→    if (toolResult.solicitudActivation) {
   284→      const sol = toolResult.solicitudActivation;
   285→      console.log(`[Tools] Solicitud activation from ${tc.name}:`, sol.type, sol.solicitudKey);
   286→      
   287→      controller.enqueue(createSSEJSON({
   288→        type: 'solicitud_activation',
   289→        toolName: tc.name,
   290→        activationType: sol.type,
   291→        solicitudKey: sol.solicitudKey,
   292→        targetCharacterId: sol.targetCharacterId,
   293→        targetCharacterName: sol.targetCharacterName,
   294→        fromCharacterId: sol.fromCharacterId,
   295→        fromCharacterName: sol.fromCharacterName,
   296→        description: sol.description,
   297→        completionDescription: sol.completionDescription,
   298→        peticionKey: sol.peticionKey,
   299→      }));
   300→    }
   301→
   302→    // Check for memory activation (sync to client-side Character Memory)
   303→    if (toolResult.memoryActivation) {
   304→      const mem = toolResult.memoryActivation;
   305→      console.log(`[Tools] Memory activation from ${tc.name}:`, mem.type);
   306→      
   307→      controller.enqueue(createSSEJSON({
   308→        type: 'memory_activation',
   309→        toolName: tc.name,
   310→        activationType: mem.type,
   311→        characterId: mem.characterId,
   312→        eventData: mem.eventData,
   313→        relationshipData: mem.relationshipData,
   314→        noteContent: mem.noteContent,
   315→        deleteEventId: mem.deleteEventId,
   316→        deleteEmbeddingId: mem.deleteEmbeddingId,
   317→      }));
   318→    }
   319→
   320→    console.log(`[Tools] Tool ${tc.name}: success=${toolResult.success} duration=${toolResult.duration}ms`, toolResult.displayMessage);
   321→
   322→    toolResults.push({ success: toolResult.success, displayMessage: toolResult.displayMessage });
   323→    toolsUsed.push({
   324→      name: tc.name,
   325→      label: toolDef?.label || tc.name,
   326→      icon: toolDef?.icon || 'Wrench',
   327→      success: toolResult.success,
   328→    });
   329→    if (toolResult.displayMessage) {
   330→      allDisplayMessages += (allDisplayMessages ? '\n' : '') + toolResult.displayMessage;
   331→    }
   332→  }
   333→
   334→  // Return the display messages as content, tool results, and signal that a follow-up call is needed
   335→  return {
   336→    newContent: allDisplayMessages,
   337→    shouldContinue: true, // Always continue to get the LLM's natural response
   338→    toolResults,
   339→    questActivations,
   340→    toolsUsed,
   341→  };
   342→}

   344→export async function POST(request: NextRequest) {
   345→  try {
   346→    // Capture auth headers from Z.ai gateway for token resolution
   347→    // Only use actual JWT tokens - session IDs are NOT valid X-Tokens
   348→    const incomingXToken = request.headers.get('X-Token');
   349→    const fcSecurityToken = request.headers.get('x-fc-security-token');
   350→
   351→    const body = await request.json();
   352→
   353→    // Validate request (automatically detects request type)
   354→    const validation = validateRequest(null, body);
   355→    if (!validation.success) {
   356→      return createErrorResponse(validation.error, 400);
   357→    }
   358→
   359→    const {
   360→      message,
   361→      character,
   362→      messages = [],
   363→      llmConfig,
   364→      userName = 'User',
   365→      persona,
   366→      sessionStats,
   367→      allCharacters, // All characters for peticiones/solicitudes resolution
   368→    } = validation.data;
   369→
   370→    // Extract lorebooks from body (not validated by validation.ts)
   371→    const lorebooks: Lorebook[] = body.lorebooks || [];
   372→
   373→    // Extract HUD context from body
   374→    const hudContext: HUDContextConfig | undefined = body.hudContext;
   375→
   376→    // Extract Quest data for pre-LLM integration (NEW FORMAT)
   377→    const questTemplates: QuestTemplate[] = body.questTemplates || [];
   378→    const sessionQuests: SessionQuestInstance[] = body.sessionQuests || [];
   379→    const questSettings: QuestSettings = {
   380→      ...DEFAULT_QUEST_SETTINGS,
   381→      ...(body.questSettings || {})
   382→    };
   383→
   384→    // Extract Sound data for {{sonidos}} key resolution
   385→    const soundTriggers: SoundTrigger[] = body.soundTriggers || [];
   386→    const soundSettings = body.settings?.sound;
   387→
   388→    // Extract summary for memory/context compression (single summary from session)
   389→    const summary: SessionSummary | undefined = body.summary;
   390→    
   391→    // Extract embeddings chat settings for automatic context retrieval
   392→    const embeddingsChat: Partial<EmbeddingsChatSettings> = body.embeddingsChat || {};
   393→    const sessionId: string | undefined = body.sessionId;
   394→    const characterId: string | undefined = body.characterId;
   395→    
   396→    // Extract character memory (events, relationships, notes from Zustand store)
   397→    const characterMemory: CharacterMemory | undefined = body.characterMemory;
   398→    
   399→    // Extract tools settings for tool/action system (native tool calling only)
   400→    const toolsSettings: ToolsSettings = {
   401→      enabled: body.toolsSettings?.enabled ?? true,
   402→      maxToolCallsPerTurn: body.toolsSettings?.maxToolCallsPerTurn ?? 4,
   403→      characterConfigs: body.toolsSettings?.characterConfigs || [],
   404→      usePromptBasedFallback: body.toolsSettings?.usePromptBasedFallback ?? false,
   405→      disabledTools: body.toolsSettings?.disabledTools || [],
   406→    };
   407→
   408→    // Extract inventory data for Inventory V2 system
   409→    const inventoryData: InventoryPromptData | undefined = body.inventoryData;
   410→
   411→    if (!llmConfig) {
   412→      return createErrorResponse('No LLM configuration provided', 400);
   413→    }
   414→
   415→    // Validate API key for providers that require one
   416→    const providersRequiringApiKey = ['openai', 'vllm', 'lm-studio', 'custom', 'anthropic', 'grok'];
   417→    if (providersRequiringApiKey.includes(llmConfig.provider) && !llmConfig.apiKey) {
   418→      return createErrorResponse(
   419→        `API Key is required for ${llmConfig.provider} provider. Please configure it in Settings.`,
   420→        400
   421→      );
   422→    }
   423→
   424→    // Validate endpoint for providers that require one
   425→    const providersRequiringEndpoint = ['openai', 'vllm', 'lm-studio', 'custom', 'anthropic', 'ollama', 'text-generation-webui', 'koboldcpp'];
   426→    if (providersRequiringEndpoint.includes(llmConfig.provider) && !llmConfig.endpoint) {
   427→      return createErrorResponse(
   428→        `Endpoint URL is required for ${llmConfig.provider} provider. Please configure it in Settings.`,
   429→        400
   430→      );
   431→    }
   432→
   433→    // If using Z.ai provider, collect gateway-forwarded auth headers
   434→    // The SDK v0.0.17+ reads config from /etc/.z-ai-config and uses the "token" field for X-Token.
   435→    // Gateway headers are passed as runtime overrides to the provider.
   436→    let zaiRuntimeToken: string | undefined;
   437→    if (llmConfig.provider === 'z-ai') {
   438→      // Only use actual auth tokens (not session IDs)
   439→      const gatewayToken = incomingXToken || fcSecurityToken || undefined;
   440→      if (gatewayToken) {
   441→        zaiRuntimeToken = gatewayToken;
   442→        console.log(`[Stream Route] Z.ai runtime token available (${gatewayToken.length} chars, source: ${incomingXToken ? 'X-Token' : 'fc-security-token'})`);
   443→      } else {
   444→        console.log(`[Stream Route] Z.ai: no gateway token available, using config file only`);
   445→      }
   446→    }
   447→
   448→    // Sanitize user message
   449→    const sanitizedMessage = sanitizeInput(message);
   450→
   451→    // Create default character if none provided
   452→    const effectiveCharacter: CharacterCard = character || DEFAULT_CHARACTER;
   453→
   454→    // Get effective user name from persona or use provided userName
   455→    const effectiveUserName = getEffectiveUserName(persona, userName);
   456→
   457→    // Build context configuration from request or use defaults
   458→    const contextConfig: Partial<ContextConfig> = body.contextConfig || {};
   459→
   460→    // Apply sliding window to messages
   461→    const contextWindow = selectContextMessages(messages, llmConfig, contextConfig);
   462→
   463→    // Log context stats (for debugging)
   464→    const stats = getContextStats(messages);
   465→
   466→    // Process lorebooks and get matched entries
   467→    const { plan: lorebookPlan, lorebookAttributeKeys, lorebookEntryKeyMap, lorebookDebugEntries } = buildLorebookSectionForPrompt(
   468→      messages,
   469→      lorebooks,
   470→      {
   471→        scanDepth: contextConfig.scanDepth,
   472→        userName: effectiveUserName,
   473→        charName: effectiveCharacter?.name,
   474→      },
   475→      { sessionStats, characterId: effectiveCharacter?.id, characters: allCharacters }
   476→    );
   477→
   478→    // ========================================
   479→    // DEBUG: Build lorebook debug data for SSE event
   480→    // ========================================
   481→    const lorebookDebugData = {
   482→      lorebookAttributeKeys,
   483→      debugEntries: lorebookDebugEntries,
   484→      availableStats: sessionStats
   485→        ? Object.fromEntries(
   486→            Object.entries(sessionStats.characterStats || {}).map(([charId, cs]) => [
   487→              charId,
   488→              cs?.attributeValues ? { ...cs.attributeValues } : '(no values)',
   489→            ])
   490→          )
   491→        : '(no sessionStats)',
   492→    };
   493→
   494→    // ========================================
   495→    // Embeddings Context Retrieval
   496→    // ========================================
   497→    // Enrich search query with recent context for better semantic matching
   498→    const searchContextDepth = embeddingsChat.searchContextDepth || 0;
   499→    let enrichedSearchQuery = sanitizedMessage;
   500→    if (searchContextDepth > 0) {
   501→      // Collect recent messages from history for context enrichment
   502→      const recentHistory = messages
   503→        .filter(m => !m.isDeleted)
   504→        .slice(-(searchContextDepth * 2 + 1)) // user+assistant pairs
   505→        .map(m => m.content?.trim())
   506→        .filter(Boolean)
   507→        .slice(0, -1); // exclude current message (already in sanitizedMessage)
   508→      
   509→      if (recentHistory.length > 0) {
   510→        enrichedSearchQuery = recentHistory.join(' ') + ' ' + sanitizedMessage;
   511→      }
   512→    }
   513→
   514→    // Smart truncation: limit query to the embedding model's context window.
   515→    // Prioritize the user's CURRENT message over history when truncating.
   516→    // The Ollama client also handles truncation, but this avoids sending huge payloads.
   517→    try {
   518→      const { MODEL_CONTEXT_LENGTHS, DEFAULT_CONTEXT_LENGTH, CHARS_PER_TOKEN } = await import('@/lib/embeddings/types');
   519→      const { loadConfig } = await import('@/lib/embeddings/config-persistence');
   520→      const embConfig = loadConfig();
   521→      const modelKey = embConfig.model || 'bge-m3:567m';
   522→      const modelCtx = MODEL_CONTEXT_LENGTHS[modelKey]
   523→        || MODEL_CONTEXT_LENGTHS[modelKey.split(':')[0]]
   524→        || DEFAULT_CONTEXT_LENGTH;
   525→      const maxQueryChars = Math.floor(modelCtx * 0.75 * CHARS_PER_TOKEN);
   526→
   527→      if (enrichedSearchQuery.length > maxQueryChars) {
   528→        // Smart: keep the user's message intact, truncate history prefix
   529→        const userMsgLen = sanitizedMessage.length;
   530→        if (userMsgLen >= maxQueryChars) {
   531→          // User message alone exceeds limit — truncate it
   532→          enrichedSearchQuery = sanitizedMessage.slice(0, maxQueryChars);
   533→        } else {
   534→          // Keep full user message + as much recent history as fits
   535→          const budgetForHistory = maxQueryChars - userMsgLen - 1; // -1 for separator
   536→          enrichedSearchQuery = enrichedSearchQuery.slice(
   537→            Math.max(0, enrichedSearchQuery.length - maxQueryChars)
   538→          );
   539→        }
   540→        console.warn(
   541→          `[Stream Route] Search query trimmed to ${enrichedSearchQuery.length} chars ` +
   542→          `(model: ${modelKey}, context: ${modelCtx} tokens)`
   543→        );
   544→      }
   545→    } catch { /* fallback: no truncation at this level, Ollama client handles it */ }
   546→
   547→    // Retrieve relevant embeddings based on enriched query and settings
   548→    // Pass Character Memory events for deduplication (avoid duplicate memory in prompt)
   549→    const existingMemoryEvents = characterMemory?.events?.map(e => ({
   550→      content: e.content,
   551→      importance: e.importance,
   552→    }));
   553→
   554→    // Extract last assistant message for bidirectional search
   555→    const lastAssistantMsg = messages
   556→      .filter(m => !m.isDeleted && m.role === 'assistant')
   557→      .pop()?.content;
   558→
   559→    const embeddingsResult = await retrieveEmbeddingsContext(
   560→      enrichedSearchQuery,
   561→      characterId || effectiveCharacter.id,
   562→      sessionId,
   563→      embeddingsChat,
   564→      undefined, // groupId
   565→      existingMemoryEvents, // for deduplication
   566→      lastAssistantMsg, // bidirectional search with last assistant message
   567→    );
