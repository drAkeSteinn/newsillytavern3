    97→    97→export function CharacterEditor({ characterId, open, onClose }: CharacterEditorProps) {
    98→    98→  // Use individual selectors to avoid re-rendering on unrelated store changes
    99→    99→  const addCharacter = useTavernStore((s) => s.addCharacter);
   100→   100→  const updateCharacter = useTavernStore((s) => s.updateCharacter);
   101→   101→  const getCharacterById = useTavernStore((s) => s.getCharacterById);
   102→   102→  const characters = useTavernStore((s) => s.characters);
   103→   103→  const personas = useTavernStore((s) => s.personas);
   104→   104→  const activePersonaId = useTavernStore((s) => s.activePersonaId);
   105→   105→  const questTemplates = useTavernStore((s) => s.questTemplates);
   106→   106→
   107→   107→  // Active tab state
   108→   108→  const [activeTab, setActiveTab] = useState('info');
   109→   109→
   110→   110→  // Global TTS config for voice panel (OmniVoice provider awareness)
   111→   111→  const [globalTTSConfig, setGlobalTTSConfig] = useState<TTSWebUIConfig | null>(null);
   112→   112→
   113→   113→  const loadTTSConfig = useCallback(async () => {
   114→   114→    try {
   115→   115→      const response = await fetch('/api/tts/config');
   116→   116→      const data = await response.json();
   117→   117→      if (data.success && data.config?.tts) {
   118→   118→        setGlobalTTSConfig(data.config.tts);
   119→   119→      }
   120→   120→    } catch {
   121→   121→      // Silently fail — TTS config is optional for voice panel
   122→   122→    }
   123→   123→  }, []);
   124→   124→
   125→   125→  useEffect(() => {
   126→   126→    if (open) {
   127→   127→      loadTTSConfig();
   128→   128→    }
   129→   129→  }, [open, loadTTSConfig]);
   130→   130→
   131→   131→  // Reset tab when opening
   132→   132→  useEffect(() => {
   133→   133→    if (open) {
   134→   134→      setActiveTab('info');
   135→   135→    }
   136→   136→  }, [open]);
   137→   137→
   138→   138→  // Escape key handler
   139→   139→  const onCloseRef = useRef(onClose);
   140→   140→  onCloseRef.current = onClose;
   141→   141→
   142→   142→  useEffect(() => {
   143→   143→    if (!open) return;
   144→   144→    const handleKeyDown = (e: KeyboardEvent) => {
   145→   145→      if (e.key === 'Escape') {
   146→   146→        e.preventDefault();
   147→   147→        onCloseRef.current();
   148→   148→      }
   149→   149→    };
   150→   150→    document.addEventListener('keydown', handleKeyDown);
   151→   151→    return () => document.removeEventListener('keydown', handleKeyDown);
   152→   152→  }, [open]);
   153→   153→
   154→   154→  // Get active persona
   155→   155→  const activePersona = personas.find(p => p.id === activePersonaId);
   156→   156→
   157→   157→  // Get all characters except the one being edited (for target selection in invitations)
   158→   158→  // Also include the active persona if it has solicitudes configured
   159→   159→  const allCharacters = useMemo(() => {
   160→   160→    const result = characters
   161→   161→      .filter(c => c.id !== characterId)
   162→   162→      .map(c => ({
   163→   163→        id: c.id,
   164→   164→        name: c.name,
   165→   165→        solicitudDefinitions: c.statsConfig?.solicitudDefinitions || []
   166→   166→      }));
   167→   167→    
   168→   168→    // Add active persona if it has solicitudes configured
   169→   169→    if (activePersona?.statsConfig?.enabled && 
   170→   170→        (activePersona.statsConfig.solicitudDefinitions?.length || 0) > 0) {
   171→   171→      result.push({
   172→   172→        id: '__user__',
   173→   173→        name: activePersona.name || 'Usuario',
   174→   174→        solicitudDefinitions: activePersona.statsConfig.solicitudDefinitions || []
   175→   175→      });
   176→   176→    }
   177→   177→    
   178→   178→    return result;
   179→   179→  }, [characters, characterId, activePersona]);
   180→   180→
   181→   181→  // Build available targets for target_attribute rewards: other characters + persona
   182→   182→  const availableTargets = useMemo(() => {
   183→   183→    const targets: Array<{
   184→   184→      id: string;
   185→   185→      name: string;
   186→   186→      attributes: Array<{ key: string; name: string; type: 'number' | 'keyword' | 'text'; min?: number; max?: number }>;
   187→   187→      spritePacks?: Array<{ id: string; name: string; conditionalMode?: boolean; spriteCount: number }>;
   188→   188→    }> = [];
   189→   189→    // Helper: map attributes filtering out those without a valid key
   190→   190→    const mapAttrs = (attrs: Array<{ key: string; name: string; type: 'number' | 'keyword' | 'text'; min?: number; max?: number }>) =>
   191→   191→      attrs.filter(a => a.key && a.key.trim() !== '').map(a => ({
   192→   192→        key: a.key,
   193→   193→        name: a.name,
   194→   194→        type: a.type,
   195→   195→        min: a.min,
   196→   196→        max: a.max,
   197→   197→      }));
   198→   198→
   199→   199→    // Helper: map sprite packs for a character
   200→   200→    const mapPacks = (packs: Array<{ id: string; name: string; conditionalMode?: boolean; sprites: unknown[] }>) =>
   201→   201→      packs.map(p => ({ id: p.id, name: p.name, conditionalMode: p.conditionalMode, spriteCount: p.sprites.length }));
   202→   202→
   203→   203→    // Add OTHER characters with attributes (exclude the one being edited)
   204→   204→    characters.forEach(c => {
   205→   205→      const attrs = mapAttrs(c.statsConfig?.attributes || []);
   206→   206→      if (c.id !== characterId && c.statsConfig?.enabled && attrs.length > 0) {
   207→   207→        targets.push({
   208→   208→          id: c.id,
   209→   209→          name: c.name,
   210→   210→          attributes: attrs,
   211→   211→          spritePacks: mapPacks(c.spritePacksV2 || []),
   212→   212→        });
   213→   213→      }
   214→   214→    });
   215→   215→    // Add active persona with attributes
   216→   216→    if (activePersona?.statsConfig?.enabled) {
   217→   217→      const attrs = mapAttrs(activePersona.statsConfig.attributes || []);
   218→   218→      if (attrs.length > 0) {
   219→   219→        targets.push({
   220→   220→          id: '__user__',
   221→   221→          name: activePersona.name || 'Persona',
   222→   222→          attributes: attrs,
   223→   223→          spritePacks: mapPacks(activePersona.spritePacksV2 || []),
   224→   224→        });
   225→   225→      }
   226→   226→    }
   227→   227→    return targets;
   228→   228→  }, [characters, characterId, activePersona]);
   229→   229→
   230→   230→  // Initialize character data based on characterId
   231→   231→  const getInitialCharacter = () => {
   232→   232→    if (characterId) {
   233→   233→      const existing = getCharacterById(characterId);
   234→   234→      if (existing) {
   235→   235→        return existing;
   236→   236→      }
   237→   237→    }
   238→   238→    return defaultCharacter;
   239→   239→  };
   240→   240→
   241→   241→  const [character, setCharacter] = useState(getInitialCharacter);
   242→   242→  const [newTag, setNewTag] = useState('');
   243→   243→  const [uploading, setUploading] = useState(false);
   244→   244→  const fileInputRef = useRef<HTMLInputElement>(null);
   245→   245→
   246→   246→  const handleSave = () => {
   247→   247→    if (!character.name.trim()) {
   248→   248→      alert('El nombre del personaje es requerido');
   249→   249→      return;
   250→   250→    }
   251→   251→
   252→   252→    if (characterId) {
   253→   253→      updateCharacter(characterId, character);
   254→   254→    } else {
   255→   255→      addCharacter(character);
   256→   256→    }
   257→   257→    onClose();
   258→   258→  };
   259→   259→
   260→   260→  const handleAvatarUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   261→   261→    const file = e.target.files?.[0];
   262→   262→    if (!file) return;
   263→   263→
   264→   264→    // Validate file size (max 5MB)
   265→   265→    if (file.size > 5 * 1024 * 1024) {
   266→   266→      alert('La imagen es muy grande. El tamaño máximo es 5MB.');
   267→   267→      return;
   268→   268→    }
   269→   269→
   270→   270→    // Validate file type
   271→   271→    const allowedTypes = ['image/jpeg', 'image/png', 'image/gif', 'image/webp'];
   272→   272→    if (!allowedTypes.includes(file.type)) {
   273→   273→      alert('Formato no soportado. Usa JPEG, PNG, GIF o WebP.');
   274→   274→      return;
   275→   275→    }
   276→   276→
   277→   277→    setUploading(true);
   278→   278→
   279→   279→    try {
   280→   280→      // Upload via API to save as file
   281→   281→      const formData = new FormData();
   282→   282→      formData.append('file', file);
   283→   283→      formData.append('type', 'avatar');
   284→   284→
   285→   285→      const res = await fetch('/api/upload', {
   286→   286→        method: 'POST',
   287→   287→        body: formData,
   288→   288→      });
   289→   289→
   290→   290→      const data = await res.json();
   291→   291→
   292→   292→      if (res.ok && data.success) {
   293→   293→        setCharacter(prev => ({ ...prev, avatar: data.url }));
   294→   294→      } else {
   295→   295→        alert(data.error || 'Error al subir la imagen');
   296→   296→      }
   297→   297→    } catch (error) {
   298→   298→      editorLogger.error('Upload error', { error });
   299→   299→      alert('Error de conexión al subir la imagen');
   300→   300→    } finally {
   301→   301→      setUploading(false);
   302→   302→      if (fileInputRef.current) {
   303→   303→        fileInputRef.current.value = '';
   304→   304→      }
   305→   305→    }
   306→   306→  };
   307→   307→
   308→   308→  const handleAddTag = () => {
   309→   309→    if (newTag.trim() && !character.tags.includes(newTag.trim())) {
   310→   310→      setCharacter(prev => ({
   311→   311→        ...prev,
   312→   312→        tags: [...prev.tags, newTag.trim()]
   313→   313→      }));
   314→   314→      setNewTag('');
   315→   315→    }
   316→   316→  };
   317→   317→
   318→   318→  const handleRemoveTag = (tag: string) => {
   319→   319→    setCharacter(prev => ({
   320→   320→      ...prev,
   321→   321→      tags: prev.tags.filter(t => t !== tag)
   322→   322→    }));
   323→   323→  };
   324→   324→
   325→   325→  // Tab content renderers
   326→   326→  const renderInfoTab = () => (
   327→   327→    <div className="space-y-6">
   328→   328→      {/* Banner */}
   329→   329→      <div className="flex items-center gap-3 p-3 bg-gradient-to-r from-blue-500/10 to-indigo-500/10 border border-blue-500/20 rounded-lg">
   330→   330→        <Palette className="w-4 h-4 text-blue-500 shrink-0" />
   331→   331→        <p className="text-xs text-muted-foreground">
   332→   332→          Configura <strong>avatar</strong>, <strong>información básica</strong> y <strong>asignaciones</strong> del personaje.
   333→   333→        </p>
   334→   334→      </div>
   335→   335→
   336→   336→      {/* ═══ Section: Identidad ═══ */}
   337→   337→      <div className="rounded-lg border bg-card p-5">
   338→   338→        <div className="flex items-center gap-2 mb-4">
   339→   339→          <div className="flex items-center justify-center w-6 h-6 rounded-md bg-blue-500/10">
   340→   340→            <Palette className="w-3.5 h-3.5 text-blue-500" />
   341→   341→          </div>
   342→   342→          <h3 className="text-sm font-semibold">Identidad</h3>
   343→   343→        </div>
   344→   344→
   345→   345→        <div className="flex gap-6 items-start">
   346→   346→          {/* Avatar */}
   347→   347→          <div className="flex-shrink-0">
   348→   348→            <Label className="text-xs font-medium mb-2 block">Avatar</Label>
   349→   349→            <Tooltip>
   350→   350→              <TooltipTrigger asChild>
   351→   351→                <div 
   352→   352→                  className={cn(
   353→   353→                    "w-32 h-32 rounded-xl overflow-hidden bg-muted border-2 border-dashed border-muted-foreground/25 flex items-center justify-center transition-colors",
   354→   354→                    !uploading && "cursor-pointer hover:border-primary/50 hover:bg-muted/50"
   355→   355→                  )}
   356→   356→                  onClick={() => !uploading && fileInputRef.current?.click()}
   357→   357→                >
   358→   358→                  {uploading ? (
   359→   359→                    <div className="text-center text-muted-foreground">
   360→   360→                      <Loader2 className="w-7 h-7 mx-auto animate-spin" />
   361→   361→                      <span className="text-[10px] mt-1 block">Subiendo...</span>
   362→   362→                    </div>
   363→   363→                  ) : character.avatar ? (
   364→   364→                    <img 
   365→   365→                      src={character.avatar} 
   366→   366→                      alt={character.name || 'Avatar'}
   367→   367→                      className="w-full h-full object-cover"
   368→   368→                    />
   369→   369→                  ) : (
   370→   370→                    <div className="text-center text-muted-foreground">
   371→   371→                      <Camera className="w-8 h-8 mx-auto mb-1 opacity-50" />
   372→   372→                      <span className="text-[10px]">Subir avatar</span>
   373→   373→                    </div>
   374→   374→                  )}
   375→   375→                </div>
   376→   376→              </TooltipTrigger>
   377→   377→              <TooltipContent>
   378→   378→                <p>Haz clic para subir avatar (máx 5MB)</p>
   379→   379→              </TooltipContent>
   380→   380→            </Tooltip>
   381→   381→            <input
   382→   382→              type="file"
   383→   383→              ref={fileInputRef}
   384→   384→              className="hidden"
   385→   385→              accept="image/*"
   386→   386→              onChange={handleAvatarUpload}
   387→   387→              disabled={uploading}
   388→   388→            />
   389→   389→          </div>
   390→   390→
   391→   391→          {/* Name + Tags */}
   392→   392→          <div className="flex-1 min-w-0 space-y-4">
   393→   393→            {/* Name field */}
   394→   394→            <div>
   395→   395→              <div className="flex items-center gap-1.5 mb-1.5">
   396→   396→                <Label htmlFor="name" className="text-sm font-medium">Nombre *</Label>
   397→   397→                <Tooltip>
   398→   398→                  <TooltipTrigger asChild>
   399→   399→                    <HelpCircle className="w-3.5 h-3.5 text-muted-foreground cursor-help" />
   400→   400→                  </TooltipTrigger>
   401→   401→                  <TooltipContent className="max-w-xs">
   402→   402→                    <p>El nombre del personaje que se mostrará en el chat.</p>
   403→   403→                  </TooltipContent>
   404→   404→                </Tooltip>
   405→   405→              </div>
   406→   406→              <Input
   407→   407→                id="name"
   408→   408→                value={character.name}
   409→   409→                onChange={(e) => setCharacter(prev => ({ ...prev, name: e.target.value }))}
   410→   410→                placeholder="Nombre del personaje"
   411→   411→                className="h-10"
   412→   412→              />
   413→   413→            </div>
   414→   414→
   415→   415→            {/* Tags input */}
   416→   416→            <div>
   417→   417→              <div className="flex items-center gap-1.5 mb-1.5">
   418→   418→                <Label className="text-sm font-medium">Etiquetas</Label>
   419→   419→                <Tooltip>
   420→   420→                  <TooltipTrigger asChild>
   421→   421→                    <HelpCircle className="w-3.5 h-3.5 text-muted-foreground cursor-help" />
   422→   422→                  </TooltipTrigger>
   423→   423→                  <TooltipContent className="max-w-xs">
   424→   424→                    <p>Tags para organizar y buscar personajes.</p>
   425→   425→                  </TooltipContent>
   426→   426→                </Tooltip>
   427→   427→              </div>
   428→   428→              <div className="flex gap-2">
   429→   429→                <Input
   430→   430→                  value={newTag}
   431→   431→                  onChange={(e) => setNewTag(e.target.value)}
   432→   432→                  placeholder="Agregar etiqueta..."
   433→   433→                  onKeyDown={(e) => e.key === 'Enter' && handleAddTag()}
   434→   434→                  className="h-9 flex-1"
   435→   435→                />
   436→   436→                <Button variant="outline" size="sm" className="h-9 px-4" onClick={handleAddTag}>
   437→   437→                  <Plus className="w-3.5 h-3.5 mr-1" />
   438→   438→                  Agregar
   439→   439→                </Button>
   440→   440→              </div>
   441→   441→              {character.tags.length > 0 && (
   442→   442→                <div className="flex flex-wrap gap-1.5 mt-2">
   443→   443→                  {character.tags.map((tag) => (
   444→   444→                    <Badge key={tag} variant="secondary" className="gap-1 text-xs py-1 px-2.5">
   445→   445→                      {tag}
   446→   446→                      <button
   447→   447→                        type="button"
   448→   448→                        className="ml-0.5 rounded-full p-0.5 hover:bg-destructive/20 hover:text-destructive transition-colors"
   449→   449→                        onClick={(e) => { e.stopPropagation(); handleRemoveTag(tag); }}
   450→   450→                      >
   451→   451→                        <X className="w-3 h-3" />
   452→   452→                      </button>
   453→   453→                    </Badge>
   454→   454→                  ))}
   455→   455→                </div>
   456→   456→              )}
   457→   457→            </div>
   458→   458→          </div>
   459→   459→        </div>
   460→   460→      </div>
   461→   461→
   462→   462→      {/* ═══ Section: Asignaciones ═══ */}
   463→   463→      <div className="rounded-lg border bg-card p-5">
   464→   464→        <div className="flex items-center gap-2 mb-4">
   465→   465→          <div className="flex items-center justify-center w-6 h-6 rounded-md bg-amber-500/10">
   466→   466→            <Package className="w-3.5 h-3.5 text-amber-500" />
   467→   467→          </div>
