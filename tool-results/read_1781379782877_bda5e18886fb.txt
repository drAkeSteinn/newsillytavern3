    20→// Sprite List Rotation Tracker (shared with character-sprite.tsx)
    21→// Tracks rotation indices per character per state
    22→// ============================================
    23→const _spriteRotationIndex = new Map<string, Map<SpriteState, number>>();
    24→
    25→function getRotationIndex(characterId: string, state: SpriteState): number {
    26→  let charMap = _spriteRotationIndex.get(characterId);
    27→  if (!charMap) {
    28→    charMap = new Map<SpriteState, number>();
    29→    _spriteRotationIndex.set(characterId, charMap);
    30→  }
    31→  return charMap.get(state) ?? 0;
    32→}
    33→
    34→function advanceRotationIndex(characterId: string, state: SpriteState): number {
    35→  let charMap = _spriteRotationIndex.get(characterId);
    36→  if (!charMap) {
    37→    charMap = new Map<SpriteState, number>();
    38→    _spriteRotationIndex.set(characterId, charMap);
    39→  }
    40→  const current = charMap.get(state) ?? 0;
    41→  const next = current + 1;
    42→  charMap.set(state, next);
    43→  return next;
    44→}

    46→// Get sprite from State Collection V2 (uses sprite packs)
    47→function getSpriteFromStateCollectionV2(
    48→  state: SpriteState,
    49→  stateCollectionsV2: StateCollectionV2[],
    50→  spritePacksV2: SpritePackV2[],
    51→  characterId: string
    52→): { url: string | null; label: string | null } {
    53→  const stateCollection = stateCollectionsV2.find(c => c.state === state);
    54→  if (!stateCollection) return { url: null, label: null };
    55→  
    56→  const pack = spritePacksV2.find(p => p.id === stateCollection.packId);
    57→  if (!pack || pack.sprites.length === 0) return { url: null, label: null };
    58→  
    59→  switch (stateCollection.behavior) {
    60→    case 'principal':
    61→      if (stateCollection.principalSpriteId) {
    62→        const principal = pack.sprites.find(s => s.id === stateCollection.principalSpriteId);
    63→        if (principal) return { url: principal.url, label: principal.label };
    64→      }
    65→      return { url: pack.sprites[0].url, label: pack.sprites[0].label };
    66→    
    67→    case 'random': {
    68→      const randomIndex = Math.floor(Math.random() * pack.sprites.length);
    69→      const randomSprite = pack.sprites[randomIndex];
    70→      return { url: randomSprite.url, label: randomSprite.label };
    71→    }
    72→    
    73→    case 'list': {
    74→      const index = getRotationIndex(characterId, state);
    75→      const safeIndex = index % pack.sprites.length;
    76→      const listSprite = pack.sprites[safeIndex];
    77→      return { url: listSprite.url, label: listSprite.label };
    78→    }
    79→    
    80→    default:
    81→      return { url: pack.sprites[0].url, label: pack.sprites[0].label };
    82→  }
    83→}

    85→// Format milliseconds to readable time
    86→function formatTime(ms: number): string {
    87→  if (ms <= 0) return '0s';
    88→  const seconds = Math.floor(ms / 1000);
    89→  if (seconds < 60) return `${seconds}s`;
    90→  const minutes = Math.floor(seconds / 60);
    91→  const remainingSeconds = seconds % 60;
    92→  return `${minutes}m ${remainingSeconds}s`;
    93→}

   149→// Get the appropriate sprite URL based on state and config
   150→// Priority: V2 State Collections > Avatar (legacy system removed)
   151→function getSpriteUrl(
   152→  state: SpriteState,
   153→  character?: CharacterCard,
   154→  characterId?: string
   155→): { url: string; label: string | null } {
   156→  const charId = characterId || character?.id || '';
   157→  // Check for non-empty arrays (empty arrays should not be considered as "configured")
   158→  const hasV2Collections = character?.stateCollectionsV2 && character.stateCollectionsV2.length > 0;
   159→  const hasV2Packs = character?.spritePacksV2 && character.spritePacksV2.length > 0;
   160→  
   161→  // Try V2 state collections (new system)
   162→  if (hasV2Collections && hasV2Packs) {
   163→    const v2Result = getSpriteFromStateCollectionV2(
   164→      state,
   165→      character.stateCollectionsV2,
   166→      character.spritePacksV2,
   167→      charId
   168→    );
   169→    if (v2Result.url) {
   170→      return v2Result;
   171→    }
   172→  }
   173→
   174→  // Additional fallbacks for talk and thinking states - try V2 idle collection
   175→  if (state === 'talk' || state === 'thinking') {
   176→    if (hasV2Collections && hasV2Packs) {
   177→      const v2IdleResult = getSpriteFromStateCollectionV2(
   178→        'idle',
   179→        character!.stateCollectionsV2,
   180→        character!.spritePacksV2,
   181→        charId
   182→      );
   183→      if (v2IdleResult.url) {
   184→        return v2IdleResult;
   185→      }
   186→    }
   187→  }
   188→
   189→  // Final fallback to avatar
   190→  if (character?.avatar) {
   191→    return { url: character.avatar, label: 'avatar' };
   192→  }
   193→
   194→  return { url: '', label: null };
   195→}

   197→export function GroupSprites({
   198→  characters,
   199→  activeCharacterId,
   200→  isStreaming,
   201→  isTTSPlaying = false,
   202→  maxVisible = 3,
   203→  activeGroup,
   204→}: GroupSpritesProps) {
   205→  const [settings, setSettings] = useState<GroupSpriteSettings>(() => {
   206→    if (typeof window !== 'undefined') {
   207→      const saved = localStorage.getItem(STORAGE_KEY);
   208→      if (saved) {
   209→        try {
   210→          return { ...DEFAULT_SETTINGS, ...JSON.parse(saved) };
   211→        } catch {
   212→          return DEFAULT_SETTINGS;
   213→        }
   214→      }
   215→    }
   216→    return DEFAULT_SETTINGS;
   217→  });
   218→
   219→  const [positions, setPositions] = useState<SavedPositions>(() => {
   220→    if (typeof window !== 'undefined') {
   221→      const saved = localStorage.getItem(POSITIONS_KEY);
   222→      if (saved) {
   223→        try {
   224→          return JSON.parse(saved);
   225→        } catch {
   226→          return {};
   227→        }
   228→      }
   229→    }
   230→    return {};
   231→  });
   232→
   233→  // Track which sprite is being interacted with
   234→  const [draggingId, setDraggingId] = useState<string | null>(null);
   235→  const [resizingId, setResizingId] = useState<string | null>(null);
   236→  const [hoveredId, setHoveredId] = useState<string | null>(null);
   237→  
   238→  // Countdowns per character (read from store periodically)
   239→  const [countdowns, setCountdowns] = useState<Map<string, number>>(new Map());
   240→
   241→  const containerRef = useRef<HTMLDivElement>(null);
   242→  const dragStartRef = useRef({ x: 0, y: 0, posX: 0, posY: 0 });
   243→  const resizeStartRef = useRef({ x: 0, y: 0, width: 0, height: 0 });
   244→
   245→  // ============================================
   246→  // UNIFIED SYSTEM: Use store for per-character sprite state
   247→  // ============================================
   248→  const store = useTavernStore();
   249→
   250→  // Save to localStorage
   251→  useEffect(() => {
   252→    localStorage.setItem(STORAGE_KEY, JSON.stringify(settings));
   253→  }, [settings]);
   254→
   255→  useEffect(() => {
   256→    localStorage.setItem(POSITIONS_KEY, JSON.stringify(positions));
   257→  }, [positions]);
   258→
   259→  // Update countdowns every 100ms for all characters
   260→  useEffect(() => {
   261→    const interval = setInterval(() => {
   262→      const newCountdowns = new Map<string, number>();
   263→      
   264→      characters.forEach(character => {
   265→        const remaining = store.getReturnToIdleCountdownForCharacter(character.id);
   266→        if (remaining > 0) {
   267→          newCountdowns.set(character.id, remaining);
   268→        }
   269→      });
   270→
   271→      setCountdowns(newCountdowns);
   272→    }, 100);
   273→
   274→    return () => clearInterval(interval);
   275→  }, [characters, store]);
   276→
   277→  // Get position for a character
   278→  const getCharacterPosition = useCallback((character: CharacterCard, index: number, total: number): SpritePosition => {
   279→    if (positions[character.id]) {
   280→      return positions[character.id];
   281→    }
   282→    if (total === 1) {
   283→      return { ...DEFAULT_SPRITE, x: 50 };
   284→    } else if (total === 2) {
   285→      return { ...DEFAULT_SPRITE, x: index === 0 ? 25 : 75 };
   286→    } else {
   287→      const spacing = 70 / (total - 1);
   288→      return { ...DEFAULT_SPRITE, x: 15 + (spacing * index) };
   289→    }
   290→  }, [positions]);
   291→
   292→  // Update position
   293→  const updatePosition = useCallback((characterId: string, updates: Partial<SpritePosition>) => {
   294→    setPositions(prev => {
   295→      const current = prev[characterId] || DEFAULT_SPRITE;
   296→      return {
   297→        ...prev,
   298→        [characterId]: { ...current, ...updates }
   299→      };
   300→    });
   301→  }, []);
   302→
   303→  // Drag handlers
   304→  const handleDragStart = (e: React.MouseEvent, characterId: string) => {
   305→    if (resizingId) return;
   306→    const target = e.target as HTMLElement;
   307→    if (target.closest('.sprite-controls')) return;
   308→    
   309→    e.preventDefault();
   310→    e.stopPropagation();
   311→    const pos = positions[characterId] || DEFAULT_SPRITE;
   312→    setDraggingId(characterId);
   313→    dragStartRef.current = {
   314→      x: e.clientX,
   315→      y: e.clientY,
   316→      posX: pos.x,
   317→      posY: pos.y
   318→    };
   319→  };
   320→
   321→  useEffect(() => {
   322→    if (!draggingId) return;
   323→
   324→    const handleMouseMove = (e: MouseEvent) => {
   325→      const container = containerRef.current;
   326→      if (!container) return;
   327→
   328→      const rect = container.getBoundingClientRect();
   329→      const deltaX = ((e.clientX - dragStartRef.current.x) / rect.width) * 100;
   330→      const deltaY = ((e.clientY - dragStartRef.current.y) / rect.height) * 100;
   331→
   332→      const currentPos = positions[draggingId] || DEFAULT_SPRITE;
   333→      let newX = dragStartRef.current.posX + deltaX;
   334→      let newY = dragStartRef.current.posY - deltaY;
   335→
   336→      const halfWidth = currentPos.width / 2;
   337→      newX = Math.max(halfWidth, Math.min(100 - halfWidth, newX));
   338→      newY = Math.max(0, Math.min(100 - currentPos.height, newY));
   339→
   340→      updatePosition(draggingId, { x: newX, y: newY });
   341→    };
   342→
   343→    const handleMouseUp = () => {
   344→      setDraggingId(null);
   345→    };
   346→
   347→    document.addEventListener('mousemove', handleMouseMove);
   348→    document.addEventListener('mouseup', handleMouseUp);
   349→
   350→    return () => {
   351→      document.removeEventListener('mousemove', handleMouseMove);
   352→      document.removeEventListener('mouseup', handleMouseUp);
   353→    };
   354→  }, [draggingId, positions, updatePosition]);
   355→
   356→  // Resize handlers
   357→  const handleResizeStart = (e: React.MouseEvent, characterId: string) => {
   358→    e.preventDefault();
   359→    e.stopPropagation();
   360→    const pos = positions[characterId] || DEFAULT_SPRITE;
   361→    setResizingId(characterId);
   362→    resizeStartRef.current = {
   363→      x: e.clientX,
   364→      y: e.clientY,
   365→      width: pos.width,
   366→      height: pos.height
   367→    };
   368→  };
   369→
   370→  useEffect(() => {
   371→    if (!resizingId) return;
   372→
   373→    const handleMouseMove = (e: MouseEvent) => {
   374→      const container = containerRef.current;
   375→      if (!container) return;
   376→
   377→      const rect = container.getBoundingClientRect();
   378→      const deltaX = ((e.clientX - resizeStartRef.current.x) / rect.width) * 100;
   379→      const deltaY = ((e.clientY - resizeStartRef.current.y) / rect.height) * 100;
   380→
   381→      let newWidth = Math.max(15, Math.min(50, resizeStartRef.current.width - deltaX * 2));
   382→      let newHeight = Math.max(20, Math.min(80, resizeStartRef.current.height - deltaY * 2));
   383→
   384→      updatePosition(resizingId, { width: newWidth, height: newHeight });
   385→    };
   386→
   387→    const handleMouseUp = () => {
   388→      setResizingId(null);
   389→    };
   390→
   391→    document.addEventListener('mousemove', handleMouseMove);
   392→    document.addEventListener('mouseup', handleMouseUp);
   393→
   394→    return () => {
   395→      document.removeEventListener('mousemove', handleMouseMove);
   396→      document.removeEventListener('mouseup', handleMouseUp);
   397→    };
   398→  }, [resizingId, updatePosition]);
   399→
   400→  const resetPositions = () => setPositions({});
   401→  const resetSettings = () => setSettings(DEFAULT_SETTINGS);
   402→
   403→  // Filter out narrator characters if showSprite is false
   404→  const filteredCharacters = useMemo(() => {
   405→    if (!activeGroup?.narratorSettings?.showSprite) {
   406→      // Find narrator character IDs
   407→      const narratorIds = new Set(
   408→        (activeGroup?.members || [])
   409→          .filter(m => m.isNarrator)
   410→          .map(m => m.characterId)
   411→      );
   412→      // Filter out narrators
   413→      return characters.filter(c => !narratorIds.has(c.id));
   414→    }
   415→    return characters;
   416→  }, [characters, activeGroup]);
   417→
   418→  // Determine display mode
   419→  const displayMode = filteredCharacters.length <= maxVisible ? 'multi' : 'single';
   420→  const visibleCharacters = displayMode === 'multi'
   421→    ? filteredCharacters
   422→    : (filteredCharacters.find(c => c.id === activeCharacterId) ? [filteredCharacters.find(c => c.id === activeCharacterId)!] : filteredCharacters.slice(0, 1));
   423→
   424→  // ============================================
   425→  // LIST MODE ROTATION: Track state transitions per character
   426→  // ============================================
   427→  const prevStreamingActiveRef = useRef<string | null>(null);
   428→  useEffect(() => {
   429→    const prevActive = prevStreamingActiveRef.current;
   430→    const currentActive = isStreaming ? activeCharacterId : null;
   431→    prevStreamingActiveRef.current = currentActive;
   432→
   433→    // When a character STARTS streaming (talk), advance its rotation index if using list mode
   434→    if (currentActive && currentActive !== prevActive) {
   435→      const character = characters.find(c => c.id === currentActive);
   436→      if (character) {
   437→        const collection = character.stateCollectionsV2?.find(c => c.state === 'talk');
   438→        if (collection?.behavior === 'list') {
   439→          advanceRotationIndex(currentActive, 'talk');
   440→        }
   441→      }
   442→    }
   443→  }, [isStreaming, activeCharacterId, characters]);
   444→
   445→  return (
   446→    <div ref={containerRef} className="absolute inset-0 z-5">
   447→      {/* Sprites */}
   448→      {visibleCharacters.map((character, index) => {
   449→        const position = getCharacterPosition(character, index, visibleCharacters.length);
   450→        const isActive = character.id === activeCharacterId;
   451→        const isHovered = hoveredId === character.id;
   452→        const isBeingDragged = draggingId === character.id;
   453→        const isBeingResized = resizingId === character.id;
   454→        const showControls = isHovered && !isBeingDragged;
   455→
   456→        // Calculate opacity
   457→        let opacity = settings.globalOpacity;
   458→        if (displayMode === 'single') {
   459→          opacity *= settings.activeOpacity;
   460→        } else {
   461→          if (isStreaming) {
   462→            opacity *= isActive ? settings.activeOpacity : settings.inactiveOpacity;
   463→          } else if (settings.showAllEqually) {
   464→            opacity *= settings.activeOpacity;
   465→          } else {
   466→            opacity *= settings.inactiveOpacity;
   467→          }
   468→        }
   469→
   470→        // ============================================
   471→        // UNIFIED SYSTEM: Get sprite state from store
   472→        // ============================================
   473→        const charSpriteState = store.getCharacterSpriteState(character.id);
   474→        
   475→        // Determine sprite state:
   476→        // Priority: Trigger > TTS Talk > Streaming Thinking > Idle
   477→        const hasTriggerSprite = charSpriteState.triggerSpriteUrl;
   478→        let spriteState: SpriteState;
   479→        
   480→        if (hasTriggerSprite) {
   481→          spriteState = 'idle'; // Trigger active, use trigger sprite
   482→        } else if (isTTSPlaying) {
   483→          spriteState = 'talk'; // TTS playing → character is speaking
   484→        } else if (isStreaming && isActive) {
   485→          spriteState = 'thinking'; // Generating → thinking
   486→        } else {
   487→          spriteState = charSpriteState.spriteState || 'idle';
   488→        }
   489→        const countdown = countdowns.get(character.id) || 0;
   490→        
   491→        let spriteUrl: string;
   492→        let spriteLabel: string | null = null;
   493→        
   494→        if (hasTriggerSprite) {
   495→          // Use trigger sprite (highest priority)
   496→          spriteUrl = charSpriteState.triggerSpriteUrl!;
   497→          spriteLabel = charSpriteState.triggerSpriteLabel;
   498→        } else {
   499→          // Use state collection or fallback - pass full character for V2 support
   500→          const spriteResult = getSpriteUrl(spriteState, character, character.id);
   501→          spriteUrl = spriteResult.url;
   502→          spriteLabel = spriteResult.label;
   503→        }
   504→
   505→        // FASE 7: Determine active transition for this character
   506→        const activeTransition: SpriteTransitionConfig | undefined = hasTriggerSprite
   507→          ? (charSpriteState.triggerTransition ?? undefined)
   508→          : (charSpriteState.defaultTransition ?? character.defaultTransition ?? undefined);
   509→
   510→        return (
   511→          <div
   512→            key={character.id}
   513→            data-character-id={character.id}
   514→            className={cn(
   515→              "absolute select-none",
   516→              isBeingDragged ? "cursor-grabbing" : "cursor-grab"
   517→            )}
   518→            style={{
   519→              left: `${position.x}%`,
   520→              bottom: `${position.y}%`,
   521→              transform: 'translate(-50%, 0)',
   522→              width: `${position.width}%`,
   523→              height: `${position.height}%`,
   524→              opacity,
   525→              zIndex: isActive && isStreaming ? 10 : 5,
   526→              filter: (isActive || !isStreaming || !settings.grayscaleInactive) ? 'none' : 'grayscale(20%)',
   527→            }}
   528→            onMouseDown={(e) => handleDragStart(e, character.id)}
   529→            onMouseEnter={() => setHoveredId(character.id)}
   530→            onMouseLeave={() => !isBeingDragged && setHoveredId(null)}
   531→          >
   532→            {/* The actual sprite image/video */}
   533→            {spriteUrl ? (
   534→              <SpriteTransitionWrapper
   535→                src={spriteUrl}
   536→                alt={character.name}
   537→                transition={activeTransition}
   538→                className="w-full h-full drop-shadow-2xl select-none pointer-events-none"
   539→              />
   540→            ) : (
   541→              <div className="w-full h-full flex items-end justify-center">
   542→                <div
   543→                  className={cn(
   544→                    "w-20 h-20 rounded-full flex items-center justify-center transition-all duration-300",
   545→                    isActive
   546→                      ? "bg-gradient-to-br from-amber-400 to-orange-600 shadow-lg"
   547→                      : "bg-gradient-to-br from-gray-400 to-gray-600"
   548→                  )}
   549→                >
   550→                  <span className="text-white font-bold text-2xl">
   551→                    {character.name?.[0]?.toUpperCase() || '?'}
   552→                  </span>
   553→                </div>
   554→              </div>
   555→            )}
   556→
   557→            {/* Trigger countdown badge - per character */}
   558→            {hasTriggerSprite && countdown > 0 && (
   559→              <div className="sprite-controls absolute top-2 left-1/2 -translate-x-1/2 z-50">
   560→                <Badge 
   561→                  variant="secondary" 
   562→                  className="bg-blue-500/90 text-white border-blue-400 gap-1.5 px-3 py-1 shadow-lg"
   563→                >
   564→                  <Timer className="h-3 w-3" />
   565→                  <span className="text-xs font-medium">→ Idle: {formatTime(countdown)}</span>
   566→                </Badge>
   567→              </div>
   568→            )}
   569→
   570→            {/* Glow effect for active character */}
   571→            {isActive && isStreaming && (
   572→              <div
   573→                className="absolute inset-0 rounded-lg animate-pulse pointer-events-none"
   574→                style={{
   575→                  background: 'radial-gradient(ellipse at center bottom, rgba(251, 191, 36, 0.15) 0%, transparent 70%)',
   576→                  animationDuration: '2s',
   577→                }}
   578→              />
   579→            )}
   580→
   581→            {/* Border highlight for active character */}
   582→            {isActive && isStreaming && (
   583→              <div
