   327→    const file = e.target.files?.[0];
   328→    if (!file) return;

   330→    if (file.size > 5 * 1024 * 1024) {
   331→      toast({
   332→        title: 'Imagen muy grande',
   333→        description: 'El tamaño máximo es 5MB.',
   334→        variant: 'destructive'
   335→      });
   336→      return;
   337→    }
   338→
   339→    const allowedTypes = ['image/jpeg', 'image/png', 'image/gif', 'image/webp'];
   340→    if (!allowedTypes.includes(file.type)) {
   341→      toast({
   342→        title: 'Formato no soportado',
   343→        description: 'Usa JPEG, PNG, GIF o WebP.',
   344→        variant: 'destructive'
   345→      });
   346→      return;
   347→    }

   349→    setAvatarUploading(true);

   354→      formData.append('type', 'group-avatar');

   356→      const res = await fetch('/api/upload', {
   357→        method: 'POST',
   358→        body: formData,

   361→      const data = await res.json();

   363→      if (res.ok && data.success) {
   364→        setAvatarUrl(data.url);

   366→        toast({
   367→          title: 'Error al subir la imagen',
   368→          description: data.error || 'Error al subir la imagen',
   369→          variant: 'destructive'
   370→        });
   371→      }
   372→    } catch {
   373→      toast({
   374→        title: 'Error de conexión',
   375→        description: 'Error de conexión al subir la imagen',
   376→        variant: 'destructive'
   377→      });

   379→      setAvatarUploading(false);
   380→      if (avatarFileInputRef.current) {
   381→        avatarFileInputRef.current.value = '';
