   334→    setIsImporting(true);

   336→    try {
   337→      const text = await file.text();

   339→
   340→      // Validate structure
   341→      if (!data.version || !Array.isArray(data.characters)) {
   342→        toast({
   343→          title: 'Error de Importación',
   344→          description: 'El archivo no tiene el formato correcto. Debe ser un backup de TavernFlow.',
   345→          variant: 'destructive'
   346→        });

   348→      }

   350→      let importedCharacters = 0;
   351→      let importedGroups = 0;

   353→
   354→      // Build a map of old ID to new ID for updating group references
   355→      const idMapping: Record<string, string> = {};
   356→
   357→      // Import characters
   358→      for (const character of data.characters) {
   359→        // Check if character with same ID already exists
   360→        const exists = characters.some(c => c.id === character.id);
   361→        if (exists) {
   362→          skippedCharacters++;
   363→          // Map the existing ID to itself for group references
   364→          if (character.id) {
   365→            idMapping[character.id] = character.id;
   366→          }
   367→          continue;
   368→        }
   369→
   370→        // Add character using store action - preserve ALL fields including ID
   371→        // Use preserveId=true to maintain the original ID for group/reward references
   372→        addCharacter({
   373→          // Preserve original ID
   374→          id: character.id,
   375→          createdAt: character.createdAt,
   376→          // Basic fields
   377→          name: character.name || 'Personaje sin nombre',
   378→          description: character.description || '',
   379→          personality: character.personality || '',
   380→          scenario: character.scenario || '',
   381→          firstMes: character.firstMes || '',
   382→          mesExample: character.mesExample || '',
   383→          creatorNotes: character.creatorNotes || '',
   384→          characterNote: character.characterNote || '',
   385→          systemPrompt: character.systemPrompt || '',
   386→          postHistoryInstructions: character.postHistoryInstructions || '',
   387→          authorNote: character.authorNote || '',
   388→          alternateGreetings: character.alternateGreetings || [],
   389→          tags: character.tags || [],
   390→          avatar: character.avatar || '',
   391→          sprites: character.sprites || [],
   392→          voice: character.voice || null,
   393→          statsConfig: character.statsConfig,
   394→          // V2 Sprite System fields
   395→          spritePacksV2: character.spritePacksV2,
   396→          stateCollectionsV2: character.stateCollectionsV2,
   397→          triggerCollections: character.triggerCollections,
   398→          spriteLibraries: character.spriteLibraries,
   399→          spriteIndex: character.spriteIndex,
   400→          spriteConfig: character.spriteConfig,
   401→          // Legacy sprite fields
   402→          spritePacks: character.spritePacks,
   403→          spriteTriggers: character.spriteTriggers,
   404→          // Reference IDs
   405→          hudTemplateId: character.hudTemplateId,
   406→          lorebookIds: character.lorebookIds,
   407→          questTemplateIds: character.questTemplateIds,
   408→          // Additional fields
   409→          creator: character.creator,
   410→          characterVersion: character.characterVersion,
   411→          extensions: character.extensions,
   412→          chatStats: character.chatStats,
   413→          memory: character.memory,
   414→          // Embeddings
   415→          embeddingNamespaces: character.embeddingNamespaces,
   416→          // Quick Replies & Proactive Messages
   417→          quickReplies: character.quickReplies,
   418→          proactiveMessages: character.proactiveMessages,
   419→        }, true); // preserveId = true
   420→        importedCharacters++;
   421→      }
   422→
   423→      // Import groups
   424→      if (data.groups && Array.isArray(data.groups)) {
   425→        for (const group of data.groups) {
   426→          // Check if group with same ID already exists
   427→          const exists = groups.some(g => g.id === group.id);
   428→          if (exists) continue;
   429→
   430→          // Add group using store action - preserve ALL fields including ID
   431→          // Use preserveId=true to maintain the original ID
   432→          addGroup({
   433→            // Preserve original ID
   434→            id: group.id,
   435→            createdAt: group.createdAt,
   436→            name: group.name || 'Grupo sin nombre',
   437→            description: group.description || '',
   438→            characterIds: group.characterIds || [],
   439→            members: group.members || [],
   440→            avatar: group.avatar || '',
   441→            systemPrompt: group.systemPrompt || '',
   442→            activationStrategy: group.activationStrategy || 'all',
   443→            minResponsesPerTurn: group.minResponsesPerTurn ?? 1,
   444→            maxResponsesPerTurn: group.maxResponsesPerTurn ?? 3,
   445→            allowMentions: group.allowMentions ?? true,
   446→            mentionTriggers: group.mentionTriggers || [],
   447→            conversationStyle: group.conversationStyle || 'sequential',
   448→            // Reference IDs
   449→            hudTemplateId: group.hudTemplateId,
   450→            lorebookIds: group.lorebookIds,
   451→            questTemplateIds: group.questTemplateIds,
   452→            // Narrator settings
   453→            narratorSettings: group.narratorSettings,
   454→            // Embeddings
   455→            embeddingNamespaces: group.embeddingNamespaces,
   456→          }, true); // preserveId = true
   457→          importedGroups++;
   458→        }
   459→      }
   460→
   461→      toast({
   462→        title: 'Importación Completa',
   463→        description: `Importados ${importedCharacters} personajes y ${importedGroups} grupos.${skippedCharacters > 0 ? ` (${skippedCharacters} personajes omitidos por duplicados)` : ''}`
   464→      });

   468→        title: 'Error de Importación',
   469→        description: 'No se pudo leer el archivo. Asegúrate de que sea un JSON válido.',
   470→        variant: 'destructive'
   471→      });

   473→      setIsImporting(false);

   568→      charLogger.error('Import error', { error, source: 'drag-drop' });
   569→      toast({
   570→        title: 'Error de Importación',
   571→        description: 'Ocurrió un error al importar el personaje.',
   572→        variant: 'destructive'
   573→      });
   574→    } finally {
   575→      setIsImporting(false);
   576→    }
   577→  };
