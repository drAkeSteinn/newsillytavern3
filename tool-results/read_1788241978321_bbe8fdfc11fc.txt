   104→  canAfford,

   121→      </div>

   124→        <div className="flex items-center gap-1.5 text-xs text-muted-foreground">

   126→          {item.description && (
   127→            <>
   128→              <span>•</span>

   138→        </div>
   139→        <Button
   140→          size="sm"
   141→          className="h-7 px-2 text-xs"
