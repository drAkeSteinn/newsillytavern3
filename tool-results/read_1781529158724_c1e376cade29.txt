     2→// Quick Replies Panel - Character-specific quick replies
     3→// Each quick reply can optionally modify character attributes
     4→// ============================================
     5→
     6→'use client';
     7→
     8→import React, { useState } from 'react';
     9→import { Button } from '@/components/ui/button';
    10→import { Input } from '@/components/ui/input';
    11→import { Label } from '@/components/ui/label';
    12→import { Badge } from '@/components/ui/badge';
    13→import {
    14→  Select,
    15→  SelectContent,
    16→  SelectItem,
    17→  SelectTrigger,
    18→  SelectValue,
    19→} from '@/components/ui/select';
    20→import {
    21→  MessageSquare,
    22→  Plus,
    23→  Trash2,
    24→  ChevronDown,
    25→  ChevronUp,
    26→  Zap,
    27→  Settings2,
    28→  X,
    29→  Check,
    30→  ImageIcon,
    31→  Timer,
    32→  GripVertical,
    33→} from 'lucide-react';
    34→import { cn } from '@/lib/utils';
    35→import {
    36→  DndContext,
    37→  closestCenter,
    38→  KeyboardSensor,
    39→  PointerSensor,
    40→  useSensor,
    41→  useSensors,
    42→  DragEndEvent,
    43→} from '@dnd-kit/core';
    44→import {
    45→  arrayMove,
    46→  SortableContext,
    47→  sortableKeyboardCoordinates,
    48→  useSortable,
    49→  verticalListSortingStrategy,
    50→} from '@dnd-kit/sortable';
    51→import { CSS } from '@dnd-kit/utilities';
    52→import type {
    53→  CharacterQuickReply,
    54→  QuickReplyAttributeModifier,
    55→  QuickReplyModifierOperation,
    56→  QuickReplySpriteActivation,
    57→  QuickReplySpriteFallbackMode,
    58→  AttributeDefinition,
    59→  CharacterStatsConfig,
    60→  SpritePackV2,
    61→  TriggerCollection,
    62→} from '@/types';
    63→
    64→interface QuickRepliesPanelProps {
    65→  quickReplies: CharacterQuickReply[] | undefined;
    66→  statsConfig: CharacterStatsConfig | undefined;
    67→  /** Available sprite packs for sprite activation */
    68→  spritePacksV2?: SpritePackV2[];
    69→  /** Available trigger collections for sprite activation */
    70→  triggerCollections?: TriggerCollection[];
    71→  onChange: (quickReplies: CharacterQuickReply[]) => void;
    72→}
    73→
    74→// Operation labels for display
    75→const OPERATION_LABELS: Record<QuickReplyModifierOperation, { label: string; symbol: string; description: string }> = {
    76→  set: { label: 'Establecer', symbol: '=', description: 'Reemplaza el valor actual' },
    77→  add: { label: 'Sumar', symbol: '+', description: 'Agrega al valor actual' },
    78→  subtract: { label: 'Restar', symbol: '-', description: 'Reduce del valor actual' },
    79→  multiply: { label: 'Multiplicar', symbol: '×', description: 'Multiplica el valor actual' },
    80→  divide: { label: 'Dividir', symbol: '÷', description: 'Divide el valor actual' },
    81→};
    82→
    83→function generateId(): string {
    84→  return `qr_${Date.now()}_${Math.random().toString(36).substring(2, 8)}`;
    85→}
    86→
    87→export function QuickRepliesPanel({
    88→  quickReplies,
    89→  statsConfig,
    90→  spritePacksV2,
    91→  triggerCollections,
    92→  onChange,
    93→}: QuickRepliesPanelProps) {
    94→  const replies = quickReplies || [];
    95→  const attributes = statsConfig?.attributes || [];
    96→
    97→  // State for new reply form
    98→  const [newLabel, setNewLabel] = useState('');
    99→  const [newResponse, setNewResponse] = useState('');
   100→  const [newModifiers, setNewModifiers] = useState<QuickReplyAttributeModifier[]>([]);
   101→
   102→  // State for editing existing reply
   103→  const [editingId, setEditingId] = useState<string | null>(null);
   104→  const [editLabel, setEditLabel] = useState('');
   105→  const [editResponse, setEditResponse] = useState('');
   106→  const [editModifiers, setEditModifiers] = useState<QuickReplyAttributeModifier[]>([]);
   107→
   108→  // State for expanded modifiers section
   109→  const [expandedModifiers, setExpandedModifiers] = useState<string | null>(null);
   110→
   111→  // State for sprite activation in new reply
   112→  const [newSpriteActivation, setNewSpriteActivation] = useState<QuickReplySpriteActivation | undefined>(undefined);
   113→
   114→  // State for sprite activation in editing reply
   115→  const [editSpriteActivation, setEditSpriteActivation] = useState<QuickReplySpriteActivation | undefined>(undefined);
   116→
   117→  // State for expanded sprite activation section
   118→  const [expandedSpriteActivation, setExpandedSpriteActivation] = useState<string | null>(null);
   119→
   120→  // Whether sprite activation is available (has packs or collections)
   121→  const hasSpriteOptions = (spritePacksV2 && spritePacksV2.length > 0) || (triggerCollections && triggerCollections.length > 0);
   122→
   123→  const isAdding = newLabel.trim() && newResponse.trim();
   124→
   125→  const handleAdd = () => {
   126→    if (!isAdding) return;
   127→    const newReply: CharacterQuickReply = {
   128→      id: generateId(),
   129→      label: newLabel.trim(),
   130→      response: newResponse.trim(),
   131→      modifiers: newModifiers.length > 0 ? newModifiers : undefined,
   132→      spriteActivation: newSpriteActivation,
   133→    };
   134→    onChange([...replies, newReply]);
   135→    setNewLabel('');
   136→    setNewResponse('');
   137→    setNewModifiers([]);
   138→    setNewSpriteActivation(undefined);
   139→  };
   140→
   141→  const handleDelete = (id: string) => {
   142→    onChange(replies.filter((r) => r.id !== id));
   143→  };
   144→
   145→  const handleStartEdit = (reply: CharacterQuickReply) => {
   146→    setEditingId(reply.id);
   147→    setEditLabel(reply.label);
   148→    setEditResponse(reply.response);
   149→    setEditModifiers(reply.modifiers ? [...reply.modifiers] : []);
   150→    setEditSpriteActivation(reply.spriteActivation ? { ...reply.spriteActivation } : undefined);
   151→  };
   152→
   153→  const handleSaveEdit = () => {
   154→    if (!editingId || !editLabel.trim() || !editResponse.trim()) return;
   155→    onChange(
   156→      replies.map((r) =>
   157→        r.id === editingId
   158→          ? {
   159→              ...r,
   160→              label: editLabel.trim(),
   161→              response: editResponse.trim(),
   162→              modifiers: editModifiers.length > 0 ? editModifiers : undefined,
   163→              spriteActivation: editSpriteActivation,
   164→            }
   165→          : r
   166→      )
   167→    );
   168→    setEditingId(null);
   169→    setEditLabel('');
   170→    setEditResponse('');
   171→    setEditModifiers([]);
   172→    setEditSpriteActivation(undefined);
   173→  };
   174→
   175→  const handleCancelEdit = () => {
   176→    setEditingId(null);
   177→    setEditLabel('');
   178→    setEditResponse('');
   179→    setEditModifiers([]);
   180→    setEditSpriteActivation(undefined);
   181→  };
   182→
   183→  // Add a modifier to the new reply form
   184→  const addModifierToNew = () => {
   185→    if (attributes.length === 0) return;
   186→    setNewModifiers([
   187→      ...newModifiers,
   188→      { attributeKey: attributes[0].key, operation: 'add', value: 1 },
   189→    ]);
   190→  };
   191→
   192→  const updateNewModifier = (index: number, field: keyof QuickReplyAttributeModifier, value: string | number) => {
   193→    const updated = [...newModifiers];
   194→    updated[index] = { ...updated[index], [field]: value };
   195→    setNewModifiers(updated);
   196→  };
   197→
   198→  const removeNewModifier = (index: number) => {
   199→    setNewModifiers(newModifiers.filter((_, i) => i !== index));
   200→  };
   201→
   202→  // Add a modifier to the editing reply form
   203→  const addModifierToEdit = () => {
   204→    if (attributes.length === 0) return;
   205→    setEditModifiers([
   206→      ...editModifiers,
   207→      { attributeKey: attributes[0].key, operation: 'add', value: 1 },
   208→    ]);
   209→  };
   210→
   211→  const updateEditModifier = (index: number, field: keyof QuickReplyAttributeModifier, value: string | number) => {
   212→    const updated = [...editModifiers];
   213→    updated[index] = { ...updated[index], [field]: value };
   214→    setEditModifiers(updated);
   215→  };
   216→
   217→  const removeEditModifier = (index: number) => {
   218→    setEditModifiers(editModifiers.filter((_, i) => i !== index));
   219→  };
   220→
   221→  // Render modifier row
   222→  const renderModifierRow = (
   223→    modifier: QuickReplyAttributeModifier,
   224→    index: number,
   225→    onUpdate: (index: number, field: keyof QuickReplyAttributeModifier, value: string | number) => void,
   226→    onRemove: (index: number) => void
   227→  ) => {
   228→    const attr = attributes.find((a) => a.key === modifier.attributeKey);
   229→    const isTextAttr = attr?.type === 'text' || attr?.type === 'keyword';
   230→    const availableOps: QuickReplyModifierOperation[] = isTextAttr
   231→      ? ['set']
   232→      : ['set', 'add', 'subtract', 'multiply', 'divide'];
   233→
   234→    return (
   235→      <div key={index} className="flex items-center gap-1.5">
   236→        {/* Attribute selector */}
   237→        <Select
   238→          value={modifier.attributeKey}
   239→          onValueChange={(val) => onUpdate(index, 'attributeKey', val)}
   240→        >
   241→          <SelectTrigger className="h-7 text-xs flex-1 min-w-[80px]">
   242→            <SelectValue placeholder="Atributo" />
   243→          </SelectTrigger>
   244→          <SelectContent>
   245→            {attributes.map((attr) => (
   246→              <SelectItem key={attr.key} value={attr.key}>
   247→                <span className="text-xs">{attr.icon ? `${attr.icon} ` : ''}{attr.name}</span>
   248→              </SelectItem>
   249→            ))}
   250→          </SelectContent>
   251→        </Select>
   252→
   253→        {/* Operation selector */}
   254→        <Select
   255→          value={modifier.operation}
   256→          onValueChange={(val) => onUpdate(index, 'operation', val as QuickReplyModifierOperation)}
   257→        >
   258→          <SelectTrigger className="h-7 text-xs w-[90px]">
   259→            <SelectValue />
   260→          </SelectTrigger>
   261→          <SelectContent>
   262→            {availableOps.map((op) => (
   263→              <SelectItem key={op} value={op}>
   264→                <span className="text-xs">
   265→                  {OPERATION_LABELS[op].symbol} {OPERATION_LABELS[op].label}
   266→                </span>
   267→              </SelectItem>
   268→            ))}
   269→          </SelectContent>
   270→        </Select>
   271→
   272→        {/* Value input */}
   273→        <Input
   274→          type={isTextAttr ? 'text' : 'number'}
   275→          value={String(modifier.value)}
   276→          onChange={(e) => {
   277→            const val = isTextAttr ? e.target.value : (parseFloat(e.target.value) || 0);
   278→            onUpdate(index, 'value', val);
   279→          }}
   280→          className="h-7 text-xs w-[70px]"
   281→          placeholder="Valor"
   282→        />
   283→
   284→        {/* Remove button */}
   285→        <Button
   286→          variant="ghost"
   287→          size="sm"
   288→          className="h-7 w-7 p-0 text-muted-foreground hover:text-red-400 flex-shrink-0"
   289→          onClick={() => onRemove(index)}
   290→        >
   291→          <X className="w-3 h-3" />
   292→        </Button>
   293→      </div>
   294→    );
   295→  };
   296→
   297→  // Render sprite activation section
   298→  const renderSpriteActivation = (
   299→    activation: QuickReplySpriteActivation | undefined,
   300→    setActivation: (a: QuickReplySpriteActivation | undefined) => void,
   301→    sectionKey: string,
   302→  ) => {
   303→    const isExpanded = expandedSpriteActivation === sectionKey;
   304→    const isActive = !!activation;
   305→
   306→    return (
   307→      <div className="space-y-2">
   308→        <button
   309→          type="button"
   310→          onClick={() => {
   311→            setExpandedSpriteActivation(isExpanded ? null : sectionKey);
   312→            if (!isActive && !isExpanded) {
   313→              // Pre-populate with defaults
   314→              setActivation({
   315→                mode: 'sprite_pack',
   316→                targetId: '',
   317→                fallbackMode: 'idle_collection',
   318→                fallbackDelayMs: 3000,
   319→              });
   320→            }
   321→          }}
   322→          className="flex items-center gap-1.5 text-xs text-muted-foreground hover:text-foreground transition-colors"
   323→        >
   324→          <ImageIcon className="w-3.5 h-3.5 text-emerald-400" />
   325→          <span>Activación de Sprite</span>
   326→          {isActive && (
   327→            <Badge variant="secondary" className="h-4 text-[10px] px-1 bg-emerald-500/20 text-emerald-400">
   328→              {activation?.mode === 'trigger_collection' ? 'Trigger' : 'Pack'}
   329→            </Badge>
   330→          )}
   331→          {isExpanded ? (
   332→            <ChevronUp className="w-3 h-3" />
   333→          ) : (
   334→            <ChevronDown className="w-3 h-3" />
   335→          )}
   336→        </button>
   337→
   338→        {(isExpanded || isActive) && (
   339→          <div className="ml-5 space-y-2 border-l-2 border-emerald-500/20 pl-3">
   340→            {/* Enable/Disable toggle */}
   341→            <div className="flex items-center gap-2">
   342→              <label className="text-[10px] text-muted-foreground">Activar sprite:</label>
   343→              <button
   344→                type="button"
   345→                onClick={() => {
   346→                  if (isActive) {
   347→                    setActivation(undefined);
   348→                  } else {
   349→                    setActivation({
   350→                      mode: 'sprite_pack',
   351→                      targetId: '',
   352→                      fallbackMode: 'idle_collection',
   353→                      fallbackDelayMs: 3000,
   354→                    });
   355→                  }
   356→                }}
   357→                className={cn(
   358→                  "text-[10px] px-2 py-0.5 rounded-full transition-colors",
   359→                  isActive ? "bg-emerald-500/20 text-emerald-400" : "bg-muted text-muted-foreground"
   360→                )}
   361→              >
   362→                {isActive ? 'Activado' : 'Desactivado'}
   363→              </button>
   364→            </div>
   365→
   366→            {isActive && activation && (
   367→              <>
   368→                {/* Mode selector */}
   369→                <div className="flex items-center gap-2">
   370→                  <Label className="text-[10px] text-muted-foreground w-16 flex-shrink-0">Modo</Label>
   371→                  <Select
   372→                    value={activation.mode}
   373→                    onValueChange={(val) => setActivation({ ...activation, mode: val as 'trigger_collection' | 'sprite_pack', targetId: '' })}
   374→                  >
   375→                    <SelectTrigger className="h-7 text-xs flex-1">
   376→                      <SelectValue />
   377→                    </SelectTrigger>
   378→                    <SelectContent>
   379→                      <SelectItem value="sprite_pack">
   380→                        <span className="text-xs">📦 Sprite Pack (condicional)</span>
   381→                      </SelectItem>
   382→                      <SelectItem value="trigger_collection">
   383→                        <span className="text-xs">🎯 Colección de Triggers</span>
   384→                      </SelectItem>
   385→                    </SelectContent>
   386→                  </Select>
   387→                </div>
   388→
   389→                {/* Target selector */}
   390→                <div className="flex items-center gap-2">
   391→                  <Label className="text-[10px] text-muted-foreground w-16 flex-shrink-0">Objetivo</Label>
   392→                  <Select
   393→                    value={activation.targetId}
   394→                    onValueChange={(val) => setActivation({ ...activation, targetId: val })}
   395→                  >
   396→                    <SelectTrigger className="h-7 text-xs flex-1">
   397→                      <SelectValue placeholder={activation.mode === 'trigger_collection' ? 'Seleccionar trigger...' : 'Seleccionar pack...'} />
   398→                    </SelectTrigger>
   399→                    <SelectContent>
   400→                      {activation.mode === 'trigger_collection' ? (
   401→                        (triggerCollections || []).map((tc) => (
   402→                          <SelectItem key={tc.id} value={tc.id}>
   403→                            <span className="text-xs">🎯 {tc.name} (Prioridad: {tc.priority})</span>
   404→                          </SelectItem>
   405→                        ))
   406→                      ) : (
   407→                        (spritePacksV2 || []).map((pack) => (
   408→                          <SelectItem key={pack.id} value={pack.id}>
   409→                            <span className="text-xs">📦 {pack.name} {pack.conditionalMode ? '(condicional)' : ''}</span>
   410→                          </SelectItem>
   411→                        ))
   412→                      )}
   413→                    </SelectContent>
   414→                  </Select>
   415→                </div>
   416→
   417→                {/* Fallback mode */}
   418→                <div className="flex items-center gap-2">
   419→                  <Label className="text-[10px] text-muted-foreground w-16 flex-shrink-0">Al terminar</Label>
   420→                  <Select
   421→                    value={activation.fallbackMode}
   422→                    onValueChange={(val) => setActivation({ ...activation, fallbackMode: val as QuickReplySpriteFallbackMode })}
   423→                  >
   424→                    <SelectTrigger className="h-7 text-xs flex-1">
   425→                      <SelectValue />
   426→                    </SelectTrigger>
   427→                    <SelectContent>
   428→                      <SelectItem value="idle_collection">
   429→                        <span className="text-xs">🔄 Volver al estado normal</span>
   430→                      </SelectItem>
   431→                      <SelectItem value="collection_default">
   432→                        <span className="text-xs">🖼️ Sprite principal del pack</span>
   433→                      </SelectItem>
   434→                      <SelectItem value="custom_sprite">
   435→                        <span className="text-xs">✨ Sprite personalizado</span>
   436→                      </SelectItem>
   437→                    </SelectContent>
   438→                  </Select>
   439→                </div>
   440→
   441→                {/* Fallback delay */}
   442→                <div className="flex items-center gap-2">
   443→                  <Label className="text-[10px] text-muted-foreground w-16 flex-shrink-0">
   444→                    <Timer className="w-3 h-3 inline mr-1" />
   445→                    Duración
   446→                  </Label>
   447→                  <Input
   448→                    type="number"
   449→                    value={activation.fallbackDelayMs}
   450→                    onChange={(e) => setActivation({ ...activation, fallbackDelayMs: parseInt(e.target.value) || 0 })}
   451→                    className="h-7 text-xs w-[80px]"
   452→                    min={0}
   453→                    step={500}
   454→                  />
   455→                  <span className="text-[10px] text-muted-foreground">ms (0 = persistir)</span>
   456→                </div>
   457→
   458→                {/* Custom sprite for fallback (only when mode is custom_sprite) */}
   459→                {activation.fallbackMode === 'custom_sprite' && (
   460→                  <div className="flex items-center gap-2">
   461→                    <Label className="text-[10px] text-muted-foreground w-16 flex-shrink-0">Sprite fall.</Label>
   462→                    <Select
   463→                      value={activation.fallbackSpriteId || ''}
   464→                      onValueChange={(val) => setActivation({ ...activation, fallbackSpriteId: val })}
   465→                    >
   466→                      <SelectTrigger className="h-7 text-xs flex-1">
   467→                        <SelectValue placeholder="Seleccionar sprite..." />
   468→                      </SelectTrigger>
   469→                      <SelectContent>
   470→                        {(spritePacksV2 || []).flatMap((pack) =>
   471→                          pack.sprites.map((sprite) => (
   472→                            <SelectItem key={sprite.id} value={sprite.id}>
   473→                              <span className="text-xs">{pack.name} / {sprite.label}</span>
   474→                            </SelectItem>
   475→                          ))
   476→                        )}
   477→                      </SelectContent>
   478→                    </Select>
   479→                  </div>
   480→                )}
   481→              </>
   482→            )}
   483→          </div>
   484→        )}
   485→      </div>
   486→    );
   487→  };
   488→
   489→  // DnD sensors
   490→  const sensors = useSensors(
   491→    useSensor(PointerSensor, {
   492→      activationConstraint: {
   493→        distance: 5,
   494→      },
   495→    }),
   496→    useSensor(KeyboardSensor, {
   497→      coordinateGetter: sortableKeyboardCoordinates,
   498→    })
   499→  );
   500→
   501→  const handleDragEnd = (event: DragEndEvent) => {
   502→    const { active, over } = event;
   503→    if (over && active.id !== over.id) {
   504→      const oldIndex = replies.findIndex((r) => r.id === active.id);
   505→      const newIndex = replies.findIndex((r) => r.id === over.id);
   506→      if (oldIndex !== -1 && newIndex !== -1) {
   507→        onChange(arrayMove(replies, oldIndex, newIndex));
   508→      }
   509→    }
   510→  };
   511→
   512→  return (
   513→    <div className="space-y-6">
   514→      {/* Header */}
   515→      <div className="bg-gradient-to-r from-violet-500/10 to-purple-500/10 border border-violet-500/20 rounded-lg p-3">
   516→        <div className="flex items-start gap-3">
   517→          <div className="p-2 bg-violet-500/20 rounded-lg">
   518→            <MessageSquare className="w-5 h-5 text-violet-500" />
   519→          </div>
   520→          <div className="flex-1">
   521→            <h4 className="text-sm font-medium text-violet-600">Respuestas Rápidas</h4>
   522→            <p className="text-xs text-muted-foreground mt-1">
   523→              Botones de acceso rápido personalizados para este personaje.{' '}
   524→              <strong>Etiqueta</strong> es lo que se ve, <strong>Respuesta</strong> es lo que se envía.{' '}
   525→              Puedes usar <code className="text-[10px] bg-muted px-1 rounded">{'{{char}}'}</code> y{' '}
   526→              <code className="text-[10px] bg-muted px-1 rounded">{'{{user}}'}</code> para insertar nombres.
   527→            </p>
   528→            {attributes.length > 0 && (
   529→              <p className="text-xs text-muted-foreground mt-1">
   530→                Opcionalmente puedes agregar <Zap className="w-3 h-3 inline text-amber-400" /> modificadores de atributos que se aplican al usar la respuesta.
   531→              </p>
   532→            )}
   533→            {hasSpriteOptions && (
   534→              <p className="text-xs text-muted-foreground mt-1">
   535→                También puedes activar <ImageIcon className="w-3 h-3 inline text-emerald-400" /> sprites al usar la respuesta.
   536→              </p>
   537→            )}
   538→            {replies.length > 1 && (
   539→              <p className="text-xs text-muted-foreground mt-1">
   540→                <GripVertical className="w-3 h-3 inline text-muted-foreground" /> Arrastra para reordenar las respuestas.
   541→              </p>
   542→            )}
   543→          </div>
   544→        </div>
   545→      </div>
   546→
   547→      {/* Existing replies */}
   548→      {replies.length > 0 && (
   549→        <DndContext
   550→          sensors={sensors}
   551→          collisionDetection={closestCenter}
   552→          onDragEnd={handleDragEnd}
   553→        >
   554→          <SortableContext
   555→            items={replies.map((r) => r.id)}
   556→            strategy={verticalListSortingStrategy}
   557→          >
   558→            <div className="space-y-2">
   559→              {replies.map((reply) => (
   560→                <SortableQuickReplyItem
   561→                  key={reply.id}
   562→                  reply={reply}
   563→                  editingId={editingId}
   564→                  editLabel={editLabel}
   565→                  editResponse={editResponse}

   692→        </p>
   693→      )}
   694→
   695→      {replies.length === 0 && (
   696→        <div className="text-center py-6 text-muted-foreground">
   697→          <MessageSquare className="w-8 h-8 mx-auto mb-2 opacity-30" />

   699→          <p className="text-xs opacity-60">Agrega respuestas rápidas para acceso directo en el chat</p>
   700→        </div>
   701→      )}
   702→    </div>
   703→  );
