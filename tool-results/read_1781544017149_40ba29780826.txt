   459→function ExtraccionTab() {
   460→  const settings = useTavernStore((s) => s.settings);
   461→  const embeddingsChat = settings.embeddingsChat ?? DEFAULT_EMBEDDINGS_CHAT;
   462→  const updateSettings = useTavernStore((state) => state.updateSettings);
   463→
   464→  // Context settings with defaults
   465→  const contextSettings = settings.context ?? {
   466→    maxMessages: 50,
   467→    maxTokens: 4096,
   468→    keepFirstN: 1,
   469→    keepLastN: 20,
   470→  };
   471→
   472→  // Update context settings helper
   473→  const updateContextSettings = useCallback((updates: Partial<typeof contextSettings>) => {
   474→    updateSettings({
   475→      context: { ...contextSettings, ...updates }
   476→    });
   477→  }, [contextSettings, updateSettings]);
   478→
   479→  // Prompt editor state
   480→  const [activePromptTab, setActivePromptTab] = useState<'normal' | 'group'>('normal');
   481→  const [localPrompt, setLocalPrompt] = useState(() => embeddingsChat.memoryExtractionPrompt || DEFAULT_MEMORY_EXTRACTION_PROMPT);
   482→  const [localGroupPrompt, setLocalGroupPrompt] = useState(() => embeddingsChat.groupMemoryExtractionPrompt || DEFAULT_GROUP_MEMORY_EXTRACTION_PROMPT);
   483→  const [showPreview, setShowPreview] = useState(false);
   484→
   485→  const isNormal = activePromptTab === 'normal';
   486→  const currentPrompt = isNormal ? localPrompt : localGroupPrompt;
   487→  const currentStored = isNormal
   488→    ? (embeddingsChat.memoryExtractionPrompt || DEFAULT_MEMORY_EXTRACTION_PROMPT)
   489→    : (embeddingsChat.groupMemoryExtractionPrompt || DEFAULT_GROUP_MEMORY_EXTRACTION_PROMPT);
   490→  const previewData = isNormal ? NORMAL_PREVIEW : GROUP_PREVIEW;
   491→
   492→  const handleSavePrompt = () => {
   493→    if (isNormal) {
   494→      updateSettings({ embeddingsChat: { ...embeddingsChat, memoryExtractionPrompt: localPrompt } });
   495→    } else {
   496→      updateSettings({ embeddingsChat: { ...embeddingsChat, groupMemoryExtractionPrompt: localGroupPrompt } });
   497→    }
   498→  };
   499→
   500→  const handleRestoreDefault = () => {
   501→    if (isNormal) {
   502→      setLocalPrompt(DEFAULT_MEMORY_EXTRACTION_PROMPT);
   503→      updateSettings({ embeddingsChat: { ...embeddingsChat, memoryExtractionPrompt: DEFAULT_MEMORY_EXTRACTION_PROMPT } });
   504→    } else {
   505→      setLocalGroupPrompt(DEFAULT_GROUP_MEMORY_EXTRACTION_PROMPT);
   506→      updateSettings({ embeddingsChat: { ...embeddingsChat, groupMemoryExtractionPrompt: DEFAULT_GROUP_MEMORY_EXTRACTION_PROMPT } });
   507→    }
   508→  };
   509→
   510→  const handleChange = (value: string) => {
   511→    if (isNormal) setLocalPrompt(value);
   512→    else setLocalGroupPrompt(value);
   513→  };
   514→
   515→  const previewText = currentPrompt
   516→    .replace('{characterName}', previewData.characterName)
   517→    .replace('{chatContext}', previewData.chatContext)
   518→    .replace('{lastMessage}', previewData.lastMessage);
   519→
   520→  const hasChanges = currentPrompt !== currentStored;
   521→
   522→  return (
   523→    <div className="space-y-6">
   524→      {/* Note about needing Embeddings enabled */}
   525→      <div className="bg-amber-500/5 border border-amber-500/20 rounded-lg p-4">
   526→        <div className="flex items-start gap-2">
   527→          <Info className="w-4 h-4 text-amber-500 mt-0.5 shrink-0" />
   528→          <div className="space-y-1">
   529→            <p className="text-sm font-medium text-amber-600 dark:text-amber-400">Requiere Embeddings</p>
   530→            <p className="text-xs text-muted-foreground">
   531→              Estos ajustes requieren que Embeddings esté activado. Configúralo en la pestaña <strong>Base de Conocimiento</strong>.
   532→            </p>
   533→          </div>
   534→        </div>
   535→      </div>
   536→
   537→      {/* Memory Extraction Section */}
   538→      <Card>
   539→        <CardHeader className="pb-3">
   540→          <CardTitle className="flex items-center gap-2 text-base">
   541→            <Brain className="w-4 h-4 text-violet-500" />
   542→            Extracción Automática de Memoria
   543→          </CardTitle>
   544→          <CardDescription>
   545→            Extrae hechos memorables de las respuestas del personaje y los guarda como embeddings
   546→          </CardDescription>
   547→        </CardHeader>
   548→        <CardContent className="space-y-4">
   549→          <div className="flex items-center justify-between">
   550→            <div className="space-y-0.5">
   551→              <Label className="text-sm">Activar Extracción</Label>
   552→              <p className="text-[10px] text-muted-foreground">
   553→                Extrae automáticamente hechos memorables tras cada respuesta del personaje
   554→              </p>
   555→            </div>
