   146→   146→function fmtDate(iso: string): string {
   147→   147→  if (!iso) return '';
   148→   148→  const d = new Date(iso);
   149→   149→  if (isNaN(d.getTime())) return iso.slice(0, 10);
   150→   150→  return d.toLocaleDateString('es-ES', { day: '2-digit', month: '2-digit', year: 'numeric' });
   151→   151→}
