    44→/**
    45→ * Apply a single item attribute effect to SessionStats via updateCharacterStat.
    46→ * This directly modifies the session stats so the UI reflects the change.
    47→ *
    48→ * Supports both numeric and text/keyword attributes:
    49→ * - Numeric: applies arithmetic operators (+, -, *, /, =, set_min, set_max)
    50→ * - Text/Keyword: only '=' operator is valid, sets the attribute to the string value
    51→ *
    52→ * IMPORTANT: When the character stats for a target don't exist in the session
    53→ * yet (e.g., persona stats were configured after session creation), we must
    54→ * look up the default value from the statsConfig BEFORE computing the effect.
    55→ */
    56→function applyEffectToSessionStats(
    57→  stateAny: any,
    58→  effect: ItemAttributeEffect
    59→): void {
    60→  const sessionId = stateAny.activeSessionId as string | undefined;
    61→  if (!sessionId) return;
    62→
    63→  const targetId = effect.targetId || '__user__';
    64→
    65→  // Detect if this is a text/keyword attribute by checking the statsConfig
    66→  const attrType = getAttributeType(stateAny, targetId, effect.attributeKey);
    67→  const isTextAttr = attrType === 'text' || attrType === 'keyword';
    68→
    69→  // For text/keyword attributes with '=' operator, set the string value directly
    70→  if (isTextAttr && effect.operator === '=') {
    71→    const newValue = typeof effect.value === 'string' ? effect.value : String(effect.value);
    72→    stateAny.updateCharacterStat?.(sessionId, targetId, effect.attributeKey, newValue, 'trigger');
    73→    return;
    74→  }
    75→
    76→  // For numeric attributes, proceed with arithmetic operations
    77→  let currentValue = stateAny.getAttributeValue?.(sessionId, targetId, effect.attributeKey);
    78→
    79→  // If the attribute doesn't exist yet in session stats, look up the default
    80→  // value from the character's/persona's statsConfig.
    81→  if (currentValue === null || currentValue === undefined) {
    82→    const defaultValue = getDefaultAttributeValue(stateAny, targetId, effect.attributeKey);
    83→    if (defaultValue !== null) {
    84→      currentValue = defaultValue;
    85→    }
    86→  }
    87→
    88→  // If still null, default to 0 for numeric operations (truly new attribute)
    89→  const currentNum = (currentValue !== null && currentValue !== undefined)
    90→    ? (typeof currentValue === 'number' ? currentValue : parseFloat(String(currentValue)))
    91→    : 0;
    92→
    93→  if (isNaN(currentNum) && currentValue !== null && currentValue !== undefined) {
    94→    // If we can't parse as number but it's a text value with '=' operator, set directly
    95→    if (effect.operator === '=' && typeof effect.value === 'string') {
    96→      stateAny.updateCharacterStat?.(sessionId, targetId, effect.attributeKey, effect.value, 'trigger');
    97→      return;
    98→    }
    99→    return;
   100→  }
   101→
   102→  const effectNumValue = typeof effect.value === 'number' ? effect.value : parseFloat(String(effect.value));
   103→
   104→  let newValue: number | string = currentNum;
   105→  switch (effect.operator) {
   106→    case '+': newValue = currentNum + effectNumValue; break;
   107→    case '-': newValue = currentNum - effectNumValue; break;
   108→    case '*': newValue = currentNum * effectNumValue; break;
   109→    case '/': newValue = effectNumValue !== 0 ? currentNum / effectNumValue : currentNum; break;
   110→    case '=': newValue = effectNumValue; break;
   111→    case 'set_min': newValue = Math.min(currentNum, effectNumValue); break;
   112→    case 'set_max': newValue = Math.max(currentNum, effectNumValue); break;
   113→  }
   114→
   115→  newValue = Math.round((newValue as number) * 100) / 100;
   116→
   117→  stateAny.updateCharacterStat?.(sessionId, targetId, effect.attributeKey, newValue, 'trigger');
   118→}

   120→/**
   121→ * Look up the default value for an attribute from the character's/persona's statsConfig.
   122→ * Returns null if no default is found.
   123→ */
   124→function getDefaultAttributeValue(
   125→  stateAny: any,
   126→  targetId: string,
   127→  attributeKey: string
   128→): number | string | null {
   129→  let statsConfig: any = undefined;
   130→
   131→  if (targetId === '__user__') {
   132→    const activePersonaId = stateAny.activePersonaId;
   133→    const personas: any[] = stateAny.personas || [];
   134→    const activePersona = personas.find((p: any) => p.id === activePersonaId);
   135→    statsConfig = activePersona?.statsConfig;
   136→  } else {
   137→    const characters: any[] = stateAny.characters || [];
   138→    const character = characters.find((c: any) => c.id === targetId);
   139→    statsConfig = character?.statsConfig;
   140→  }
   141→
   142→  if (!statsConfig?.attributes) return null;
   143→
   144→  const attrDef = statsConfig.attributes.find((a: any) => a.key === attributeKey);
   145→  if (attrDef && attrDef.defaultValue !== undefined) {
   146→    return attrDef.defaultValue;
   147→  }
   148→
   149→  return null;
   150→}

   152→/**
   153→ * Look up the type of an attribute from the character's/persona's statsConfig.
   154→ * Returns 'number', 'text', 'keyword', or null if not found.
   155→ */
   156→function getAttributeType(
   157→  stateAny: any,
   158→  targetId: string,
   159→  attributeKey: string
   160→): 'number' | 'text' | 'keyword' | null {
   161→  let statsConfig: any = undefined;
   162→
   163→  if (targetId === '__user__') {
   164→    const activePersonaId = stateAny.activePersonaId;
   165→    const personas: any[] = stateAny.personas || [];
   166→    const activePersona = personas.find((p: any) => p.id === activePersonaId);
   167→    statsConfig = activePersona?.statsConfig;
   168→  } else {
   169→    const characters: any[] = stateAny.characters || [];
   170→    const character = characters.find((c: any) => c.id === targetId);
   171→    statsConfig = character?.statsConfig;
   172→  }
   173→
   174→  if (!statsConfig?.attributes) return null;
   175→
   176→  const attrDef = statsConfig.attributes.find((a: any) => a.key === attributeKey);
   177→  return attrDef?.type || null;
   178→}

   181→ * Apply multiple item effects to SessionStats.

   183→function applyEffectsToSessionStats(
   184→  stateAny: any,
   185→  effects: ItemAttributeEffect[]
   186→): void {
   187→  for (const effect of effects) {
   188→    applyEffectToSessionStats(stateAny, effect);

   192→/**
   193→ * Get the cycled text value for a dynamic text effect.
   194→ * For values containing `|` separators, returns the value at (activeTurns % parts.length).
   195→ * For single values, returns the value as-is.
   196→ */
   197→function getDynamicTextValue(value: string | number, activeTurns: number): string | number {
   198→  if (typeof value !== 'string') return value;
   199→  const parts = value.split('|').map(p => p.trim()).filter(Boolean);
   200→  if (parts.length <= 1) return value;
   201→  return parts[activeTurns % parts.length];
   202→}

   204→/**
   205→ * Resolve the {{slot}} template key in an item message.
   206→ * Replaces {{slot}} with the display name of the equipment slot.
   207→ * If the slot is not found or slotId is empty, removes the {{slot}} key.
   208→ *
   209→ * Example:
   210→ *   "Equipaste espada en {{slot}}" → "Equipaste espada en mano derecha"
   211→ */
   212→function resolveSlotKeyInMessage(
   213→  message: string,
   214→  slotId: string | undefined,
   215→  equipmentSlots: EquipmentSlotDefinition[]
   216→): string {
   217→  if (!message || !/\{\{slot\}\}/gi.test(message)) return message;
   218→
   219→  const slotDef = slotId ? equipmentSlots.find(s => s.id === slotId) : undefined;
   220→  const slotName = slotDef?.name || '';
   221→  return message.replace(/\{\{slot\}\}/gi, slotName);
   222→}

   224→/**
   225→ * Sync active consumable effects to the current session for per-session storage.
   226→ * This ensures effects are persisted in the session JSON and available for prompt building.
   227→ * Called after every mutation of activeConsumableEffects.
   228→ */
   229→function syncEffectsToSession(stateAny: any, effects: ActiveConsumableEffect[]): void {
   230→  const sessionId = stateAny.activeSessionId as string | undefined;
   231→  if (!sessionId) return;
   232→  try {
   233→    stateAny.updateSession?.(sessionId, { activeConsumableEffects: effects });
   234→  } catch {
   235→    // Non-critical — session might not exist yet
   236→  }
   237→}

   239→/**
   240→ * Apply a single dynamic effect for a given turn.
   241→ * - For text/keyword attributes: cycles through `|`-separated values or sets single value.
   242→ * - For numeric attributes: applies the operator to the current value (cumulative).
   243→ */
   244→function applyDynamicEffectToSessionStats(
   245→  stateAny: any,
   246→  effect: ItemAttributeEffect,
   247→  activeTurns: number
   248→): void {
   249→  const sessionId = stateAny.activeSessionId as string | undefined;
   250→  if (!sessionId) return;
   251→
   252→  const targetId = effect.targetId || '__user__';
   253→  const attrType = getAttributeType(stateAny, targetId, effect.attributeKey);
   254→  const isTextAttr = attrType === 'text' || attrType === 'keyword';
   255→
   256→  if (isTextAttr) {
   257→    // For text attributes, cycle through values or set single value
   258→    const textValue = getDynamicTextValue(effect.value, activeTurns);
   259→    const newValue = typeof textValue === 'string' ? textValue : String(textValue);
   260→    stateAny.updateCharacterStat?.(sessionId, targetId, effect.attributeKey, newValue, 'trigger');
   261→    return;
   262→  }
   263→
   264→  // For numeric attributes, apply the operator (cumulative)
   265→  applyEffectToSessionStats(stateAny, effect);
   266→}

   268→/**
   269→ * Apply a fallback value to SessionStats for a given target/attribute.
   270→ * If no fallback value is provided, reverse the operator of the given effect.
   271→ * Supports both numeric and text/keyword attributes.
   272→ */
   273→function applyFallbackToSessionStats(
   274→  stateAny: any,
   275→  targetId: string,
   276→  attributeKey: string,
   277→  fallbackValue: string | number | undefined,
   278→  effect?: ItemAttributeEffect
   279→): void {
   280→  const sessionId = stateAny.activeSessionId as string | undefined;
   281→  if (!sessionId) return;
   282→
   283→  const resolvedTargetId = targetId || '__user__';
   284→
   285→  // Detect attribute type for proper handling
   286→  const attrType = getAttributeType(stateAny, resolvedTargetId, attributeKey);
   287→  const isTextAttr = attrType === 'text' || attrType === 'keyword';
   288→
   289→  if (fallbackValue !== undefined) {
   290→    // Use explicit fallback value - for text attrs keep as string, for numeric parse
   291→    let valueToSet: string | number;
   292→    if (isTextAttr) {
   293→      valueToSet = typeof fallbackValue === 'string' ? fallbackValue : String(fallbackValue);
   294→    } else {
   295→      valueToSet = typeof fallbackValue === 'number'
   296→        ? fallbackValue
   297→        : (isNaN(Number(fallbackValue)) ? fallbackValue : Number(fallbackValue));
   298→    }
   299→    try {
   300→      stateAny.updateCharacterStat?.(sessionId, resolvedTargetId, attributeKey, valueToSet, 'trigger');
   301→    } catch {
   302→      // Non-critical
   303→    }
   304→  } else if (effect) {
   305→    // No fallback - for text/keyword attributes with '=', we can't reverse without fallback
   306→    if (isTextAttr) {
   307→      // Text attributes cannot be auto-reversed, skip silently
   308→      return;
   309→    }
   310→
   311→    // No fallback - reverse the operator for numeric attributes
   312→    const currentValue = stateAny.getAttributeValue?.(sessionId, resolvedTargetId, attributeKey);
   313→    if (currentValue === null || currentValue === undefined) return;
   314→
   315→    const currentNum = typeof currentValue === 'number' ? currentValue : parseFloat(String(currentValue));
   316→    if (isNaN(currentNum)) return;
   317→
   318→    const effectNumValue = typeof effect.value === 'number' ? effect.value : parseFloat(String(effect.value));
   319→
   320→    let newValue = currentNum;
   321→    switch (effect.operator) {
   322→      case '+': newValue = currentNum - effectNumValue; break; // reverse add
   323→      case '-': newValue = currentNum + effectNumValue; break; // reverse subtract
   324→      case '*': newValue = effectNumValue !== 0 ? currentNum / effectNumValue : currentNum; break; // reverse multiply
   325→      case '/': newValue = currentNum * effectNumValue; break; // reverse divide
   326→      case '=': break; // can't reverse a set without fallback
   327→      case 'set_min': break; // can't reverse without fallback
   328→      case 'set_max': break; // can't reverse without fallback
   329→      default: break;
   330→    }
   331→
   332→    if (newValue !== currentNum) {
   333→      newValue = Math.round(newValue * 100) / 100;
   334→      try {
   335→        stateAny.updateCharacterStat?.(sessionId, resolvedTargetId, attributeKey, newValue, 'trigger');
   336→      } catch {
   337→        // Non-critical
   338→      }
   339→    }
   340→  }
   341→}

   490→
   491→export const createInventorySlice: StateCreator<InventorySlice, [], [], InventorySlice> = (set, get) => ({
   492→  // Initial State
   493→  items: [],
   494→  activeConsumableEffects: [],
   495→  inventorySettings: DEFAULT_INVENTORY_V2_SETTINGS,
   496→  inventoryNotifications: [],
   497→  pendingItemMessage: null,
   498→  pendingEquipAction: null,
   499→  pendingFallbacks: [],
   500→  dynamicEquipmentState: {},
   501→
   502→  // ===== Item Registry Actions =====
   503→  addItem: (item) => set((state) => ({
   504→    items: [...state.items, item]
   505→  })),
   506→
   507→  updateItem: (id, updates) => set((state) => ({
   508→    items: state.items.map(item =>
   509→      item.id === id ? { ...item, ...updates, updatedAt: new Date().toISOString() } : item
   510→    )
   511→  })),
   512→
   513→  deleteItem: (id) => {
   514→    const stateAny = get() as any;
   515→    const item = get().getItemById(id);
   516→
   517→    // Reverse effects and clean up personas that have this item
   518→    if (item) {
   519→      const personas = stateAny.personas as Array<{ id: string; inventoryItems?: PersonaInventoryEntry[] }> | undefined;
   520→      if (personas) {
   521→        for (const persona of personas) {
   522→          const entry = persona.inventoryItems?.find(e => e.itemId === id);
   523→          if (entry) {
   524→            // If the item is equipped, reverse its effects first
   525→            if (entry.equipped && item.type === 'equipment' && item.attributeEffects) {
   526→              for (const ae of item.attributeEffects) {
   527→                const effectTargetId = entry.targetOverrideId || ae.targetId;
   528→                applyFallbackToSessionStats(stateAny, effectTargetId, ae.attributeKey, ae.fallbackValue, ae);
   529→              }
   530→            }
   531→            // Remove the item from this persona's inventory
   532→            const updatedItems = persona.inventoryItems?.filter(e => e.itemId !== id) || [];
   533→            stateAny.updatePersona(persona.id, { inventoryItems: updatedItems });
   534→          }
   535→        }
   536→      }
   537→    }
   538→
   539→    // Also reverse active consumable effects for this item
   540→    const activeEffects = get().activeConsumableEffects.filter(e => e.itemId === id);

   958→  tickEffects: (personaId) => {
   959→    const expiredMessages: string[] = [];
   960→    const stateAny = get() as any;
   961→
   962→    // 1. Apply dynamic consumable effects before decrementing
   963→    const consumableEffects = get().activeConsumableEffects.filter(e => e.personaId === personaId);
   964→    for (const activeEffect of consumableEffects) {
   965→      const dynamicEffects = activeEffect.effects.filter(e => e.mode === 'dynamic');
   966→      const activeTurns = activeEffect.totalTurns - activeEffect.remainingTurns;
   967→      for (const dynEffect of dynamicEffects) {
   968→        applyDynamicEffectToSessionStats(stateAny, dynEffect, activeTurns);
   969→      }
   970→    }
   971→
   972→    // 2. Apply dynamic equipment effects and increment turn counters
   973→    const dynamicEqState = { ...get().dynamicEquipmentState };
   974→    const equippedItems = get().getEquippedItems(personaId);
   975→    for (const { entry, item } of equippedItems) {
   976→      const dynamicEffects = (item.attributeEffects || []).filter((e: ItemAttributeEffect) => e.mode === 'dynamic');
   977→      if (dynamicEffects.length > 0) {
   978→        const stateKey = `${personaId}:${item.id}`;
   979→        const currentState = dynamicEqState[stateKey] || { activeTurns: 0, appliedAt: new Date().toISOString() };
   980→        for (const dynEffect of dynamicEffects) {
   981→          const resolvedEffect = entry.targetOverrideId ? { ...dynEffect, targetId: entry.targetOverrideId } : dynEffect;
   982→          applyDynamicEffectToSessionStats(stateAny, resolvedEffect, currentState.activeTurns);
   983→        }
   984→        dynamicEqState[stateKey] = { ...currentState, activeTurns: currentState.activeTurns + 1 };
   985→      }
   986→    }
   987→
   988→    // 3. Decrement consumable turns
   989→    set((state) => ({
   990→      activeConsumableEffects: state.activeConsumableEffects.map(effect => {
   991→        if (effect.personaId !== personaId) return effect;
   992→        const newRemaining = effect.remainingTurns - 1;
   993→        if (newRemaining <= 0) {
   994→          const msg = effect.expireMessage || `El efecto de ${effect.itemName} ha expirado`;
   995→          expiredMessages.push(msg);
   996→        }
   997→        return { ...effect, remainingTurns: Math.max(0, newRemaining) };
   998→      }),
   999→      dynamicEquipmentState: dynamicEqState,
  1000→    }));
  1001→
  1002→    // Sync to session for per-session storage
  1003→    syncEffectsToSession(stateAny, get().activeConsumableEffects);
  1004→
  1005→    return expiredMessages;
  1006→  },

  1011→
  1012→    set((state) => {
  1013→      const expired = state.activeConsumableEffects.filter(
  1014→        e => e.personaId === personaId && e.remainingTurns <= 0
  1015→      );
  1016→      for (const e of expired) {
  1017→        const msg = e.expireMessage || `El efecto de ${e.itemName} ha expirado`;

  1022→          itemId: e.itemId,
  1023→          itemName: e.itemName,
  1024→          quantity: 1,
  1025→          message: msg,
  1026→        });
  1027→      }
  1028→
  1029→      // Apply fallback values directly to SessionStats for expired effects
  1030→      for (const effect of expired) {
  1031→        const item = get().getItemById(effect.itemId);
  1032→        if (!item?.attributeEffects) continue;
  1033→        for (const ae of item.attributeEffects) {
  1034→          // Use the effect's overridden targetId if it was set (from executeUseWithTarget)
  1035→          const activeEffect = effect.effects.find(e => e.attributeKey === ae.attributeKey);
  1036→          const effectTargetId = activeEffect?.targetId || ae.targetId;
  1037→          applyFallbackToSessionStats(stateAny, effectTargetId, ae.attributeKey, ae.fallbackValue, ae);
  1038→        }
  1039→      }
  1040→
  1041→      return {
  1042→        activeConsumableEffects: state.activeConsumableEffects.filter(
  1043→          e => !(e.personaId === personaId && e.remainingTurns <= 0)
  1044→        ),
  1045→        pendingFallbacks: state.pendingFallbacks || [], // Keep for backward compat but no new additions
  1046→      };
  1047→    });
  1048→
  1049→    // Sync to session for per-session storage
  1050→    syncEffectsToSession(stateAny, get().activeConsumableEffects);
  1051→
  1052→    return expiredMessages;

  1065→  removeEffect: (effectId) => {

  1370→  executeUseWithTarget: (personaId, itemId, targetOverrideId) => {

  1554→
  1555→    // Sync imported effects to session for per-session storage
  1556→    const stateAny = get() as any;
  1557→    syncEffectsToSession(stateAny, get().activeConsumableEffects);
  1558→  },
  1559→});

  1657→/**
  1658→ * Apply item effects to session stats
  1659→ * This modifies session stats based on active item effects (equipment + consumables)
  1660→ * Returns the modified stats
  1661→ */
  1662→export function applyItemEffectsToStats(
  1663→  baseStats: Record<string, { value: number | string }>,
  1664→  effects: ItemAttributeEffect[],
  1665→  targetId: string
  1666→): Record<string, { value: number | string; modified: boolean; modifier?: string }> {
  1667→  const result: Record<string, { value: number | string; modified: boolean; modifier?: string }> = {};
  1668→
  1669→  // Copy base stats
  1670→  for (const [key, stat] of Object.entries(baseStats)) {
  1671→    result[key] = { ...stat, modified: false };
  1672→  }
  1673→
  1674→  // Apply effects
  1675→  for (const effect of effects) {
  1676→    if (effect.targetId !== targetId) continue;
  1677→
  1678→    const stat = result[effect.attributeKey];
  1679→    if (!stat || typeof stat.value !== 'number') continue;
  1680→
  1681→    const originalValue = stat.value;
  1682→    let newValue = originalValue;
  1683→
  1684→    switch (effect.operator) {
  1685→      case '+':
  1686→        newValue = originalValue + effect.value;
  1687→        break;
  1688→      case '-':
  1689→        newValue = originalValue - effect.value;
  1690→        break;
  1691→      case '*':
  1692→        newValue = originalValue * effect.value;
  1693→        break;
  1694→      case '/':
  1695→        newValue = effect.value !== 0 ? originalValue / effect.value : originalValue;
  1696→        break;
  1697→      case '=':
  1698→        newValue = effect.value;
  1699→        break;
  1700→      case 'set_min':
  1701→        newValue = Math.min(originalValue, effect.value);
  1702→        break;
  1703→      case 'set_max':
  1704→        newValue = Math.max(originalValue, effect.value);
  1705→        break;
  1706→    }
  1707→
  1708→    stat.value = newValue;
  1709→    stat.modified = true;
  1710→    stat.modifier = `${effect.operator}${effect.value}`;
  1711→  }
  1712→
  1713→  return result;
  1714→}

  1716→/**
  1717→ * Build inventory prompt section
  1718→ * Creates a text block showing current inventory, active effects, and currency
  1719→ */
  1720→export function buildInventoryPromptSectionV2(
  1721→  personaId: string,
  1722→  getPersonaItems: () => Array<{ entry: PersonaInventoryEntry; item: Item }>,
  1723→  getEquippedItems: () => Array<{ entry: PersonaInventoryEntry; item: Item }>,
  1724→  activeConsumableEffects: ActiveConsumableEffect[],
  1725→  currency: number,
  1726→  currencyName: string,
  1727→  template: string
  1728→): string {
  1729→  // Build items list
  1730→  const personaItems = getPersonaItems();
  1731→  const equippedItems = getEquippedItems();
  1732→  const personaEffects = activeConsumableEffects.filter(e => e.personaId === personaId);
  1733→
  1734→  const itemLines = personaItems.map(({ entry, item }) => {
  1735→    const qty = entry.quantity > 1 ? ` x${entry.quantity}` : '';
  1736→    const eq = entry.equipped ? ' [Equipado]' : '';
  1737→    const effects = (item.attributeEffects?.length ?? 0) > 0
  1738→      ? ` (${item.attributeEffects!.map(e => `${e.operator}${e.value} ${e.attributeKey}`).join(', ')})`
  1739→      : '';
  1740→    return `- ${item.icon || ''} ${item.name}${qty}${eq}${effects}`;
  1741→  }).join('\n');
  1742→
  1743→  // Build active effects list
  1744→  const effectLines = personaEffects.map(e => {
  1745→    const turnsLeft = e.remainingTurns > 0 ? ` (${e.remainingTurns}/${e.totalTurns} turnos)` : '';
  1746→    const effectDesc = e.effects.map(ef =>
  1747→      `${ef.operator}${ef.value} ${ef.attributeKey}${ef.targetId !== '__user__' ? ` → ${ef.targetName || ef.targetId}` : ''}`
  1748→    ).join(', ');
  1749→    return `- ${e.itemName}: ${effectDesc}${turnsLeft}`;
  1750→  }).join('\n');
  1751→
  1752→  // Build equipped items list
  1753→  const equipLines = equippedItems.map(({ item }) => {
  1754→    const effects = (item.attributeEffects ?? []).map(e => `${e.operator}${e.value} ${e.attributeKey}`).join(', ');
  1755→    return `- ${item.icon || ''} ${item.name}${item.slot ? ` [${item.slot}]` : ''}${effects ? ` → ${effects}` : ''}`;
  1756→  }).join('\n');
  1757→
  1758→  // Build currency
  1759→  const currencyLine = `${currencyName}: ${currency}`;
  1760→
  1761→  return template
  1762→    .replace('{{activeItems}}', itemLines || 'Vacío')
  1763→    .replace('{{activeEffects}}', effectLines || 'Ninguno')
  1764→    .replace('{{equippedItems}}', equipLines || 'Ninguno')
  1765→    .replace('{{currency}}', currencyLine);

  1772→/**
  1773→ * Apply inventory item effects (equipment + consumables) to session stats.
  1774→ * Returns a DEEP COPY of sessionStats with attribute values modified by item effects.
  1775→ * This is called BEFORE resolveStats() so that item effects are reflected in {{key}} templates.
  1776→ *
  1777→ * Item effects can target:
  1778→ * - '__user__' (persona) — applies to persona's attribute values
  1779→ * - characterId — applies to that character's attribute values
  1780→ */
  1781→export function applyInventoryEffectsToSessionStats(
  1782→  sessionStats: SessionStats | undefined,
  1783→  equippedItems: Array<{ entry: PersonaInventoryEntry; item: Item }>,
  1784→  activeEffects: ActiveConsumableEffect[],
  1785→  pendingFallbacks?: Array<{ targetId: string; attributeKey: string; fallbackValue: string | number }>,
  1786→): SessionStats | undefined {
  1787→  if (!sessionStats?.characterStats) return sessionStats;
  1788→
  1789→  // Deep copy sessionStats to avoid mutation
  1790→  const modified: SessionStats = JSON.parse(JSON.stringify(sessionStats));
  1791→
  1792→  // Apply pending fallbacks first (set attribute directly to fallback value)
  1793→  if (pendingFallbacks && pendingFallbacks.length > 0) {
  1794→    for (const fb of pendingFallbacks) {
  1795→      const targetId = fb.targetId || '__user__';
  1796→      const charStats = modified.characterStats[targetId];
  1797→      if (!charStats?.attributeValues) continue;
  1798→
  1799→      const currentValue = charStats.attributeValues[fb.attributeKey];
  1800→      if (currentValue === undefined) continue;
  1801→
  1802→      charStats.attributeValues[fb.attributeKey] = typeof fb.fallbackValue === 'number'
  1803→        ? fb.fallbackValue
  1804→        : (isNaN(Number(fb.fallbackValue)) ? fb.fallbackValue : Number(fb.fallbackValue));
  1805→    }
  1806→  }
  1807→
  1808→  // Collect ALL effects: equipment (permanent while equipped) + consumable (temporary)
  1809→  const allEffects: ItemAttributeEffect[] = [];
  1810→
  1811→  // Equipment effects (from equipped items) — respect targetOverrideId from entry
  1812→  for (const { entry, item } of equippedItems) {
  1813→    if (item.attributeEffects) {
  1814→      for (const effect of item.attributeEffects) {
  1815→        // If the entry has a targetOverrideId, override the effect's targetId
  1816→        const effectiveTargetId = entry.targetOverrideId || effect.targetId;
  1817→        const effectiveTargetName = entry.targetOverrideId
  1818→          ? (entry.targetOverrideId === '__user__' ? 'Persona' : effect.targetName)
  1819→          : effect.targetName;
  1820→        allEffects.push({ ...effect, targetId: effectiveTargetId, targetName: effectiveTargetName });
  1821→      }
  1822→    }
  1823→  }
  1824→
  1825→  // Consumable effects (from active effects)
  1826→  for (const effect of activeEffects) {
  1827→    allEffects.push(...effect.effects);
  1828→  }
  1829→
  1830→  if (allEffects.length === 0 && (!pendingFallbacks || pendingFallbacks.length === 0)) return sessionStats;
  1831→
  1832→  // Group effects by targetId
  1833→  const effectsByTarget = new Map<string, ItemAttributeEffect[]>();
  1834→  for (const effect of allEffects) {
  1835→    const targetId = effect.targetId || '__user__';
  1836→    if (!effectsByTarget.has(targetId)) {
  1837→      effectsByTarget.set(targetId, []);
  1838→    }
  1839→    effectsByTarget.get(targetId)!.push(effect);
  1840→  }
  1841→
  1842→  // Apply effects for each target
  1843→  for (const [targetId, targetEffects] of effectsByTarget) {
  1844→    const charStats = modified.characterStats[targetId];
  1845→    if (!charStats?.attributeValues) continue;
  1846→
  1847→    for (const effect of targetEffects) {
  1848→      const currentValue = charStats.attributeValues[effect.attributeKey];
  1849→      if (currentValue === undefined) continue;
  1850→
  1851→      const currentNum = typeof currentValue === 'number' ? currentValue : parseFloat(String(currentValue));
  1852→      if (isNaN(currentNum)) continue;
  1853→
  1854→      let newValue = currentNum;
  1855→      switch (effect.operator) {
  1856→        case '+': newValue = currentNum + effect.value; break;
  1857→        case '-': newValue = currentNum - effect.value; break;
  1858→        case '*': newValue = currentNum * effect.value; break;
  1859→        case '/': newValue = effect.value !== 0 ? currentNum / effect.value : currentNum; break;
  1860→        case '=': newValue = effect.value; break;
  1861→        case 'set_min': newValue = Math.min(currentNum, effect.value); break;
  1862→        case 'set_max': newValue = Math.max(currentNum, effect.value); break;
  1863→      }
  1864→
  1865→      charStats.attributeValues[effect.attributeKey] = Math.round(newValue * 100) / 100;
  1866→    }
  1867→  }
  1868→
  1869→  return modified;
  1870→}
